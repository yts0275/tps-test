```python
def partition(arr, low, high):
    pivot = arr[high]          # 배열의 마지막 원소를 피벗으로 선택
    i = low - 1                # 피벗보다 작은 원소들의 경계 인덱스 (아직 없으므로 low-1)

    for j in range(low, high): # low부터 high-1까지 순회하며 피벗과 비교
        if arr[j] <= pivot:    # 현재 원소가 피벗 이하이면 왼쪽 파티션에 포함
            i += 1             # 경계 인덱스를 오른쪽으로 한 칸 이동
            arr[i], arr[j] = arr[j], arr[i]  # 경계 원소와 현재 원소를 교환

    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 최종 위치(경계 바로 오른쪽)로 이동
    return i + 1               # 피벗의 최종 인덱스를 반환


def quicksort(arr, low, high):
    if low < high:             # 부분 배열의 크기가 2 이상일 때만 재귀 진행
        pi = partition(arr, low, high)   # 파티션을 수행하고 피벗의 위치(pi)를 구함
        quicksort(arr, low, pi - 1)      # 피벗 왼쪽 부분 배열을 재귀적으로 정렬
        quicksort(arr, pi + 1, high)     # 피벗 오른쪽 부분 배열을 재귀적으로 정렬


if __name__ == "__main__":
    arr = [10, 7, 8, 9, 1, 5]  # 정렬할 예시 배열
    n = len(arr)               # 배열의 길이
    quicksort(arr, 0, n - 1)   # 배열 전체에 대해 퀵소트 실행 (인덱스 0 ~ n-1)
    print("정렬 결과:", arr)    # 정렬된 배열 출력
```

**구조 설명:**

- `partition()` — 피벗(마지막 원소)을 기준으로 배열을 두 파티션으로 나누고, 피벗의 최종 위치를 반환합니다.
- `quicksort()` — 피벗 위치를 기준으로 왼쪽/오른쪽을 재귀적으로 정렬합니다.

**시간 복잡도:** 평균 O(n log n), 최악 O(n²) (이미 정렬된 배열 + 마지막 원소 피벗 시)
