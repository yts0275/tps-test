```python
def quicksort(arr):                                    # 퀵소트 함수 정의, arr: 정렬할 리스트
    if len(arr) <= 1:                                  # 원소가 0개 또는 1개이면 이미 정렬된 상태
        return arr                                     # 그대로 반환 (재귀 종료 조건)

    pivot = arr[len(arr) // 2]                         # 중간 인덱스 원소를 피벗으로 선택
    left = [x for x in arr if x < pivot]              # 피벗보다 작은 원소들을 왼쪽 배열에 수집
    middle = [x for x in arr if x == pivot]           # 피벗과 같은 원소들을 중간 배열에 수집 (중복 처리)
    right = [x for x in arr if x > pivot]             # 피벗보다 큰 원소들을 오른쪽 배열에 수집

    return quicksort(left) + middle + quicksort(right) # 왼쪽/오른쪽 재귀 정렬 후 합쳐서 반환


if __name__ == "__main__":
    data = [3, 6, 8, 10, 1, 2, 1]                     # 테스트용 입력 배열
    print("정렬 전:", data)                             # 정렬 전 배열 출력
    result = quicksort(data)                           # 퀵소트 실행
    print("정렬 후:", result)                           # 정렬 후 배열 출력
```

**동작 방식:**
1. **피벗 선택** — 배열 중간값을 피벗으로 사용
2. **분할** — 피벗 기준으로 left / middle / right 세 그룹으로 분리
3. **재귀** — left와 right를 재귀적으로 정렬 후 합산

**시간복잡도:** 평균 O(n log n), 최악 O(n²)
