아래는 퀵소트 구현 코드입니다.

```python
def quicksort(arr):                                    # quicksort 함수 정의: 리스트를 인자로 받음
    if len(arr) <= 1:                                  # 원소가 0개 또는 1개면 이미 정렬된 상태
        return arr                                     # 그대로 반환 (재귀의 기저 조건)

    pivot = arr[len(arr) // 2]                         # 중간 인덱스의 원소를 피벗으로 선택
    left = [x for x in arr if x < pivot]              # 피벗보다 작은 원소들을 왼쪽 리스트에 수집
    middle = [x for x in arr if x == pivot]            # 피벗과 같은 원소들을 중간 리스트에 수집 (중복 처리)
    right = [x for x in arr if x > pivot]             # 피벗보다 큰 원소들을 오른쪽 리스트에 수집

    return quicksort(left) + middle + quicksort(right) # 왼쪽 재귀정렬 + 중간 + 오른쪽 재귀정렬을 합쳐서 반환


def quicksort_inplace(arr, low, high):                 # 제자리(in-place) 퀵소트 정의: 배열과 범위 인덱스를 받음
    if low < high:                                     # 정렬 범위가 2개 이상의 원소를 포함할 때만 진행
        pivot_idx = partition(arr, low, high)          # 분할 수행 후 피벗의 최종 위치 인덱스를 받음
        quicksort_inplace(arr, low, pivot_idx - 1)     # 피벗 왼쪽 부분 배열을 재귀적으로 정렬
        quicksort_inplace(arr, pivot_idx + 1, high)    # 피벗 오른쪽 부분 배열을 재귀적으로 정렬


def partition(arr, low, high):                         # 분할 함수 정의: 피벗을 기준으로 배열을 두 부분으로 나눔
    pivot = arr[high]                                  # 마지막 원소를 피벗으로 선택
    i = low - 1                                        # i는 피벗보다 작은 원소들의 경계 인덱스 (초기값: 범위 시작 전)

    for j in range(low, high):                         # j를 범위 시작부터 피벗 직전까지 순회
        if arr[j] <= pivot:                            # 현재 원소가 피벗 이하이면
            i += 1                                     # 경계를 한 칸 오른쪽으로 이동
            arr[i], arr[j] = arr[j], arr[i]            # 경계 원소와 현재 원소를 교환하여 작은 쪽으로 이동

    arr[i + 1], arr[high] = arr[high], arr[i + 1]      # 피벗을 최종 위치(경계 다음 자리)로 이동
    return i + 1                                       # 피벗의 최종 인덱스를 반환


if __name__ == "__main__":                             # 직접 실행할 때만 아래 코드 수행
    data = [3, 6, 8, 10, 1, 2, 1]                     # 테스트용 정렬되지 않은 리스트 정의

    result = quicksort(data)                           # 새 리스트를 반환하는 방식으로 정렬 수행
    print("새 리스트 방식:", result)                    # 정렬 결과 출력

    data2 = [3, 6, 8, 10, 1, 2, 1]                    # 제자리 정렬을 위한 별도 테스트 리스트
    quicksort_inplace(data2, 0, len(data2) - 1)        # 원본 리스트를 직접 수정하는 방식으로 정렬
    print("제자리 방식:  ", data2)                     # 정렬된 원본 리스트 출력
```

---

두 가지 구현 방식을 제공했습니다.

**`quicksort`** — 새 리스트를 생성하는 함수형 방식. 코드가 직관적이고 중복 원소를 자연스럽게 처리합니다.

**`quicksort_inplace`** — 원본 배열을 직접 수정하는 방식. 메모리 할당 없이 동작하므로 공간 효율이 좋습니다 (`O(log n)` 스택 공간만 사용).
