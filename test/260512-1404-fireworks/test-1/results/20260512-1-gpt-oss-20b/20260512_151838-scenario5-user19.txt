```python
# ------------------------------------------------------------------------------------------------
# 빠른 정렬(퀵소트) 알고리즘을 구현하고, 각 줄마다 주석을 달았습니다.
# ------------------------------------------------------------------------------------------------

def quicksort(arr):
    # quicksort 함수: 재귀적으로 배열을 정렬합니다. 
    # 배열을 그대로 수정하고 싶다면, in-place 방식을 따르는 것이 좋지만, 
    # 이 구현은 새로운 리스트를 반환합니다. 
    if len(arr) <= 1:          # 배열이 0개 또는 1개라면 이미 정렬된 상태이므로 그대로 반환
        return arr             # 정렬된 배열을 반환

    pivot = arr[-1]            # 피벗을 선택: 보통 배열의 마지막 원소를 사용합니다.
    left = []                  # 피벗 보다 작거나 같은 값을 저장할 리스트
    right = []                 # 피벗 보다 큰 값을 저장할 리스트

    # 피벗을 제외한 나머지 원소를 순회하며 두 리스트로 분류합니다.
    for x in arr[:-1]:         # arr[:-1] 은 피벗을 제외한 모든 원소를 가져옵니다.
        if x <= pivot:        # 현재 원소가 피벗 이하라면 left 리스트에 추가
            left.append(x)
        else:                 # 그렇지 않으면 right 리스트에 추가
            right.append(x)

    # 재귀 호출을 이용해 두 부분을 정렬하고, 피벗을 중앙에 놓아 최종 정렬된 배열을 반환합니다.
    return quicksort(left) + [pivot] + quicksort(right)

# ------------------------------------------------------------------------------------------------
# 아래는 quicksort 함수를 테스트하기 위한 예시입니다.
# ------------------------------------------------------------------------------------------------
if __name__ == "__main__":
    unsorted = [3, 6, 8, 10, 1, 2, 1]   # 정렬되지 않은 배열
    print("정렬 전:", unsorted)        # 원본 배열 출력
    sorted_arr = quicksort(unsorted)     # quicksort 함수 호출
    print("정렬 후:", sorted_arr)       # 정렬된 배열 출력
```
