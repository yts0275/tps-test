```python
# 퀵소트(QuickSort) 구현 예제
# 각 줄마다 주석을 달아 단계별 동작을 설명합니다.

def quick_sort(arr, low, high):
    """
    퀵소트 함수
    arr : 정렬할 리스트
    low : 시작 인덱스 (보통 0)
    high: 끝 인덱스 (보통 len(arr)-1)
    """
    # 현재 배열 범위에 2개 이상의 요소가 있는지 확인.
    # 하나 이하이면 이미 정렬된 상태라서 종료.
    if low < high:
        # 피벗(구분자) 선택 및 분할
        # partition 함수는 피벗을 기준으로 왼쪽은 피벗보다 작은 값,
        # 오른쪽은 피벗보다 큰 값으로 재배치하고,
        # 피벗의 최종 위치 인덱스를 반환한다.
        pivot_index = partition(arr, low, high)
        
        # 피벗 앞부분(좌측)을 재귀적으로 정렬
        quick_sort(arr, low, pivot_index - 1)
        # 피벗 뒷부분(우측)을 재귀적으로 정렬
        quick_sort(arr, pivot_index + 1, high)

def partition(arr, low, high):
    """
    파티션(분할) 함수
    - arr[low] 를 피벗으로 사용
    - 피벗보다 작은 값은 왼쪽, 큰 값은 오른쪽에 배치
    - 완성된 뒤 피벗이 위치한 인덱스를 반환
    """
    # 첫 번째 요소를 피벗으로 지정
    pivot = arr[low]
    
    # 왼쪽(보조 인덱스)과 오른쪽(트래킹 인덱스) 포인터
    left = low + 1
    right = high
    
    # while 루프 내에서 왼쪽과 오른쪽 포인터가 교차할 때까지 이동
    while True:
        # 피벗보다 큰 값을 찾을 때까지 왼쪽 포인터를 오른쪽으로 이동
        while left <= right and arr[left] <= pivot:
            left += 1
        # 피벗보다 작은 값을 찾을 때까지 오른쪽 포인터를 왼쪽으로 이동
        while left <= right and arr[right] > pivot:
            right -= 1
        
        # 두 포인터가 교차하면 더 이상 교환할 필요가 없다.
        if left > right:
            break
        # 교차하지 않았다면 (left, right) 위치의 원소를 서로 교환
        arr[left], arr[right] = arr[right], arr[left]
    
    # 피벗을 적절한 위치에 삽입
    arr[low], arr[right] = arr[right], arr[low]
    
    # 피벗이 정렬된 최종 인덱스 반환
    return right

# 예시: 퀵소트를 사용해 리스트 정렬
if __name__ == "__main__":
    # 테스트용 리스트
    lst = [8, 3, 5, 4, 7, 6, 1, 2]
    print("정렬 전:", lst)
    
    # 퀵소트 호출: 인덱스 범위는 0부터 len(lst)-1까지
    quick_sort(lst, 0, len(lst) - 1)
    
    print("정렬 후 :", lst)
```

**설명**  
- `quick_sort` 함수는 재귀적으로 호출되며, `partition` 함수를 통해 배열을 피벗을 중심으로 두 부분으로 나눈 뒤, 필요한 부분을 다시 정렬합니다.  
- `partition` 함수는 피벗을 기준으로 왼쪽에는 작은 값, 오른쪽에는 큰 값을 배치하는 전형적인 방법을 사용합니다.  
- 최종적으로 `if __name__ == "__main__":` 블록에서 예시 리스트를 정렬해 보입니다.  
- 모든 줄에 주석을 달았으며, 각 주석은 해당 줄이 수행하는 역할을 한국어로 설명합니다.
