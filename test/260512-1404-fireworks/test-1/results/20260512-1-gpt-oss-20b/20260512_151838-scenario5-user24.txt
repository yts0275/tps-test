```python
# 보조 함수: 피봇 위치를 맞춘 뒤, 피봇보다 작은 값은 앞쪽에 정렬한다.
def partition(arr, low, high):  # 함수 정의
    pivot = arr[high]          # 피봇을 배열의 마지막 요소로 설정
    i = low - 1                # i는 작은 값들이 모인 영역의 마지막 인덱스
    for j in range(low, high): # j로 배열을 순회하면서 피봇보다 작은 값을 찾아
        if arr[j] <= pivot:    # 조건을 만족하면 i를 증가시키고 두 위치를 스왑
            i += 1            
            arr[i], arr[j] = arr[j], arr[i]
    arr[i+1], arr[high] = arr[high], arr[i+1] # 피봇을 올바른 위치에 두고 반환
    return i + 1                                      # 피봇 인덱스를 반환

# 퀵소트 메인 함수: 재귀적으로 정렬을 수행한다.
def quicksort(arr, low, high):          # 함수 정의
    if low < high:                      # 재귀 종료 조건: 최소 두 개의 원소가 있을 때만 진행
        pi = partition(arr, low, high) # 피봇 위치 찾고 배열 분할
        quicksort(arr, low, pi - 1)    # 피봇 앞부분 정렬
        quicksort(arr, pi + 1, high)   # 피봇 뒷부분 정렬

# 사용 예시: 리스트를 정렬하고 결과를 출력한다.
if __name__ == "__main__":           # 직접 실행될 때만 동작
    array = [10, 7, 8, 9, 1, 5]      # 정렬할 원본 배열
    quicksort(array, 0, len(array) - 1) # 퀵소트 호출
    print("정렬된 배열:", array)          # 정렬 결과 출력
```
