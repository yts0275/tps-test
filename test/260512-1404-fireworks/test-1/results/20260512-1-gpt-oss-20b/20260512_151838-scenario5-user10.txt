```python
def quicksort(arr):  # quicksort 함수 정의: 입력 배열 arr을 정렬된 리스트로 반환
    if len(arr) <= 1:  # 배열이 0개 또는 1개 원소인 경우 이미 정렬된 상태다
        return arr  # 그대로 반환
    pivot = arr[len(arr) // 2]  # 피벗 선택: 중간 원소를 사용
    left = [x for x in arr if x < pivot]  # 피벗보다 작은 원소들 모음
    middle = [x for x in arr if x == pivot]  # 피벗과 같은 원소들 모음
    right = [x for x in arr if x > pivot]  # 피벗보다 큰 원소들 모음
    return quicksort(left) + middle + quicksort(right)  # 재귀 호출로 정렬 후 합치기
```
