```python
def quicksort(arr):  # quicksort 함수 정의: arr 리스트를 정렬
    if len(arr) <= 1:  # 기본 케이스: 리스트 길이가 1 이하이면 이미 정렬됨
        return arr     # 정렬된 리스트를 그대로 반환

    pivot = arr[len(arr)//2]  # 피벗 선택: 리스트의 중앙 요소를 기준으로 사용
    left = [x for x in arr if x < pivot]    # 피벗보다 작은 모든 요소를 모은 리스트
    middle = [x for x in arr if x == pivot] # 피벗과 동일한 값(중복이 있을 경우)를 모은 리스트
    right = [x for x in arr if x > pivot]   # 피벗보다 큰 모든 요소를 모은 리스트

    # 왼쪽 리스트를 재귀적으로 정렬하고, 피벗(중간)과 오른쪽 리스트를 정렬한 뒤 합쳐서 반환
    return quicksort(left) + middle + quicksort(right)
```
