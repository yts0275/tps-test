def quick_sort(arr):
    # 함수 시작: 인자 arr을 정렬해 반환
    if len(arr) <= 1:
        # 조건: 리스트 길이가 1 이하이면 정렬이 필요 없으므로 그대로 반환
        return arr
    # 피벗을 배열의 마지막 원소로 설정
    pivot = arr[-1]
    # 피벗보다 작거나 같은 요소를 저장할 리스트
    left = []
    # 피벗보다 큰 요소를 저장할 리스트
    right = []
    # 피벗을 제외한 모든 원소를 순회
    for x in arr[:-1]:
        # x가 피벗보다 작거나 같으면 left에 추가
        if x <= pivot:
            left.append(x)
        # 그렇지 않으면 right에 추가
        else:
            right.append(x)
    # 재귀적으로 left를 정렬하고, 피벗, 그리고 정렬된 right를 연결해 반환
    return quick_sort(left) + [pivot] + quick_sort(right)

if __name__ == "__main__":
    # 테스트용 배열 정의
    arr = [3, 6, 8, 10, 1, 2, 1]
    # quick_sort 함수 호출
    sorted_arr = quick_sort(arr)
    # 정렬된 결과 출력
    print(sorted_arr)
