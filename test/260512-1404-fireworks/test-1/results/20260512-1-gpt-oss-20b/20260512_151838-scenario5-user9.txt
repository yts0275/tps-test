def quicksort(arr, low, high):  # 퀵소트 함수: arr[low~high]까지 정렬
    if low < high:  # 정렬 범위가 두 개 이상의 원소를 포함하면
        pi = partition(arr, low, high)  # 피벗을 기준으로 배열을 분할하고 피벗이 끝날 인덱스를 반환
        quicksort(arr, low, pi - 1)  # 피벗 앞 부분을 재귀적으로 정렬
        quicksort(arr, pi + 1, high)  # 피벗 뒤 부분을 재귀적으로 정렬


def partition(arr, low, high):  # 피벗을 기준으로 배열을 분할하는 함수
    pivot = arr[high]  # 배열의 가장 우측 원소를 피벗으로 선택
    i = low - 1  # i는 피벗보다 작은 원소의 마지막 위치를 가리킴
    for j in range(low, high):  # j는 정렬 범위 안의 모든 인덱스를 순회
        if arr[j] <= pivot:  # 현재 원소가 피벗보다 작거나 같다면
            i += 1  # i를 한 칸 오른쪽으로 이동
            arr[i], arr[j] = arr[j], arr[i]  # i와 j에 있는 원소를 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 올바른 위치에 옮김
    return i + 1  # 피벗이 최종적으로 위치한 인덱스를 반환


# 예시 사용법
arr = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 배열
quicksort(arr, 0, len(arr)-1)  # 퀵소트 호출
print(arr)  # 정렬된 결과 출력
