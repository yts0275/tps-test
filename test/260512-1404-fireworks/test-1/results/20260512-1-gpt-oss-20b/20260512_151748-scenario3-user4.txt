```python
def quicksort(arr):  # quicksort 함수 정의, 입력은 정렬할 리스트
    if len(arr) <= 1:  # 리스트 길이가 1 이하라면 그대로 반환
        return arr  # 정렬이 필요 없으므로 원본 리스트 반환
    pivot = arr[len(arr)//2]  # 리스트의 중앙값을 피벗으로 선택
    left = [x for x in arr if x < pivot]  # 피벗보다 작은 원소들
    middle = [x for x in arr if x == pivot]  # 피벗과 같은 원소들
    right = [x for x in arr if x > pivot]  # 피벗보다 큰 원소들
    return quicksort(left) + middle + quicksort(right)  # 재귀 호출로 정렬된 결과 합치기

if __name__ == "__main__":  # 스크립트가 직접 실행될 때만 아래 코드를 수행
    arr = [3, 6, 8, 10, 1, 2, 1]  # 예시 배열 정의
    print("Original:", arr)  # 원본 배열 출력
    sorted_arr = quicksort(arr)  # quicksort 함수 호출
    print("Sorted:", sorted_arr)  # 정렬된 배열 출력
```
