```python
# quicksort.py
# -----------------------------------------------------------
# 핵심: 퀵소트(Quick Sort) 알고리즘을 재귀적으로 구현하고
#       각 줄마다 주석을 달아 이해를 돕는다.
# -----------------------------------------------------------

def quicksort(arr):
    # 1. 입력 배열의 길이가 0 또는 1이면 이미 정렬된 상태이므로 그대로 반환
    if len(arr) <= 1:
        return arr        # 재귀 종료 조건

    # 2. 가장 오른쪽 원소를 피벗(pivot) 으로 선택한다
    pivot = arr[-1]       # 피벗 선택

    # 3. 피벗보다 작은 값들만 모아 리스트 `less` 를 만든다
    less = [x for x in arr[:-1] if x < pivot]   # 피벗보다 작은 원소들

    # 4. 피벗과 같은 값들만 모아 리스트 `equal` 을 만든다
    equal = [x for x in arr if x == pivot]      # 피벗과 같은 원소들

    # 5. 피벗보다 큰 값들만 모아 리스트 `greater` 를 만든다
    greater = [x for x in arr[:-1] if x > pivot]  # 피벗보다 큰 원소들

    # 6. 재귀적으로 `less` 와 `greater` 를 각각 정렬한 뒤,
    #    `equal` 을 사이에 넣어 최종 정렬된 리스트를 반환한다
    return quicksort(less) + equal + quicksort(greater)   # 정렬된 리스트 합치기


# -----------------------------------------------------------
# 테스트용 메인 코드
# -----------------------------------------------------------
if __name__ == "__main__":
    # 7. 예시 배열을 정의한다
    sample = [3, 6, 1, 8, 4, 5, 5, 0, -1, 10]

    # 8. quicksort 함수를 호출하고 결과를 출력한다
    sorted_array = quicksort(sample)      # 퀵소트 수행
    print("원본 배열:", sample)           # 원본 배열 표시
    print("정렬된 배열:", sorted_array)   # 정렬 결과 표시
```

각 줄에 `#` 뒤에 간결한 설명이 주석으로 달려 있어, 퀵소트의 흐름과 각 부분이 수행하는 역할을 한눈에 파악할 수 있습니다.  
코드를 실행하면 원본 배열과 정렬된 배열이 출력됩니다.
