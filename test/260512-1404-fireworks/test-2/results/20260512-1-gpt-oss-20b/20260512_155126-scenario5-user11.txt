```python
def quicksort(arr):
    # 1️⃣ 길이가 0 또는 1인 배열은 이미 정렬돼 있으므로 그대로 반환
    if len(arr) <= 1:
        return arr

    # 2️⃣ 피벗을 선택 (여기서는 배열의 중간 원소를 사용)
    pivot = arr[len(arr) // 2]

    # 3️⃣ 피벗보다 작은 값들을 모음 (리스트 컴프리헨션 사용)
    left = [x for x in arr if x < pivot]
    # 4️⃣ 피벗과 같은 값들을 모음 (고루 같은 값이 있을 경우 중복을 유지)
    middle = [x for x in arr if x == pivot]
    # 5️⃣ 피벗보다 큰 값들을 모음
    right = [x for x in arr if x > pivot]

    # 6️⃣ 재귀적으로 정렬한 결과를 연결해서 반환
    return quicksort(left) + middle + quicksort(right)
```
