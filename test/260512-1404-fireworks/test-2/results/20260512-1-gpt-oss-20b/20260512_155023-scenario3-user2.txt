```python
# QuickSort를 구현하는 파이썬 코드
# (프로그램의 모든 행에 주석을 달았습니다.)

def quick_sort(arr, low, high):
    # 재귀 종료 조건: 배열의 절반 이하가 남으면 종료
    if low < high:
        # 파티션(분할) 수행 후 피벗 인덱스를 반환
        pivot_index = partition(arr, low, high)
        # 피벗 앞부분을 재귀적으로 정렬
        quick_sort(arr, low, pivot_index - 1)
        # 피벗 뒷부분을 재귀적으로 정렬
        quick_sort(arr, pivot_index + 1, high)

def partition(arr, low, high):
    # 마지막 원소를 피벗으로 선택
    pivot = arr[high]
    # 피벗보다 작은 값을 왼쪽에 모으기 위한 인덱스
    i = low - 1
    # low부터 high-1까지 순회
    for j in range(low, high):
        # 현재 원소가 피벗보다 작으면
        if arr[j] < pivot:
            # i를 1 증가시켜요
            i += 1
            # arr[j]와 arr[i]를 교환
            arr[i], arr[j] = arr[j], arr[i]
    # 피벗을 올바른 위치(인덱스 i+1)로 이동
    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    # 피벗의 최종 인덱스를 반환
    return i + 1

# 예시 메인 실행 부분
if __name__ == "__main__":
    # 정수 리스트 입력 예시
    data = [34, 7, 23, 32, 5, 62]
    # 리스트 길이와 초기 인덱스, 마지막 인덱스를 얻어 QuickSort 호출
    quick_sort(data, 0, len(data) - 1)
    # 정렬 결과 출력
    print("정렬된 리스트:", data)
```

이 코드는 각 줄마다 한국어로 주석을 달아 이해를 돕고, 재귀적으로 QuickSort를 수행합니다.
