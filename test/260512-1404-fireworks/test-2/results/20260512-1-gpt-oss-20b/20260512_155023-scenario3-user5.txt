def quicksort(arr, low, high):  # 퀵소트 함수: arr 배열을 low부터 high까지 정렬한다
    if low < high:  # 재귀 종료 조건
        pivot_index = partition(arr, low, high)  # 피벗 위치를 찾는다
        quicksort(arr, low, pivot_index-1)  # 피벗 왼쪽 부분을 정렬한다
        quicksort(arr, pivot_index+1, high)  # 피벗 오른쪽 부분을 정렬한다

def partition(arr, low, high):  # 파티션 함수: 피벗을 기준으로 배열을 재배치한다
    pivot = arr[high]  # 마지막 원소를 피벗으로 선택한다
    i = low - 1  # 피벗보다 작은 원소의 가장 오른쪽 인덱스
    for j in range(low, high):  # 피벗을 제외한 모든 원소를 검사한다
        if arr[j] <= pivot:  # 현재 원소가 피벗보다 작거나 같으면
            i += 1  # 작은 원소 영역을 확장한다
            arr[i], arr[j] = arr[j], arr[i]  # i와 j 위치의 원소를 교환한다
    arr[i+1], arr[high] = arr[high], arr[i+1]  # 피벗을 올바른 위치에 놓는다
    return i+1  # 피벗의 최종 인덱스를 반환한다

if __name__ == "__main__":  # 스크립트가 직접 실행될 때만 실행된다
    arr = [10, 7, 8, 9, 1, 5]  # 예시 배열
    quicksort(arr, 0, len(arr)-1)  # 배열을 정렬한다
    print("Sorted array:", arr)  # 정렬된 배열을 출력한다
