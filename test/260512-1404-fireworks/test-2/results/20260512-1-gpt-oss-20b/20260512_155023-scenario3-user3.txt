```python
def quicksort(arr):  # 퀵소트 함수 정의: 주어진 리스트를 재귀적으로 정렬한다.
    if len(arr) <= 1:  # 리스트 길이가 1 이하이면 정렬할 필요가 없으므로 그대로 반환한다.
        return arr  # 정렬이 필요 없는 리스트를 그대로 반환
    pivot = arr[len(arr)//2]  # 리스트의 중간 요소를 피벗으로 선택한다.
    left = [x for x in arr if x < pivot]  # 피벗보다 작은 모든 요소를 모아 새로운 리스트를 만든다.
    middle = [x for x in arr if x == pivot]  # 피벗과 동일한 값들을 모아 새로운 리스트를 만든다.
    right = [x for x in arr if x > pivot]  # 피벗보다 큰 모든 요소를 모아 새로운 리스트를 만든다.
    # 재귀적으로 left와 right를 정렬한 뒤, 정렬된 리스트를 중간 리스트와 합쳐 최종 정렬 결과를 반환한다.
    return quicksort(left) + middle + quicksort(right)
```
