def quicksort(arr):  # 함수 정의: 입력 배열을 정렬해서 반환한다.
    if len(arr) <= 1:  # 길이가 0 또는 1이면 이미 정렬된 상태이므로 그대로 반환
        return arr  # 정렬된 배열을 그대로 반환한다.
    pivot = arr[len(arr) // 2]  # 배열의 중앙 요소를 피벗으로 선택한다.
    left = [x for x in arr if x < pivot]  # 피벗보다 작은 값들로만 이루어진 리스트를 만든다.
    middle = [x for x in arr if x == pivot]  # 피벗과 같은 값들로만 이루어진 리스트를 만든다.
    right = [x for x in arr if x > pivot]  # 피벗보다 큰 값들로만 이루어진 리스트를 만든다.
    return quicksort(left) + middle + quicksort(right)  # 왼쪽, 중앙, 오른쪽 리스트를 재귀적으로 정렬한 뒤 합친다.

# --- 테스트 예시 ----------------------------------------------------------------
if __name__ == "__main__":  # 스크립트가 직접 실행될 때만 아래 코드가 실행된다.
    arr = [3, 6, 8, 10, 1, 2, 1]  # 정렬되지 않은 예시 배열
    sorted_arr = quicksort(arr)  # 퀵소트 함수 호출
    print(sorted_arr)  # 정렬된 배열 출력
