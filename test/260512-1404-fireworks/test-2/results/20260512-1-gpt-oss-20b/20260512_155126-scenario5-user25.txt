```python
# 퀵소트(QuickSort) 구현 – Python 3

def quicksort(arr):
    # 재귀의 기본 상황: 크기가 0이거나 1인 배열은 이미 정렬돼 있음
    if len(arr) <= 1:
        # 그대로 반환
        return arr

    # 피벗(pivot) 요소를 선택 (보통 중앙값을 사용)
    pivot = arr[len(arr) // 2]

    # 피벗보다 작은 모든 원소들을 모음
    left = [x for x in arr if x < pivot]
    # 피벗과 같은 값들(같은 원소들이 있으면 그들을 한 묶음으로 처리)
    middle = [x for x in arr if x == pivot]
    # 피벗보다 큰 모든 원소들을 모음
    right = [x for x in arr if x > pivot]

    # 재귀적으로 정렬한 왼쪽, 중간, 오른쪽 배열을 순서대로 이어서 반환
    return quicksort(left) + middle + quicksort(right)

# 예시 사용
if __name__ == "__main__":
    data = [33, 10, 55, 71, 29, 3, 18]
    print("원본:", data)
    sorted_data = quicksort(data)
    print("정렬된 결과:", sorted_data)
```
