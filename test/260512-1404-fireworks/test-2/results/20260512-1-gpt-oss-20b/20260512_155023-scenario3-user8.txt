def quicksort(arr, low, high):  # Quicksort 함수: arr의 low~high 구간을 정렬
    if low < high:  # 정렬할 구간이 두 개 이상 있을 때만 실행
        pivot_index = partition(arr, low, high)  # 피벗 위치 찾기
        quicksort(arr, low, pivot_index - 1)  # 피벗 앞부분 재귀 정렬
        quicksort(arr, pivot_index + 1, high)  # 피벗 뒷부분 재귀 정렬

def partition(arr, low, high):  # 배열을 피벗을 기준으로 두 부분으로 나누는 함수
    pivot = arr[high]  # 가장 오른쪽 값을 피벗으로 선택
    i = low - 1  # i는 피벗보다 작은 요소의 마지막 인덱스를 가리킴
    for j in range(low, high):  # j는 현재 검사 중인 요소를 가리킴
        if arr[j] <= pivot:  # 현재 요소가 피벗보다 작거나 같으면
            i += 1  # i를 한 칸 오른쪽으로 이동
            arr[i], arr[j] = arr[j], arr[i]  # i와 j 위치의 요소를 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 올바른 위치에 놓음
    return i + 1  # 피벗의 최종 인덱스를 반환

# 예시 실행
if __name__ == "__main__":  # 직접 실행할 때만 아래 코드 실행
    data = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 배열
    quicksort(data, 0, len(data) - 1)  # quicksort 호출
    print("정렬된 배열:", data)  # 정렬 결과 출력
