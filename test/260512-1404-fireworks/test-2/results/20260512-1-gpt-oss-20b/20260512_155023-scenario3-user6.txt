def quicksort(arr, low, high):                     # # 재귀적으로 배열을 정렬하는 함수
    if low < high:                                 # # 만약 왼쪽 인덱스가 오른쪽 인덱스보다 작을 때만 진행
        pi = partition(arr, low, high)             # # 배열을 분할하고 피벗 인덱스를 가져옴
        quicksort(arr, low, pi - 1)                # # 피벗 왼쪽 부분 재귀 호출
        quicksort(arr, pi + 1, high)               # # 피벗 오른쪽 부분 재귀 호출

def partition(arr, low, high):                     # # 배열을 피벗을 기준으로 분할하는 함수
    pivot = arr[high]                              # # 피벗 원소를 배열의 마지막 원소로 선택
    i = low - 1                                    # # i는 피벗보다 작은 값이 위치할 인덱스를 가리킴
    for j in range(low, high):                     # # j를 사용해 배열을 한 번 순회
        if arr[j] <= pivot:                        # # 현재 원소가 피벗보다 작거나 같다면
            i += 1                                 # # i를 한 칸 뒤로 이동
            arr[i], arr[j] = arr[j], arr[i]        # # i와 j가 가리키는 원소를 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1] # # 피벗을 올바른 위치(정렬된 위치)로 이동
    return i + 1                                   # # 분할 인덱스 반환

# 예시 사용법
arr = [3, 6, 8, 10, 1, 2, 1]                     # # 정렬할 배열 정의
quicksort(arr, 0, len(arr) - 1)                  # # 배열 전체를 빠른 정렬
print(arr)                                       # # 정렬된 배열 출력
