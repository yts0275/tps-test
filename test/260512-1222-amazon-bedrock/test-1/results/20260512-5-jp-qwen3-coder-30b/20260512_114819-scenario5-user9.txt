```python
def quicksort(arr, low, high):
    """
    퀵정렬 함수: 배열을 재귀적으로 정렬합니다.
    - arr: 정렬할 배열
    - low: 정렬 범위의 시작 인덱스
    - high: 정렬 범위의 끝 인덱스
    """
    # 기저 조건: low가 high보다 크거나 같으면 더 이상 정렬할 필요가 없습니다.
    if low < high:
        # 피벗을 기준으로 배열을 분할하고, 피벗의 올바른 위치를 반환받습니다.
        pi = partition(arr, low, high)

        # 피벗의 왼쪽 부분 배열을 재귀적으로 정렬합니다.
        quicksort(arr, low, pi - 1)

        # 피벗의 오른쪽 부분 배열을 재귀적으로 정렬합니다.
        quicksort(arr, pi + 1, high)


def partition(arr, low, high):
    """
    파티션 함수: 피벗을 기준으로 배열을 두 그룹으로 나눕니다.
    - arr: 정렬할 배열
    - low: 파티션 범위의 시작 인덱스
    - high: 파티션 범위의 끝 인덱스
    """
    # 배열의 마지막 요소를 피벗으로 선택합니다.
    pivot = arr[high]

    # i는 작은 요소들의 인덱스를 나타냅니다.
    i = low - 1

    # j는 현재 검사 중인 요소의 인덱스입니다.
    for j in range(low, high):
        # 만약 현재 요소가 피벗보다 작거나 같다면
        if arr[j] <= pivot:
            # i를 증가시키고, 현재 요소와 i 위치의 요소를 교환합니다.
            i += 1
            arr[i], arr[j] = arr[j], arr[i]

    # 피벗을 올바른 위치로 이동시킵니다.
    arr[i + 1], arr[high] = arr[high], arr[i + 1]

    # 피벗이 있는 새로운 위치를 반환합니다.
    return i + 1


# 사용 예제
if __name__ == "__main__":
    # 정렬할 배열을 생성합니다.
    arr = [10, 7, 8, 9, 1, 5]
    
    # 배열의 길이를 구합니다.
    n = len(arr)
    
    # 퀵정렬 함수를 호출하여 배열을 정렬합니다.
    quicksort(arr, 0, n - 1)
    
    # 정렬된 배열을 출력합니다.
    print("정렬된 배열:", arr)
```

### 설명:

- `quicksort()` 함수는 주어진 배열을 재귀적으로 정렬합니다.  
- `partition()` 함수는 피벗을 기준으로 배열을 잘라 두 개의 부분 배열로 나누고, 피벗이 들어갈 위치를 반환합니다.

이 구현은 평균적으로 O(n log n) 시간 복잡도를 가지며, 공간 복잡도는 O(log n)입니다.
