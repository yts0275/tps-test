```python
# 퀵정렬(QuickSort) 알고리즘 구현

def quicksort(arr, low, high):
    # 재귀적인 호출을 종료하는 조건: low가 high보다 크거나 같을 때
    if low < high:
        # 파티션 함수를 호출하여 기준값(pivot)의 올바른 위치를 찾음
        pivot_index = partition(arr, low, high)
        
        # 기준값 왼쪽 부분 배열에 대해 재귀적으로 퀵정렬 수행
        quicksort(arr, low, pivot_index - 1)
        
        # 기준값 오른쪽 부분 배열에 대해 재귀적으로 퀵정렬 수행
        quicksort(arr, pivot_index + 1, high)

def partition(arr, low, high):
    # 마지막 요소를 기준값(pivot)으로 설정
    pivot = arr[high]
    
    # i는 작은 원소들의 인덱스를 나타내며, 초기에는 low-1로 시작함
    i = low - 1
    
    # j는 배열을 순회하면서 처리할 인덱스
    for j in range(low, high):
        # 현재 원소가 기준값보다 작거나 같으면
        if arr[j] <= pivot:
            # i를 증가시킨 후, i와 j의 값을 교환
            i += 1
            arr[i], arr[j] = arr[j], arr[i]
    
    # 기준값을 올바른 위치로 이동 (i+1 위치에 배치)
    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    
    # 기준값이 들어갈 정확한 위치인 인덱스를 반환
    return i + 1

# 예제 실행 코드
if __name__ == "__main__":
    # 정렬할 배열 정의
    example_array = [10, 7, 8, 9, 1, 5]
    
    print("정렬 전 배열:", example_array)
    
    # 퀵정렬 실행
    quicksort(example_array, 0, len(example_array) - 1)
    
    print("정렬 후 배열:", example_array)
```

이 코드는 다음과 같은 동작을 합니다:

1. **`quicksort` 기능**: 재귀적으로 배열을 나누고 정렬합니다.
2. **`partition` 기능**: 기준값(pivot)을 중심으로 배열을 두 개의 그룹으로 분할하고, 기준값의 최종 위치를 반환합니다.

해당 코드는 평균적으로 $O(n \log n)$ 시간복잡도를 가지며, 공간복잡도는 $O(\log n)$입니다.
