```python
# 퀵정렬(Quick Sort) 알고리즘 구현

def quicksort(arr, low, high):
    # 배열의 하위 범위가 유효한 경우에만 재귀 호출
    if low < high:
        # 파티션 함수를 호출하여 피벗의 올바른 위치를 찾고, 해당 인덱스를 반환받음
        pi = partition(arr, low, high)
        
        # 피벗의 왼쪽 부분 배열에 대해 재귀적으로 퀵정렬 수행
        quicksort(arr, low, pi - 1)
        
        # 피벗의 오른쪽 부분 배열에 대해 재귀적으로 퀵정렬 수행
        quicksort(arr, pi + 1, high)

def partition(arr, low, high):
    # 가장 오른쪽 요소를 피벗으로 설정
    pivot = arr[high]
    
    # i는 작은 요소들의 인덱스를 나타내며, 초기값은 low-1로 설정
    i = low - 1
    
    # low부터 high-1까지 반복하며 비교 및 교환 수행
    for j in range(low, high):
        # 현재 요소가 피벗보다 작거나 같으면
        if arr[j] <= pivot:
            # i를 증가시킨 후, arr[i]와 arr[j]를 교환
            i += 1
            arr[i], arr[j] = arr[j], arr[i]
    
    # 피벗을 올바른 위치로 이동시키기 위해, 피벗과 arr[high]를 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    
    # 피벗이 위치하게 되는 인덱스를 반환합니다.
    return i + 1

# 예제 실행
if __name__ == "__main__":
    # 정렬할 배열 선언
    arr = [10, 7, 8, 9, 1, 5]
    
    # 배열의 길이를 계산
    n = len(arr)
    
    print("정렬 전:", arr)
    
    # 퀵정렬 실행
    quicksort(arr, 0, n - 1)
    
    print("정렬 후:", arr)
```

이 코드는 퀵정렬 알고리즘을 구현한 것입니다.  
`quicksort` 함수는 재귀적으로 배열의 일부분을 정렬하며, `partition` 함수는 피벗을 기준으로 배열을 분할하는 역할을 합니다.  
모든 줄에 대해 설명하는 주석이 포함되어 있어 이해에 도움이 됩니다.  
테스트로 `[10, 7, 8, 9, 1, 5]`라는 배열을 정렬하는 예시가 포함되었습니다.
