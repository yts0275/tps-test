```python
# 퀵정렬을 수행하는 함수 정의
def quicksort(arr):
    # 배열이 1개 이하이면 그대로 반환 (기저 조건)
    if len(arr) <= 1:
        return arr
    
    # 피벗을 배열의 첫 번째 요소로 선택
    pivot = arr[0]
    
    # 피벗보다 작은 요소들을 저장할 리스트
    left = []
    
    # 피벗보다 큰 요소들을 저장할 리스트
    right = []
    
    # 피벗과 같은 요소들을 저장할 리스트
    equal = []
    
    # 배열의 모든 요소를 순회하며, 피벗과 비교하여 분류
    for x in arr:
        if x < pivot:
            left.append(x)      # 피벗보다 작으면 왼쪽 리스트에 추가
        elif x > pivot:
            right.append(x)     # 피벗보다 크면 오른쪽 리스트에 추가
        else:
            equal.append(x)     # 피벗과 같으면 중간 리스트에 추가
    
    # 재귀적으로 왼쪽과 오른쪽 부분 배열을 정렬하고, 결과를 합치기
    return quicksort(left) + equal + quicksort(right)

# 예제 사용법
arr = [3, 6, 8, 10, 1, 2, 1]
sorted_arr = quicksort(arr)
print(sorted_arr)  # 출력: [1, 1, 2, 3, 6, 8, 10]
```

이 코드는 퀵정렬 알고리즘을 구현한 것입니다.  
- 기저 조건으로 배열 길이가 1 이하인 경우는 그대로 반환됩니다.
- 피벗을 선택한 후, 다른 요소들을 피벗과 비교하여 세 그룹(작은 값, 같은 값, 큰 값)으로 나눕니다.
- 재귀적으로 각 그룹을 정렬한 후 병합합니다.
