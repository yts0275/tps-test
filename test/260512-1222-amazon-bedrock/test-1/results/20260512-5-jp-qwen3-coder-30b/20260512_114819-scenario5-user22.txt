```python
# 퀵정렬(QuickSort) 알고리즘을 구현하는 함수
def quicksort(arr):
    # 배열의 길이가 1 이하일 경우, 정렬할 필요 없음 -> 그대로 반환
    if len(arr) <= 1:
        return arr
    
    # 배열의 중간 인덱스를 선택하여 피벗(pivot)으로 설정
    pivot = arr[len(arr) // 2]
    
    # 피벗보다 작은 원소들만 모은 리스트 (왼쪽 부분)
    left = [x for x in arr if x < pivot]
    
    # 피벗과 같은 원소들만 모은 리스트 (중간 부분)
    middle = [x for x in arr if x == pivot]
    
    # 피벗보다 큰 원소들만 모은 리스트 (오른쪽 부분)
    right = [x for x in arr if x > pivot]
    
    # 재귀적으로 왼쪽, 중간, 오른쪽 배열을 정렬하고 합침
    return quicksort(left) + middle + quicksort(right)

# 예시: 정렬할 배열
arr = [3, 6, 8, 10, 1, 2, 1]

# 퀵정렬 실행 후 결과 출력
print("정렬 전:", arr)
print("정렬 후:", quicksort(arr))
```

### 설명:
- `quicksort` 함수는 주어진 배열을 분할하고 병합하면서 정렬합니다.
- 피벗을 선택하게 되고, 피벗 기준으로 모든 원소들을 세 가지 그룹으로 나누어 정렬합니다.
- 마지막에는 정렬된 하위 배열들을 합쳐 전체 배열을 정렬된 상태로 만들어줍니다.
