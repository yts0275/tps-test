```python
# 퀵정렬(Quiicksort) 알고리즘 구현
# 파티셔닝 함수: 배열을 기준값(pivot)을 기준으로 두 부분으로 나누는 함수
def partition(arr, low, high):
    # 마지막 요소를 기준값으로 설정 (pivot)
    pivot = arr[high]
    
    # i는 작은 요소들의 인덱스를 나타냄
    i = low - 1
    
    # 배열의 첫 번째 요소부터 끝까지 반복
    for j in range(low, high):
        # 현재 요소가 기준값보다 작거나 같으면
        if arr[j] <= pivot:
            # i를 증가시키고, 현재 요소를 i 위치로 이동
            i += 1
            arr[i], arr[j] = arr[j], arr[i]  # 교환
    
    # 기준값을 올바른 위치로 이동시킴 (작은 요소들 끝과 기준값 위치 교환)
    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    
    # 기준값이 있는 위치 반환
    return i + 1

# 퀵정렬 함수: 재귀적으로 배열을 정렬하는 함수
def quicksort(arr, low, high):
    # 재귀의 종료 조건: 시작 인덱스가 끝 인덱스보다 작거나 같을 때까지만 수행
    if low < high:
        # 파티셔닝 함수를 호출하여 기준값의 올바른 위치를 찾음
        pi = partition(arr, low, high)
        
        # 기준값 왼쪽 부분에 대해 재귀적으로 퀵정렬 수행
        quicksort(arr, low, pi - 1)
        
        # 기준값 오른쪽 부분에 대해 재귀적으로 퀵정렬 수행
        quicksort(arr, pi + 1, high)

# 예제 배열 생성
arr = [10, 7, 8, 9, 1, 5]

# 정렬 전 배열 출력
print("정렬 전:", arr)

# 퀵정렬 실행
quicksort(arr, 0, len(arr) - 1)

# 정렬 후 배열 출력
print("정렬 후:", arr)
```

### 설명:
- **partition 함수**: 기준값(pivot)을 기준으로 배열을 분할합니다.
- **quicksort 함수**: 재귀적으로 배열을 분할하고 정렬합니다.
- 시간 복잡도: 평균 O(n log n), 최악 O(n²)
- 공간 복잡도: O(log n)

이 코드는 입력 배열 `[10, 7, 8, 9, 1, 5]`를 정렬하여 `[1, 5, 7, 8, 9, 10]`로 출력합니다.
