```python
def quicksort(arr, low, high):
    # 배열의 하위 인덱스와 상위 인덱스를 기준으로 퀵소트 수행
    if low < high:
        # 파티션 함수를 호출하여 피벗 위치를 가져옴
        pi = partition(arr, low, high)
        
        # 피벗의 왼쪽 부분에 대해 재귀적으로 퀵소트 호출
        quicksort(arr, low, pi - 1)
        
        # 피벗의 오른쪽 부분에 대해 재귀적으로 퀵소트 호출
        quicksort(arr, pi + 1, high)

def partition(arr, low, high):
    # 가장 오른쪽 요소를 피벗으로 설정
    pivot = arr[high]
    
    # 작은 요소들의 인덱스를 초기화
    i = low - 1
    
    # 배열의 모든 요소를 순회하며 비교
    for j in range(low, high):
        # 현재 요소가 피벗보다 작거나 같으면
        if arr[j] <= pivot:
            # 인덱스를 증가시키고,
            i += 1
            # 현재 요소와 작은 요소들의 위치를 교환
            arr[i], arr[j] = arr[j], arr[i]
    
    # 피벗을 올바른 위치로 이동시킴
    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    
    # 피벗이 위치한 인덱스 반환
    return i + 1

# 예제 사용법
if __name__ == "__main__":
    # 정렬할 배열 생성
    arr = [10, 7, 8, 9, 1, 5]
    
    # 배열의 길이 저장
    n = len(arr)
    
    # 퀵소트 수행
    quicksort(arr, 0, n - 1)
    
    # 결과 출력
    print("정렬된 배열:", arr)
```

### 코드 설명:

- `quicksort` 함수는 재귀적으로 배열을 정렬하는 주요 함수입니다.
- `partition` 함수는 배열을 피벗을 기준으로 두 부분으로 나누는 일을 합니다.
- 모든 인덱스는 0-based로 처리되며, 배열의 원소들이 분할과 교환 과정을 통해 정렬됩니다.
