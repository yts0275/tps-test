```python
# 퀵소트(QuickSort) 알고리즘을 구현하는 함수입니다.
def quicksort(arr):
    # 기저 조건: 배열의 길이가 1 이하이면 이미 정렬된 상태이므로 그대로 반환합니다.
    if len(arr) <= 1:
        return arr
    
    # 피벗(pivot) 요소를 배열의 중간에 위치한 요소로 선택합니다.
    pivot = arr[len(arr) // 2]
    
    # 피벗보다 작은 요소들을 모은 리스트입니다.
    left = [x for x in arr if x < pivot]
    
    # 피벗과 같은 값을 모은 리스트입니다.
    middle = [x for x in arr if x == pivot]
    
    # 피벗보다 큰 요소들을 모은 리스트입니다.
    right = [x for x in arr if x > pivot]
    
    # 재귀적으로 왼쪽과 오른쪽 부분을 정렬한 후, 중간 부분과 결합하여 결과를 반환합니다.
    return quicksort(left) + middle + quicksort(right)

# 예제 배열을 생성합니다.
arr = [3, 6, 8, 10, 1, 2, 1]

# 퀵소트를 적용하여 정렬합니다.
sorted_arr = quicksort(arr)

# 정렬된 결과를 출력합니다.
print("정렬된 배열:", sorted_arr)
```
