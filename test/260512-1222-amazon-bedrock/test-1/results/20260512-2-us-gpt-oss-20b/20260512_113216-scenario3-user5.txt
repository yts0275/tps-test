```python
# 퀵소트(QuickSort) 구현 – 각 줄에 한 줄씩 주석을 달았습니다.

def quicksort(arr):                         # 1️⃣ 배열을 정렬해서 반환하는 함수 정의
    if len(arr) <= 1:                       # 2️⃣ 배열의 길이가 0 또는 1이면 더 이상 나눌 필요가 없으므로 그대로 반환
        return arr                           # 3️⃣ 정렬이 완료된 배열 반환

    pivot = arr[len(arr)//2]                # 4️⃣ 피벗(pivot) 선택: 배열의 중간 인덱스 요소 사용
    left = [x for x in arr if x < pivot]    # 5️⃣ 피벗보다 작은 요소들만 모아서 왼쪽(왼쪽 부분집합) 리스트 생성
    middle = [x for x in arr if x == pivot] # 6️⃣ 피벗과 같은 요소들만 모아서 가운데(middle) 리스트 생성
    right = [x for x in arr if x > pivot]   # 7️⃣ 피벗보다 큰 요소들만 모아서 오른쪽(right) 리스트 생성

    return quicksort(left) + middle + quicksort(right)  # 8️⃣ 왼쪽, 가운데, 오른쪽을 재귀 정렬 후 합쳐서 반환
```

위 코드는 재귀적인 방식으로 리스트를 분할하고 정렬합니다.  
`left`, `middle`, `right` 리스트를 이용해 피벗을 기준으로 세 구간으로 나누고,  
각 구간을 재귀적으로 정렬한 뒤 다시 합쳐서 완전한 정렬 결과를 얻습니다.
