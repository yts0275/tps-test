```python
# QuickSort 함수를 정의합니다.
def quicksort(arr):
    # 배열의 길이가 1 이하인 경우 재귀 종료
    if len(arr) <= 1:
        return arr

    # 피벗(pivot)으로 중앙값을 선택합니다.
    pivot = arr[len(arr) // 2]

    # 피벗보다 작은 값들을 모읍니다.
    lows = [x for x in arr if x < pivot]

    # 피벗보다 큰 값들을 모읍니다.
    highs = [x for x in arr if x > pivot]

    # 피벗과 같은 값들(중복된 값들)을 모읍니다.
    pivots = [x for x in arr if x == pivot]

    # 재귀적으로 낮은 값 부분, 피벗, 높은 값 부분을 정렬한 뒤 합칩니다.
    return quicksort(lows) + pivots + quicksort(highs)


# 예시: 정렬이 필요한 숫자 리스트를 선언합니다.
numbers = [3, 6, 8, 10, 1, 2, 1]

# QuickSort를 호출하고 결과를 출력합니다.
sorted_numbers = quicksort(numbers)
print(sorted_numbers)   # 출력: [1, 1, 2, 3, 6, 8, 10]
```
