```python
# 퀵소트(Quick Sort) 구현 (Python)
# 각 줄마다 주석을 달아 이해를 돕도록 하겠습니다.

def quicksort(arr):   # 1. 리스트 arr을 매개변수로 받는 함수 정의
    if len(arr) <= 1: # 2. 리스트 길이가 0 또는 1이면 이미 정렬된 상태이므로 그대로 반환
        return arr     # 3. 정렬되지 않은 그대로 반환
    # 4. 중앙값을 피벗(pivot)으로 선택
    pivot = arr[len(arr) // 2]   # 5. 피벗은 리스트 중간 위치의 요소

    # 6. 피벗보다 작은 값들만 담은 새 리스트 left 생성
    left = [x for x in arr if x < pivot]
    # 7. 피벗과 같은 값들만 담은 새 리스트 middle 생성
    middle = [x for x in arr if x == pivot]
    # 8. 피벗보다 큰 값들만 담은 새 리스트 right 생성
    right = [x for x in arr if x > pivot]

    # 9. 재귀적으로 왼쪽과 오른쪽 리스트를 정렬한 뒤, left, middle, right 순서로 합침
    return quicksort(left) + middle + quicksort(right)  # 10. 최종 정렬 결과 반환

# 11. 테스트용 예시 리스트 선언
if __name__ == "__main__":
    sample_list = [3, 6, 8, 10, 1, 2, 1]  # 12. 예시 데이터 (정렬 전 리스트)
    sorted_list = quicksort(sample_list)  # 13. quicksort 함수 호출하여 정렬 수행
    print("정렬 전:", sample_list)        # 14. 원본 리스트 출력
    print("정렬 후:", sorted_list)        # 15. 정렬 결과 출력
```

위 코드는 피벗으로 리스트 중앙값을 고르고, 피벗을 기준으로 작은 값, 같은 값, 큰 값으로 나눈 뒤 재귀적으로 정렬하는 전형적인 퀵소트 구현입니다.  
각 줄마다 주석을 달았으므로 원하는 부분을 쉽게 이해할 수 있습니다.
