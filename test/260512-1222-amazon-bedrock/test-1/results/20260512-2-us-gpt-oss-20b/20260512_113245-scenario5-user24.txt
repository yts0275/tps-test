```python
# 퀵소트(QuickSort) 구현 예시 (파이썬)
# ---------------------------------
# 각 코드 라인 뒤에 주석으로 설명을 달았습니다.

def quicksort(arr, low, high):  # 퀵소트 함수: 배열 arr과 정렬 범위(low~high)를 인수로 받음
    if low < high:  # 정렬할 범위가 1개 이하이면 종료 (이미 정렬된 상태)
        pi = partition(arr, low, high)  # 피벗을 기준으로 배열을 나누고 피벗이 놓일 최종 인덱스 반환
        quicksort(arr, low, pi - 1)  # 피벗 왼쪽 부분을 재귀적으로 정렬
        quicksort(arr, pi + 1, high)  # 피벗 오른쪽 부분을 재귀적으로 정렬

def partition(arr, low, high):  # 피벗을 기준으로 배열을 분할하는 함수
    pivot = arr[high]  # 가장 오른쪽 원소를 피벗으로 선택
    i = low - 1  # i는 < 피벗인 원소의 가장 오른쪽 인덱스
    for j in range(low, high):  # low부터 high-1까지 순회
        if arr[j] <= pivot:  # 현재 원소가 피벗보다 작거나 같으면
            i += 1  # i를 한 칸 오른쪽으로 이동
            arr[i], arr[j] = arr[j], arr[i]  # i와 j 위치의 원소 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 올바른 위치(i+1)로 이동
    return i + 1  # 피벗이 놓인 최종 인덱스를 반환

# 사용 예시
if __name__ == "__main__":  # 직접 실행할 때만 아래 코드 실행
    arr = [10, 7, 8, 9, 1, 5]  # 정렬할 배열
    n = len(arr)  # 배열 길이
    print("정렬 전 배열:", arr)  # 정렬 전 배열 출력
    quicksort(arr, 0, n - 1)  # 퀵소트 호출
    print("정렬 후 배열:", arr)  # 정렬 후 배열 출력
```

위 코드는 배열을 **in‑place** 로 정렬합니다.  
각 줄마다 주석을 달아 구현 로직과 흐름을 명확히 설명했습니다.
