```python
def quick_sort(arr):  # quick_sort 함수 정의: 입력 리스트를 정렬하여 새 리스트 반환
    if len(arr) <= 1:  # 리스트 길이가 1 이하이면 이미 정렬된 상태라 그대로 반환
        return arr  # 정렬된 리스트 반환
    pivot = arr[len(arr) // 2]  # 리스트의 중간 값을 피벗으로 선택
    left = [x for x in arr if x < pivot]  # 피벗보다 작은 원소들을 모음
    mid = [x for x in arr if x == pivot]  # 피벗과 같은 원소들을 모음
    right = [x for x in arr if x > pivot]  # 피벗보다 큰 원소들을 모음
    return quick_sort(left) + mid + quick_sort(right)  # 재귀적으로 각 부분을 정렬 후 합침

# 예시 사용:
arr = [3, 6, 8, 10, 1, 2, 1]  # 정렬하고자 하는 리스트
sorted_arr = quick_sort(arr)  # quick_sort 함수 호출
print(sorted_arr)  # 정렬 결과 출력
```
