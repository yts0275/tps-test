```python
def quicksort(arr):                     # 재귀적으로 배열을 정렬하는 함수 정의
    if len(arr) <= 1:                    # 배열이 1개 이하이면 이미 정렬된 상태라 그대로 반환
        return arr                       # 정렬이 끝난 배열을 반환
    pivot = arr[-1]                      # 배열의 마지막 원소를 피벗(분할 기준)으로 선택
    left = []                            # 피벗보다 작은 값들을 담을 리스트 초기화
    right = []                           # 피벗보다 큰 값들을 담을 리스트 초기화
    for x in arr[:-1]:                   # 피벗을 제외한 모든 원소를 순회
        if x < pivot:                   # 현재 원소가 피벗보다 작으면 left에 추가
            left.append(x)              # left 리스트에 원소 저장
        else:                          # 그렇지 않으면 right에 추가
            right.append(x)             # right 리스트에 원소 저장
    # 왼쪽 부분 배열을 재귀적으로 정렬한 뒤, 피벗, 그리고 오른쪽 부분 배열을 이어 붙임
    return quicksort(left) + [pivot] + quicksort(right)   # 정렬 결과를 반환

# 사용 예시
if __name__ == "__main__":
    arr = [10, 5, 2, 3, 90, 7, 8, 1]
    sorted_arr = quicksort(arr)          # quicksort 함수 호출로 정렬 수행
    print(sorted_arr)                    # 정렬된 배열 출력: [1, 2, 3, 5, 7, 8, 10, 90]
```

> **주의**  
> * 각 줄 끝에 `#` 뒤에 한국어 주석을 넣었으며,  
> * 재귀 호출이 많아 큰 배열에서는 호출 깊이 초과가 발생할 수 있으므로 주의합니다.  
> * 피벗을 항상 마지막 원소로 고정했지만, 다른 선택 방법(중앙값 등)도 가능합니다.
