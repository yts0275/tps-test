```python
# 순수 파이썬으로 구현한 퀵소트(재귀 버전)  
def quicksort(arr, low, high):  # 정렬할 배열, 정렬 시작 인덱스, 종료 인덱스를 인수로 받음
    if low < high:  # 종료 조건: 정렬 범위가 하나 이하이면 더 이상 나눠지지 않음
        pivot_index = partition(arr, low, high)  # 파티션 함수를 호출해 피벗 인덱스를 얻음
        quicksort(arr, low, pivot_index - 1)  # 피벗 왼쪽 부분을 재귀적으로 정렬
        quicksort(arr, pivot_index + 1, high)  # 피벗 오른쪽 부분을 재귀적으로 정렬

def partition(arr, low, high):  # 파티션 함수: 피벗을 기준으로 배열을 재배열
    pivot = arr[high]  # 피벗 원소를 배열의 끝에 있는 값으로 정함
    i = low - 1  # i는 피벗보다 작은 값이 저장될 인덱스를 가리킴
    for j in range(low, high):  # low부터 high-1까지 순회
        if arr[j] < pivot:  # 현재 원소가 피벗보다 작으면
            i += 1  # i를 하나 옮겨 주고
            arr[i], arr[j] = arr[j], arr[i]  # i와 j를 교환해 작은 값들을 앞쪽에 모음
    arr[i+1], arr[high] = arr[high], arr[i+1]  # 피벗을 적절한 위치에 삽입
    return i + 1  # 피벗이 놓인 최종 인덱스를 반환

# 메인 실행 블록: 예시 배열을 정렬하고 결과를 출력
if __name__ == "__main__":  # 스크립트가 직접 실행될 때만 아래 코드 실행
    arr = [5, 3, 8, 4, 2]  # 정렬할 숫자 배열
    quicksort(arr, 0, len(arr)-1)  # 전체 배열을 정렬
    print(arr)  # 정렬된 결과 출력: [2, 3, 4, 5, 8]
```
