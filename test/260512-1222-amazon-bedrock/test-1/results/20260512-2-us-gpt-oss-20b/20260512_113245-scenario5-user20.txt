```python
def quick_sort(arr):  # 퀵소트 함수 정의
    if len(arr) <= 1:  # 배열이 비어있거나 한 개의 원소만 있을 때는 정렬이 필요 없음
        return arr  # 그대로 반환
    pivot = arr[len(arr) // 2]  # 중앙 원소를 피벗으로 선택
    left = [x for x in arr if x < pivot]  # 피벗보다 작은 원소들을 모음
    middle = [x for x in arr if x == pivot]  # 피벗과 같은 원소들을 모음
    right = [x for x in arr if x > pivot]  # 피벗보다 큰 원소들을 모음
    return quick_sort(left) + middle + quick_sort(right)  # 재귀 호출한 결과들을 합쳐 반환

# 사용 예시
if __name__ == "__main__":  # 이 파일이 직접 실행될 때만 아래 코드 실행
    arr = [3, 6, 8, 10, 1, 2, 1]  # 테스트용 배열
    print("원본 배열:", arr)  # 원본 배열 출력
    sorted_arr = quick_sort(arr)  # 퀵소트 실행
    print("정렬된 배열:", sorted_arr)  # 정렬 결과 출력
```
