```python
# 리스트를 인수로 받아 정렬된 리스트를 반환하는 퀵소트 함수
def quicksort(arr, low, high):
    if low < high:                        # 1) 재귀가 진행될 조건
        pi = partition(arr, low, high)    # 2) 파티션을 수행하고 피벗 인덱스를 받는다
        quicksort(arr, low, pi - 1)       # 3) 피벗 좌측 부분을 재귀적으로 정렬한다
        quicksort(arr, pi + 1, high)      # 4) 피벗 우측 부분을 재귀적으로 정렬한다

# 비슷한 요소를 한 쪽으로 모으는 파티션 함수
def partition(arr, low, high):
    pivot = arr[high]              # 5) 가장 오른쪽 요소를 피벗으로 선택한다
    i = low - 1                    # 6) 비교 대상보다 작은 요소를 찾을 인덱스
    for j in range(low, high):     # 7) 피벗 이전까지 순회
        if arr[j] <= pivot:        # 8) 현재 요소가 피벗보다 작으면
            i += 1                 # 9) i를 한 칸 오른쪽으로 이동
            arr[i], arr[j] = arr[j], arr[i]  # 10) i와 j에 해당하는 값을 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 11) 피벗을 올바른 위치에 삽입
    return i + 1                    # 12) 피벗의 최종 인덱스를 반환

# 사용 예시
if __name__ == "__main__":
    numbers = [10, 7, 8, 9, 1, 5]  # 13) 정렬할 원본 리스트
    quicksort(numbers, 0, len(numbers) - 1)  # 14) 퀵소트 실행
    print(numbers)                  # 15) 정렬된 리스트 출력
```
