def quicksort(arr, low=0, high=None):                     # 1️⃣ 퀵소트 함수 정의, 근거 인덱스 low와 high 지정
    if high is None:                                      # 2️⃣ 고정된 high가 없으면 배열 길이로 초기화
        high = len(arr) - 1                               # 3️⃣ 배열 마지막 인덱스를 high로 설정
    if low < high:                                        # 4️⃣ 종료 조건: 두 포인터가 만나야 종료
        pivot_index = partition(arr, low, high)           # 5️⃣ 피벗 위치를 찾는 파티션 호출
        quicksort(arr, low, pivot_index - 1)              # 6️⃣ 피벗 왼쪽 부분 배열 재귀 정렬
        quicksort(arr, pivot_index + 1, high)             # 7️⃣ 피벗 오른쪽 부분 배열 재귀 정렬

def partition(arr, low, high):                             # 8️⃣ 파티션 함수 정의, 요소들을 재배치
    pivot = arr[high]                                      # 9️⃣ 피벗을 가장 오른쪽 원소로 지정
    i = low - 1                                            # 🔟 i를 low-1로 초기화 (작은 값 저장 위치)
    for j in range(low, high):                             # ㊙️ j를 low부터 high-1까지 순회
        if arr[j] <= pivot:                                # 🔢 현재 원소가 피벗보다 작거나 같은 경우
            i += 1                                         # ㊙️ i 증가
            arr[i], arr[j] = arr[j], arr[i]                 # ㊙️ 두 원소 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1]         # 🝺 피벗을 올바른 위치로 이동
    return i + 1                                           # 🔚 피벗이 최종적으로 놓인 인덱스 반환

if __name__ == "__main__":                                 # ㊙️ 직접 실행 시 실행
    data = [3, 6, 8, 10, 1, 2, 1]                         # ㊙️ 예시 데이터 배열 정의
    print("원본 배열:", data)                             # ㊙️ 원본 배열 출력
    quicksort(data)                                       # ㊙️ 퀵소트 함수 호출
    print("정렬된 배열:", data)                           # ㊙️ 정렬된 배열 출력
