### 🟦 퀵소트(Quick Sort) – Python 구현

```python
# 퀵소트를 수행하는 재귀 함수
def quick_sort(arr):
    # 만약 배열의 길이가 0 또는 1이면 재귀를 종료
    if len(arr) <= 1:
        return arr

    # 피벗(pivot)을 선택: 여기서는 첫 번째 요소를 사용
    pivot = arr[0]

    # 피벗보다 작은 요소들을 담을 리스트
    left = []
    
    # 피벗과 같은 요소들을 담을 리스트 (중복 제거를 위해서는 이 부분을 생략 또는 수정 가능)
    middle = []
    
    # 피벗보다 큰 요소들을 담을 리스트
    right = []

    # 배열의 모든 요소에 대해 비교 수행
    for i in range(len(arr)):
        if arr[i] < pivot:     # 피벗 보다 작으면 왼쪽 리스트에 추가
            left.append(arr[i])
        elif arr[i] == pivot:  # 피벗과 같으면 중앙 리스트에 추가
            middle.append(arr[i])
        else:                  # 피벗 보다 크면 오른쪽 리스트에 추가
            right.append(arr[i])

    # 재귀적으로 퀵소트를 수행한 후 왼쪽 + 중간 + 오른쪽을 결합하여 반환
    return quick_sort(left) + middle + quick_sort(right)
```

---

### 🟨 실행 예시

```python
# 예제 배열
unsorted_list = [3, 6, 8, 10, 1, 2, 1]

# 퀵소트 실행
sorted_list = quick_sort(unsorted_list)

# 정렬 결과 출력
print("정렬된 배열:", sorted_list)
```
