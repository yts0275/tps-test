def quicksort(arr, low, high):  # 재귀 함수 정의: 배열 arr에서 low부터 high까지 정렬
    if low < high:  # 정렬 범위가 하나의 요소보다 크면(정렬 필요 시)  
        pi = partition(arr, low, high)  # 파티션 수행, 피벗(고려된 기준값) 위치 반환
        quicksort(arr, low, pi-1)  # 피벗 왼쪽 부분 정렬
        quicksort(arr, pi+1, high)  # 피벗 오른쪽 부분 정렬

def partition(arr, low, high):  # 파티션 함수 정의: 피벗을 기준으로 배열 분할
    pivot = arr[high]  # 가장 마지막 요소를 피벗으로 선택
    i = low - 1  # i를 low-1로 초기화(작은 값이 놓일 위치를 가리킴)
    for j in range(low, high):  # low부터 high-1까지 순회
        if arr[j] <= pivot:  # 현재 요소가 피벗보다 작거나 같으면
            i += 1  # i를 한 칸 오른쪽으로 이동
            arr[i], arr[j] = arr[j], arr[i]  # i와 j 위치 교환
    arr[i+1], arr[high] = arr[high], arr[i+1]  # 피벗을 올바른 위치에 놓음
    return i+1  # 피벗 인덱스 반환

def quick_sort(arr):  # 편의 함수: 배열 전체를 정렬하는 래퍼
    if arr:  # 배열이 비어있지 않으면
        quicksort(arr, 0, len(arr)-1)  # 전체 범위에 대해 재귀 정렬 호출
    return arr  # 정렬된 배열 반환

if __name__ == "__main__":  # 직접 실행 시 테스트 코드
    array = [10, 7, 8, 9, 1, 5]  # 예시 배열
    print("원본 배열:", array)  # 원본 배열 출력
    sorted_array = quick_sort(array)  # 빠른 정렬 수행
    print("정렬된 배열:", sorted_array)  # 정렬 결과 출력
