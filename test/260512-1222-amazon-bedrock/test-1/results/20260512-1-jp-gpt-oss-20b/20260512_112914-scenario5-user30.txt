```python
def quicksort(arr):                     # 리스트(arr)를 퀵소트로 정렬하는 함수  
    if len(arr) <= 1:                  # 리스트가 비었거나 원소 1개인 경우 그대로 반환  
        return arr                    # 정렬이 필요 없으므로 입력 리스트를 그대로 돌려준다  

    pivot = arr[0]                     # 가장 첫 번째 원소를 피벗으로 선택  
    left = [x for x in arr[1:] if x <= pivot]   # 피벗보다 작거나 같은 원소들을 왼쪽 리스트에 모음  
    right = [x for x in arr[1:] if x > pivot]    # 피벗보다 큰 원소들을 오른쪽 리스트에 모음  

    # 재귀 호출: 왼쪽 리스트와 오른쪽 리스트를 각각 퀵소트로 정렬 후 피벗과 합침  
    return quicksort(left) + [pivot] + quicksort(right)  
```

이 코드는 각 줄에 대한 설명을 주석으로 달아, 어떻게 동작하는지 단계별로 이해할 수 있도록 구성되었습니다. 빠른 정렬(퀵소트)은 평균적으로 O(n log n) 시간복잡도를 가지며, 작은 입력에 대해서도 매우 효율적으로 동작합니다.
