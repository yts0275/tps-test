```python
# quicksort 알고리즘 구현: 배열을 재귀적으로 정렬합니다.
def quicksort(arr, low, high):   # 배열 arr, 정렬 구간의 시작 인덱스 low, 끝 인덱스 high를 인수로 받는 함수 정의
    if low < high:                # 구간이 유효하면(시작 인덱스가 끝 인덱스보다 작으면) 재귀 호출 수행
        pivot_index = partition(arr, low, high)  # 현재 구간을 분할하여 피벗 위치를 반환받음
        quicksort(arr, low, pivot_index - 1)     # 피벗 왼쪽 부분에 대해 빠른 정렬 재귀 호출
        quicksort(arr, pivot_index + 1, high)    # 피벗 오른쪽 부분에 대해 빠른 정렬 재귀 호출

def partition(arr, low, high):   # partition 함수 정의: 피벗을 기준으로 배열을 분할
    pivot = arr[high]           # 가장 오른쪽 원소를 피벗 값으로 선택
    i = low - 1                 # 피벗보다 작은 원소를 모을 인덱스 i를 시작 인덱스보다 한 칸 앞에서 초기화
    for j in range(low, high):  # 피벗을 제외한 구간( low 부터 high-1 까지 )을 순회
        if arr[j] <= pivot:     # 현재 원소가 피벗 이하이면
            i += 1              # i를 한 칸 옮겨 피벗보다 작은 원소를 저장할 위치를 표시
            arr[i], arr[j] = arr[j], arr[i]  # 작은 원소와 현재 원소를 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 가장 큰 원소보다 한 칸 뒤인 위치에 놓음
    return i + 1                # 피벗의 최종 인덱스를 반환

# 사용 예시
arr = [10, 7, 8, 9, 1, 5]        # 정렬할 대상 배열 정의
quicksort(arr, 0, len(arr) - 1)  # quicksort 함수를 호출해 배열 정렬
print("정렬된 배열:", arr)        # 정렬된 결과를 출력
```
