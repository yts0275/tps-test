def quicksort(arr, low, high):  # 빠른 정렬을 위한 재귀 함수
    if low < high:  # 두 개 이상의 원소가 있을 때만 정렬을 수행
        pivot_index = partition(arr, low, high)  # 피벗을 기준으로 배열을 분할
        quicksort(arr, low, pivot_index - 1)  # 피벗 앞쪽 서브 배열 재귀 정렬
        quicksort(arr, pivot_index + 1, high)  # 피벗 뒤쪽 서브 배열 재귀 정렬

def partition(arr, low, high):  # 파티션을 수행하는 함수
    pivot = arr[high]  # 가장 오른쪽 원소를 피벗으로 설정
    i = low - 1  # 작은 값이 저장될 인덱스를 초기화
    for j in range(low, high):  # 배열을 순회
        if arr[j] <= pivot:  # 피벗보다 작거나 같을 때
            i += 1  # 작은 값이 놓일 인덱스 증가
            arr[i], arr[j] = arr[j], arr[i]  # 작은 원소를 앞쪽으로 스왑
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 정렬된 위치로 이동
    return i + 1  # 피벗 인덱스 반환

if __name__ == "__main__":  # 스크립트가 메인으로 실행될 때만 아래 코드 실행
    arr = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 예시 배열
    print("원본 배열:", arr)  # 원본 배열 출력
    quicksort(arr, 0, len(arr) - 1)  # 전체 배열 정렬
    print("정렬된 배열:", arr)  # 정렬 결과 출력
