def quicksort(arr, low, high):  # quicksort 함수 정의: 배열, 시작 인덱스(low), 끝 인덱스(high)를 인수로 받음
    if low < high:  # 시작 인덱스가 끝 인덱스보다 작으면 정렬을 진행
        pi = partition(arr, low, high)  # 파티션 함수를 호출해 피벗(index)를 구함
        quicksort(arr, low, pi - 1)  # 피벗 왼쪽 부분을 재귀적으로 정렬
        quicksort(arr, pi + 1, high)  # 피벗 오른쪽 부분을 재귀적으로 정렬

def partition(arr, low, high):  # 파티션 함수 정의: 피벗을 기준으로 배열을 재배치
    pivot = arr[high]  # 가장 오른쪽 원소를 피벗으로 선택
    i = low - 1  # i는 피벗보다 작은 값이 끝나는 위치를 가리킴
    for j in range(low, high):  # j는 피벗 전까지 순회
        if arr[j] < pivot:  # 현재 원소가 피벗보다 작으면
            i += 1  # i를 한 칸 앞쪽으로 옮김
            arr[i], arr[j] = arr[j], arr[i]  # i와 j의 위치를 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 올바른 위치(끝+1)에 넣음
    return i + 1  # 피벗의 최종 인덱스를 반환

if __name__ == "__main__":  # 스크립트를 직접 실행할 때만 아래 코드가 실행됨
    data = [3, 6, 8, 10, 1, 2, 1]  # 정렬이 필요한 데이터 리스트
    quicksort(data, 0, len(data)-1)  # 전체 배열에 대해 quicksort 호출
    print(data)  # 정렬 결과를 출력
}
