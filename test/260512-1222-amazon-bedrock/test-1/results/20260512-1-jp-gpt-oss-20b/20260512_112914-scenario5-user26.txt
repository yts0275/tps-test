```python
# 반전된 재귀 퀵소트 구현 예시

def quicksort(arr):  # quicksort 함수 정의: 리스트 arr를 인자로 받음
    if len(arr) <= 1:  # 리스트의 길이 0 또는 1이면 이미 정렬된 상태, 그대로 반환
        return arr  # 정렬된 리스트 반환
    pivot = arr[len(arr) // 2]  # 중앙값을 피벗으로 선택 (리스트 요소 번호가 0부터 시작)
    left = [x for x in arr if x < pivot]  # 피벗보다 작은 모든 요소를 모은 서브리스트
    middle = [x for x in arr if x == pivot]  # 피벗과 같은 값들(중복인 경우)만 모은 서브리스트
    right = [x for x in arr if x > pivot]  # 피벗보다 큰 모든 요소를 모은 서브리스트
    return quicksort(left) + middle + quicksort(right)  # 재귀적으로 정렬된 왼쪽, 중간, 오른쪽을 합쳐 반환

# 테스트용 코드
if __name__ == "__main__":  # 스크립트가 직접 실행될 때만 아래 코드를 실행
    sample = [9, 4, 6, 2, 8, 1, 3, 7, 5]  # 정렬하려는 예시 리스트
    print("원본 리스트:", sample)  # 원본 리스트 출력
    sorted_list = quicksort(sample)  # quicksort 함수 호출
    print("정렬된 리스트:", sorted_list)  # 정렬 결과 출력
```
