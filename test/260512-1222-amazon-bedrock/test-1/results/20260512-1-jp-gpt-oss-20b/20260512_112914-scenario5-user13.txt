```python
# --------------------------  퀵소트 구현 (Python)  --------------------------
def quick_sort(arr):                                            # 1. quick_sort( ) 함수를 정의한다.
    if len(arr) <= 1:                                          # 2. 리스트가 1개 이하이면 이미 정렬된 상태라 바로 반환한다.
        return arr                                              # 3. 정렬된 리스트를 그대로 반환한다.
    # 피벗을 리스트 중앙의 값으로 정한다(공통적인 방법 중 하나)   
    pivot = arr[len(arr) // 2]                                 # 4. 피벗(pivot)을 중앙값으로 설정한다.
    # 피벗보다 작은 값, 같은 값, 큰 값을 구분하기 위한 임시 리스트를 미리 만든다.
    left = []                                                   # 5. 피벗보다 작은 값들을 저장할 리스트
    middle = []                                                 # 6. 피벗과 같은 값들을 저장할 리스트
    right = []                                                  # 7. 피벗보다 큰 값들을 저장할 리스트
    # 원본 리스트의 모든 원소를 순회하면서 피벗과 비교 후 적절한 리스트에 추가한다.
    for x in arr:                                               # 8. 리스트의 모든 원소를 반복한다.
        if x < pivot:                                          # 9. 현재 원소가 피벗보다 작으면 left 리스트에 추가
            left.append(x)                                     #10. left 리스트에 x를 삽입
        elif x == pivot:                                       #11. 현재 원소가 피벗과 같으면 middle 리스트에 추가
            middle.append(x)                                   #12. middle 리스트에 x를 삽입
        else:                                                   #13. 현재 원소가 피벗보다 크면 right 리스트에 추가
            right.append(x)                                    #14. right 리스트에 x를 삽입
    # 재귀적으로 왼쪽과 오른쪽 부분을 정렬한 뒤, 정렬된 왼쪽 + 중간값 + 정렬된 오른쪽 순서로 합친다.
    return quick_sort(left) + middle + quick_sort(right)          #15. 재귀 호출과 리스트 합성을 통해 정렬된 결과를 반환

# --------------------------  사용 예시  --------------------------
if __name__ == "__main__":                                       #16. 직접 실행했을 때를 위한 메인 블록
    unsorted_list = [38, 27, 43, 3, 9, 82, 10]                   #17. 정렬이 필요한 예시 리스트
    print("원본 리스트:", unsorted_list)                        #18. 원본 리스트를 출력
    sorted_list = quick_sort(unsorted_list)                       #19. quick_sort 함수를 호출하여 정렬
    print("정렬된 리스트:", sorted_list)                         #20. 정렬된 결과를 출력
```

위 코드에서 **각 줄마다** 한 줄씩 주석을 달아 이해를 돕고, 퀵소트 알고리즘이 어떻게 동작하는지 한눈에 볼 수 있습니다.  
`quick_sort` 함수는 재귀적으로 자신을 호출하면서 `left`, `middle`, `right` 세 리스트를 이용해 피벗을 기준으로 분할하고, 최종적으로 합쳐서 정렬된 리스트를 반환합니다.
