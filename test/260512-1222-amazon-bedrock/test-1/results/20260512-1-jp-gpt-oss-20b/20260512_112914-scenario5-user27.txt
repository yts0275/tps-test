아래는 각 줄마다 한국어 주석이 달린 파이썬 퀵소트 구현입니다.

```python
def quicksort(arr):                                     # 함수 정의 : 배열을 정렬해 반환
    if len(arr) <= 1:                                   # 기본 사상(기저 사상) : 길이가 0 또는 1인 배열은 이미 정렬됨
        return arr                                      # 기저 사상일 때 배열 그대로 반환
    pivot = arr[-1]                                     # 피벗 선택 : 배열의 마지막 원소를 피벗으로 사용
    less = []                                          # 피벗보다 작은 원소들을 저장할 리스트
    greater = []                                       # 피벗보다 크거나 같은 원소들을 저장할 리스트
    for x in arr[:-1]:                                 # 피벗을 제외한 모든 원소에 대해 반복
        if x < pivot:                                  # 피벗보다 작으면 less 리스트에 추가
            less.append(x)                              # less 리스트에 원소 삽입
        else:                                          # 그 외(피벗보다 크거나 같은 경우) greater 리스트에 추가
            greater.append(x)                           # greater 리스트에 원소 삽입
    return quicksort(less) + [pivot] + quicksort(greater)   # 재귀적으로 정렬된 less, 피벗, greater를 합쳐 반환
```

```python
if __name__ == "__main__":                             # 스크립트 실행 시 바로 아래 코드 실행
```
