```python
# 퀵정렬(Quick Sort) 알고리즘을 구현하는 함수
def quicksort(arr, low, high):
    # 만약 low가 high보다 크거나 같다면 더 이상 정렬할 필요가 없음
    if low < high:
        # 피벗을 기준으로 배열을 분할하고, 피벗의 위치를 반환
        pi = partition(arr, low, high)
        
        # 피벗의 왼쪽 부분에 대해 재귀적으로 퀵정렬 수행
        quicksort(arr, low, pi - 1)
        
        # 피벗의 오른쪽 부분에 대해 재귀적으로 퀵정렬 수행
        quicksort(arr, pi + 1, high)

# 배열을 피벗을 기준으로 분할하는 함수
def partition(arr, low, high):
    # 가장 오른쪽 요소를 피벗으로 설정
    pivot = arr[high]
    
    # i는 작은 요소들을 배치할 위치를 나타냄
    i = low - 1
    
    # low에서 high-1까지 반복하면서 요소들을 비교
    for j in range(low, high):
        # 만약 현재 요소가 피벗보다 작거나 같다면
        if arr[j] <= pivot:
            # i를 증가시키고, arr[i]와 arr[j]를 교환
            i += 1
            arr[i], arr[j] = arr[j], arr[i]
    
    # 피벗을 올바른 위치로 이동시킴 (i+1 위치에 배치)
    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    
    # 피벗이 위치한 인덱스를 반환
    return i + 1

# 테스트를 위한 예제
if __name__ == "__main__":
    # 정렬할 배열
    arr = [10, 7, 8, 9, 1, 5]
    
    # 배열의 길이
    n = len(arr)
    
    # 퀵정렬 실행 (시작 인덱스: 0, 끝 인덱스: n-1)
    quicksort(arr, 0, n - 1)
    
    # 정렬 결과 출력
    print("정렬된 배열:", arr)
```
