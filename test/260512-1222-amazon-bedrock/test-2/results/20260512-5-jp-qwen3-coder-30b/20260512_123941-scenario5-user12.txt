```python
# 퀵정렬(Quck Sort) 알고리즘 구현

# 배열을 분할하는 함수 - 피벗을 기준으로 배열을 나눔
def partition(arr, low, high):
    # 배열의 마지막 원소를 피벗으로 선택
    pivot = arr[high]
    # 작은 요소들이 위치할 인덱스를 설정 (초기값은 low-1)
    i = low - 1
    
    # 배열을 순회하며 피벗과 비교
    for j in range(low, high):
        # 현재 요소가 피벗보다 작거나 같은 경우
        if arr[j] <= pivot:
            # i를 증가시키고, 현재 요소와 교환
            i += 1
            arr[i], arr[j] = arr[j], arr[i]
    
    # 피벗을 올바른 위치로 이동 (작은 요소들 집합과 큰 요소들 집합 사이에 위치)
    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    # 피벗이 위치한 인덱스 반환
    return i + 1

# 퀵정렬의 메인 함수 - 재귀적으로 배열을 정렬함
def quickSort(arr, low, high):
    # 기본 조건: low가 high보다 크거나 같으면 종료
    if low < high:
        # 피벗을 기준으로 배열을 분할하고, 피벗의 인덱스를 가져옴
        pi = partition(arr, low, high)
        
        # 피벗 왼쪽 부분 배열에 대해 재귀 호출 (피벗보다 작은 원소들)
        quickSort(arr, low, pi - 1)
        
        # 피벗 오른쪽 부분 배열에 대해 재귀 호출 (피벗보다 큰 원소들)
        quickSort(arr, pi + 1, high)

# 사용 예제
if __name__ == "__main__":
    # 정렬할 배열 정의
    arr = [10, 7, 8, 9, 1, 5]
    n = len(arr)
    
    # 퀵정렬 실행
    quickSort(arr, 0, n - 1)
    
    # 정렬 후 출력
    print("정렬된 배열:", arr)
```
