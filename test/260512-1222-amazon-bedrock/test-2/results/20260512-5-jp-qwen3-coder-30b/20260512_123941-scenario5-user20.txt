```python
# 퀵정렬을 수행하는 함수 정의
def quicksort(arr):
    # 기저 조건: 배열의 길이가 1 이하이면 그대로 반환
    if len(arr) <= 1:
        return arr

    # 피벗(pivot)을 배열의 중간 요소로 선택
    pivot = arr[len(arr) // 2]

    # 피벗보다 작은 요소들을 저장할 리스트
    left = [x for x in arr if x < pivot]

    # 피벗과 같은 요소들을 저장할 리스트
    middle = [x for x in arr if x == pivot]

    # 피벗보다 큰 요소들을 저장할 리스트
    right = [x for x in arr if x > pivot]

    # 재귀적으로 왼쪽과 오른쪽 부분 배열을 정렬하고, 중간 부분과 합칩니다.
    return quicksort(left) + middle + quicksort(right)

# 예시 배열
arr = [3, 6, 8, 10, 1, 2, 1]

# 퀵정렬 수행 및 결과 출력
print("정렬 전:", arr)
sorted_arr = quicksort(arr)
print("정렬 후:", sorted_arr)
```
