```python
def quicksort(arr, low, high):
    # 배열의 하위 범위가 유효한지 확인 (low < high)
    if low < high:
        # 분할 및 정렬을 위한 인덱스를 얻습니다.
        pi = partition(arr, low, high)

        # 피벗의 왼쪽 부분에 대해 퀵소트 재귀 호출
        quicksort(arr, low, pi - 1)

        # 피벗의 오른쪽 부분에 대해 퀵소트 재귀 호출
        quicksort(arr, pi + 1, high)


def partition(arr, low, high):
    # 가장 오른쪽 요소를 피벗으로 선택합니다.
    pivot = arr[high]

    # 작은 요소들의 인덱스 초기화
    i = low - 1

    # 배열의 모든 요소를 순회하면서 피벗과 비교합니다.
    for j in range(low, high):
        # 현재 요소가 피벗보다 작거나 같으면
        if arr[j] <= pivot:
            # 인덱스 증가
            i += 1
            # 현재 요소와 작은 요소 영역의 다음 위치를 교환합니다.
            arr[i], arr[j] = arr[j], arr[i]

    # 피벗을 올바른 위치에 배치합니다.
    arr[i + 1], arr[high] = arr[high], arr[i + 1]

    # 피벗이 위치한 인덱스를 반환합니다.
    return i + 1


# 예제 사용법
if __name__ == "__main__":
    # 정렬할 배열 정의
    arr = [10, 7, 8, 9, 1, 5]
    
    # 배열의 길이를 얻습니다.
    n = len(arr)
    
    # 퀵소트 함수 호출
    quicksort(arr, 0, n - 1)
    
    # 정렬된 배열 출력
    print("정렬된 배열:", arr)
```
이 코드는 퀵소트 알고리즘을 구현한 것입니다. `quicksort` 함수는 배열을 재귀적으로 분할하고 정렬하며, `partition` 함수는 피벗을 기준으로 배열을 분할하는 역할을 합니다. 각 줄마다 주석을 통해 코드의 동작을 설명하였습니다.
