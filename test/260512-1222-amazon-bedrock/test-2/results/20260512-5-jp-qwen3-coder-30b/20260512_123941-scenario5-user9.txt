```python
# 퀵 정렬을 구현하는 함수입니다.
def quicksort(arr):
    # 배열의 길이가 1 이하인 경우, 이미 정렬된 상태이므로 그대로 반환합니다.
    if len(arr) <= 1:
        return arr
    
    # 배열의 중간에 있는 원소를 기준점(pivot)으로 설정합니다.
    pivot = arr[len(arr) // 2]
    
    # 기준점보다 작은 원소들로 구성된 배열을 만듭니다.
    left = [x for x in arr if x < pivot]
    
    # 기준점과 같은 원소들로 구성된 배열을 만듭니다.
    middle = [x for x in arr if x == pivot]
    
    # 기준점보다 큰 원소들로 구성된 배열을 만듭니다.
    right = [x for x in arr if x > pivot]
    
    # 재귀적으로 왼쪽과 오른쪽 배열을 정렬하고, 중간 배열과 합쳐서 결과를 반환합니다.
    return quicksort(left) + middle + quicksort(right)

# 예제 입력 배열을 정의합니다.
arr = [3, 6, 8, 10, 1, 2, 1]

# 퀵 정렬을 실행하고 결과를 출력합니다.
print(quicksort(arr))
```

### 설명:
- 위 코드는 파이썬으로 구현된 퀵 정렬 알고리즘입니다.
- `quicksort` 함수는 입력 배열을 기준점(pivot)을 이용해 세 부분으로 나누고, 재귀적으로 각각을 정렬한 후 합칩니다.
- 기준점은 배열의 중간 값으로 선택됩니다.
- 시간 복잡도는 평균적으로 O(n log n)이며, 최악의 경우 O(n²)입니다.
