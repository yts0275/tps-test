```python
def quicksort(arr, low, high):
    """
    퀵정렬 알고리즘을 사용하여 배열을 정렬하는 함수입니다.
    :param arr: 정렬할 배열
    :param low: 정렬 범위의 시작 인덱스
    :param high: 정렬 범위의 끝 인덱스
    """
    # 시작 인덱스가 끝 인덱스보다 작거나 같을 때만 정렬 수행
    if low < high:
        # 파티션 함수를 호출하여 피벗의 올바른 위치를 찾고, 그 위치를 pivot_index에 저장
        pivot_index = partition(arr, low, high)
        
        # 피벗 왼쪽 부분 배열에 대해 재귀적으로 퀵정렬 실행
        quicksort(arr, low, pivot_index - 1)
        
        # 피벗 오른쪽 부분 배열에 대해 재귀적으로 퀵정렬 실행
        quicksort(arr, pivot_index + 1, high)


def partition(arr, low, high):
    """
    배열을 피벗 기준으로 분할하는 함수입니다.
    :param arr: 분할할 배열
    :param low: 분할 범위의 시작 인덱스
    :param high: 분할 범위의 끝 인덱스
    :return: 피벗이 최종적으로 위치할 인덱스
    """
    # 가장 오른쪽 요소를 피벗으로 설정
    pivot = arr[high]
    
    # i는 작은 요소들의 끝 위치를 가리키는 인덱스
    i = low - 1
    
    # 배열의 모든 요소를 순회하며,
    for j in range(low, high):
        # 현재 요소가 피벗보다 작거나 같은 경우
        if arr[j] <= pivot:
            # i를 증가시키고, 현재 요소와 i번째 요소를 교환
            i += 1
            arr[i], arr[j] = arr[j], arr[i]  # 두 요소의 위치를 바꿈
    
    # 피벗을 올바른 위치로 이동시킴 (i+1 위치에 배치)
    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    
    # 피벗이 최종적으로 위치할 인덱스 반환
    return i + 1


# 예제 사용법
if __name__ == "__main__":
    # 정렬할 배열 생성
    sample_array = [10, 7, 8, 9, 1, 5]
    
    print("정렬 전 배열:", sample_array)
    
    # 퀵정렬 실행
    quicksort(sample_array, 0, len(sample_array) - 1)
    
    print("정렬 후 배열:", sample_array)
```

### 설명:
- `quicksort` 메서드는 재귀적으로 배열을 정렬합니다.
- `partition` 메서드는 피벗을 기준으로 배열을 분할하고, 피벗이 위치해야 할 인덱스를 반환합니다.

이 코드는 일반적인 퀵정렬의 구현이며, 평균적인 시간 복잡도는 $O(n \log n)$입니다.
