```python
# quicksort 구현 예시 (Python), 각 줄마다 주석을 달았습니다.

def quick_sort(arr, low, high):
    # quick_sort 함수: 배열 arr의 low 인덱스부터 high 인덱스까지 재귀적으로 정렬
    if low < high:  # 정렬이 필요한 구간인지 확인 (low가 high보다 작을 때만 정렬)
        # 분할 지점(index)를 찾는다. `partition` 함수가 피벗을 기준으로 배열을 나누고 그 위치를 반환
        pi = partition(arr, low, high)
        # 피벗보다 작은 부분(왼쪽)을 재귀적으로 정렬
        quick_sort(arr, low, pi-1)
        # 피벗보다 큰 부분(오른쪽)을 재귀적으로 정렬
        quick_sort(arr, pi+1, high)

def partition(arr, low, high):
    # partition 함수: 피벗(보통 가장 오른쪽 원소)을 기준으로 배열을 두 부분으로 나누고
    # 피벗이 올바른 위치가 되도록 원소들을 교환한다.
    pivot = arr[high]          # 피벗은 배열의 가장 오른쪽 원소로 설정
    i = low - 1                # i는 피벗보다 작은 원소가 끝나는 인덱스를 가리킴
    for j in range(low, high): # j는 left부터 right-1까지 순회
        if arr[j] <= pivot:    # 현재 원소가 피벗보다 작거나 같으면
            i += 1              # i를 한 칸 뒤로 이동
            # arr[i]와 arr[j]를 교환해 arr[i] 위치에 피벗보다 작은 원소를 배치
            arr[i], arr[j] = arr[j], arr[i]
    # 피벗을 i+1 위치에 놓음. 이때 피벗은 자신보다 작은 모든 원소의 오른쪽에 위치
    arr[i+1], arr[high] = arr[high], arr[i+1]
    return i + 1                # 피벗이 위치한 인덱스를 반환

if __name__ == "__main__":
    # 테스트용 배열 정의 (정렬 전 상태)
    arr = [10, 7, 8, 9, 1, 5]
    # quick_sort 호출: 배열 전체(0부터 len(arr)-1까지)를 정렬
    quick_sort(arr, 0, len(arr)-1)
    # 정렬 결과를 출력
    print("정렬된 배열:", arr)
```
