```python
# 퀵소트(Quick Sort) 구현 (Python)
# 각 줄마다 한국어 주석을 달았습니다.

def quicksort(arr, low, high):  # 재귀적 퀵소트 함수 정의
    if low < high:  # 정렬할 범위가 있을 때
        pi = partition(arr, low, high)  # 배열을 피벗 기준으로 분할해 피벗 인덱스를 반환
        quicksort(arr, low, pi - 1)  # 피벗 왼쪽 부분을 재귀적으로 정렬
        quicksort(arr, pi + 1, high)  # 피벗 오른쪽 부분을 재귀적으로 정렬

def partition(arr, low, high):  # 피벗 기준으로 배열을 분할
    pivot = arr[high]  # 가장 오른쪽 원소를 피벗으로 선택
    i = low - 1  # 피벗보다 작은 원소가 저장될 인덱스 초기화
    for j in range(low, high):  # 범위 안의 모든 원소를 순회
        if arr[j] <= pivot:  # 피벗보다 작거나 같은 원소를 찾으면
            i += 1  # 작은 영역의 끝 인덱스를 증가
            arr[i], arr[j] = arr[j], arr[i]  # 현재 원소와 i+1번째 원소를 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 올바른 위치(정렬된 위치)로 이동
    return i + 1  # 피벗의 최종 인덱스를 반환

def quick_sort(array):  # 사용자가 호출할 함수
    if not array:  # 배열이 비어 있으면 바로 반환
        return
    quicksort(array, 0, len(array) - 1)  # 전체 배열을 정렬
    return array

if __name__ == "__main__":  # 프로그램이 직접 실행될 때
    arr = [10, 7, 8, 9, 1, 5]  # 정렬할 예시 배열
    print("원본 배열:", arr)  # 원본 배열 출력
    quick_sort(arr)  # Quick Sort 호출
    print("정렬된 배열:", arr)  # 정렬 결과 출력
```
