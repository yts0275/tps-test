```python
# ------------------------------------------------------------------
# 퀵소트(Quick Sort) 구현 (Python)
# 각 줄마다 한 줄 주석을 달아 동작 원리를 설명합니다.
# ------------------------------------------------------------------

def quick_sort(arr):
    # 인풋 배열을 그대로 수정하므로, 배열을 복사하지 않고 인자로 받은 배열을 사용합니다.
    
    def _quick_sort(left, right):
        # 재귀 함수: left ~ right 인덱스 범위에 대해 정렬을 수행합니다.
        if left >= right:
            # 정렬할 부분이 하나이거나 비어 있는 경우 바로 반환합니다.
            return
        
        # 파티셔닝(Partition) 단계
        pivot_index = partition(left, right)
        # 왼쪽 부분( pivot보다 작은 값들 )에 대해 재귀적으로 정렬합니다.
        _quick_sort(left, pivot_index - 1)
        # 오른쪽 부분( pivot보다 큰 값들 )에 대해 재귀적으로 정렬합니다.
        _quick_sort(pivot_index + 1, right)
    
    def partition(left, right):
        # 현재 범위의 가장 끝( right )에 있는 값을 피벗(pivot)으로 선택합니다.
        pivot = arr[right]
        # i 포인터는 "피벗보다 작은 값들이 모여 있는 영역의 끝"을 가리킵니다.
        i = left - 1
        # j 포인터를 left 부터 right-1 까지 순회하면서, 피벗보다 작은 값이 나오면 i 를 한 칸 오른쪽으로 끌고,
        # 그 값과 arr[i] 를 서로 바꿉니다.
        for j in range(left, right):
            if arr[j] < pivot:
                i += 1
                arr[i], arr[j] = arr[j], arr[i]
        # 마지막에 피벗을 i+1 위치에 넣어서, 피벗보다 작은 값은 왼쪽, 큰 값은 오른쪽에 정렬됩니다.
        arr[i + 1], arr[right] = arr[right], arr[i + 1]
        return i + 1
    
    # 재귀 호출을 시작합니다. (배열 전체를 정렬)
    _quick_sort(0, len(arr) - 1)
    # 함수형 스타일로 배열을 반환해 주면 사용하기 편합니다.
    return arr

# ------------------------------------------------------------------
# 예시 사용
# ------------------------------------------------------------------
if __name__ == "__main__":
    sample = [10, 7, 8, 9, 1, 5]
    print("정렬 전:", sample)
    sorted_arr = quick_sort(sample)
    print("정렬 후 :", sorted_arr)
```

**설명**  
- `quick_sort` 함수는 입력 배열을 인자로 받아 재귀적으로 파티션(분할) 과정과 정렬을 수행합니다.  
- `partition` 함수는 피벗을 기준으로 작은 값은 왼쪽, 큰 값은 오른쪽에 모으는 핵심 과정을 담당합니다.  
- 모든 조건문, 루프, 스왑 동작에 대해 한 줄씩 주석을 넣어 초보자도 흐름을 쉽게 따라올 수 있게 구성했습니다.  
- 예시 코드에서는 정수 배열을 정렬하여 결과를 출력합니다.  
- **주의**: 대용량 배열이나 이미 거의 정렬된 배열에서 재귀 깊이가 깊어질 수 있으니, 필요에 따라 최적화(깊이 제한, 삽입 정렬 전환 등)를 고려하세요.
