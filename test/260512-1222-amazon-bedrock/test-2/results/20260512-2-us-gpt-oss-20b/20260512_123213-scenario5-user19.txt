```python
# quicksort 구현 (Python 코드, 각 줄마다 주석 추가)

def quick_sort(arr):
    # 외부 함수: quick_sort에서 내부 재귀를 호출
    def _quick_sort(low, high):
        # 재귀 함수: 배열의 low~high 범위 안에서 정렬 완료
        if low < high:
            # 파티션을 호출해 피벗 인덱스를 구함
            pivot_index = partition(low, high)
            # 피벗 앞 범위 재귀 정렬
            _quick_sort(low, pivot_index - 1)
            # 피벗 뒤 범위 재귀 정렬
            _quick_sort(pivot_index + 1, high)

    def partition(low, high):
        # 파티션 함수: 왼쪽 끝 요소를 피벗으로 선택하고 배열을 분할
        pivot = arr[low]                       # 피벗 값 저장
        left = low + 1                          # 왼쪽 포인터 초기화
        right = high                            # 오른쪽 포인터 초기화

        while True:                             # 포인터가 교차할 때까지 반복
            # 왼쪽 포인터가 피벗보다 큰 값을 찾을 때까지 이동
            while left <= right and arr[left] <= pivot:
                left += 1
            # 오른쪽 포인터가 피벗보다 작은 값을 찾을 때까지 이동
            while left <= right and arr[right] >= pivot:
                right -= 1
            # 포인터가 교차하면 분할 종료
            if left > right:
                break
            # 교차한 값들을 서로 교환
            arr[left], arr[right] = arr[right], arr[left]

        # 피벗을 올바른 위치(요소보다 큰 좌우를 정렬)로 이동
        arr[low], arr[right] = arr[right], arr[low]
        # 피벗의 최종 인덱스를 반환
        return right

    # 재귀 정렬 시작 (전체 배열 기준)
    _quick_sort(0, len(arr) - 1)
    # 정렬된 배열 반환
    return arr


# 테스트 코드 (demo)
if __name__ == "__main__":
    # 정렬할 데이터 정의
    data = [3, 6, 8, 10, 1, 2, 1]
    print("정렬 전:", data)
    # quick_sort 호출 및 결과 출력
    sorted_data = quick_sort(data)
    print("정렬 후:", sorted_data)
```

이 코드는 **quicksort** 알고리즘을 재귀적으로 구현한 것이며, 각 코드 라인에 대해 한국어 주석을 통해 단계별 동작을 설명하고 있습니다.
