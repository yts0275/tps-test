아래는 파이썬으로 구현한 퀵소트(Quick Sort) 알고리즘이며, **각 줄마다 주석을 달았습니다**.  
주석은 한국어로 작성되어 있어 이해하시기 편합니다.

```python
def quick_sort(arr, low, high):
    # quick_sort 함수: arr 배열의 low 인덱스부터 high 인덱스까지 정렬한다.
    if low < high:
        # 재귀의 종료 조건: 두 개 이하의 원소가 남으면 더이상 정렬 필요 없음
        pi = partition(arr, low, high)   # 파티션을 수행하고 피벗(중앙) 인덱스를 받는다
        quick_sort(arr, low, pi - 1)     # 피벗보다 작은 부분 배열을 재귀 정렬
        quick_sort(arr, pi + 1, high)    # 피벗보다 큰 부분 배열을 재귀 정렬

def partition(arr, low, high):
    # partition 함수: high 인덱스에 있는 요소를 피벗으로 사용해 배열을 분할한다.
    pivot = arr[high]                   # 피벗(중앙) 요소 선택
    i = low - 1                         # i는 피벗보다 작은 요소가 있는 가장 오른쪽 인덱스
    for j in range(low, high):          # j는 배열의 왼쪽부터 피벗 바로 앞까지 탐색
        if arr[j] < pivot:              # 현재 요소가 피벗보다 작으면
            i += 1                     # i를 한 칸 오른쪽으로 이동
            arr[i], arr[j] = arr[j], arr[i]  # 작은 요소를 왼쪽으로 옮김
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 마지막으로 피벗을 중앙에 삽입
    return i + 1                         # 피벗의 최종 인덱스를 반환

if __name__ == "__main__":
    # 예시 배열 선언
    arr = [10, 7, 8, 9, 1, 5]
    n = len(arr)                        # 배열 길이 저장
    quick_sort(arr, 0, n - 1)           # 전체 배열에 퀵소트 수행
    print("Sorted array:", arr)         # 정렬 결과 출력
```
