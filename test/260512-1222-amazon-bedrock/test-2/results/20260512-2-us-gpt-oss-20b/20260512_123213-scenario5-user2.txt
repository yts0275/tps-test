def quick_sort(arr):  # 주어진 리스트 arr를 정렬하여 반환합니다.
    return _quick_sort(arr, 0, len(arr)-1)  # 재귀 퀵소트 호출

def _quick_sort(arr, low, high):  # 내부 재귀 퀵소트 구현
    if low < high:  # 정렬 범위가 1 이상일 때
        pivot_index = partition(arr, low, high)  # 피벗 자리 찾기
        _quick_sort(arr, low, pivot_index - 1)  # 피벗 왼쪽 영역 정렬
        _quick_sort(arr, pivot_index + 1, high)  # 피벗 오른쪽 영역 정렬

def partition(arr, low, high):  # 피벗을 기준으로 배열을 분할
    pivot = arr[high]  # 마지막 원소를 피벗으로 선택
    i = low - 1  # i는 피벗보다 작은 요소의 마지막 인덱스
    for j in range(low, high):  # 고정 피벗을 제외한 범위 탐색
        if arr[j] <= pivot:  # 현재 원소가 피벗 이하일 때
            i += 1  # 작은 영역 한 칸 확장
            arr[i], arr[j] = arr[j], arr[i]  # 원소 교환
    arr[i+1], arr[high] = arr[high], arr[i+1]  # 피벗을 적절한 위치로 이동
    return i+1  # 피벗 인덱스를 반환

if __name__ == "__main__":  # 실행 시 테스트
    arr = [3, 6, 8, 10, 1, 2, 1]  # 정렬 전 배열
    quick_sort(arr)  # 배열 정렬 함수 호출
    print("정렬된 배열:", arr)  # 정렬 결과 출력
