```python
def quicksort(arr):  # 함수 정의: 퀵소트를 수행할 리스트를 매개변수로 받는다.
    if len(arr) <= 1:  # 리스트 길이가 0 또는 1이면 이미 정렬된 상태이므로 그대로 반환한다.
        return arr  # 정렬된 리스트를 반환한다.
    pivot = arr[-1]  # 기준점(pivot)을 리스트의 마지막 요소로 선택한다.
    less = [x for x in arr[:-1] if x <= pivot]  # 피벗 이하의 요소들로 구성된 리스트를 만든다.
    greater = [x for x in arr[:-1] if x > pivot]  # 피벗 보다 큰 요소들로 구성된 리스트를 만든다.
    # 피벗을 기준으로 좌우 부분을 재귀적으로 정렬한 뒤 합쳐 최종 정렬 결과를 만든다.
    return quicksort(less) + [pivot] + quicksort(greater)  # 재귀 호출 결과를 결합하여 반환한다.

if __name__ == "__main__":  # 스크립트가 직접 실행될 때만 아래 코드를 실행한다.
    unsorted = [3, 6, 8, 10, 1, 2, 1]  # 정렬되지 않은 예시 배열을 정의한다.
    sorted_arr = quicksort(unsorted)  # 퀵소트 함수를 호출하여 정렬된 배열을 얻는다.
    print(sorted_arr)  # 정렬된 결과를 콘솔에 출력한다.
```
