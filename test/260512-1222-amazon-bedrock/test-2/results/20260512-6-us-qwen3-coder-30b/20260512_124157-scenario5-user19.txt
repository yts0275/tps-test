```python
# 퀵정렬(Quck Sort) 알고리즘을 구현하는 코드입니다.
# 이 알고리즘은 분할 정복(Divide and Conquer) 기법을 사용하여 배열을 정렬합니다.

def quicksort(arr, low, high):
    # 재귀 호출의 기저 조건: low가 high보다 크거나 같으면 더 이상 정렬할 필요가 없습니다.
    if low < high:
        # 파티션 함수를 호출하여 피벗을 올바른 위치에 배치하고,
        # 피벗의 인덱스를 반환받습니다.
        pi = partition(arr, low, high)
        
        # 피벗 왼쪽 부분 배열에 대해 재귀적으로 퀵정렬을 수행합니다.
        quicksort(arr, low, pi - 1)
        
        # 피벗 오른쪽 부분 배열에 대해 재귀적으로 퀵정렬을 수행합니다.
        quicksort(arr, pi + 1, high)

def partition(arr, low, high):
    # 마지막 요소를 피벗으로 선택합니다.
    pivot = arr[high]
    
    # i는 작은 요소들의 인덱스를 나타냅니다.
    i = low - 1
    
    # 배열의 모든 요소를 순회하면서 비교합니다.
    for j in range(low, high):
        # 현재 요소가 피벗보다 작거나 같으면,
        if arr[j] <= pivot:
            # i를 증가시키고, 현재 요소와 arr[i]를 교환합니다.
            i += 1
            arr[i], arr[j] = arr[j], arr[i]
    
    # 피벗을 올바른 위치로 이동시킵니다.
    # 즉, 피벗보다 작은 요소들이 모두 왼쪽에, 큰 요소들이 모두 오른쪽에 있도록 합니다.
    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    
    # 피벗이 위치한 인덱스를 반환합니다.
    return i + 1

# 테스트할 배열 예시입니다.
arr = [10, 7, 8, 9, 1, 5]

# 퀵정렬을 적용하기 전 배열 상태를 출력합니다.
print("정렬 전 배열:", arr)

# 퀵정렬을 수행합니다. 인덱스는 0부터 시작하여 배열의 끝까지입니다.
quicksort(arr, 0, len(arr) - 1)

# 퀵정렬을 적용한 후 결과를 출력합니다.
print("정렬 후 배열:", arr)
