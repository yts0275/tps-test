```python
# 퀵정렬(Quiicksort) 알고리즘을 구현하는 함수입니다.
def quicksort(arr):
    # 배열의 길이가 1 이하인 경우, 정렬할 필요가 없으므로 그대로 반환합니다.
    if len(arr) <= 1:
        return arr
    
    # 기준 원소(pivot)를 배열의 중간에 위치한 원소로 설정합니다.
    pivot = arr[len(arr) // 2]
    
    # 기준 원소보다 작은 원소들을 저장할 리스트입니다.
    left = [x for x in arr if x < pivot]
    
    # 기준 원소와 같은 원소들을 저장할 리스트입니다.
    middle = [x for x in arr if x == pivot]
    
    # 기준 원소보다 큰 원소들을 저장할 리스트입니다.
    right = [x for x in arr if x > pivot]
    
    # 재귀적으로 왼쪽과 오른쪽 부분 배열을 정렬한 후, 중간 배열과 함께 결합합니다.
    return quicksort(left) + middle + quicksort(right)

# 예제 배열을 정의합니다.
arr = [3, 6, 8, 10, 1, 2, 1]

# 퀵정렬 함수를 호출하여 정렬된 배열을 출력합니다.
print(quicksort(arr))
```
