```python
# 퀵정렬(Quick Sort) 알고리즘을 구현합니다.
def quicksort(arr, low, high):
    # 재귀적으로 정렬을 수행할 때 종료 조건입니다.
    # low가 high보다 크거나 같으면 더 이상 정렬할 요소가 없습니다.
    if low < high:
        # 파티션 함수를 호출하여 피벗의 올바른 위치를 찾습니다.
        pi = partition(arr, low, high)
        
        # 피벗의 왼쪽 부분 배열에 대해 재귀적으로 퀵정렬을 수행합니다.
        quicksort(arr, low, pi - 1)
        
        # 피벗의 오른쪽 부분 배열에 대해 재귀적으로 퀵정렬을 수행합니다.
        quicksort(arr, pi + 1, high)

# 파티션 함수: 배열을 피벗을 기준으로 두 개의 부분 배열로 분할합니다.
def partition(arr, low, high):
    # 마지막 요소를 피벗으로 선택합니다.
    pivot = arr[high]
    
    # i는 작은 요소들을 가리키는 인덱스입니다.
    i = low - 1
    
    # j는 현재 검토하는 요소를 가리킵니다.
    for j in range(low, high):
        # 현재 요소가 피벗보다 작거나 같으면,
        if arr[j] <= pivot:
            # i를 증가시키고,
            i += 1
            # 배열의 i번째와 j번째 요소를 swap합니다.
            arr[i], arr[j] = arr[j], arr[i]
    
    # 피벗을 올바른 위치로 이동시킵니다.
    # 피벗은 현재 i+1 위치에 있습니다.
    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    
    # 피벗이 위치한 인덱스를 반환합니다.
    return i + 1

# 예제 사용법
if __name__ == "__main__":
    # 정렬할 배열을 정의합니다.
    arr = [10, 7, 8, 9, 1, 5]
    
    # 배열의 길이를 계산합니다.
    n = len(arr)
    
    # 퀵정렬을 실행합니다.
    quicksort(arr, 0, n - 1)
    
    # 결과를 출력합니다.
    print("정렬된 배열:", arr)
```
### 설명:

1. `quicksort` 함수는 재귀적으로 배열을 분할하고 정렬합니다.
2. `partition` 함수는 피벗을 기준으로 배열을 나누고, 피벗이 올바른 위치에 있게 만듭니다.
3. 코드 상에서 주석을 통해 각 줄의 동작을 쉽게 이해할 수 있도록 했습니다.
