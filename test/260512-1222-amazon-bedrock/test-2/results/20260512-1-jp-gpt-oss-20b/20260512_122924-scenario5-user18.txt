```python
# 빠른 정렬(QuickSort) 구현 (Python)
def quicksort(arr):                      # 정렬할 리스트를 입력으로 받음
    if len(arr) <= 1:                    # 리스트가 비었거나 한 개의 원소만 남았으면 (이미 정렬됨) 그 리스트를 반환
        return arr                      # 정렬이 끝난 리스트를 반환
    pivot = arr[len(arr) // 2]           # 중앙에 있는 원소를 피벗으로 선택
    left = [x for x in arr if x < pivot] # 피벗보다 작은 원소들만 모음
    middle = [x for x in arr if x == pivot]  # 피벗과 같은 값들을 모음 (복수개가 있을 수 있음)
    right = [x for x in arr if x > pivot]  # 피벗보다 큰 원소들만 모음
    return quicksort(left) + middle + quicksort(right)  # 재귀적으로 정렬한 좌우 리스트를 합쳐서 반환

# 사용 예시
unsorted_list = [8, 3, 9, 1, 4, 7, 2, 6, 5]
sorted_list = quicksort(unsorted_list)   # quicksort 함수 호출
print(sorted_list)                       # 정렬된 결과 출력
```

각 줄마다 `#` 뒤에 한국어 주석을 달아 이해를 돕도록 하였습니다.
