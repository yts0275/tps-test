```python
# quicksort 함수 정의: arr(정렬할 배열), low(시작 인덱스), high(끝 인덱스)를 인자로 받음
def quicksort(arr, low, high):
    # low가 high보다 작으면 재귀적으로 분할 및 정렬을 진행한다
    if low < high:
        # partition 함수를 호출해 피벗 위치를 반환하고, 배열을 피벗 기준으로 재배치한다
        pi = partition(arr, low, high)
        # 피벗 앞(왼쪽) 부분을 재귀적으로 quicksort 한다
        quicksort(arr, low, pi - 1)
        # 피벗 뒤(오른쪽) 부분을 재귀적으로 quicksort 한다
        quicksort(arr, pi + 1, high)

# partition 함수 정의: 피벗을 기준으로 배열을 재배치하고 피벗의 최종 인덱스를 반환한다
def partition(arr, low, high):
    # 마지막 원소를 피벗(pivot)으로 선택한다
    pivot = arr[high]
    # 피벗 이하 원소의 최종 위치를 가리키는 인덱스 i를 low-1로 초기화한다
    i = low - 1
    # low부터 high-1까지 인덱스 j를 순회하며, arr[j]가 피벗 이하인 경우에만 교환한다
    for j in range(low, high):
        if arr[j] <= pivot:
            i += 1
            arr[i], arr[j] = arr[j], arr[i]   # i와 j에 위치한 원소를 교환
    # 피벗을 올바른 위치 i+1에 배치한다 (피벗이 가장 큰 값이 될 때)
    arr[i+1], arr[high] = arr[high], arr[i+1]
    # 피벗의 최종 인덱스(정렬된 위치)를 반환한다
    return i + 1

# 예시 사용법: 스크립트가 직접 실행될 때 동작
if __name__ == "__main__":
    data = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 배열을 초기화
    quicksort(data, 0, len(data)-1)  # quicksort 함수를 호출해 정렬
    print(data)  # 정렬된 배열을 출력
```
