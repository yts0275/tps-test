def quicksort(arr, low, high):  # quicksort 함수: arr를 low~high까지 정렬
    if low < high:  # 재귀 종료 조건인 low가 high보다 작을 때만 진행
        pivot_index = partition(arr, low, high)  # 배열을 파티션하여 피벗 위치 반환
        quicksort(arr, low, pivot_index - 1)  # 피벗 왼쪽 부분 정렬
        quicksort(arr, pivot_index + 1, high)  # 피벗 오른쪽 부분 정렬

def partition(arr, low, high):  # 파티션 함수: 피벗을 기준으로 배열 분할
    pivot = arr[high]  # 마지막 원소를 피벗으로 선택
    i = low - 1  # i는 작은 원소가 들어갈 마지막 인덱스
    for j in range(low, high):  # low~high-1 범위에서 반복
        if arr[j] <= pivot:  # 현재 원소가 피벗 이하일 때
            i += 1  # i를 한 칸 앞으로
            arr[i], arr[j] = arr[j], arr[i]  # i와 j 위치를 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 적절한 위치로 이동
    return i + 1  # 피벗의 최종 인덱스 반환

# 사용 예시
if __name__ == "__main__":  # 메인 실행문
    arr = [34, 7, 23, 32, 5, 62]  # 정렬할 배열
    quicksort(arr, 0, len(arr) - 1)  # 인덱스 범위 지정
    print(arr)  # 정렬 결과 출력
