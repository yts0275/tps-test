```python
# 퀵소트 구현 (파이썬)

def quick_sort(arr):
    # 배열의 길이가 0 또는 1이면 그대로 반환(이미 정렬돼 있음)
    if len(arr) <= 1:
        return arr
    # 피벗 값은 배열의 중간 원소로 선택 (안정적 성능)
    pivot = arr[len(arr) // 2]
    # 피벗보다 작은 원소들로 새 리스트 생성
    less = [x for x in arr if x < pivot]
    # 피벗과 같은 원소들로 새 리스트 생성 (같은 값들끼리 한 묶음)
    equal = [x for x in arr if x == pivot]
    # 피벗보다 큰 원소들로 새 리스트 생성
    greater = [x for x in arr if x > pivot]
    # 재귀적으로 less와 greater를 정렬하고, equal은 그대로 붙여 최종 정렬 결과 반환
    return quick_sort(less) + equal + quick_sort(greater)

# 사용 예시
if __name__ == "__main__":
    # 한 줄 입력에서 정수 리스트를 가져와서 정렬 실행
    arr = list(map(int, input("정렬할 정수 리스트 입력 (공백으로 구분) : ").split()))  # 입력 처리
    sorted_arr = quick_sort(arr)  # 퀵소트 호출
    print("정렬된 결과:", sorted_arr)  # 결과 출력
```
