def quicksort(arr, low, high):  # quicksort 함수 정의: arr(배열), low(시작 인덱스), high(끝 인덱스)
    if low < high:  # 재귀 종료 기준: low가 high보다 작으면 재귀 실행
        pivot_index = partition(arr, low, high)  # partition 호출 후 피벗 인덱스 받기
        quicksort(arr, low, pivot_index - 1)  # 파티션 왼쪽 부분 재귀 호출
        quicksort(arr, pivot_index + 1, high)  # 파티션 오른쪽 부분 재귀 호출

def partition(arr, low, high):  # 파티션 함수 정의: 배열을 두 부분으로 나누기
    pivot = arr[high]  # 마지막 요소를 피벗으로 선택
    i = low - 1  # 작은 요소를 위한 인덱스 i를 low - 1로 초기화
    for j in range(low, high):  # low부터 high-1까지 순회
        if arr[j] <= pivot:  # 현재 원소가 피벗보다 작거나 같으면
            i += 1  # 인덱스 i 증가
            arr[i], arr[j] = arr[j], arr[i]  # i와 j 위치의 요소 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 올바른 위치로 이동
    return i + 1  # 피벗 인덱스 반환

if __name__ == "__main__":  # 스크립트가 직접 실행될 때만 아래 코드 실행
    arr = [9, 3, 7, 1, 6, 8, 2, 5, 4]  # 정렬할 배열 초기화
    quicksort(arr, 0, len(arr) - 1)  # quicksort 호출
    print("정렬된 배열:", arr)  # 정렬 결과 출력
