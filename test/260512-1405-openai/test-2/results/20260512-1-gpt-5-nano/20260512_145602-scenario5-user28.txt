다음은 파이썬으로 구현한 퀵소트 예시입니다. 각 줄마다 간단한 주석을 달았습니다.

def quicksort(arr):  # 주어진 배열을 제자리에서 오름차순으로 정렬하는 진입점
    _quicksort(arr, 0, len(arr) - 1)  # 전체 구간에 대해 재귀적으로 정렬 수행
    return arr  # 정렬된 배열을 반환

def _quicksort(arr, low, high):  # 내부 재귀 함수: 구간 [low, high]를 정렬
    if low < high:  # 구간에 원소가 하나 이상 있을 때만 진행
        p = partition(arr, low, high)  # 피벗을 기준으로 분할 지점 얻기
        _quicksort(arr, low, p - 1)  # 좌측 부분 배열 재귀적으로 정렬
        _quicksort(arr, p + 1, high)  # 우측 부분 배열 재귀적으로 정렬

def partition(arr, low, high):  # Lomuto 파티션: 피벗(pivot)를 기준으로 분할
    pivot = arr[high]  # 보통 마지막 원소를 피벗으로 선택
    i = low - 1  # i는 피벗보다 작아야 하는 구간의 끝 인덱스

    for j in range(low, high):  # 구간 [low, high-1]을 순회
        if arr[j] <= pivot:  # 현재 원소가 피벗보다 작거나 같으면
            i += 1  # 작아진 구간의 끝을 한 칸 확장
            arr[i], arr[j] = arr[j], arr[i]  # 교환하여 작아진 구간으로 보냄

    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 구간의 중간으로 이동
    return i + 1  # 피벗의 최종 위치를 반환

# 예시 사용 방법
if __name__ == "__main__":  # 스크립트로 실행될 때만 아래 코드가 동작
    data = [9, 3, 7, 1, 5, 3]  # 테스트 배열
    print("원본:", data)  # 원본 출력
    quicksort(data)  # 배열을 제자리에서 정렬
    print("정렬:", data)  # 정렬 결과 출력
