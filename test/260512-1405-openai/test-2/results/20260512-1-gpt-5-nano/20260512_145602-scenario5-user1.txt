다음은 모든 줄에 주석이 달린 파이썬 퀵소트 구현입니다.

```python
# 퀵소트 구현 예시 - 각 줄마다 주석을 달았습니다
def partition(arr, low, high):  # 파티션 함수: 피봇은 배열의 마지막 원소
    pivot = arr[high]  # 피봇으로 마지막 원소 선택
    i = low - 1  # i는 작은 원소의 끝 위치를 나타냄
    for j in range(low, high):  # j는 현재 비교 원소의 인덱스
        if arr[j] <= pivot:  # 현재 원소가 피봇보다 작거나 같으면
            i += 1  # 작아진 원소 구간의 끝을 한 칸 늘림
            arr[i], arr[j] = arr[j], arr[i]  # 원소를 앞쪽으로 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피봇을 올바른 위치로 이동
    return i + 1  # 피봇의 최종 인덱스 반환

def quicksort(arr, low, high):  # 퀵소트 메인 재귀 함수
    if low < high:  # 범위에 원소가 두 개 이상 있을 때만 수행
        pi = partition(arr, low, high)  # 피봇 인덱스 얻기
        quicksort(arr, low, pi - 1)  # 왼쪽 부분 배열 재귀적으로 정렬
        quicksort(arr, pi + 1, high)  # 오른쪽 부분 배열 재귀적으로 정렬

if __name__ == "__main__":  # 독립 실행시만 실행되도록 하는 보호 구문
    data = [10, 7, 8, 9, 1, 5]  # 테스트 데이터 배열
    quicksort(data, 0, len(data) - 1)  # 전체 배열에 대해 퀵소트 수행
    print(data)  # 정렬된 결과 출력
```
