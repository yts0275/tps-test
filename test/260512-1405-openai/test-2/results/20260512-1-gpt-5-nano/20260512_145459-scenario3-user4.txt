def quicksort(arr, low, high):  # 부분 구간 [low, high]를 정렬하기 위한 재귀 함수
    if low < high:  # 구간에 최소 2개의 원소가 있으면
        pi = partition(arr, low, high)  # 피봇 위치를 구하는 파티션 수행
        quicksort(arr, low, pi - 1)  # 왼쪽 부분 구간 재귀적으로 정렬
        quicksort(arr, pi + 1, high)  # 오른쪽 부분 구간 재귀적으로 정렬

def partition(arr, low, high):  # 파티션을 수행하여 피봇의 위치를 반환하는 함수
    pivot = arr[high]  # 가장 오른쪽 원소를 피봇으로 선정
    i = low - 1  # 작은 원소의 구역의 끝 인덱스 초기화
    for j in range(low, high):  # 구간 [low, high-1]의 원소들 순회
        if arr[j] <= pivot:  # 현재 원소가 피봇보다 작거나 같으면
            i += 1  # 작은 원소 구역의 끝을 한 칸 확장
            arr[i], arr[j] = arr[j], arr[i]  # 원소 위치를 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피봇을 정렬된 위치로 이동
    return i + 1  # 피봇의 최종 인덱스 반환

def quicksort_list(arr):  # 전체 배열을 QuickSort로 정렬하는 편의 함수
    if arr is None:  # 입력이 None인 경우 처리
        return None  # None 반환
    quicksort(arr, 0, len(arr) - 1)  # 배열 전체 구간에 대해 정렬 수행
    return arr  # 정렬된 배열 반환
