def quicksort(arr, low=0, high=None):  # QuickSort 엔트리 포인트
    if high is None:  # 처음 호출일 때 high를 배열의 마지막 인덱스로 초기화
        high = len(arr) - 1  # 배열의 최댓값 인덱스로 설정
    if low < high:  # 서브배열에 두 개 이상의 원소가 남아있다면
        p = partition(arr, low, high)  # 피봇 인덱스를 기준으로 분할
        quicksort(arr, low, p - 1)  # 왼쪽 부분배열 재귀적으로 정렬
        quicksort(arr, p + 1, high)  # 오른쪽 부분배열 재귀적으로 정렬
    return arr  # 정렬된 배열을 반환 (제자리에서 정렬)

def partition(arr, low, high):  # 피봇을 중심으로 서브배열을 분할하는 함수
    pivot = arr[high]  # 마지막 원소를 피봇으로 선택
    i = low - 1  # 작은 원소 구간의 끝을 가리키는 포인터
    for j in range(low, high):  # 피봇 제외한 서브배열을 순회
        if arr[j] <= pivot:  # 현재 원소가 피봇보다 작거나 같으면
            i += 1  # 작은 원소 구간의 경계 확장
            arr[i], arr[j] = arr[j], arr[i]  # 원소 위치 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피봇을 작은 원소들 뒤에 배치
    return i + 1  # 피봇의 최종 위치 반환

lst = [3, 6, 8, 10, 1, 2, 1]  # 샘플 비정렬 리스트
quicksort(lst)  # 제자리에서 리스트 정렬
print(lst)  # 정렬된 리스트 출력
