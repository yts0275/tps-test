다음은 각 줄에 한 줄 주석이 달린 파이썬 퀵소트 구현입니다.

```python
# 파이썬으로 구현한 in-place 퀵소트
def quicksort(arr, low, high):  # 배열의 부분 구간 [low, high]에 대해 QuickSort 수행
    if low < high:  # 구간에 원소가 두 개 이상 있을 때만
        pi = partition(arr, low, high)  # 피벗 위치를 구해 분할
        quicksort(arr, low, pi - 1)  # 피벗 왼쪽 부분 재귀적으로 정렬
        quicksort(arr, pi + 1, high)  # 피벗 오른쪽 부분 재귀적으로 정렬

def partition(arr, low, high):  # 부분 구간을 분할하고 피벗 인덱스 반환
    pivot = arr[high]  # 마지막 원소를 피벗으로 선택
    i = low - 1  # 작은 원소 영역의 경계 인덱스
    for j in range(low, high):  # 피벗 제외한 구간 순회
        if arr[j] <= pivot:  # 현재 원소가 피벗보다 작거나 같으면
            i += 1  # 작은 원소 영역 확장
            arr[i], arr[j] = arr[j], arr[i]  # 원소 위치 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 작은 원소 뒤에 배치
    return i + 1  # 새로운 피벗 인덱스 반환

# 사용 예시
data = [34, 7, 23, 32, 5, 62]  # 정렬할 샘플 배열
quicksort(data, 0, len(data) - 1)  # 배열 전체를 제자리에서 정렬
print(data)  # 정렬된 배열 출력
```
