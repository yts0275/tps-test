def quicksort(arr, low, high):  # QuickSort 재귀 함수: 배열의 구간 [low, high]를 제자리에서 정렬
    if low < high:  # 구간의 길이가 2 이상일 때만 분할 필요
        pi = partition(arr, low, high)  # 피벗의 최종 위치를 얻음
        quicksort(arr, low, pi - 1)  # 피벗 왼쪽 부분 배열 재귀 정렬
        quicksort(arr, pi + 1, high)  # 피벗 오른쪽 부분 배열 재귀 정렬

def partition(arr, low, high):  # Lomuto 파티션: 피벗을 배열의 마지막 원소로 선택
    pivot = arr[high]  # 피벗은 구간의 마지막 원소
    i = low - 1  # 작은 원소들의 구분 인덱스
    for j in range(low, high):  # low부터 high-1까지 순회
        if arr[j] <= pivot:  # 현재 원소가 피벗보다 작거나 같으면
            i += 1  # 작은 원소의 구분 인덱스 증가
            arr[i], arr[j] = arr[j], arr[i]  # 교환: 작게는 앞쪽으로
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 정렬된 위치로 이동
    return i + 1  # 피벗의 최종 위치 반환

def quicksort_inplace(arr):  # 배열을 제 자리에서 퀵정렬하는 쉬운 진입점
    if arr:  # 배열이 비지 않으면
        quicksort(arr, 0, len(arr) - 1)  # 전체 구간에 대해 QuickSort 실행

if __name__ == "__main__":  # 스크립트로 실행될 때 예시 수행
    data = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 예시 배열
    print("원본:", data)  # 원본 출력
    quicksort_inplace(data)  # 배열을 빠르게 정렬
    print("정렬:", data)  # 정렬된 배열 출력
