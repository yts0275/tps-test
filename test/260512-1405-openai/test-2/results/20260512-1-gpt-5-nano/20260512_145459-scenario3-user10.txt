def quick_sort(arr):  # 외부 API: 배열을 받아 제자리에서 정렬한다
    quick_sort_in_place(arr, 0, len(arr) - 1)  # 구간 [0, len(arr)-1]에 대해 정렬 시작
    return arr  # 정렬된 배열을 반환한다

def quick_sort_in_place(a, low, high):  # 인-플레이스(배열 자체를 수정) 퀵소트의 재귀 함수
    if low < high:  # 구간에 원소가 두 개 이상 남았는지 확인한다
        p = partition(a, low, high)  # 피벗의 최종 위치를 얻는다
        quick_sort_in_place(a, low, p - 1)  # 왼쪽 부분구간 재귀적으로 정렬
        quick_sort_in_place(a, p + 1, high)  # 오른쪽 부분구간 재귀적으로 정렬

def partition(a, low, high):  # 피벗을 기준으로 배열을 분할하는 함수
    pivot = a[high]  # 가장 오른쪽 원소를 피벗으로 삼는다
    i = low - 1  # 작은 원소의 끝을 가리키는 인덱스를 초기화한다
    for j in range(low, high):  # 구간 [low, high-1]을 순회한다
        if a[j] <= pivot:  # 현재 원소가 피벗보다 작거나 같으면
            i += 1  # 작은 원소의 끝을 한 칸 앞으로 이동
            a[i], a[j] = a[j], a[i]  # 현재 원소를 앞쪽으로 교환한다
    a[i + 1], a[high] = a[high], a[i + 1]  # 피벗을 최종 위치로 옮긴다
    return i + 1  # 피벗의 인덱스를 반환한다
