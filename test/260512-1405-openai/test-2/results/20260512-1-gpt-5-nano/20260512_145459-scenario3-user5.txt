# 아래는 주석이 달린 파이썬 QuickSort 구현 예시입니다.  # 참고용 주석
def quicksort(arr, low, high):  # QuickSort 메인 함수: 배열과 정렬할 인덱스 범위를 받음
    if low < high:  # 범위에 최소 두 원소가 있는지 확인
        pi = partition(arr, low, high)  # 피벗 위치를 구하기 위한 파티션 수행
        quicksort(arr, low, pi - 1)  # 피벗 왼쪽 부분 재귀적으로 정렬
        quicksort(arr, pi + 1, high)  # 피벗 오른쪽 부분 재귀적으로 정렬

def partition(arr, low, high):  # 배열을 피벗 기준으로 분할하는 함수
    pivot = arr[high]  # 피벗으로 사용할 가장 오른쪽 원소 선택
    i = low - 1  # i는 피벗 위치의 왼쪽에 있어야 할 인덱스 초기화
    for j in range(low, high):  # low부터 high-1까지 순회
        if arr[j] <= pivot:  # 현재 원소가 피벗보다 작거나 같으면
            i += 1  # i를 한 칸 앞으로 이동
            arr[i], arr[j] = arr[j], arr[i]  # 현재 원소와 i 위치의 원소를 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 최종 위치로 이동
    return i + 1  # 피벗의 최종 인덱스 반환

if __name__ == "__main__":  # 이 스크립트가 직접 실행될 때만 아래 코드 실행
    data = [33, 10, 55, 71, 29, 3, 18, 92, 5]  # 테스트 데이터 준비
    quicksort(data, 0, len(data) - 1)  # QuickSort 실행
    print(data)  # 정렬된 결과 출력
