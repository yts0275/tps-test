def quicksort(arr, low, high):  # 구간 [low, high]를 퀵소트로 정렬
    if low < high:  # 구간에 2개 이상 요소가 있을 때만 진행
        pi = partition(arr, low, high)  # 피벗 위치를 얻고
        quicksort(arr, low, pi - 1)  # 왼쪽 부분 재귀적으로 정렬
        quicksort(arr, pi + 1, high)  # 오른쪽 부분 재귀적으로 정렬
def partition(arr, low, high):  # 파티션 수행
    pivot = arr[high]  # 피봇은 구간의 마지막 원소로 선택
    i = low - 1  # 작은 영역의 끝 인덱스 초기화
    for j in range(low, high):  # low부터 high-1까지 순회
        if arr[j] <= pivot:  # 현재 원소가 피봇보다 작으면
            i += 1  # 작은 영역 크기를 확장
            arr[i], arr[j] = arr[j], arr[i]  # 원소 위치 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피봇을 가운데로 배치
    return i + 1  # 피봇의 최종 인덱스 반환
def quicksort_inplace(arr):  # 배열 전체를 퀵소트로 정렬
    if arr:  # 배열이 비어있지 않으면
        quicksort(arr, 0, len(arr) - 1)  # 전체 구간에 대해 퀵소트 시작
# 사용 예시
data = [9, 3, 7, 1, 5, 2, 6, 4, 8]  # 테스트용 배열
print("원본:", data)  # 원본 출력
quicksort_inplace(data)  # 정렬 수행
print("정렬 결과:", data)  # 정렬 결과 출력
