다음은 각 줄마다 주석을 단 파이썬 퀵소트 구현입니다.

```python
def quicksort(arr):
    # QuickSort 래퍼: 배열을 제자리에서 정렬합니다.
    def _quicksort(a, low, high):
        # 재귀적으로 부분 배열을 정렬합니다.
        if low < high:
            # partition 함수를 이용해 피벗의 최종 위치를 얻습니다.
            p = partition(a, low, high)
            # 피벗 기준 왼쪽 부분 재귀적으로 정렬
            _quicksort(a, low, p - 1)
            # 피벗 기준 오른쪽 부분 재귀적으로 정렬
            _quicksort(a, p + 1, high)

    def partition(a, low, high):
        # 마지막 원소를 피벗으로 선택합니다.
        pivot = a[high]
        # 피벗보다 작아야 하는 요소들의 구간 끝 인덱스
        i = low - 1
        # low부터 high-1까지 순회합니다.
        for j in range(low, high):
            # 현재 원소가 피벗보다 작거나 같은 경우
            if a[j] <= pivot:
                # 작은 원소 구간의 경계를 한 칸 늘립니다.
                i += 1
                # 현재 원소를 작은 쪽으로 이동시킵니다.
                a[i], a[j] = a[j], a[i]
        # 피벗을 중앙으로 이동합니다.
        a[i + 1], a[high] = a[high], a[i + 1]
        # 피벗의 최종 인덱스를 반환합니다.
        return i + 1

    # 배열이 비어있지 않으면 전체 구간에 대해 정렬을 시작합니다.
    if arr:
        _quicksort(arr, 0, len(arr) - 1)

# 예시 사용
arr = [34, 7, 23, 32, 5, 62]  # 정렬할 배열
quicksort(arr)                # 배열을 제자리에서 퀵소트로 정렬
print(arr)                      # 정렬 결과 출력
```
