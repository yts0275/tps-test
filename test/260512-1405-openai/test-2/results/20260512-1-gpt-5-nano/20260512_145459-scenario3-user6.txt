다음은 각 줄에 주석을 다는 파이썬 퀵소트 구현 예시입니다.

```python
def quicksort(arr, low, high):  # 배열의 구간 [low, high]를 퀵소트로 정렬
    if low < high:  # 구간의 길이가 2 이상인지 확인
        p = partition(arr, low, high)  # 피벗 위치를 구하는 함수 호출
        quicksort(arr, low, p - 1)  # 왼쪽 서브배열 재귀적으로 정렬
        quicksort(arr, p + 1, high)  # 오른쪽 서브배열 재귀적으로 정렬

def partition(arr, low, high):  # 피벗으로 배열을 분할하는 함수
    pivot = arr[high]  # 마지막 원소를 피벗으로 선택
    i = low - 1  # 작은 원소 구간의 경계 인덱스 초기화
    for j in range(low, high):  # low부터 high-1까지 순회
        if arr[j] <= pivot:  # 현재 원소가 피벗보다 작거나 같으면
            i += 1  # 작은 영역의 경계를 한 칸 확장
            arr[i], arr[j] = arr[j], arr[i]  # 두 원소를 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 올바른 위치로 이동
    return i + 1  # 피벗의 최종 위치 반환

if __name__ == "__main__":  # 스크립트 진입점
    arr = [3, 6, 8, 2, 10, 1, 4]  # 예시 배열
    quicksort(arr, 0, len(arr) - 1)  # 전체 배열에 대해 퀵소트 실행
    print(arr)  # 정렬 결과 출력
```
