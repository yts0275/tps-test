다음은 각 줄에 주석을 달아 작성한 파이썬 기반의 퀵소트 구현 예시입니다.

def quicksort(arr):
    # 주어진 배열을 제자리에서 퀵소트로 정렬한다.
    def _quicksort(a, low, high):
        # 재귀적으로 부분 배열을 정렬한다.
        if low < high:
            # 분할 지점을 얻는다.
            p = _partition(a, low, high)
            # 피벗 왼쪽 부분 배열 재귀적으로 정렬
            _quicksort(a, low, p - 1)
            # 피벗 오른쪽 부분 배열 재귀적으로 정렬
            _quicksort(a, p + 1, high)

    def _partition(a, low, high):
        # 피벗으로 배열의 마지막 원소를 선택한다.
        pivot = a[high]
        # 피봇보다 작거나 같은 원소의 구역 경계 인덱스
        i = low - 1
        # low부터 high-1까지 순회하면서
        for j in range(low, high):
            # 현재 원소가 피봇보다 작거나 같으면
            if a[j] <= pivot:
                # 구역 경계를 한 칸 확장하고 원소를 교환
                i += 1
                a[i], a[j] = a[j], a[i]
        # 피봇을 최종 위치로 옮긴다.
        a[i + 1], a[high] = a[high], a[i + 1]
        # 피봇의 최종 인덱스를 반환한다.
        return i + 1

    # 배열 전체를 재귀적으로 정렬한다.
    _quicksort(arr, 0, len(arr) - 1)

# 아래는 예시 실행용 코드(필요 시 사용)
if __name__ == "__main__":
    # 테스트 배열을 만든다.
    arr = [3, 6, 8, 10, 1, 2, 1]
    # 배열을 퀵소트로 정렬한다.
    quicksort(arr)
    # 정렬된 배열을 출력한다.
    print(arr)
