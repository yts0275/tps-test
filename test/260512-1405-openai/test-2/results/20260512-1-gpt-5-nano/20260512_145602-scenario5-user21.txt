다음은 파이썬으로 작성한 퀵소트 예제입니다. 각 줄마다 주석을 달았습니다.

```python
def quicksort(arr):  # 입력 리스트를 받아 정렬된 새로운 리스트를 반환하는 함수
    if len(arr) <= 1:  # 배열의 길이가 1 이하이면 이미 정렬된 상태
        return arr  # 입력 배열을 그대로 반환
    pivot = arr[len(arr) // 2]  # 피벗으로 배열의 중간 값 선택
    left = [x for x in arr if x < pivot]  # 피벗보다 작은 원소들로 구성된 리스트 생성
    middle = [x for x in arr if x == pivot]  # 피벗과 같은 원소들로 구성된 리스트 생성
    right = [x for x in arr if x > pivot]  # 피벗보다 큰 원소들로 구성된 리스트 생성
    return quicksort(left) + middle + quicksort(right)  # 좌우 부분 배열 재귀 정렬 후 합쳐 반환

if __name__ == "__main__":  # 모듈로 임포트되지 않고 실행될 때만 동작
    data = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 샘플 데이터
    print("Original:", data)  # 원래 리스트를 출력
    sorted_data = quicksort(data)  # 퀵소트 실행
    print("Sorted:", sorted_data)  # 정렬 결과를 출력
```
