# 퀵소트(퀵소트) 구현 - 파이썬 예시
def _partition(arr, lo, hi):  # 피봇으로 마지막 요소를 선택하는 Lomuto 파티션
    pivot = arr[hi]  # 피봇 피벗으로 hi 인덱스의 요소를 선택
    i = lo - 1  # 작게 구성된 부분 배열의 끝 인덱스
    for j in range(lo, hi):  # lo부터 hi-1까지 순회하면서
        if arr[j] <= pivot:  # 현재 원소가 피봇보다 작거나 같으면
            i += 1  # 작은 원소의 구분점을 한 칸 옮김
            arr[i], arr[j] = arr[j], arr[i]  # 두 원소를 서로 교환
    arr[i+1], arr[hi] = arr[hi], arr[i+1]  # 피봇을 올바른 위치로 이동
    return i + 1  # 피봇의 최종 위치 인덱스 반환
def quicksort(arr, low=None, high=None):  # in-place 퀵소트 구현
    if low is None:  # 시작 인덱스가 주어지지 않았다면
        low = 0  # 배열의 시작으로 초기화
    if high is None:  # 끝 인덱스가 주어지지 않았다면
        high = len(arr) - 1  # 배열의 마지막 인덱스로 초기화
    if low < high:  # 구간에 두 원소 이상이 존재하면
        pi = _partition(arr, low, high)  # 파티션을 수행하고 피봇 위치를 얻음
        quicksort(arr, low, pi - 1)  # 피봇 왼쪽 부분 배열 재귀적으로 정렬
        quicksort(arr, pi + 1, high)  # 피봇 오른쪽 부분 배열 재귀적으로 정렬
lst = [3, 6, 8, 10, 1, 2, 1]  # 테스트를 위한 입력 리스트
quicksort(lst)  # 리스트를 제자리에서 정렬
print(lst)  # 정렬 결과 출력
