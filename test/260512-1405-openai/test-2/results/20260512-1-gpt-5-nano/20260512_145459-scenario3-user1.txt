아래는 파이썬으로 작성한 퀵소트 구현입니다. 각 줄에 주석을 달아 두었습니다.

# QuickSort 구현 (배열을 제자리에서 정렬) - 각 줄에 주석 달기
def partition(arr, low, high):  # 파티션 함수: 피벗을 기준으로 두 부분으로 나눔
    pivot = arr[high]  # 피벗으로 배열의 마지막 원소 선택
    i = low - 1  # 작은 요소의 구분 인덱스 초기화
    for j in range(low, high):  # low에서 high-1까지 순회
        if arr[j] <= pivot:  # 현재 원소가 피벗보다 작거나 같으면
            i += 1  # 작은 요소의 구분 인덱스 증가
            arr[i], arr[j] = arr[j], arr[i]  # 현재 원소를 앞쪽으로 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 올바른 위치로 이동
    return i + 1  # 피벗의 최종 인덱스 반환

def quicksort(arr, low, high):  # 배열의 구간 [low, high]를 퀵소트
    if low < high:  # 구간에 두 원소 이상이 있으면
        pi = partition(arr, low, high)  # 파티션 수행 후 피벗 인덱스 얻기
        quicksort(arr, low, pi - 1)  # 피벗 왼쪽 부분 배열 재귀 정렬
        quicksort(arr, pi + 1, high)  # 피벗 오른쪽 부분 배열 재귀 정렬

def quicksort_arr(arr):  # 사용하기 쉬운 래퍼 함수
    if arr is None or len(arr) <= 1:  # 배열이 비었거나 길이가 1 이하면 이미 정렬
        return arr  # 입력 배열을 그대로 반환
    quicksort(arr, 0, len(arr) - 1)  # 전체 배열에 대해 퀵소트 수행
    return arr  # 정렬된 배열 반환

# 예시 사용
lst = [3, 6, 8, 2, 10, 1, 4]  # 정렬할 리스트
quicksort_arr(lst)  # 리스트를 오름차순으로 정렬
print(lst)  # 정렬 결과 출력
