다음은 파이썬으로 구현한 퀵소트이며, 각 줄에 주석을 달았습니다.

```python
def quicksort(arr):  # 퀵소트: 피봇 기반으로 배열을 재귀적으로 정렬하는 함수
    if len(arr) <= 1:  # 배열의 길이가 1 이하이면 이미 정렬된 상태
        return arr  # 입력 배열을 그대로 반환
    pivot = arr[len(arr) // 2]  # 피봇으로 배열의 중간 원소를 선택
    left = [x for x in arr if x < pivot]  # 피봇보다 작은 원소들로 왼쪽 부분 배열 생성
    middle = [x for x in arr if x == pivot]  # 피봇과 같은 원소들로 가운데 부분 배열 생성
    right = [x for x in arr if x > pivot]  # 피봇보다 큰 원소들로 오른쪽 부분 배열 생성
    return quicksort(left) + middle + quicksort(right)  # 왼쪽 정렬 + 피봇들 + 오른쪽 정렬로 병합

arr = [3, 6, 8, 10, 1, 2, 1]  # 테스트용 입력 배열 선언
print(quicksort(arr))  # 정렬된 결과 출력
```
