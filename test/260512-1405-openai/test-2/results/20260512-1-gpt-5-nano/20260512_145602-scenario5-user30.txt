다음은 파이썬으로 구현한 퀵소트(QuickSort) 예시입니다. 각 줄마다 한국어 주석을 달았습니다.

```python
# 퀵소트(QuickSort) - 제자리에서 정렬하는 구현
def quicksort(arr, low, high):  # 배열의 특정 구간을 퀵소트로 정렬하는 재귀 함수
    if low < high:  # 구간에 두 개 이상의 원소가 있으면만 진행
        pi = partition(arr, low, high)  # 파티션 함수를 통해 피벗 인덱스 얻기
        quicksort(arr, low, pi - 1)  # 피벗 왼쪽 부분 배열 재귀적으로 정렬
        quicksort(arr, pi + 1, high)  # 피벗 오른쪽 부분 배열 재귀적으로 정렬

def partition(arr, low, high):  # 파티션 함수: 피벗을 기준으로 배열 분할
    pivot = arr[high]  # 마지막 원소를 피벗으로 선택
    i = low - 1  # 작은 원소의 경계 인덱스 초기화
    for j in range(low, high):  # low부터 high-1까지 순회
        if arr[j] <= pivot:  # 현재 원소가 피벗보다 작거나 같으면
            i += 1  # 작은 원소 경계 한 칸 앞으로 이동
            arr[i], arr[j] = arr[j], arr[i]  # 원소 위치 교환으로 정렬 유지
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 최종 위치에 배치
    return i + 1  # 피벗의 최종 인덱스 반환

# 사용 예시
arr = [10, 7, 8, 9, 1, 5]  # 정렬할 배열 예시
quicksort(arr, 0, len(arr) - 1)  # 배열 전체를 퀵소트로 정렬 실행
print(arr)  # [1, 5, 7, 8, 9, 10] 형태로 정렬된 결과 출력
```
