def quicksort(arr):  # QuickSort wrapper: 배열을 제자리에서 오름차순 정렬
    """배열을 제자리에서 빠르게 정렬합니다."""  # 간단한 설명
    def _quicksort(a, low, high):  # 재귀적 정렬을 담당하는 내부 함수
        if low < high:  # 범위에 두 원소 이상이 있을 때만 수행
            pi = partition(a, low, high)  # 피벗 위치 계산
            _quicksort(a, low, pi - 1)  # 피벗 왼쪽 부분 재귀 정렬
            _quicksort(a, pi + 1, high)  # 피벗 오른쪽 부분 재귀 정렬

    def partition(a, low, high):  # Lomuto 파티션 구현
        pivot = a[high]  # 피벗으로 배열의 마지막 원소 선택
        i = low - 1  # 작은 원소 구간의 끝 인덱스
        for j in range(low, high):  # low ~ high-1까지 순회
            if a[j] <= pivot:  # 현재 원소가 피벗보다 작으면
                i += 1  # 작은 원소 구간 확장
                a[i], a[j] = a[j], a[i]  # 원소 위치 교환
        a[i + 1], a[high] = a[high], a[i + 1]  # 피벗을 가운데로 이동
        return i + 1  # 피벗의 최종 인덱스 반환

    _quicksort(arr, 0, len(arr) - 1)  # 배열 전체 범위에 대해 정렬 시작

if __name__ == "__main__":  # 스크립트로 실행될 때만 아래 예시를 실행
    data = [7, 2, 5, 3, 9, 1, 4]  # 테스트용 정렬 데이터
    quicksort(data)  # 배열을 제자리에서 정렬 실행
    print(data)  # 정렬 결과 출력
