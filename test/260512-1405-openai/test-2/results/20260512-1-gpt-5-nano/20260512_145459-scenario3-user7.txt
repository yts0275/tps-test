다음은 파이썬으로 작성한 in-place 퀵소트 예시입니다. 각 줄마다 한국어 주석을 달았습니다.

```python
def quicksort(arr, low, high):  # 배열과 구간의 시작 인덱스, 끝 인덱스를 받는 함수
    if low < high:  # 구간에 원소가 둘 이상 있는지 확인
        pi = partition(arr, low, high)  # 파티션 인덱스를 구함
        quicksort(arr, low, pi - 1)  # 좌측 부분 배열 재귀적으로 정렬
        quicksort(arr, pi + 1, high)  # 우측 부분 배열 재귀적으로 정렬

def partition(arr, low, high):  # Lomuto 파티션 스킴 구현 함수
    pivot = arr[high]  # 피봇으로 구간의 마지막 원소를 선택
    i = low - 1  # 피봇보다 작아야 하는 배열 끝의 인덱스를 초기화
    for j in range(low, high):  # 구간의 나머지 원소들을 순회
        if arr[j] <= pivot:  # 현재 원소가 피봇보다 작거나 같으면
            i += 1  # 작아진 영역의 끝을 한 칸 확장
            arr[i], arr[j] = arr[j], arr[i]  # 작은 원소를 앞쪽으로 이동시키기 위해 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피봇을 올바른 위치로 이동
    return i + 1  # 피봇의 최종 인덱스를 반환

arr = [10, 7, 8, 9, 1, 5]  # 정렬할 배열
quicksort(arr, 0, len(arr) - 1)  # 전체 배열에 대해 퀵소트 실행
print("정렬 결과:", arr)  # 정렬된 배열 출력
```
