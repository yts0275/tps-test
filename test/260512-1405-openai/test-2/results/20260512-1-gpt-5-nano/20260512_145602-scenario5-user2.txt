다음은 파이썬으로 작성한 퀵소트 예시입니다. 각 줄마다 주석이 달려 있습니다.

def quicksort(arr):  # 정렬할 배열을 받아 새 배열로 반환
    if len(arr) <= 1:  # 기저 조건: 길이가 0 또는 1이면 이미 정렬됨
        return arr  # 입력 배열을 그대로 반환
    pivot = arr[len(arr) // 2]  # 피벗으로 배열의 중간 값을 선택
    left = [x for x in arr if x < pivot]  # 피벗보다 작은 원소들을 모아 왼쪽 리스트 생성
    middle = [x for x in arr if x == pivot]  # 피벗과 같은 원소들을 모아 가운데 리스트 생성
    right = [x for x in arr if x > pivot]  # 피벗보다 큰 원소들을 모아 오른쪽 리스트 생성
    return quicksort(left) + middle + quicksort(right)  # 왼쪽 재귀 정렬 + 피벗들 + 오른쪽 재귀 정렬 결합

if __name__ == "__main__":  # 스크립트가 직접 실행될 때의 진입점
    data = [3, 6, 8, 10, 1, 2, 1]  # 테스트 데이터
    print("원본:", data)  # 원본 배열 출력
    sorted_data = quicksort(data)  # 퀵소트 수행
    print("정렬 결과:", sorted_data)  # 정렬된 배열 출력
