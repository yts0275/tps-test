다음은 파이썬으로 구현한 퀵소트 예시입니다. 각 줄에 주석을 달았습니다.

# 퀵소트 구현(파이썬) - 배열을 제자리에서 정렬하는 버전
def quicksort(arr, low, high):  # 배열, 왼쪽 인덱스, 오른쪽 인덱스
    if low < high:  # 구간의 길이가 2 이상일 때에만 정렬 진행
        pi = partition(arr, low, high)  # 파티션 함수로 피벗 인덱스 얻기
        quicksort(arr, low, pi - 1)  # 피벗 왼쪽 부분 재귀적으로 정렬
        quicksort(arr, pi + 1, high)  # 피벗 오른쪽 부분 재귀적으로 정렬

def partition(arr, low, high):  # 파티션 함수: 피벗을 기준으로 배열 분할
    pivot = arr[high]  # 가장 오른쪽 원소를 피벗으로 선택
    i = low - 1  # i는 피벗보다 작아지는 구간의 끝 인덱스
    for j in range(low, high):  # low부터 high-1까지 순회
        if arr[j] <= pivot:  # 현재 원소가 피벗보다 작거나 같으면
            i += 1  # 작은 원소의 구간 확장
            arr[i], arr[j] = arr[j], arr[i]  # 위치 교환으로 앞쪽으로 모음
    arr[i+1], arr[high] = arr[high], arr[i+1]  # 피벗을 중간 위치로 이동
    return i + 1  # 피벗의 최종 인덱스 반환

# 사용 예
data = [12, 7, 14, 27, 3, 9, 11]  # 정렬할 배열 예시
print("원본:", data)  # 원본 배열 출력
quicksort(data, 0, len(data) - 1)  # 배열 전체를 정렬
print("퀵소트 결과:", data)  # 정렬된 배열 출력
