다음은 파이썬으로 작성한 인플레이스 퀵소트입니다. 각 줄에 주석을 달았습니다.

```python
def quicksort(arr, low=0, high=None):  # 퀵소트 메인 함수: 배열과 좌우 인덱스 범위를 받음
    if high is None:  # high를 배열의 마지막 인덱스로 초기화
        high = len(arr) - 1
    if low < high:  # 정렬할 구간이 존재하는지 확인
        pi = partition(arr, low, high)  # 피벗 인덱스 얻기
        quicksort(arr, low, pi - 1)  # 피벗 기준 왼쪽 부분 재귀 정렬
        quicksort(arr, pi + 1, high)  # 피벗 기준 오른쪽 부분 재귀 정렬
    return arr  # 정렬된 배열 반환

def partition(arr, low, high):  # Lomuto 분할 함수
    pivot = arr[high]  # 피벗으로 배열의 마지막 원소 선택
    i = low - 1  # 구분자 인덱스 초기화
    for j in range(low, high):  # low부터 high-1까지 순회
        if arr[j] <= pivot:  # 현재 원소가 피벗보다 작거나 같으면
            i += 1  # 구분자 증가
            arr[i], arr[j] = arr[j], arr[i]  # 원소들 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 올바른 위치로 이동
    return i + 1  # 피벗의 최종 위치 반환

if __name__ == "__main__":  # 스크립트로 실행될 때만 아래 예시 실행
    data = [3, 6, 8, 10, 1, 2, 1]  # 예시 데이터
    quicksort(data)  # 데이터 정렬 실행
    print(data)  # 정렬 결과 출력
```
