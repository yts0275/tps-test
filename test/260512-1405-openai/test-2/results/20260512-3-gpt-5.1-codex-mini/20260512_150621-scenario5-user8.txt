```python
def quicksort(arr):  # 주어진 리스트를 퀵소트로 정렬하기 위한 함수 정의
    if len(arr) <= 1:  # 리스트 길이가 1 이하이면 이미 정렬된 상태이므로
        return arr  # 그대로 리스트 반환
    pivot = arr[len(arr) // 2]  # 피벗을 리스트 중앙 값으로 선택
    left = [x for x in arr if x < pivot]  # 피벗보다 작은 값들을 왼쪽 리스트에 수집
    middle = [x for x in arr if x == pivot]  # 피벗과 같은 값을 중간 리스트에 수집
    right = [x for x in arr if x > pivot]  # 피벗보다 큰 값들을 오른쪽 리스트에 수집
    return quicksort(left) + middle + quicksort(right)  # 왼쪽, 중간, 오른쪽을 재귀적으로 합쳐 반환
```
