```python
def quicksort(arr):  # 퀵소트를 수행하는 함수를 정의합니다.
    if len(arr) <= 1:  # 배열 길이가 1 이하이면 이미 정렬된 상태입니다.
        return arr  # 그대로 배열을 반환합니다.
    pivot = arr[len(arr) // 2]  # 배열의 가운데 값을 피벗으로 선택합니다.
    left = [x for x in arr if x < pivot]  # 피벗보다 작은 요소들을 모읍니다.
    middle = [x for x in arr if x == pivot]  # 피벗과 같은 요소들을 모읍니다.
    right = [x for x in arr if x > pivot]  # 피벗보다 큰 요소들을 모읍니다.
    return quicksort(left) + middle + quicksort(right)  # 왼쪽, 가운데, 오른쪽을 순차적으로 정렬해서 합칩니다.
```
