```python
def quicksort(arr):  # 리스트를 입력받아 퀵소트를 수행하는 함수를 정의합니다.
    if len(arr) <= 1:  # 리스트의 길이가 1 이하이면 이미 정렬된 상태이므로 그대로 반환합니다.
        return arr  # 조건에 해당하면 입력된 리스트를 그대로 반환합니다.
    pivot = arr[len(arr) // 2]  # 리스트 가운데 값을 피벗으로 선택합니다.
    left = [x for x in arr if x < pivot]  # 피벗보다 작은 값들로 구성된 왼쪽 부분 리스트를 만듭니다.
    middle = [x for x in arr if x == pivot]  # 피벗과 같은 값들을 모아 가운데 부분 리스트를 만듭니다.
    right = [x for x in arr if x > pivot]  # 피벗보다 큰 값들로 구성된 오른쪽 부분 리스트를 만듭니다.
    return quicksort(left) + middle + quicksort(right)  # 왼쪽과 오른쪽 부분 리스트를 재귀적으로 정렬한 뒤 결합하여 반환합니다.
```
