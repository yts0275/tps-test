def quick_sort(arr):  # 퀵소트 알고리즘을 수행할 함수를 정의한다.
    if len(arr) <= 1:  # 배열의 길이가 1 이하이면 이미 정렬된 상태이므로 그대로 반환한다.
        return arr  # 정렬된 배열을 반환한다.
    pivot = arr[len(arr) // 2]  # 피벗으로 배열의 중간 요소를 선택한다.
    left = [x for x in arr if x < pivot]  # 피벗보다 작은 요소들로 구성된 왼쪽 부분 배열을 만든다.
    middle = [x for x in arr if x == pivot]  # 피벗과 같은 요소들을 모은 중간 부분 배열을 만든다.
    right = [x for x in arr if x > pivot]  # 피벗보다 큰 요소들로 구성된 오른쪽 부분 배열을 만든다.
    return quick_sort(left) + middle + quick_sort(right)  # 왼쪽과 오른쪽 부분 배열에 재귀적으로 퀵소트를 적용한 뒤 연결하여 반환한다.
