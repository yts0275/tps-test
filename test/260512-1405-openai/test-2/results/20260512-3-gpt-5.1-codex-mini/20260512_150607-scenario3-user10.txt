다음은 각 줄에 주석을 단 퀵소트 구현 예제입니다:

```python
def quick_sort(arr):  # 퀵소트 함수 정의
    if len(arr) <= 1:  # 배열 길이가 1 이하이면 이미 정렬된 상태이므로 그대로 반환
        return arr  # 기존 배열 반환
    pivot = arr[len(arr) // 2]  # 피벗을 배열의 중앙값으로 설정
    left = [x for x in arr if x < pivot]  # 피벗보다 작은 요소들을 왼쪽 리스트에 담음
    middle = [x for x in arr if x == pivot]  # 피벗과 같은 요소들을 가운데 리스트에 담음
    right = [x for x in arr if x > pivot]  # 피벗보다 큰 요소들을 오른쪽 리스트에 담음
    return quick_sort(left) + middle + quick_sort(right)  # 재귀적으로 정렬된 결과를 합쳐 반환함
```
