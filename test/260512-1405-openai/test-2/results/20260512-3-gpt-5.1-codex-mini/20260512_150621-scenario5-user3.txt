```python
def quicksort(arr):  # 배열을 입력받아 퀵소트를 수행하는 함수 정의
    if len(arr) <= 1:  # 배열 길이가 0이나 1이면 이미 정렬된 상태이므로
        return arr  # 그대로 반환하여 재귀 종료
    pivot = arr[len(arr) // 2]  # 피벗은 배열의 중간 요소로 선정
    left = [x for x in arr if x < pivot]  # 피벗보다 작은 요소들을 왼쪽에 모음
    middle = [x for x in arr if x == pivot]  # 피벗과 같은 요소들을 중앙에 모음
    right = [x for x in arr if x > pivot]  # 피벗보다 큰 요소들을 오른쪽에 모음
    return quicksort(left) + middle + quicksort(right)  # 왼쪽과 오른쪽을 재귀적으로 정렬한 후 합침
```
