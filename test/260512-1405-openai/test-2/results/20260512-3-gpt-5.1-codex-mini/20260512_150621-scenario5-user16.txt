```python
def quicksort(arr):  # 퀵소트를 수행하는 함수를 정의합니다
    if len(arr) <= 1:  # 배열 길이가 1 이하이면 더 이상 분할할 필요가 없음을 확인합니다
        return arr  # 이미 정렬된 배열을 그대로 반환합니다
    pivot = arr[len(arr) // 2]  # 중앙 위치의 값을 피벗으로 설정합니다
    less = [x for x in arr if x < pivot]  # 피벗보다 작은 원소들을 모아 새로운 리스트를 생성합니다
    equal = [x for x in arr if x == pivot]  # 피벗과 값이 같은 원소들을 모아서 평준화를 위해 사용합니다
    greater = [x for x in arr if x > pivot]  # 피벗보다 큰 원소들을 따로 모아 새로운 리스트를 생성합니다
    return quicksort(less) + equal + quicksort(greater)  # 재귀적으로 정렬된 부분들을 결합하여 최종 정렬 결과를 반환합니다
```
