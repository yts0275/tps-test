```python
def quicksort(arr):  # 배열을 인자로 받아 퀵소트를 수행하는 함수 정의
    if len(arr) <= 1:  # 배열 길이가 1 이하이면 이미 정렬된 상태이므로
        return arr  # 입력 배열 그대로 반환
    pivot = arr[len(arr) // 2]  # pivot을 배열 중간 값으로 결정
    left = [x for x in arr if x < pivot]  # pivot보다 작은 원소들을 왼쪽 그룹에 모음
    middle = [x for x in arr if x == pivot]  # pivot과 같은 값들을 중간 그룹에 모음
    right = [x for x in arr if x > pivot]  # pivot보다 큰 원소들을 오른쪽 그룹에 모음
    return quicksort(left) + middle + quicksort(right)  # 재귀적으로 왼쪽과 오른쪽 정렬 후 합쳐서 반환
```
