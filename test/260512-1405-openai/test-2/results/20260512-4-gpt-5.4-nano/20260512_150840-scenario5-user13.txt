아래는 **퀵소트(Quick Sort)**를 파이썬으로 구현한 예시이며, **각 줄마다 주석**을 달았습니다.

```python
def quicksort(arr):  # 퀵소트 함수 정의(입력: arr)
    if len(arr) <= 1:  # 원소가 0개 또는 1개면 이미 정렬된 상태
        return arr  # 그대로 반환

    pivot = arr[0]  # 피벗을 첫 번째 원소로 선택
    less = []  # 피벗보다 작은 값들을 담을 리스트
    equal = []  # 피벗과 같은 값들을 담을 리스트
    greater = []  # 피벗보다 큰 값들을 담을 리스트

    for x in arr:  # 배열의 각 원소를 순회
        if x < pivot:  # 현재 원소가 피벗보다 작으면
            less.append(x)  # less 리스트에 추가
        elif x > pivot:  # 현재 원소가 피벗보다 크면
            greater.append(x)  # greater 리스트에 추가
        else:  # 현재 원소가 피벗과 같으면
            equal.append(x)  # equal 리스트에 추가

    return quicksort(less) + equal + quicksort(greater)  # 재귀적으로 정렬 후 합쳐서 반환


# 테스트용 예시
data = [3, 6, 1, 8, 2, 5, 7, 4]  # 정렬할 데이터 리스트
sorted_data = quicksort(data)  # 퀵소트 실행하여 정렬된 결과 저장
print(sorted_data)  # 정렬 결과 출력
```
