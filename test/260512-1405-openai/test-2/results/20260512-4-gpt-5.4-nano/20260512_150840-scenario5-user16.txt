아래는 **퀵소트(Quick Sort)** 구현 예시이며, **각 줄마다 주석**을 달았습니다. (파이썬)

```python
def quicksort(arr):  # 퀵소트 함수를 정의한다.
    if len(arr) <= 1:  # 배열의 길이가 0 또는 1이면 이미 정렬된 상태이므로
        return arr  # 배열을 그대로 반환한다.
    
    pivot = arr[len(arr) // 2]  # 가운데 원소를 피벗(pivot)으로 선택한다.
    left = []  # 피벗보다 작은 값들을 담을 리스트를 만든다.
    middle = []  # 피벗과 같은 값들을 담을 리스트를 만든다.
    right = []  # 피벗보다 큰 값들을 담을 리스트를 만든다.
    
    for x in arr:  # 배열의 모든 원소 x에 대해 반복한다.
        if x < pivot:  # x가 피벗보다 작으면
            left.append(x)  # left에 x를 추가한다.
        elif x == pivot:  # x가 피벗과 같으면
            middle.append(x)  # middle에 x를 추가한다.
        else:  # x가 피벗보다 크면
            right.append(x)  # right에 x를 추가한다.
    
    return quicksort(left) + middle + quicksort(right)  # 왼쪽 정렬 결과 + 가운데 + 오른쪽 정렬 결과를 합쳐 반환한다.


# 예시 실행 코드(원하면 삭제해도 됨)
data = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 예제 리스트를 만든다.
sorted_data = quicksort(data)  # 퀵소트를 실행해 정렬 결과를 얻는다.
print(sorted_data)  # 정렬된 결과를 출력한다.
```
