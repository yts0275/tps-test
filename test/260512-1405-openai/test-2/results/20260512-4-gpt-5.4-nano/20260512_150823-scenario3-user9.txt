아래는 **퀵소트(Quick Sort) 구현 예시**입니다. 요청하신 대로 **각 줄마다 주석**을 달았습니다. (파이썬)

```python
def quicksort(arr):  # 퀵소트를 수행할 함수를 정의함
    if len(arr) <= 1:  # 배열 길이가 0 또는 1이면 이미 정렬된 상태이므로
        return arr  # 그대로 반환함
    pivot = arr[len(arr) // 2]  # 가운데 원소를 피벗으로 선택함
    left = []  # 피벗보다 작은 값들을 담을 리스트를 생성함
    middle = []  # 피벗과 같은 값들을 담을 리스트를 생성함
    right = []  # 피벗보다 큰 값들을 담을 리스트를 생성함
    
    for x in arr:  # 배열의 모든 원소를 순회함
        if x < pivot:  # 현재 값이 피벗보다 작으면
            left.append(x)  # left에 추가함
        elif x == pivot:  # 현재 값이 피벗과 같으면
            middle.append(x)  # middle에 추가함
        else:  # 현재 값이 피벗보다 크면
            right.append(x)  # right에 추가함
    
    return quicksort(left) + middle + quicksort(right)  # 정렬된 left + middle + 정렬된 right를 합쳐 반환함


# 사용 예시(테스트 코드)
data = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 입력 배열을 만듦
sorted_data = quicksort(data)  # 퀵소트를 실행하여 정렬 결과를 얻음
print(sorted_data)  # 정렬된 배열을 출력함
```
