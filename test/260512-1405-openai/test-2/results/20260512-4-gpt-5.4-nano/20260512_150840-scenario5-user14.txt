아래는 **퀵소트(Quick Sort)** 구현 예시이며, **각 줄마다 주석**을 달아 드렸습니다. (파이썬)

```python
def quick_sort(arr):  # 퀵소트 함수를 정의합니다. (입력: arr)
    if len(arr) <= 1:  # 원소가 0개 또는 1개이면 이미 정렬된 상태이므로
        return arr  # 그대로 반환합니다.
    
    pivot = arr[len(arr) // 2]  # 가운데 원소를 피벗(pivot)으로 선택합니다.
    left = []  # 피벗보다 작은 원소들을 담을 리스트를 만듭니다.
    right = []  # 피벗보다 큰 원소들을 담을 리스트를 만듭니다.
    middle = []  # 피벗과 같은 원소들을 담을 리스트를 만듭니다. (중복 처리용)

    for x in arr:  # 배열의 각 원소 x에 대해 반복합니다.
        if x < pivot:  # x가 피벗보다 작으면
            left.append(x)  # left에 x를 추가합니다.
        elif x > pivot:  # x가 피벗보다 크면
            right.append(x)  # right에 x를 추가합니다.
        else:  # x가 피벗과 같으면
            middle.append(x)  # middle에 x를 추가합니다.

    return quick_sort(left) + middle + quick_sort(right)  # 정렬된 left + pivot들(middle) + 정렬된 right를 합쳐 반환합니다.


# 테스트용 예시입니다.
data = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 입력 배열을 만듭니다.
sorted_data = quick_sort(data)  # 퀵소트를 호출하여 정렬을 수행합니다.
print(sorted_data)  # 정렬 결과를 출력합니다.
```
