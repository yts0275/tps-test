아래는 **퀵소트(Quick Sort) 구현 예시**이며, **각 줄마다 주석**을 달았습니다. (파이썬 예시)

```python
def quicksort(arr):  # 퀵소트 함수를 정의합니다. (입력: 리스트 arr)
    if len(arr) <= 1:  # 원소가 0개 또는 1개이면 이미 정렬된 상태이므로 그대로 반환합니다.
        return arr  # 정렬된 배열을 반환합니다.

    pivot = arr[len(arr) // 2]  # 가운데 원소를 피벗(pivot)으로 선택합니다.
    left = []  # 피벗보다 작은 값들을 담을 리스트를 만듭니다.
    middle = []  # 피벗과 같은 값들을 담을 리스트를 만듭니다.
    right = []  # 피벗보다 큰 값들을 담을 리스트를 만듭니다.

    for x in arr:  # 입력 배열의 각 원소 x에 대해 반복합니다.
        if x < pivot:  # x가 피벗보다 작으면
            left.append(x)  # left 리스트에 x를 추가합니다.
        elif x == pivot:  # x가 피벗과 같으면
            middle.append(x)  # middle 리스트에 x를 추가합니다.
        else:  # x가 피벗보다 크면
            right.append(x)  # right 리스트에 x를 추가합니다.

    return quicksort(left) + middle + quicksort(right)  # 왼쪽 정렬 + 피벗(동일값) + 오른쪽 정렬 결과를 합쳐 반환합니다.


# 예시 실행
data = [3, 7, 2, 5, 1, 4, 6]  # 정렬할 데이터 리스트를 만듭니다.
sorted_data = quicksort(data)  # 퀵소트를 호출해서 정렬된 결과를 얻습니다.
print(sorted_data)  # 정렬된 결과를 출력합니다.
```
