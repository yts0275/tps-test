아래는 **퀵소트(Quicksort) 구현 예시**와 **각 줄마다 주석**을 단 코드입니다. (파이썬)

```python
def quicksort(arr):  # 퀵소트 함수를 정의(입력: 리스트 arr)
    if len(arr) <= 1:  # 원소가 0개 또는 1개이면 이미 정렬된 상태이므로
        return arr  # 그대로 반환
    
    pivot = arr[-1]  # 피벗을 마지막 원소로 선택
    left = []  # 피벗보다 작은 값들을 담을 리스트
    right = []  # 피벗보다 크거나 같은 값들을 담을 리스트
    
    for x in arr[:-1]:  # 피벗을 제외한 나머지 원소들을 순회
        if x < pivot:  # 현재 원소 x가 피벗보다 작으면
            left.append(x)  # left 리스트에 추가
        else:  # 아니면(피벗보다 크거나 같으면)
            right.append(x)  # right 리스트에 추가
    
    return quicksort(left) + [pivot] + quicksort(right)  # 재귀적으로 정렬한 결과를 합쳐 반환


# 간단한 테스트
data = [3, 6, 8, 10, 1, 2, 1]  # 예시 입력 리스트
print(quicksort(data))  # 퀵소트 실행 후 정렬 결과 출력
```
