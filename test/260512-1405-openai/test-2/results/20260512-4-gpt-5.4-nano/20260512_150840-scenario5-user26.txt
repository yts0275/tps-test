아래는 **퀵소트(Quick Sort)** 구현 예시입니다. 요청하신 대로 **각 줄마다 주석**을 달았습니다. (파이썬)

```python
def quick_sort(arr, low=0, high=None):  # 퀵소트를 함수로 정의 (arr: 정렬할 배열, low/high: 범위)
    if high is None:  # high가 지정되지 않았다면
        high = len(arr) - 1  # 배열의 마지막 인덱스로 설정
    
    if low >= high:  # 정렬 범위가 0 또는 음수 크기면 (원소 0개/1개)
        return arr  # 이미 정렬된 상태이므로 배열을 그대로 반환
    
    pivot = arr[high]  # 피벗을 마지막 원소로 선택
    i = low - 1  # pivot보다 작은 영역의 끝 인덱스를 가리키는 변수 초기화
    
    for j in range(low, high):  # low부터 high-1까지 탐색
        if arr[j] <= pivot:  # 현재 원소가 피벗보다 작거나 같으면
            i += 1  # 작은 영역을 확장하기 위해 i를 1 증가
            arr[i], arr[j] = arr[j], arr[i]  # 작은 원소들을 앞으로 모이도록 swap
            
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 제자리로 이동
    pivot_index = i + 1  # 피벗이 최종 위치한 인덱스를 기록
    
    quick_sort(arr, low, pivot_index - 1)  # 피벗 왼쪽 부분(작은 값들)을 재귀적으로 정렬
    quick_sort(arr, pivot_index + 1, high)  # 피벗 오른쪽 부분(큰 값들)을 재귀적으로 정렬
    return arr  # 정렬 완료된 배열 반환


# 사용 예시
data = [10, 7, 8, 9, 1, 5]  # 정렬할 데이터
result = quick_sort(data)  # 퀵소트 실행 (제자리 정렬)
print(result)  # 정렬 결과 출력
```
