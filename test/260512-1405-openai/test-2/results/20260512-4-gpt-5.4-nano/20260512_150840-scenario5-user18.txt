아래는 **퀵소트(Quick Sort) 구현 예시**입니다. 요청하신 대로 **각 줄마다 주석**을 달았습니다. (파이썬)

```python
def quicksort(arr):  # 퀵소트 함수를 정의한다. (입력: arr)
    if len(arr) <= 1:  # 원소가 0개 또는 1개면 이미 정렬된 상태이므로 그대로 반환한다.
        return arr  # 정렬된 배열을 그대로 반환한다.

    pivot = arr[len(arr) // 2]  # 가운데 원소를 피벗(pivot)으로 선택한다.
    
    left = [x for x in arr if x < pivot]  # 피벗보다 작은 원소들을 left 리스트로 모은다.
    middle = [x for x in arr if x == pivot]  # 피벗과 같은 원소들을 middle 리스트로 모은다.
    right = [x for x in arr if x > pivot]  # 피벗보다 큰 원소들을 right 리스트로 모은다.

    return quicksort(left) + middle + quicksort(right)  # left를 정렬하고, middle은 그대로 두고, right를 정렬한 뒤 합친다.


# 테스트용 코드 시작
data = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 예시 배열을 만든다.
sorted_data = quicksort(data)  # 퀵소트를 수행해 정렬 결과를 얻는다.
print(sorted_data)  # 정렬된 배열을 출력한다.
```
