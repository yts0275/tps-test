아래는 **퀵소트(Quick Sort)** 구현 예시입니다. **각 줄마다 주석**을 달아 드릴게요.

```python
def quick_sort(arr):  # 퀵소트 함수를 정의합니다.
    if len(arr) <= 1:  # 원소가 0개 또는 1개면 정렬이 필요 없으므로 그대로 반환합니다.
        return arr  # 정렬된 배열(자기 자신)을 반환합니다.

    pivot = arr[len(arr) // 2]  # 가운데 원소를 피벗(pivot)으로 선택합니다.
    
    left = []  # 피벗보다 작은 원소들을 담을 리스트를 만듭니다.
    middle = []  # 피벗과 같은 원소들을 담을 리스트를 만듭니다.
    right = []  # 피벗보다 큰 원소들을 담을 리스트를 만듭니다.

    for x in arr:  # 배열의 각 원소 x를 순회합니다.
        if x < pivot:  # x가 피벗보다 작으면
            left.append(x)  # left 리스트에 x를 추가합니다.
        elif x > pivot:  # x가 피벗보다 크면
            right.append(x)  # right 리스트에 x를 추가합니다.
        else:  # x가 피벗과 같으면
            middle.append(x)  # middle 리스트에 x를 추가합니다.

    return quick_sort(left) + middle + quick_sort(right)  # 왼쪽 정렬 + 피벗과 같은 것 + 오른쪽 정렬 결과를 합칩니다.


# 테스트 코드: 예시 배열을 하나 준비합니다.
data = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 입력 배열입니다.

print(quick_sort(data))  # 퀵소트를 실행한 결과를 출력합니다.
```
