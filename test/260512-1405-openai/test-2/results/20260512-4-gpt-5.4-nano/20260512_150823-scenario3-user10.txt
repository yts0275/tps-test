아래는 **퀵소트(Quicksort) 구현 예시(파이썬)**이고, **코드의 각 줄마다 주석**을 달았습니다.

```python
def quicksort(arr):  # 퀵소트 함수 정의(배열 arr을 정렬해 반환)
    if len(arr) <= 1:  # 원소가 0개 또는 1개면 정렬할 필요가 없으므로 그대로 반환
        return arr  # 그대로 반환

    pivot = arr[len(arr) // 2]  # 피벗을 배열의 가운데 원소로 선택
    left = []  # 피벗보다 작은 원소들을 담을 리스트
    middle = []  # 피벗과 같은 원소들을 담을 리스트
    right = []  # 피벗보다 큰 원소들을 담을 리스트

    for x in arr:  # 배열의 각 원소 x에 대해 반복
        if x < pivot:  # x가 피벗보다 작으면
            left.append(x)  # left 리스트에 x 추가
        elif x > pivot:  # x가 피벗보다 크면
            right.append(x)  # right 리스트에 x 추가
        else:  # x가 피벗과 같으면
            middle.append(x)  # middle 리스트에 x 추가

    return quicksort(left) + middle + quicksort(right)  # 정렬된 left + middle + 정렬된 right 결합하여 반환
                                                            # (재귀적으로 left와 right을 계속 퀵소트)

# 예시 사용
arr = [3, 7, 2, 5, 1, 4, 6]  # 정렬할 원본 배열 예시
print(quicksort(arr))  # 퀵소트 결과를 출력
```
