아래는 **퀵소트(Quick Sort)**를 예시로 구현한 코드입니다. 요청하신 대로 **각 줄마다 주석**을 달았습니다. (파이썬)

```python
def quicksort(arr, left, right):  # 퀵소트 함수를 정의한다 (배열, 시작 인덱스, 끝 인덱스를 받음)
    if left >= right:  # 원소가 0개 또는 1개이면 더 이상 정렬할 필요가 없으므로 종료 조건
        return  # 재귀를 종료한다

    pivot = arr[(left + right) // 2]  # 피벗(pivot)을 가운데 값으로 선택한다
    i = left  # i를 왼쪽 포인터로 초기화한다
    j = right  # j를 오른쪽 포인터로 초기화한다

    while i <= j:  # i가 j보다 크면 파티션 분할이 끝났으므로 반복 종료
        while arr[i] < pivot:  # i 위치의 값이 피벗보다 작으면 i를 오른쪽으로 이동한다
            i += 1  # i를 1 증가시킨다
        while arr[j] > pivot:  # j 위치의 값이 피벗보다 크면 j를 왼쪽으로 이동한다
            j -= 1  # j를 1 감소시킨다

        if i <= j:  # 두 포인터가 교차하지 않았을 때(또는 같을 때) 교환을 수행한다
            arr[i], arr[j] = arr[j], arr[i]  # i와 j의 값을 서로 swap(교환)한다
            i += 1  # 교환 후 i를 1 증가시킨다
            j -= 1  # 교환 후 j를 1 감소시킨다

    quicksort(arr, left, j)  # 피벗보다 작은 구간(left ~ j)을 재귀적으로 퀵소트한다
    quicksort(arr, i, right)  # 피벗보다 큰 구간(i ~ right)을 재귀적으로 퀵소트한다


# --- 테스트 코드(예시) ---

data = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 예시 배열을 만든다
quicksort(data, 0, len(data) - 1)  # 배열 전체를 퀵소트로 정렬한다
print(data)  # 정렬 결과를 출력한다
```
