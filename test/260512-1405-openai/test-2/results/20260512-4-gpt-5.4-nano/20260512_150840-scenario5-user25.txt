아래는 **퀵소트(Quick Sort)** 구현 예시와, **각 줄마다 주석**을 단 코드입니다. (파이썬)

```python
def quicksort(arr, low=0, high=None):  # 퀵정렬 함수 정의: arr에 대해 low~high 구간을 정렬
    if high is None:  # 사용자가 high를 안 주면
        high = len(arr) - 1  # high를 배열의 마지막 인덱스로 설정

    if low >= high:  # 정렬할 원소가 0개 또는 1개면
        return arr  # 더 이상 정렬할 게 없으므로 그대로 반환

    pivot = arr[high]  # 피벗을 구간의 맨 끝 원소로 선택
    i = low - 1  # 피벗보다 작은 값들의 마지막 인덱스를 표시할 변수(i)를 설정

    for j in range(low, high):  # j를 low부터 high-1까지 순회
        if arr[j] <= pivot:  # 현재 값이 피벗 이하이면
            i += 1  # 작은 값 영역을 확장하기 위해 i를 증가
            arr[i], arr[j] = arr[j], arr[i]  # arr[i]와 arr[j]를 교환하여 작은 값을 앞으로 모음

    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 올바른 위치(i+1)에 놓기 위해 교환
    p = i + 1  # 피벗의 최종 위치를 p로 저장

    quicksort(arr, low, p - 1)  # 피벗 왼쪽 부분을 재귀로 정렬
    quicksort(arr, p + 1, high)  # 피벗 오른쪽 부분을 재귀로 정렬

    return arr  # 정렬 완료된 배열 반환


# 사용 예시
data = [10, 7, 8, 9, 1, 5]  # 정렬할 데이터
result = quicksort(data)  # 퀵소트 실행
print(result)  # 정렬 결과 출력
```
