아래는 **퀵소트(Quick Sort) 구현 코드**이며, **각 줄마다 주석**을 달아 설명했습니다. (파이썬 예시)

```python
def quick_sort(arr):  # 퀵소트 함수를 정의한다. 매개변수로 정렬할 리스트 arr을 받는다.
    if len(arr) <= 1:  # 원소가 0개 또는 1개면 이미 정렬되어 있으므로 arr을 그대로 반환한다.
        return arr  # 기본 종료 조건: 길이가 1 이하이면 그대로 반환.

    pivot = arr[len(arr) // 2]  # 피벗(pivot)을 배열의 가운데 값으로 선택한다.
    left = []  # 피벗보다 작은 값들을 담을 리스트를 만든다.
    middle = []  # 피벗과 같은 값들을 담을 리스트를 만든다.
    right = []  # 피벗보다 큰 값들을 담을 리스트를 만든다.

    for x in arr:  # 입력 배열의 각 원소 x를 순회한다.
        if x < pivot:  # x가 피벗보다 작으면
            left.append(x)  # left 리스트에 x를 추가한다.
        elif x > pivot:  # x가 피벗보다 크면
            right.append(x)  # right 리스트에 x를 추가한다.
        else:  # x가 피벗과 같으면
            middle.append(x)  # middle 리스트에 x를 추가한다.

    return quick_sort(left) + middle + quick_sort(right)  # 왼쪽 정렬 + 피벗(동일 값) + 오른쪽 정렬 결과를 합쳐 반환한다.


# 테스트용 실행 코드 시작
data = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 예시 배열을 만든다.
print(quick_sort(data))  # 퀵소트를 실행한 결과를 출력한다.
```
