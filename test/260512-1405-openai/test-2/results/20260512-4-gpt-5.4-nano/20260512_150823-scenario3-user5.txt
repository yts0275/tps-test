아래는 **퀵소트(Quick Sort)** 를 예제로 구현한 코드입니다. **각 줄마다 주석**을 달았습니다. (파이썬 버전)

```python
def quicksort(arr):  # 퀵소트 함수를 정의함 (입력: 배열 arr)
    if len(arr) <= 1:  # 배열 길이가 0 또는 1이면 이미 정렬된 상태이므로
        return arr  # 그대로 반환함

    pivot = arr[len(arr) // 2]  # 피벗(기준값)을 배열의 가운데 값으로 선택함
    left = []  # 피벗보다 작은 값들을 담을 리스트를 준비함
    middle = []  # 피벗과 같은 값들을 담을 리스트를 준비함
    right = []  # 피벗보다 큰 값들을 담을 리스트를 준비함

    for x in arr:  # 배열의 각 원소 x를 순회함
        if x < pivot:  # x가 피벗보다 작으면
            left.append(x)  # left 리스트에 x를 추가함
        elif x > pivot:  # x가 피벗보다 크면
            right.append(x)  # right 리스트에 x를 추가함
        else:  # x가 피벗과 같다면
            middle.append(x)  # middle 리스트에 x를 추가함

    return quicksort(left) + middle + quicksort(right)  # 재귀적으로 정렬한 결과를 이어 붙여 반환함


# 예시 실행을 위한 코드
data = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 입력 배열을 예시로 정의함
sorted_data = quicksort(data)  # 퀵소트를 적용하여 정렬 결과를 얻음
print(sorted_data)  # 정렬된 배열을 출력함
```
