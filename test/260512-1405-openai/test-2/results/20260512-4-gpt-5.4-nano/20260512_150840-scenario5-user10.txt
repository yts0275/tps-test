아래는 **퀵소트(Quick Sort)**를 구현한 예시입니다. 코드 **각 줄마다 주석**을 달았습니다.

```python
def quick_sort(arr, start=0, end=None):                 # 퀵소트 함수를 정의 (배열과 시작/끝 인덱스)
    if end is None:                                      # end를 주지 않았다면(초기 호출이면) 전체 범위를 쓰도록 처리
        end = len(arr) - 1                              # end를 배열의 마지막 인덱스로 설정

    if start >= end:                                     # 정렬할 원소가 0개 또는 1개면 이미 정렬됨
        return arr                                       # 정렬된 배열을 반환

    pivot = arr[end]                                     # 피벗을 end 위치의 값으로 선택
    i = start - 1                                        # 피벗보다 작은 구간의 끝 인덱스를 가리키도록 i를 초기화

    for j in range(start, end):                           # j를 start부터 end-1까지 순회(피벗은 제외)
        if arr[j] <= pivot:                              # 현재 원소가 피벗 이하라면
            i += 1                                       # 작은 값들이 모이는 구간을 한 칸 확장
            arr[i], arr[j] = arr[j], arr[i]             # arr[i]와 arr[j]를 교환해서 작은 값들을 앞으로 모음

    arr[i + 1], arr[end] = arr[end], arr[i + 1]          # 피벗을 최종 위치(i+1)로 보내기 위해 교환
    pi = i + 1                                            # 피벗의 최종 인덱스를 pi에 저장

    quick_sort(arr, start, pi - 1)                       # 피벗 왼쪽 부분을 재귀적으로 퀵소트
    quick_sort(arr, pi + 1, end)                         # 피벗 오른쪽 부분을 재귀적으로 퀵소트

    return arr                                            # 정렬이 완료된 배열 반환


# 사용 예시: 배열을 하나 준비
data = [10, 7, 8, 9, 1, 5]                               # 테스트용 입력 배열
sorted_data = quick_sort(data)                           # 퀵소트를 실행해 정렬
print(sorted_data)                                       # 정렬 결과 출력
```
