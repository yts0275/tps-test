아래는 **퀵소트(Quick Sort)** 구현 예시와 **각 줄마다 주석**을 달아둔 코드입니다. (파이썬)

```python
def quick_sort(arr):  # 퀵소트를 수행할 배열 arr을 입력받는 함수 정의
    if len(arr) <= 1:  # 배열 길이가 0 또는 1이면 이미 정렬된 상태이므로
        return arr  # 그대로 반환

    pivot = arr[len(arr) // 2]  # 피벗을 배열의 가운데 원소로 선택
    left = []  # 피벗보다 작은 값들을 담을 리스트 생성
    middle = []  # 피벗과 같은 값들을 담을 리스트 생성
    right = []  # 피벗보다 큰 값들을 담을 리스트 생성

    for x in arr:  # 배열의 모든 원소 x에 대해 반복
        if x < pivot:  # x가 피벗보다 작다면
            left.append(x)  # left 리스트에 추가
        elif x > pivot:  # x가 피벗보다 크다면
            right.append(x)  # right 리스트에 추가
        else:  # x가 피벗과 같다면
            middle.append(x)  # middle 리스트에 추가

    return quick_sort(left) + middle + quick_sort(right)  # 재귀적으로 좌/우 정렬 후 합치고 middle을 가운데에 둬서 반환


# 예시 실행 코드: 아래는 테스트용입니다.
data = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 입력 배열
sorted_data = quick_sort(data)  # 퀵소트를 수행해 정렬된 결과를 sorted_data에 저장
print(sorted_data)  # 정렬 결과 출력
```
