아래는 **퀵소트(Quicksort) 구현 예시**입니다. 요청하신 대로 **각 줄마다 주석을** 달았습니다. (파이썬)

```python
def quicksort(arr):  # 퀵소트 함수를 정의: 입력 배열 arr을 정렬해 반환
    if len(arr) <= 1:  # 배열 길이가 0 또는 1이면 이미 정렬 완료이므로 그대로 반환
        return arr  # 정렬된 배열(원본 그대로)을 반환

    pivot = arr[len(arr) // 2]  # 피벗을 배열의 가운데 원소로 선택
    left = []  # 피벗보다 작은 값들을 담을 리스트를 생성
    middle = []  # 피벗과 같은 값들을 담을 리스트를 생성
    right = []  # 피벗보다 큰 값들을 담을 리스트를 생성

    for x in arr:  # 입력 배열의 각 원소 x에 대해 반복
        if x < pivot:  # x가 피벗보다 작으면
            left.append(x)  # left에 x를 추가
        elif x > pivot:  # x가 피벗보다 크면
            right.append(x)  # right에 x를 추가
        else:  # x가 피벗과 같으면
            middle.append(x)  # middle에 x를 추가

    return quicksort(left) + middle + quicksort(right)  # 재귀적으로 left와 right 정렬 후 middle과 합쳐 반환


# 테스트 코드: 예시 배열
data = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 데이터 예시
sorted_data = quicksort(data)  # 퀵소트를 실행해 정렬 결과를 저장
print(sorted_data)  # 정렬된 결과 출력
```
