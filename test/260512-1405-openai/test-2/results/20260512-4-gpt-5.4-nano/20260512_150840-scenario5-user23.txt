아래는 **퀵소트(Quick Sort) 구현 예시**와 **각 줄마다 주석**을 단 코드입니다. (파이썬 기준)

```python
def quicksort(arr, low, high):  # 퀵소트를 수행하는 함수 정의(배열, 시작 인덱스, 끝 인덱스 받음)
    if low >= high:  # 원소가 0개 또는 1개면 정렬할 필요가 없으므로 종료
        return  # 재귀 종료
    
    pivot = arr[high]  # 피벗을 배열의 마지막 원소로 선택
    i = low - 1  # 피벗보다 작은 영역의 끝을 가리키는 포인터(초기값은 low-1)
    
    for j in range(low, high):  # low부터 high-1까지 순회하며 원소들을 비교
        if arr[j] <= pivot:  # 현재 원소가 피벗보다 작거나 같으면
            i += 1  # 작은 값 영역을 한 칸 확장
            arr[i], arr[j] = arr[j], arr[i]  # 작은 값 영역 끝에 현재 원소를 배치하도록 스왑
    
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 i+1 위치로 옮겨 최종 위치 고정
    pivot_index = i + 1  # 피벗의 최종 인덱스를 기록
    
    quicksort(arr, low, pivot_index - 1)  # 피벗 왼쪽 부분 배열에 대해 재귀 호출
    quicksort(arr, pivot_index + 1, high)  # 피벗 오른쪽 부분 배열에 대해 재귀 호출


def quicksort_main(arr):  # 사용자에게 노출될 간단한 진입 함수(전체 배열 정렬)
    quicksort(arr, 0, len(arr) - 1)  # 전체 범위를 대상으로 퀵소트 수행
    return arr  # 정렬된 배열을 반환


# 실행 예시
data = [10, 7, 8, 9, 1, 5]  # 예시 입력 배열 생성
sorted_data = quicksort_main(data)  # 퀵소트로 정렬 수행
print(sorted_data)  # 정렬 결과 출력
```
