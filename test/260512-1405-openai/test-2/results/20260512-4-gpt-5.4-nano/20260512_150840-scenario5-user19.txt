아래는 **퀵소트(Quick Sort)** 구현 예시(파이썬)이고, **각 줄마다 주석**을 달았습니다.

```python
def quicksort(arr):  # 입력 배열 arr을 퀵소트로 정렬하는 함수 정의
    if len(arr) <= 1:  # 배열 길이가 0 또는 1이면 이미 정렬되어 있음
        return arr  # 그대로 반환

    pivot = arr[len(arr) // 2]  # 가운데 원소를 피벗(pivot)으로 선택
    left = []  # 피벗보다 작은 값들을 담을 리스트
    middle = []  # 피벗과 같은 값들을 담을 리스트
    right = []  # 피벗보다 큰 값들을 담을 리스트

    for x in arr:  # 배열의 각 원소 x에 대해 반복
        if x < pivot:  # x가 피벗보다 작으면
            left.append(x)  # left에 추가
        elif x > pivot:  # x가 피벗보다 크면
            right.append(x)  # right에 추가
        else:  # x가 피벗과 같으면
            middle.append(x)  # middle에 추가

    return quicksort(left) + middle + quicksort(right)  # 왼쪽 정렬 + 가운데 + 오른쪽 정렬 결과 결합해서 반환


# 예시 실행 부분
data = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 예제 배열 생성
sorted_data = quicksort(data)  # 퀵소트로 정렬 수행
print(sorted_data)  # 정렬 결과 출력
```
