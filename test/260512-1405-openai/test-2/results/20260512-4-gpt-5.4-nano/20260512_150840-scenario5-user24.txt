아래는 **퀵소트(Quick Sort)** 를 파이썬으로 구현한 예시이며, **각 줄마다 주석**을 달았습니다.

```python
def quick_sort(arr):  # 입력 배열을 퀵소트로 정렬하는 함수 정의
    if len(arr) <= 1:  # 배열 길이가 0 또는 1이면 이미 정렬된 상태이므로 그대로 반환
        return arr  # 정렬 결과(원본)를 반환

    pivot = arr[len(arr) // 2]  # 가운데 원소를 피벗(pivot)으로 선택
    left = []  # 피벗보다 작은 값들을 담을 리스트 생성
    middle = []  # 피벗과 같은 값들을 담을 리스트 생성
    right = []  # 피벗보다 큰 값들을 담을 리스트 생성

    for x in arr:  # 배열의 각 원소 x를 순회
        if x < pivot:  # x가 피벗보다 작으면
            left.append(x)  # left 리스트에 추가
        elif x == pivot:  # x가 피벗과 같으면
            middle.append(x)  # middle 리스트에 추가
        else:  # x가 피벗보다 크면
            right.append(x)  # right 리스트에 추가

    return quick_sort(left) + middle + quick_sort(right)  # 재귀적으로 정렬 후 합쳐 반환


# 예시 실행 코드: 테스트용으로만 사용
data = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 예시 배열
print(quick_sort(data))  # 퀵소트 결과를 출력
```
