다음은 파이썬으로 작성한 퀵소트 구현이며, 각 줄마다 한국어 주석을 달아 설명한 코드입니다.

```python
def quicksort(arr):  # 배열을 입력받아 정렬된 동일 배열을 반환하는 래퍼 함수 정의
    def _quicksort(a, low, high):  # 내부 재귀 함수: a의 인덱스 low..high 구간을 정렬
        if low < high:  # 정렬할 구간의 시작이 끝보다 작을 때만 분할 및 재귀 수행
            p = partition(a, low, high)  # 피벗을 기준으로 분할하고 피벗의 최종 인덱스 p를 받음
            _quicksort(a, low, p - 1)  # 피벗 왼쪽 구간을 재귀적으로 정렬
            _quicksort(a, p + 1, high)  # 피벗 오른쪽 구간을 재귀적으로 정렬
        # 만약 low >= high이면 해당 구간은 이미 정렬된 상태이므로 아무 작업도 하지 않음

    def partition(a, low, high):  # Lomuto 파티션 방식으로 a[low..high]를 분할
        pivot = a[high]  # 피벗으로 구간의 마지막 요소를 선택
        i = low - 1  # 작은 요소들의 끝 인덱스를 나타내는 포인터를 초기화
        for j in range(low, high):  # low부터 high-1까지 현재 요소들을 순회
            if a[j] <= pivot:  # 현재 요소가 피벗보다 작거나 같으면
                i += 1  # 작은 요소 영역을 한 칸 확장
                a[i], a[j] = a[j], a[i]  # 현재 요소를 작은 요소 영역의 끝으로 교환
        a[i + 1], a[high] = a[high], a[i + 1]  # 피벗을 작은 요소 영역 바로 다음으로 옮겨 위치 확정
        return i + 1  # 피벗의 최종 인덱스를 반환

    n = len(arr)  # 입력 배열의 길이를 계산
    _quicksort(arr, 0, n - 1)  # 배열 전체를 정렬하도록 내부 함수 호출
    return arr  # 정렬된 배열(원본 배열 자체)을 반환
```

사용 예:
- quicksort([3, 6, 1, 8, 4])  # -> [1, 3, 4, 6, 8]
