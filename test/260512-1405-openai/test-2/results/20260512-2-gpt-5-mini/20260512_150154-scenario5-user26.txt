다음은 파이썬으로 구현한 퀵소트 예제입니다. 각 줄마다 한국어 주석을 달아 동작을 설명했습니다.

```python
def quicksort(seq):                                  # 입력 시퀀스를 받아 정렬된 새 리스트를 반환하는 진입 함수
    a = list(seq)                                    # 원본을 변경하지 않기 위해 입력을 복사하여 사용
    def _quicksort(a, lo, hi):                       # 실제 재귀 정렬을 수행하는 내부 함수 정의
        if lo < hi:                                  # 정렬할 구간이 유효한지(원소가 2개 이상인지) 검사
            p = partition(a, lo, hi)                 # 분할을 수행하여 피벗의 최종 인덱스를 얻음
            _quicksort(a, lo, p - 1)                 # 피벗 왼쪽 부분을 재귀적으로 정렬
            _quicksort(a, p + 1, hi)                 # 피벗 오른쪽 부분을 재귀적으로 정렬
    def partition(a, lo, hi):                        # Lomuto 분할 방식으로 피벗 위치를 결정하는 함수
        pivot = a[hi]                                # 구간의 마지막 원소를 피벗으로 선택
        i = lo                                       # 작은 요소들이 모일 경계 인덱스 초기화
        for j in range(lo, hi):                      # 피벗을 제외한 구간을 순회하며 비교
            if a[j] <= pivot:                        # 현재 원소가 피벗보다 작거나 같으면
                a[i], a[j] = a[j], a[i]              # 작은 원소를 경계 왼쪽으로 교환
                i += 1                               # 경계를 오른쪽으로 한 칸 확장
        a[i], a[hi] = a[hi], a[i]                    # 피벗을 경계 위치로 옮겨 올바른 위치로 만듦
        return i                                     # 피벗의 최종 인덱스 반환
    _quicksort(a, 0, len(a) - 1)                     # 전체 배열을 대상으로 재귀 정렬 시작
    return a                                         # 정렬된 배열을 반환

if __name__ == "__main__":                           # 이 파일을 직접 실행할 때만 아래 예시를 수행
    data = [3, 6, 8, 10, 1, 2, 1]                    # 정렬할 예시 데이터 정의
    print(quicksort(data))                           # 정렬 결과를 출력
```

원하시면 in-place(원본 변경) 버전, 다른 분할 방식(Hoare), 혹은 반복적(비재귀) 구현으로도 보여드릴게요.
