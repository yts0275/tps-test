아래는 각 줄마다 주석을 단 파이썬 퀵소트(람오토 분할)를 구현한 예시입니다.

```python
def quicksort(arr, low=0, high=None):  # 퀵소트 함수 정의; 기본적으로 전체 리스트를 정렬하도록 설계함
    if high is None:  # high가 주어지지 않았으면
        high = len(arr) - 1  # 배열의 마지막 인덱스를 high로 설정함
    if low < high:  # 정렬할 구간이 유효한(원소가 2개 이상인) 경우에만 처리함
        pivot_index = partition(arr, low, high)  # 분할 함수로 피벗의 최종 위치를 얻음
        quicksort(arr, low, pivot_index - 1)  # 피벗 왼쪽 부분을 재귀적으로 정렬함
        quicksort(arr, pivot_index + 1, high)  # 피벗 오른쪽 부분을 재귀적으로 정렬함
    return arr  # 정렬된 배열(참조)을 반환함


def partition(arr, low, high):  # 람오토(Lomuto) 분할 알고리즘을 사용한 분할 함수 정의
    pivot = arr[high]  # 피벗을 구간의 마지막 원소로 선택함
    i = low - 1  # 작은 요소들의 끝 인덱스를 가리키는 포인터를 초기화함
    for j in range(low, high):  # low부터 high-1까지 순회하면서 피벗보다 작은 원소를 찾음
        if arr[j] <= pivot:  # 현재 원소가 피벗 이하라면
            i += 1  # 작은 요소들의 끝 인덱스를 한 칸 증가시킴
            arr[i], arr[j] = arr[j], arr[i]  # 현재 원소를 작은 요소들 뒤로 이동시켜 교환함
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 중간 위치(i+1)로 옮겨서 분할을 완료함
    return i + 1  # 피벗의 최종 위치 인덱스를 반환함
```
