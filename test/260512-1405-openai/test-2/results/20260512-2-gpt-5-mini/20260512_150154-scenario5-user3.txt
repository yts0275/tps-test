아래는 Lomuto 분할 방식을 이용한 제자리(in-place) 퀵소트 구현입니다. 각 코드 줄마다 한국어 주석을 달아두었습니다.

```python
def quicksort_inplace(arr, low=0, high=None):  # arr를 제자리로 퀵소트하는 함수, low와 high는 정렬 범위 인덱스
    if high is None:  # high가 주어지지 않으면
        high = len(arr) - 1  # high를 배열의 마지막 인덱스로 설정
    if low < high:  # 정렬할 부분이 두 개 이상의 원소를 가지는 경우에만
        p = partition(arr, low, high)  # 분할을 수행하여 피벗의 최종 위치 인덱스를 얻음
        quicksort_inplace(arr, low, p - 1)  # 피벗 왼쪽 부분을 재귀적으로 정렬
        quicksort_inplace(arr, p + 1, high)  # 피벗 오른쪽 부분을 재귀적으로 정렬
    return arr  # 정렬된 배열(같은 객체)을 반환

def partition(arr, low, high):  # Lomuto 분할 알고리즘을 구현한 함수
    pivot = arr[high]  # 피벗으로 범위의 마지막 원소를 선택
    i = low - 1  # 작은(<= pivot) 요소들의 경계 인덱스를 초기화
    for j in range(low, high):  # low부터 high-1까지 반복하면서
        if arr[j] <= pivot:  # 현재 요소가 피벗 이하이면
            i += 1  # 작은 영역의 경계를 오른쪽으로 한 칸 이동
            arr[i], arr[j] = arr[j], arr[i]  # 현재 요소를 작은 영역의 끝으로 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 작은 영역의 바로 다음 위치로 교환
    return i + 1  # 피벗의 최종 위치 인덱스를 반환

# 사용 예시
arr = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 예제 배열
sorted_arr = quicksort_inplace(arr)  # 제자리 퀵소트 호출
print(sorted_arr)  # 정렬 결과 출력
```
