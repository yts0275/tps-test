아래는 각 줄마다 주석을 단 Python으로 구현한 퀵소트입니다.

```python
def quicksort(arr):  # 배열을 제자리(in-place)로 정렬하는 공개 함수
    _quicksort(arr, 0, len(arr) - 1)  # 내부 재귀 함수로 전체 구간을 정렬 시작

def _quicksort(arr, low, high):  # 배열의 [low, high] 구간을 재귀적으로 정렬하는 함수
    if low < high:  # 구간에 두 개 이상의 요소가 있는지 확인 (기저 조건)
        p = partition(arr, low, high)  # 피벗을 기준으로 분할하고 피벗의 최종 인덱스 반환
        _quicksort(arr, low, p - 1)  # 피벗 왼쪽 부분을 재귀적으로 정렬
        _quicksort(arr, p + 1, high)  # 피벗 오른쪽 부분을 재귀적으로 정렬

def partition(arr, low, high):  # Lomuto 분할 방법으로 [low, high]를 분할하는 함수
    pivot = arr[high]  # 구간의 마지막 요소를 피벗으로 선택
    i = low - 1  # 피벗보다 작거나 같은 요소들의 경계 인덱스(i)를 초기화
    for j in range(low, high):  # low부터 high-1까지의 인덱스를 순회
        if arr[j] <= pivot:  # 현재 요소가 피벗 이하이면
            i += 1  # 경계 인덱스를 하나 이동하여 작은 요소를 위한 자리 확보
            arr[i], arr[j] = arr[j], arr[i]  # 작은 요소를 왼쪽 부분으로 스왑
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 경계 바로 다음 위치로 이동
    return i + 1  # 피벗의 최종 위치를 반환

if __name__ == "__main__":  # 모듈을 직접 실행할 때만 아래 코드가 실행됨
    data = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 예시 리스트 정의
    print("정렬 전:", data)  # 정렬 전 리스트 출력
    quicksort(data)  # 퀵소트 호출 (in-place 정렬)
    print("정렬 후:", data)  # 정렬 후 리스트 출력
```
