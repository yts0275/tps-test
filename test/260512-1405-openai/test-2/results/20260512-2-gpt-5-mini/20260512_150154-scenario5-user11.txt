아래는 Python으로 구현한 퀵소트(롬토 분할) 예제입니다. 코드의 각 줄에 주석을 달아 설명했습니다.

# 퀵소트(롬토 분할) 구현 예제입니다.  # 예제 목적 설명
def quicksort(arr):  # 리스트를 인플레이스(in-place)로 정렬하는 공개 함수 정의
    def _quicksort(low, high):  # 배열의 [low, high] 구간을 재귀적으로 정렬하는 내부 함수 정의
        if low < high:  # 구간에 정렬할 요소가 2개 이상 있는지 확인
            p = partition(low, high)  # 분할을 수행하여 피벗의 최종 위치 인덱스를 받음
            _quicksort(low, p-1)  # 피벗 왼쪽 부분을 재귀적으로 정렬
            _quicksort(p+1, high)  # 피벗 오른쪽 부분을 재귀적으로 정렬
    def partition(low, high):  # 롬토(Lomuto) 분할 함수 정의: 마지막 요소를 피벗으로 사용
        pivot = arr[high]  # 피벗 값으로 구간의 마지막 요소 선택
        i = low - 1  # 작은 요소들의 끝을 나타내는 인덱스 초기화
        for j in range(low, high):  # low부터 high-1까지 모든 요소를 순회
            if arr[j] <= pivot:  # 현재 요소가 피벗보다 작거나 같으면
                i += 1  # 작은 요소들의 끝 인덱스를 하나 증가
                arr[i], arr[j] = arr[j], arr[i]  # 작은 그룹에 현재 요소를 넣기 위해 스왑
        arr[i+1], arr[high] = arr[high], arr[i+1]  # 피벗을 가운데 위치(i+1)로 이동(스왑)
        return i+1  # 피벗의 최종 인덱스 반환
    _quicksort(0, len(arr)-1)  # 전체 리스트를 정렬하기 위해 내부 재귀 호출 시작

# 사용 예시                             
data = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 샘플 리스트 정의
quicksort(data)  # 주어진 리스트를 인플레이스로 정렬 수행
print(data)  # 정렬된 결과 출력 (오름차순)
