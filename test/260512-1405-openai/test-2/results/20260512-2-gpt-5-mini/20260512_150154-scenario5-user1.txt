아래는 Python으로 구현한 퀵소트 예제입니다. 모든 코드 줄마다 한국어 주석을 달았습니다.

```python
def quicksort(arr, low=0, high=None):  # 퀵소트 함수 정의: 정렬할 리스트와 범위 인덱스 받음
    if high is None:  # high가 지정되지 않았으면
        high = len(arr) - 1  # 전체 리스트의 마지막 인덱스로 설정
    if low < high:  # 정렬할 구간이 유효한지(하한 < 상한) 확인
        p = partition(arr, low, high)  # 파티션 수행하여 피벗의 최종 인덱스 얻음
        quicksort(arr, low, p - 1)  # 피벗 왼쪽 구간을 재귀적으로 정렬
        quicksort(arr, p + 1, high)  # 피벗 오른쪽 구간을 재귀적으로 정렬
    return arr  # 정렬된(또는 변화된) 리스트 반환

def partition(arr, low, high):  # 파티션 함수 정의: Lomuto 파티션 사용
    pivot = arr[high]  # 피벗으로 구간의 마지막 요소 선택
    i = low - 1  # 작은 요소들의 경계 인덱스 초기화
    for j in range(low, high):  # low부터 high-1까지 각 요소 검사
        if arr[j] <= pivot:  # 현재 요소가 피벗 이하이면
            i += 1  # 작은 요소 경계를 하나 옮김
            arr[i], arr[j] = arr[j], arr[i]  # 작은 요소를 경계 다음 위치로 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 경계 다음 위치로 이동
    return i + 1  # 피벗의 최종 인덱스 반환

if __name__ == "__main__":  # 스크립트를 직접 실행할 때만 아래 코드 실행
    data = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 예시 리스트 정의
    print("정렬 전:", data)  # 정렬 전 리스트 출력
    sorted_data = quicksort(data)  # 퀵소트 호출하여 정렬 수행 (제자리 정렬)
    print("정렬 후:", sorted_data)  # 정렬 결과 출력
```
