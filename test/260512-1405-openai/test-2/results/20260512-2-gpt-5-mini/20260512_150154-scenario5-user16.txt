아래는 Python으로 구현한 퀵소트입니다. 각 코드 줄마다 한국어 주석을 달았습니다.

```python
def quicksort(arr, low=0, high=None):  # 퀵소트 함수 정의: arr을 정렬, low와 high는 정렬할 구간의 인덱스
    if high is None:  # high가 주어지지 않았으면
        high = len(arr) - 1  # 전체 배열의 마지막 인덱스를 high로 설정
    if low < high:  # 정렬할 구간이 유효한(두 개 이상의 원소가 있는) 경우에만 처리
        p = partition(arr, low, high)  # 파티션 함수를 호출해 피벗의 최종 위치를 얻음
        quicksort(arr, low, p - 1)  # 피벗 왼쪽 부분을 재귀적으로 정렬
        quicksort(arr, p + 1, high)  # 피벗 오른쪽 부분을 재귀적으로 정렬
    return arr  # 정렬된(또는 부분 정렬된) 배열을 반환


def partition(arr, low, high):  # 파티션 함수 정의: Lomuto 파티션 사용
    pivot = arr[high]  # 피벗으로 구간의 마지막 원소를 선택
    i = low  # i는 작은 원소들의 경계(다음 작은 원소가 들어갈 위치)
    for j in range(low, high):  # low부터 high-1까지 인덱스를 순회
        if arr[j] < pivot:  # 현재 원소가 피벗보다 작으면
            arr[i], arr[j] = arr[j], arr[i]  # 현재 원소를 작은 원소 영역의 끝으로 이동(교환)
            i += 1  # 작은 원소 영역의 경계를 오른쪽으로 한 칸 이동
    arr[i], arr[high] = arr[high], arr[i]  # 피벗을 중간 경계로 옮겨 최종 위치 확정
    return i  # 피벗의 최종 인덱스를 반환


if __name__ == "__main__":  # 이 파일을 직접 실행할 때만 아래 코드 실행
    a = [3, 6, 8, 10, 1, 2, 1]  # 예시 정렬 대상 배열 선언
    print("정렬 전:", a)  # 정렬 전 배열 출력
    quicksort(a)  # 제자리(in-place)로 퀵소트 수행
    print("정렬 후:", a)  # 정렬 후 배열 출력
```
