아래는 파이썬으로 구현한 퀵소트 코드입니다. 각 코드 줄마다 한국어 주석을 달아 동작을 설명했습니다.

```python
def quicksort(arr):  # 퀵소트 함수의 공개 래퍼 함수로, 리스트를 받아 제자리 정렬합니다.
    def _quicksort(low, high):  # 실제 재귀 호출을 수행하는 내부 함수로, 부분 배열의 인덱스 범위를 받습니다.
        if low < high:  # 부분 배열에 정렬할 요소가 2개 이상 있는지 확인합니다.
            p = partition(low, high)  # 피벗을 기준으로 분할하고 피벗의 최종 위치 인덱스를 얻습니다.
            _quicksort(low, p - 1)  # 피벗 왼쪽 부분 배열을 재귀적으로 정렬합니다.
            _quicksort(p + 1, high)  # 피벗 오른쪽 부분 배열을 재귀적으로 정렬합니다.

    def partition(low, high):  # 분할 함수로, lomuto 파티셔닝을 사용합니다.
        pivot = arr[high]  # 피벗으로 현재 부분배열의 마지막 요소를 선택합니다.
        i = low - 1  # 작은 요소들의 끝 인덱스를 나타내는 포인터를 초기화합니다.
        for j in range(low, high):  # low부터 high-1까지 순회하며 피벗보다 작은 요소를 찾습니다.
            if arr[j] <= pivot:  # 현재 요소가 피벗 이하이면
                i += 1  # 작은 요소들의 끝을 한 칸 이동합니다.
                arr[i], arr[j] = arr[j], arr[i]  # 현재 요소를 작은 그룹의 뒤로 스왑합니다.
        arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 작은 그룹 다음 위치로 옮깁니다.
        return i + 1  # 피벗의 최종 위치 인덱스를 반환합니다.

    _quicksort(0, len(arr) - 1)  # 전체 배열 범위에 대해 내부 재귀 함수를 호출하여 정렬을 시작합니다.

# 사용 예시
if __name__ == "__main__":  # 이 파일을 직접 실행할 때만 아래 예시 코드가 동작하도록 합니다.
    data = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 예시 리스트를 정의합니다.
    print("정렬 전:", data)  # 정렬 전 리스트를 출력합니다.
    quicksort(data)  # 리스트를 제자리에서 퀵소트로 정렬합니다.
    print("정렬 후:", data)  # 정렬된 리스트를 출력합니다.
```
