아래는 파이썬으로 구현한 퀵소트 예제입니다. 각 코드 줄마다 한국어 주석을 추가했습니다.

```python
# 퀵소트(QuickSort) 알고리즘을 구현한 코드입니다. 각 줄에 주석이 있습니다.
def quicksort(arr, low, high):  # arr 리스트를 low..high 구간에 대해 퀵소트 수행하는 함수 정의
    if low < high:  # 정렬할 구간이 1개 초과인 경우에만 분할 및 정복 수행
        p = partition(arr, low, high)  # 피벗을 기준으로 분할하고 피벗의 최종 위치 인덱스 p를 받음
        quicksort(arr, low, p - 1)  # 피벗 왼쪽 부분 배열을 재귀적으로 퀵소트 수행
        quicksort(arr, p + 1, high)  # 피벗 오른쪽 부분 배열을 재귀적으로 퀵소트 수행

def partition(arr, low, high):  # arr의 low..high 구간을 피벗 기준으로 분할하는 함수 정의
    pivot = arr[high]  # 편의상 구간의 마지막 원소를 피벗으로 선택
    i = low - 1  # 피벗보다 작거나 같은 요소들의 마지막 위치를 가리키는 인덱스 초기화
    for j in range(low, high):  # low부터 high-1까지 순회하면서 현재 원소들을 검사
        if arr[j] <= pivot:  # 현재 원소가 피벗보다 작거나 같으면 왼쪽 파트에 포함시켜야 함
            i += 1  # 왼쪽 파트의 경계 인덱스를 한 칸 오른쪽으로 이동
            arr[i], arr[j] = arr[j], arr[i]  # 현재 원소와 경계 다음 위치의 원소를 교환하여 왼쪽 파트에 추가
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 왼쪽 파트의 바로 다음 위치로 옮겨 최종 위치 확보
    return i + 1  # 피벗의 최종 위치 인덱스를 반환

if __name__ == "__main__":  # 이 스크립트가 직접 실행될 때만 아래 코드 블록을 실행
    data = [3, 6, 8, 10, 1, 2, 1]  # 예시로 정렬할 리스트를 정의
    quicksort(data, 0, len(data) - 1)  # 리스트 전체 구간에 대해 퀵소트 호출
    print(data)  # 정렬된 결과를 출력
```
