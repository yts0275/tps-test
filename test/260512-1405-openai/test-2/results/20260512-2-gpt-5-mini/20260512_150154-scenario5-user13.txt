아래는 각 코드 줄마다 한글 주석을 단 in-place 퀵소트 구현입니다.

```python
# 퀵소트 구현 (in-place, Lomuto 분할법 사용)
def quicksort(arr, low, high):  # 주어진 리스트 arr의 인덱스 low..high 범위를 정렬하는 재귀 함수
    if low < high:  # 부분 배열에 두 개 이상의 원소가 있을 때만 정렬을 진행
        p = partition(arr, low, high)  # 파티션을 수행하여 피벗의 최종 위치 인덱스를 얻음
        quicksort(arr, low, p - 1)  # 피벗 왼쪽 부분 배열을 재귀적으로 정렬
        quicksort(arr, p + 1, high)  # 피벗 오른쪽 부분 배열을 재귀적으로 정렬

def partition(arr, low, high):  # Lomuto 분할법을 사용하는 파티션 함수
    pivot = arr[high]  # 마지막 원소를 피벗으로 선택
    i = low - 1  # 작은 요소들의 경계 인덱스를 초기화 (경계는 i)
    for j in range(low, high):  # low부터 high-1까지 현재 검사할 인덱스 j를 순회
        if arr[j] <= pivot:  # 현재 요소가 피벗보다 작거나 같으면
            i += 1  # 경계 인덱스를 한 칸 오른쪽으로 이동
            arr[i], arr[j] = arr[j], arr[i]  # 작은 요소를 경계 바로 다음 위치로 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 경계 바로 다음 위치로 이동하여 분할 완료
    return i + 1  # 피벳의 최종 위치 인덱스를 반환

def quicksort_inplace(arr):  # 리스트 전체를 정렬하기 위한 래퍼 함수
    if len(arr) <= 1:  # 길이가 0이나 1이면 이미 정렬된 상태이므로 바로 반환
        return  # 아무 작업도 하지 않고 종료
    quicksort(arr, 0, len(arr) - 1)  # 실제 퀵소트 함수를 호출하여 전체 범위를 정렬

# 예시 사용법 (직접 실행할 때만 동작)
if __name__ == "__main__":  # 이 파일을 직접 실행할 때만 아래 코드가 실행되도록 함
    data = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 예시 리스트
    print("정렬 전:", data)  # 정렬 전 리스트를 출력
    quicksort_inplace(data)  # 퀵소트로 리스트를 제자리(in-place) 정렬
    print("정렬 후:", data)  # 정렬된 리스트를 출력
```
