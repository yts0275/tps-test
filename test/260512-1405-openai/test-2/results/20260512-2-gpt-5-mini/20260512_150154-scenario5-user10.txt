아래는 파이썬으로 구현한 퀵소트(제자리 정렬) 예제입니다. 각 줄마다 한국어 주석을 달아 설명했습니다.

```python
def quicksort(arr):  # 리스트 arr를 받아 제자리(in-place)로 퀵소트하고 정렬된 리스트를 반환하는 함수 정의
    def partition(lo, hi):  # 배열의 일부분 [lo..hi]에 대해 분할 작업을 수행하는 보조 함수 정의
        pivot = arr[hi]  # 피벗으로 구간의 마지막 원소를 선택함
        i = lo - 1  # 작은 요소들을 가리킬 인덱스 i를 초기화함 (작은 그룹의 끝)
        for j in range(lo, hi):  # lo부터 hi-1까지 반복하며 현재 요소를 검사함
            if arr[j] <= pivot:  # 현재 요소가 피벗 이하이면 작은 그룹에 포함해야 함
                i += 1  # 작은 그룹의 경계를 하나 오른쪽으로 이동
                arr[i], arr[j] = arr[j], arr[i]  # 현재 요소를 작은 그룹 뒤로 스왑하여 포함시킴
        arr[i+1], arr[hi] = arr[hi], arr[i+1]  # 피벗을 작은 그룹의 바로 다음 위치로 옮겨 제자리에 둠
        return i + 1  # 피벗의 최종 위치(분할 인덱스)를 반환함
    def _quicksort(lo, hi):  # 재귀적으로 구간 [lo..hi]를 정렬하는 내부 함수 정의
        if lo < hi:  # 유효한 구간(원소가 2개 이상)이면 정렬을 진행함
            p = partition(lo, hi)  # 현재 구간을 분할하고 피벗의 인덱스를 얻음
            _quicksort(lo, p - 1)  # 피벗 왼쪽 구간을 재귀적으로 정렬함
            _quicksort(p + 1, hi)  # 피벗 오른쪽 구간을 재귀적으로 정렬함
    _quicksort(0, len(arr) - 1)  # 전체 배열을 대상으로 내부 재귀 함수를 호출하여 정렬 시작
    return arr  # 정렬된 배열을 반환함

# 사용 예시
if __name__ == "__main__":  # 이 파일이 직접 실행될 때만 아래 코드를 실행함
    data = [33, 10, 55, 71, 29, 3, 18]  # 정렬할 예시 리스트를 정의함
    print("정렬 전:", data)  # 정렬 전 리스트를 출력함
    sorted_data = quicksort(data)  # 퀵소트를 호출하여 정렬함
    print("정렬 후:", sorted_data)  # 정렬 후 결과를 출력함
```
