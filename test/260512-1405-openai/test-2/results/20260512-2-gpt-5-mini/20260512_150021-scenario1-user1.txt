아래는 Python으로 작성한 퀵소트 구현입니다. 각 코드 줄마다 한국어 주석을 달아 동작을 설명해두었습니다.

```python
# 퀵소트(Quick Sort) 구현 예제입니다. 각 줄에 주석을 달아 설명합니다.
def quicksort(arr, low, high):  # arr를 부분구간 [low, high] 기준으로 퀵소트 실행하는 재귀함수
    if low < high:  # 정렬할 구간의 시작 인덱스가 끝 인덱스보다 작을 때만 처리
        p = partition(arr, low, high)  # 피벗을 기준으로 분할하고 피벗의 최종 위치를 p에 받음
        quicksort(arr, low, p - 1)  # 피벗 왼쪽 부분을 재귀적으로 정렬
        quicksort(arr, p + 1, high)  # 피벗 오른쪽 부분을 재귀적으로 정렬

def partition(arr, low, high):  # Lomuto 분할 방식을 사용하는 파티션 함수
    pivot = arr[high]  # 구간의 마지막 요소를 피벗으로 선택
    i = low - 1  # 작은 요소들의 끝 인덱스를 가리키는 포인터를 초기화
    for j in range(low, high):  # low에서 high-1까지 순회하면서 피벗보다 작거나 같은 요소 찾기
        if arr[j] <= pivot:  # 현재 요소가 피벗 이하이면
            i += 1  # 작은 요소들의 끝 인덱스를 한 칸 이동
            arr[i], arr[j] = arr[j], arr[i]  # 현재 요소를 작은 그룹의 다음 위치로 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 작은 그룹의 바로 다음 위치로 옮겨 최종 위치 확보
    return i + 1  # 피벗의 최종 인덱스를 반환

def quicksort_inplace(arr):  # 외부에서 호출하기 쉬운 in-place 퀵소트 래퍼 함수
    quicksort(arr, 0, len(arr) - 1)  # 전체 배열 범위에 대해 퀵소트 수행
    return arr  # 정렬된 배열(같은 객체)을 반환

# 예제 실행 코드 (모듈로 사용할 때는 제거하거나 주석 처리하세요)
if __name__ == "__main__":  # 스크립트로 직접 실행할 때만 아래 코드가 실행되도록 함
    data = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 샘플 데이터 목록을 정의
    print("정렬 전:", data)  # 정렬 전 상태 출력
    sorted_data = quicksort_inplace(data)  # in-place 퀵소트 호출하여 정렬 수행
    print("정렬 후:", sorted_data)  # 정렬 후 결과 출력
```
