아래는 각 줄마다 주석을 단 퀵소트(파이썬, in-place Lomuto 분할) 구현입니다.

def quicksort(arr, low, high): # 지정한 범위 [low, high]를 퀵소트로 정렬하는 함수 정의
    if low < high: # 정렬할 부분의 크기가 2개 이상인지 검사
        p = partition(arr, low, high) # 분할하여 피벗의 최종 위치 인덱스를 얻음
        quicksort(arr, low, p - 1) # 피벗 왼쪽 부분을 재귀적으로 정렬
        quicksort(arr, p + 1, high) # 피벗 오른쪽 부분을 재귀적으로 정렬
def partition(arr, low, high): # Lomuto 분할 알고리즘을 사용하는 분할 함수 정의
    pivot = arr[high] # 피벗으로 배열의 마지막 요소를 선택
    i = low - 1 # 작은 요소들을 모을 위치의 바로 이전 인덱스를 초기화
    for j in range(low, high): # low부터 high-1까지 각 요소를 검사
        if arr[j] <= pivot: # 검사한 요소가 피벗보다 작거나 같으면
            i += 1 # 작은 요소 영역의 경계를 한 칸 앞으로 이동
            arr[i], arr[j] = arr[j], arr[i] # 현재 요소를 작은 영역 끝으로 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1] # 피벗을 작은 요소들 바로 다음 위치로 옮김
    return i + 1 # 피벗의 최종 인덱스를 반환
def quicksort_inplace(arr): # 외부에서 간편히 호출할 수 있는 포장 함수 정의
    quicksort(arr, 0, len(arr) - 1) # 배열 전체 범위에 대해 퀵소트를 호출
# 사용 예시: 아래 코드는 이 파일을 직접 실행할 때만 동작함
if __name__ == "__main__": # 이 모듈이 직접 실행될 때만 예시 코드 실행
    data = [3, 6, 8, 10, 1, 2, 1] # 정렬할 예제 리스트를 정의
    print("정렬 전:", data) # 정렬 전 리스트 출력
    quicksort_inplace(data) # 리스트를 제자리에서 퀵소트로 정렬
    print("정렬 후:", data) # 정렬 후 리스트 출력
