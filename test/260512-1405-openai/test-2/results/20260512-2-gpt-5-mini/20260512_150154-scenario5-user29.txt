아래는 각 줄마다 한국어 주석을 단 퀵소트 구현 예제입니다.

def quicksort(arr): # 입력 리스트를 복사하여 정렬된 새 리스트를 반환하는 함수 정의
    a = arr[:] # 원본을 변경하지 않기 위해 리스트를 복사함
    def _quicksort(a, lo, hi): # 내부 재귀 함수: a[lo:hi+1] 구간을 정렬함
        if lo < hi: # 구간에 두 개 이상의 요소가 있을 때만 정렬을 수행함
            p = partition(a, lo, hi) # 분할 함수를 호출하여 피벗 인덱스를 얻음
            _quicksort(a, lo, p-1) # 피벗 왼쪽 부분을 재귀적으로 정렬함
            _quicksort(a, p+1, hi) # 피벗 오른쪽 부분을 재귀적으로 정렬함
    def partition(a, lo, hi): # Lomuto 분할법을 사용한 분할 함수 정의
        pivot = a[hi] # 구간의 마지막 요소를 피벗으로 선택함
        i = lo # 작은 요소들을 채워 넣을 위치 인덱스를 초기화함
        for j in range(lo, hi): # lo부터 hi-1까지 반복하면서 피벗보다 작은 요소를 발견하면 이동시킴
            if a[j] < pivot: # 현재 요소가 피벗보다 작으면
                a[i], a[j] = a[j], a[i] # 현재 요소를 왼쪽 파티션의 끝으로 교환함
                i += 1 # 왼쪽 파티션의 끝 위치를 한 칸 뒤로 옮김
        a[i], a[hi] = a[hi], a[i] # 피벗을 중간 위치로 옮겨서 분할을 완성함
        return i # 피벗의 최종 위치 인덱스를 반환함
    _quicksort(a, 0, len(a)-1) # 전체 배열 범위에 대해 재귀 정렬을 시작함
    return a # 정렬된 복사본을 반환함

# 사용 예시: 아래 두 줄은 테스트용 코드이며 필요 없으면 지워도 됨
arr = [3, 6, 8, 10, 1, 2, 1] # 테스트할 입력 리스트 예시
print(quicksort(arr)) # 정렬 결과를 출력함 (원본은 변경되지 않음)
