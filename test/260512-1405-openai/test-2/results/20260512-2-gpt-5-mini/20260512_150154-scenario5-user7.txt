아래는 파이썬으로 구현한 퀵소트 예제입니다. 각 코드 줄마다 한글 주석을 달아 동작을 설명했습니다.

```python
def quicksort(arr, low, high):  # 퀵소트 재귀 함수 정의: 배열과 정렬할 구간의 시작/끝 인덱스를 받음
    if low < high:  # 구간에 두 개 이상의 원소가 있을 때만 정렬을 수행함
        p = partition(arr, low, high)  # 피벗을 기준으로 분할하고 피벗의 최종 인덱스를 받음
        quicksort(arr, low, p - 1)  # 피벗 왼쪽 부분을 재귀적으로 정렬함
        quicksort(arr, p + 1, high)  # 피벗 오른쪽 부분을 재귀적으로 정렬함

def partition(arr, low, high):  # 분할 함수 정의: 마지막 원소를 피벗으로 사용함 (Lomuto 방식)
    pivot = arr[high]  # 피벗을 배열의 마지막 원소로 정함
    i = low - 1  # 작은 원소들의 끝 인덱스를 추적하기 위해 i를 초기화함
    for j in range(low, high):  # low부터 high-1까지 반복하면서 피벗보다 작은 원소를 찾아 위치를 교환함
        if arr[j] <= pivot:  # 현재 원소가 피벗보다 작거나 같으면
            i += 1  # 작은 원소들의 구간을 확장하기 위해 i를 증가시킴
            arr[i], arr[j] = arr[j], arr[i]  # 현재 원소를 작은 원소들의 다음 위치로 스왑함
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 올바른 위치(i+1)로 이동시킴
    return i + 1  # 피벗의 최종 인덱스를 반환함

if __name__ == "__main__":  # 이 파일을 직접 실행할 때만 아래 예제를 수행하도록 함
    arr = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 예제 배열을 정의함
    quicksort(arr, 0, len(arr) - 1)  # 배열 전체를 퀵소트로 정렬함
    print(arr)  # 정렬된 배열을 출력함
```
