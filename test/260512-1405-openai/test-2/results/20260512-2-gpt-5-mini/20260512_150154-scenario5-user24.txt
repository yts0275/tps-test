아래는 Python으로 구현한 퀵소트(루모토 파티션) 예제입니다. 각 코드 줄마다 설명 주석을 붙였습니다.

```python
def quicksort(arr, low=0, high=None):  # 배열 arr와 정렬할 구간 low..high를 받아 퀵소트 수행하는 함수 정의
    if high is None:  # 호출 시 high가 주어지지 않으면
        high = len(arr) - 1  # high를 배열의 마지막 인덱스로 설정
    if low < high:  # 정렬할 범위에 두 개 이상의 원소가 있을 때만 처리
        p = partition(arr, low, high)  # partition 함수를 호출해 피벗의 최종 위치 인덱스 p를 얻음
        quicksort(arr, low, p - 1)  # 피벗 왼쪽 부분을 재귀적으로 정렬
        quicksort(arr, p + 1, high)  # 피벗 오른쪽 부분을 재귀적으로 정렬

def partition(arr, low, high):  # 루모토(Lomuto) 파티션 방식으로 분할하는 함수 정의
    pivot = arr[high]  # 구간의 마지막 원소를 피벗으로 선택
    i = low - 1  # i는 피벗보다 작거나 같은 요소들의 끝 인덱스를 가리키게 함
    for j in range(low, high):  # low부터 high-1까지 각 요소를 검사
        if arr[j] <= pivot:  # 현재 요소가 피벗 이하이면
            i += 1  # 작은 그룹의 끝을 한 칸 늘림
            arr[i], arr[j] = arr[j], arr[i]  # 현재 요소를 작은 그룹의 끝으로 스왑
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 작은 그룹 바로 뒤 위치로 이동(정렬된 위치)
    return i + 1  # 피벗의 최종 인덱스를 반환

# 사용 예시
arr = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 예시 배열
quicksort(arr)  # 퀵소트 호출하여 arr을 제자리(in-place)로 정렬
print(arr)  # 정렬된 배열 출력
```
