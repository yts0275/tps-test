다음은 각 줄마다 주석을 단 파이썬 퀵소트 구현입니다.

```python
# 퀵소트 구현 (in-place, Lomuto 파티션)을 각 줄마다 주석과 함께 제공합니다.
def quicksort(arr, low=0, high=None):  # 재귀적 in-place 퀵소트 함수 정의, high를 None으로 기본 설정
    if high is None:  # 처음 호출할 때 high가 None이면
        high = len(arr) - 1  # high를 배열의 마지막 인덱스로 설정
    if low < high:  # 정렬할 구간에 원소가 2개 이상 있을 때만 작업 수행
        p = partition(arr, low, high)  # 파티션을 수행해서 피벗의 최종 인덱스를 받음
        quicksort(arr, low, p - 1)  # 피벗 왼쪽 부분을 재귀적으로 정렬
        quicksort(arr, p + 1, high)  # 피벗 오른쪽 부분을 재귀적으로 정렬
    return arr  # (선택적) 정렬된 배열을 반환

def partition(arr, low, high):  # Lomuto 분할법을 사용하는 파티션 함수 정의
    pivot = arr[high]  # 피벗으로 구간의 마지막 원소를 선택
    i = low - 1  # 작은 요소들의 끝 인덱스(i)를 초기화
    for j in range(low, high):  # low부터 high-1까지 각 원소를 검사
        if arr[j] <= pivot:  # 현재 원소가 피벗보다 작거나 같으면
            i += 1  # 작은 그룹의 경계를 한 칸 이동
            arr[i], arr[j] = arr[j], arr[i]  # 작은 그룹 뒤로 현재 원소를 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 작은 그룹과 경계 사이로 이동
    return i + 1  # 피벗의 최종 인덱스를 반환

if __name__ == "__main__":  # 이 파일을 직접 실행할 때만 아래 예제를 수행
    data = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 예제 리스트 정의
    print("정렬 전:", data)  # 정렬 전 리스트 출력
    sorted_data = quicksort(data)  # 퀵소트 호출 (in-place로 정렬되며 반환값도 있음)
    print("정렬 후:", sorted_data)  # 정렬 후 리스트 출력
```
