아래는 파이썬으로 작성한 퀵소트 구현 예제입니다. 각 코드 줄마다 한국어 주석을 추가했습니다.

```python
def quicksort(arr):  # 퀵소트의 공개 함수로, 리스트를 인플레이스(in-place)로 정렬한다.
    def partition(lo, hi):  # 파티션 함수: arr[lo:hi] 구간을 피벗 기준으로 분할하고 피벗의 최종 인덱스를 반환한다.
        pivot = arr[hi]  # 피벗을 구간의 마지막 요소로 선택한다.
        i = lo - 1  # 작은 요소들의 경계 인덱스를 설정한다 (초기값은 lo-1).
        for j in range(lo, hi):  # lo부터 hi-1까지 순회하며 피벗보다 작거나 같은 요소를 찾는다.
            if arr[j] <= pivot:  # 현재 요소가 피벗 이하이면
                i += 1  # 작은 요소들의 경계를 한 칸 오른쪽으로 이동한다.
                arr[i], arr[j] = arr[j], arr[i]  # 경계 위치와 현재 요소를 교환한다.
        arr[i + 1], arr[hi] = arr[hi], arr[i + 1]  # 피벗을 경계 바로 다음 위치로 옮겨 제자리에 둔다.
        return i + 1  # 피벗의 최종 인덱스를 반환한다.

    def _quicksort(lo, hi):  # 재귀적으로 퀵소트를 수행하는 내부 함수이다.
        if lo < hi:  # 구간에 두 개 이상의 요소가 있을 때만 정렬을 수행한다.
            p = partition(lo, hi)  # 파티션을 수행하여 피벗의 최종 위치를 얻는다.
            _quicksort(lo, p - 1)  # 피벗의 왼쪽 부분을 재귀적으로 정렬한다.
            _quicksort(p + 1, hi)  # 피벗의 오른쪽 부분을 재귀적으로 정렬한다.

    _quicksort(0, len(arr) - 1)  # 전체 배열 범위에 대해 내부 퀵소트 함수를 호출한다.

# 사용 예제
if __name__ == "__main__":  # 스크립트로 직접 실행할 때만 아래 코드를 수행한다.
    data = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 예제 리스트를 정의한다.
    print("정렬 전:", data)  # 정렬 전 배열 상태를 출력한다.
    quicksort(data)  # 퀵소트 함수로 리스트를 정렬한다.
    print("정렬 후:", data)  # 정렬된 배열을 출력한다.
```
