아래는 각 줄마다 한글 주석을 단 파이썬 퀵소트 구현입니다.

```python
def quicksort(arr):  # 입력 리스트 arr를 제자리(in-place)로 정렬하는 최상위 함수 정의
    def _quicksort(a, low, high):  # 내부적으로 재귀 호출을 수행하는 헬퍼 함수 정의
        if low < high:  # 분할할 구간이 1개 이상 원소를 포함하는지 확인
            p = partition(a, low, high)  # 피벗 기준으로 배열을 분할하고 피벗의 최종 인덱스를 받음
            _quicksort(a, low, p-1)  # 피벗 왼쪽 부분 배열을 재귀적으로 정렬
            _quicksort(a, p+1, high)  # 피벗 오른쪽 부분 배열을 재귀적으로 정렬
    def partition(a, low, high):  # Lomuto 분할 방식으로 피벗 위치를 결정하는 함수 정의
        pivot = a[high]  # 배열의 마지막 요소를 피벗으로 선택
        i = low - 1  # 작은 요소들의 경계를 나타내는 인덱스를 초기화
        for j in range(low, high):  # 피벗을 제외한 요소들을 순회
            if a[j] <= pivot:  # 현재 요소가 피벗 이하이면
                i += 1  # 작은 요소 영역의 경계를 한 칸 오른쪽으로 이동
                a[i], a[j] = a[j], a[i]  # 현재 요소를 작은 영역 끝으로 스왑
        a[i+1], a[high] = a[high], a[i+1]  # 피벗을 작은 요소들 바로 다음 위치로 옮김
        return i+1  # 피벗의 최종 인덱스를 반환
    _quicksort(arr, 0, len(arr)-1)  # 전체 범위(0부터 len-1)로 퀵소트 시작
    return arr  # 정렬된 동일 리스트를 반환

if __name__ == "__main__":  # 이 파일을 직접 실행할 때만 아래 코드가 실행되도록 함
    data = [3, 6, 8, 10, 1, 2, 1]  # 예시로 사용할 정렬 전 리스트를 정의
    print("before:", data)  # 정렬 전 상태 출력
    print("after:", quicksort(data))  # 퀵소트를 호출하고 정렬 후 결과를 출력
```
