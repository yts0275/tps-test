아래는 파이썬으로 구현한 퀵소트이며, 각 줄마다 한국어 주석을 달아 설명을 넣었습니다.

```python
import random  # 랜덤 피벗 선택을 위해 random 모듈을 불러옵니다

def quicksort(arr):  # 외부에서 호출할 수 있는 퀵소트 함수(배열을 제자리에서 정렬함)
    if arr is None:  # 입력이 None일 경우 아무 작업도 하지 않도록 처리합니다
        return  # None 입력이면 즉시 반환합니다
    _quicksort(arr, 0, len(arr) - 1)  # 내부 재귀 함수로 전체 범위(0~len-1)를 정렬하도록 호출합니다

def _quicksort(arr, low, high):  # 실제 재귀 정렬을 수행하는 내부 함수(부분 배열 [low..high] 정렬)
    if low < high:  # 정렬할 구간에 원소가 2개 이상(즉 low < high)일 때만 작업합니다
        pivot_index = _partition(arr, low, high)  # 분할 과정을 수행하고 피벗의 최종 인덱스를 얻습니다
        _quicksort(arr, low, pivot_index - 1)  # 피벗보다 작은 부분 배열을 재귀적으로 정렬합니다
        _quicksort(arr, pivot_index + 1, high)  # 피벗보다 큰 부분 배열을 재귀적으로 정렬합니다

def _partition(arr, low, high):  # Lomuto 분할 방식을 사용하는 파티션 함수(피벗을 배열 끝으로 옮겨 처리)
    pivot_index = random.randint(low, high)  # low~high 범위에서 임의의 피벗 인덱스를 선택합니다
    arr[pivot_index], arr[high] = arr[high], arr[pivot_index]  # 선택한 피벗을 배열의 끝(high)으로 이동시킵니다
    pivot = arr[high]  # 이제 배열 끝에 있는 값을 피벗 값으로 사용합니다
    i = low  # i는 피벗보다 작은 구간의 경계(다음 작은 원소가 들어갈 위치)를 가리킵니다
    for j in range(low, high):  # low부터 high-1까지 순회하며 피벗과 비교합니다
        if arr[j] < pivot:  # 현재 원소가 피벗보다 작으면
            arr[i], arr[j] = arr[j], arr[i]  # 작은 구간의 끝으로 원소를 옮기기 위해 교환합니다
            i += 1  # 작은 구간의 경계를 한 칸 오른쪽으로 이동합니다
    arr[i], arr[high] = arr[high], arr[i]  # 피벗을 작은 구간의 바로 다음 위치로 옮겨 최종 위치를 확보합니다
    return i  # 피벗의 최종 인덱스를 반환합니다

if __name__ == "__main__":  # 이 파일을 직접 실행할 때만 아래 예시 코드가 동작하도록 합니다
    data = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 예시 배열을 정의합니다
    print("정렬 전:", data)  # 정렬 전 배열 상태를 출력합니다
    quicksort(data)  # 퀵소트 함수로 배열을 정렬합니다(제자리 정렬)
    print("정렬 후:", data)  # 정렬 후 배열 상태를 출력합니다
```
