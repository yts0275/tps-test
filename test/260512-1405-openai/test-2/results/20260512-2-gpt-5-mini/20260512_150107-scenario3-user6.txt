아래는 in-place Lomuto 분할법과 무작위 피벗을 사용하는 퀵소트 구현입니다; 각 코드 줄마다 한국어로 주석을 달아두었습니다.

```python
import random  # 난수 생성을 위해 random 모듈을 불러옵니다.
def quicksort(arr):  # 리스트를 받아서 정렬된 같은 리스트를 반환하는 함수 정의입니다.
    def _quicksort(lo, hi):  # 재귀적으로 부분 배열을 정렬하는 내부 함수입니다.
        if lo < hi:  # 정렬할 구간에 원소가 2개 이상일 때만 정렬을 진행합니다.
            p = partition(lo, hi)  # 분할을 수행해 피벗의 최종 위치를 얻습니다.
            _quicksort(lo, p - 1)  # 피벗 왼쪽 부분을 재귀적으로 정렬합니다.
            _quicksort(p + 1, hi)  # 피벗 오른쪽 부분을 재귀적으로 정렬합니다.
    def partition(lo, hi):  # Lomuto 분할을 사용해 구간을 분할하는 함수입니다.
        pivot_index = random.randint(lo, hi)  # 구간 [lo, hi]에서 무작위로 피벗 인덱스를 선택합니다.
        arr[pivot_index], arr[hi] = arr[hi], arr[pivot_index]  # 선택한 피벗을 구간의 끝(hi)으로 보냅니다.
        pivot = arr[hi]  # 피벗 값을 변수에 저장합니다.
        i = lo  # 작은 원소들이 모일 위치의 시작 인덱스를 설정합니다.
        for j in range(lo, hi):  # lo부터 hi-1까지 반복하며 피벗과 비교합니다.
            if arr[j] <= pivot:  # 현재 원소가 피벗 이하이면
                arr[i], arr[j] = arr[j], arr[i]  # 작은 그룹의 끝에 현재 원소를 넣기 위해 교환합니다.
                i += 1  # 작은 그룹의 경계 인덱스를 하나 증가시킵니다.
        arr[i], arr[hi] = arr[hi], arr[i]  # 피벗을 작은 그룹의 바로 다음 위치로 옮겨 최종 위치로 만듭니다.
        return i  # 피벗의 최종 인덱스를 반환합니다.
    _quicksort(0, len(arr) - 1)  # 전체 배열을 정렬하도록 내부 함수를 호출합니다.
    return arr  # 정렬된 배열(같은 객체)을 반환합니다.

# 사용 예시
if __name__ == "__main__":  # 이 파일을 직접 실행할 때만 아래 코드를 실행합니다.
    data = [3, 6, 8, 10, 1, 2, 1]  # 예시로 정렬할 리스트를 정의합니다.
    print("정렬 전:", data)  # 정렬 전 리스트를 출력합니다.
    sorted_data = quicksort(data)  # 퀵소트로 정렬을 수행합니다.
    print("정렬 후:", sorted_data)  # 정렬 결과를 출력합니다.
```
