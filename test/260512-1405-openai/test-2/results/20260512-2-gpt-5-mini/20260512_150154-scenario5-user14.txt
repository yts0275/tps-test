다음은 Python으로 구현한 퀵소트입니다. 각 코드 줄마다 한국어 주석을 달았습니다.

```python
def quicksort(arr, low, high):  # 퀵소트 함수 정의, arr: 정렬할 리스트, low/high: 정렬할 구간의 시작/끝 인덱스
    if low < high:  # 부분 배열의 크기가 1보다 클 때만 재귀로 정렬을 수행
        p = partition(arr, low, high)  # 분할하여 피벗의 최종 위치 인덱스를 받음
        quicksort(arr, low, p-1)  # 피벗 왼쪽 부분 배열을 재귀적으로 정렬
        quicksort(arr, p+1, high)  # 피벗 오른쪽 부분 배열을 재귀적으로 정렬

def partition(arr, low, high):  # Lomuto 분할 방식으로 파티셔닝하는 함수 정의
    pivot = arr[high]  # 피벗으로 구간의 마지막 요소를 선택
    i = low - 1  # 더 작은 요소들의 끝을 가리키는 포인터를 초기화
    for j in range(low, high):  # low부터 high-1까지 순회하며 피벗과 비교
        if arr[j] <= pivot:  # 현재 요소가 피벗 이하이면
            i += 1  # 더 작은 영역을 하나 확장
            arr[i], arr[j] = arr[j], arr[i]  # 현재 요소를 더 작은 영역의 끝으로 교환
    arr[i+1], arr[high] = arr[high], arr[i+1]  # 피벗을 올바른 위치로 이동시킴
    return i+1  # 피벗의 최종 인덱스를 반환

if __name__ == "__main__":  # 이 파일이 직접 실행될 때만 아래 예시 코드 실행
    data = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 예시 리스트
    print("정렬 전:", data)  # 정렬 전 리스트 출력
    quicksort(data, 0, len(data)-1)  # 퀵소트 호출로 전체 리스트를 정렬
    print("정렬 후:", data)  # 정렬된 리스트 출력
```
