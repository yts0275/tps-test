아래는 파이썬으로 구현한 퀵소트(내부 정렬, Lomuto 분할)를 각 줄마다 한글 주석을 단 코드입니다.

```python
def quicksort(arr, low, high):                       # 퀵소트 재귀 함수: arr 배열의 low..high 범위를 정렬한다
    if low < high:                                  # 정렬할 범위가 유효한지(원소가 2개 이상인지) 확인한다
        p = partition(arr, low, high)               # 분할함수로 피벗의 최종 위치 인덱스를 얻는다
        quicksort(arr, low, p - 1)                  # 피벗 왼쪽 부분을 재귀적으로 정렬한다
        quicksort(arr, p + 1, high)                 # 피벗 오른쪽 부분을 재귀적으로 정렬한다

def partition(arr, low, high):                      # Lomuto 분할: 마지막 원소를 피벗으로 사용한다
    pivot = arr[high]                               # 피벗 값을 배열의 마지막 원소로 정한다
    i = low                                         # 작은 원소들을 쌓아둘 경계 인덱스를 초기화한다
    for j in range(low, high):                      # low부터 high-1까지 순회하며 피벗과 비교한다
        if arr[j] <= pivot:                         # 현재 원소가 피벗 이하이면
            arr[i], arr[j] = arr[j], arr[i]         # 현재 원소를 경계 위치로 스왑한다
            i += 1                                  # 경계 인덱스를 오른쪽으로 이동시킨다
    arr[i], arr[high] = arr[high], arr[i]           # 피벗을 경계 위치로 옮겨 제자리를 만든다
    return i                                        # 피벗의 최종 인덱스를 반환한다

def quicksort_inplace(arr):                         # 사용 편의를 위한 래퍼 함수: 전체 배열 정렬
    quicksort(arr, 0, len(arr) - 1)                 # 전체 범위(0..len-1)를 대상으로 퀵소트를 수행한다

if __name__ == "__main__":                         # 직접 실행할 때만 아래 예제를 수행한다
    data = [3, 6, 8, 10, 1, 2, 1]                  # 예시로 정렬할 리스트를 정의한다
    print("원본:", data)                            # 정렬 전 원본을 출력한다
    quicksort_inplace(data)                         # 리스트를 제자리(in-place)에서 정렬한다
    print("정렬:", data)                            # 정렬 결과를 출력한다
```
