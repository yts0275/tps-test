다음은 Python으로 구현한 퀵소트 예제입니다. 요청하신 대로 각 줄마다 한 줄씩 주석을 달았습니다.

```python
def quicksort(arr, low, high):  # 퀵소트 함수 정의: arr는 리스트, low와 high는 정렬할 구간의 시작과 끝 인덱스
    if low < high:  # 구간의 크기가 1보다 크면(정렬할 필요가 있으면)
        p = partition(arr, low, high)  # 분할 작업을 수행하여 피벗의 최종 위치 인덱스를 받음
        quicksort(arr, low, p-1)  # 피벗 왼쪽 부분을 재귀적으로 정렬
        quicksort(arr, p+1, high)  # 피벗 오른쪽 부분을 재귀적으로 정렬

def partition(arr, low, high):  # 분할 함수 정의: 피벗을 기준으로 작은 요소와 큰 요소로 나눔
    pivot = arr[high]  # 피벗을 구간의 마지막 요소로 선택
    i = low - 1  # 작은 요소들의 경계를 나타내는 인덱스를 초기화
    for j in range(low, high):  # low부터 high-1까지의 모든 요소를 순회
        if arr[j] <= pivot:  # 현재 요소가 피벗보다 작거나 같으면
            i += 1  # 작은 요소 영역의 경계 인덱스를 한 칸 이동
            arr[i], arr[j] = arr[j], arr[i]  # 현재 요소를 작은 영역의 끝으로 교환
    arr[i+1], arr[high] = arr[high], arr[i+1]  # 피벗을 작은 영역 바로 다음 위치로 이동(제자리에 놓음)
    return i+1  # 피벗의 최종 위치 인덱스를 반환

if __name__ == "__main__":  # 이 파일을 직접 실행할 때만 아래 예제를 실행
    data = [33, 10, 55, 71, 29, 5]  # 정렬할 예시 리스트를 정의
    print("정렬 전:", data)  # 정렬 전 리스트를 출력
    quicksort(data, 0, len(data)-1)  # 리스트 전체에 대해 퀵소트 호출
    print("정렬 후:", data)  # 정렬 결과를 출력
```
