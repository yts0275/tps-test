아래는 각 줄마다 한국어 주석을 단 파이썬 퀵소트 구현입니다.

```python
# 퀵소트 함수 정의: 리스트를 받아 정렬된 새 리스트를 반환합니다.
def quicksort(arr):  # 함수 시그니처: 정렬할 리스트 arr를 인자로 받습니다.
    if len(arr) <= 1:  # 길이가 0 또는 1이면 이미 정렬된 상태입니다.
        return arr  # 그대로 반환하여 재귀 종료 조건을 만족합니다.
    pivot = arr[len(arr) // 2]  # 피벗은 리스트 중앙의 원소로 선택합니다.
    left = [x for x in arr if x < pivot]  # 피벗보다 작은 원소들을 모아 left 리스트를 만듭니다.
    middle = [x for x in arr if x == pivot]  # 피벗과 같은 원소들을 모아 middle 리스트를 만듭니다.
    right = [x for x in arr if x > pivot]  # 피벗보다 큰 원소들을 모아 right 리스트를 만듭니다.
    return quicksort(left) + middle + quicksort(right)  # 재귀로 left와 right를 정렬해 합쳐 반환합니다.

# 아래는 함수 동작 예시입니다.
if __name__ == "__main__":  # 이 파일을 직접 실행할 때만 아래 코드가 실행되도록 합니다.
    data = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 예시 리스트를 정의합니다.
    print(quicksort(data))  # 정렬 결과를 출력합니다.
```
