아래는 파이썬으로 구현한 퀵소트이며, 각 줄마다 한국어 주석을 달았습니다.

```python
def quicksort(arr, low, high):  # 배열 arr의 인덱스 low부터 high까지를 퀵소트로 정렬하는 함수
    if low < high:  # 정렬할 구간에 원소가 2개 이상 있는지 확인
        p = partition(arr, low, high)  # 파티션을 나누어 피벗의 최종 위치를 얻음
        quicksort(arr, low, p-1)  # 피벗 왼쪽 부분을 재귀적으로 정렬
        quicksort(arr, p+1, high)  # 피벗 오른쪽 부분을 재귀적으로 정렬

# 함수와 함수 사이 구분 주석
def partition(arr, low, high):  # Lomuto 방식 파티션 함수: 마지막 원소를 피벗으로 사용
    pivot = arr[high]  # 피벗을 배열의 마지막 원소로 설정
    i = low - 1  # 작은 원소들의 경계 인덱스를 초기화
    for j in range(low, high):  # low부터 high-1까지 모든 원소를 순회
        if arr[j] <= pivot:  # 현재 원소가 피벗 이하이면
            i += 1  # 작은 원소 영역을 확장
            arr[i], arr[j] = arr[j], arr[i]  # 현재 원소를 작은 영역의 끝으로 교환
    arr[i+1], arr[high] = arr[high], arr[i+1]  # 피벗을 중간 위치로 이동시켜 분할 완성
    return i+1  # 피벗의 최종 인덱스를 반환

# 사용 예시 시작 주석
arr = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 배열을 정의
quicksort(arr, 0, len(arr) - 1)  # 배열 전체 범위를 퀵소트로 정렬
print(arr)  # 정렬된 배열을 출력
```
