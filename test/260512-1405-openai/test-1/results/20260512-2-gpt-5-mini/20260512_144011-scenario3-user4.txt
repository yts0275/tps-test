아래는 파이썬으로 구현한 퀵소트(Lomuto 분할)입니다. 코드의 각 줄마다 한국어 주석을 달았습니다.

```python
# 퀵소트 구현 (Lomuto 분할 방식) - 각 줄에 주석 포함
def partition(arr, low, high):  # partition 함수 정의: 배열과 부분 범위 인덱스 받음
    pivot = arr[high]  # 피벗으로 범위의 마지막 원소 선택
    i = low - 1  # 작은 원소들의 경계 인덱스를 low-1로 초기화
    for j in range(low, high):  # low부터 high-1까지 각 원소를 검사
        if arr[j] <= pivot:  # 현재 원소가 피벗 이하이면
            i += 1  # 작은 원소 경계를 한 칸 오른쪽으로 이동
            arr[i], arr[j] = arr[j], arr[i]  # 경계 위치와 현재 원소를 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 올바른 위치(i+1)로 교환
    return i + 1  # 피벗의 최종 인덱스 반환

def quicksort(arr, low, high):  # 퀵소트 재귀 함수 정의: 배열과 부분 범위 인덱스 받음
    if low < high:  # 재귀를 계속할 조건: 범위에 두 개 이상의 원소가 있으면
        p = partition(arr, low, high)  # 분할 수행 후 피벗의 최종 인덱스 받음
        quicksort(arr, low, p - 1)  # 피벗 왼쪽 부분을 재귀적으로 정렬
        quicksort(arr, p + 1, high)  # 피벗 오른쪽 부분을 재귀적으로 정렬

def quicksort_inplace(arr):  # 외부에서 호출하기 쉬운 래퍼 함수 정의
    quicksort(arr, 0, len(arr) - 1)  # 전체 배열(0부터 len-1) 범위로 퀵소트 호출

if __name__ == "__main__":  # 이 파일을 직접 실행할 때만 아래 코드 실행
    data = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 예시 리스트 정의
    print("정렬 전:", data)  # 정렬 전 리스트 출력
    quicksort_inplace(data)  # 제자리 정렬 수행
    print("정렬 후:", data)  # 정렬 후 결과 출력
```
