아래는 파이썬으로 구현한 퀵소트(제자리 정렬) 코드입니다. 각 코드 줄마다 한국어 주석을 추가했습니다.

```python
import random  # 무작위 피벗 선택을 위해 random 모듈을 가져옵니다.
def quicksort(arr):  # 리스트 arr을 제자리에서 퀵소트로 정렬하는 공개 함수입니다.
    _quicksort(arr, 0, len(arr) - 1)  # 내부 재귀 함수로 전체 구간(0부터 len-1) 정렬을 시작합니다.
def _quicksort(arr, low, high):  # 배열의 부분구간 [low, high]를 재귀적으로 정렬하는 보조 함수입니다.
    if low < high:  # 구간에 원소가 2개 이상 있을 때만 분할 및 재귀를 수행합니다.
        p = partition(arr, low, high)  # 피벗을 기준으로 분할하여 피벗의 최종 인덱스를 반환받습니다.
        _quicksort(arr, low, p - 1)  # 피벗의 왼쪽 부분구간을 재귀적으로 정렬합니다.
        _quicksort(arr, p + 1, high)  # 피벗의 오른쪽 부분구간을 재귀적으로 정렬합니다。
def partition(arr, low, high):  # 배열의 구간 [low, high]를 분할하고 피벗의 최종 인덱스를 반환합니다.
    pivot_index = random.randint(low, high)  # 성능의 최악 사례를 줄이기 위해 구간 내에서 무작위 피벗 인덱스를 선택합니다.
    arr[pivot_index], arr[high] = arr[high], arr[pivot_index]  # 선택한 피벗을 구간의 끝(high)으로 이동시킵니다.
    pivot = arr[high]  # 끝에 옮긴 피벗 값을 pivot 변수에 저장합니다.
    i = low - 1  # 작은 원소들의 끝을 가리키는 포인터를 초기화합니다(초기값은 low-1).
    for j in range(low, high):  # low부터 high-1까지 순회하며 피벗보다 작거나 같은 원소를 왼쪽으로 모읍니다.
        if arr[j] <= pivot:  # 현재 원소가 피벗보다 작거나 같으면
            i += 1  # 작은 원소들의 끝 포인터를 한 칸 이동합니다.
            arr[i], arr[j] = arr[j], arr[i]  # 현재 원소를 작은 원소들 영역의 다음 위치로 교환합니다.
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 작은 원소들 바로 다음 위치로 옮겨 제자리를 만듭니다.
    return i + 1  # 피벗의 최종 인덱스를 반환합니다.
if __name__ == "__main__":  # 이 파일을 직접 실행할 때만 아래 예시 코드를 수행합니다.
    arr = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 예시 리스트를 정의합니다.
    print("정렬 전:", arr)  # 정렬 전 상태를 출력합니다.
    quicksort(arr)  # 정의한 퀵소트로 리스트를 제자리에서 정렬합니다.
    print("정렬 후:", arr)  # 정렬 후 결과를 출력합니다.
```
