아래는 파이썬으로 구현한 퀵소트 예시입니다. 각 코드 줄에 한국어 주석을 붙였습니다.

```python
# 파이썬으로 구현한 퀵소트(재귀, 중앙 피벗 선택) 예제입니다.
def quicksort(arr):  # arr 리스트를 입력받아 정렬된 새 리스트를 반환하는 함수 정의
    if len(arr) <= 1:  # 리스트 길이가 0 또는 1이면 이미 정렬되어 있다고 판단
        return arr  # 그대로 반환하여 재귀 종료
    pivot = arr[len(arr) // 2]  # 피벗을 리스트의 가운데 원소로 선택
    left = [x for x in arr if x < pivot]  # 피벗보다 작은 원소들로 이루어진 리스트 생성
    middle = [x for x in arr if x == pivot]  # 피벗과 같은 원소들을 모은 리스트 생성
    right = [x for x in arr if x > pivot]  # 피벗보다 큰 원소들로 이루어진 리스트 생성
    return quicksort(left) + middle + quicksort(right)  # 왼쪽과 오른쪽을 재귀 정렬한 뒤 합쳐서 반환

if __name__ == "__main__":  # 이 파일이 직접 실행될 때만 아래 코드 블록을 실행
    data = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 예제 리스트 정의
    print("원본:", data)  # 원본 리스트 출력
    sorted_data = quicksort(data)  # 퀵소트 함수 호출하여 정렬된 리스트 받기
    print("정렬:", sorted_data)  # 정렬 결과 출력
```
