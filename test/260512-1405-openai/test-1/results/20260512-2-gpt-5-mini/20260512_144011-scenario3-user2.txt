아래는 로뮤토(Lomuto) 파티션을 사용한 퀵소트 구현이며, 코드의 각 줄에 주석을 달아 놓았습니다.

```python
def quicksort(arr, low, high):  # 퀵소트 재귀 함수: arr의 low..high 인덱스를 정렬한다
    if low < high:  # 정렬할 구간이 2개 이상 원소를 포함하는지 검사한다
        pi = partition(arr, low, high)  # 파티션을 수행해 피벗의 최종 위치 인덱스를 얻는다
        quicksort(arr, low, pi - 1)  # 피벗의 왼쪽 부분을 재귀적으로 정렬한다
        quicksort(arr, pi + 1, high)  # 피벗의 오른쪽 부분을 재귀적으로 정렬한다

def partition(arr, low, high):  # 로뮤토(Lomuto) 파티션 함수: 마지막 원소를 피벗으로 사용한다
    pivot = arr[high]  # 피벗 값을 배열의 마지막 원소로 선택한다
    i = low - 1  # i는 피벗보다 작은 원소들의 마지막 인덱스를 가리키도록 초기화한다
    for j in range(low, high):  # low부터 high-1까지 순회하며 요소들을 검사한다
        if arr[j] <= pivot:  # 현재 요소가 피벗 이하이면 작은 영역에 포함시킨다
            i += 1  # 작은 영역의 끝 인덱스를 증가시킨다
            arr[i], arr[j] = arr[j], arr[i]  # 현재 요소를 작은 영역의 끝으로 스왑한다
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 작은 영역의 다음 위치로 옮긴다
    return i + 1  # 피벗의 최종 인덱스를 반환한다

def quicksort_inplace(arr):  # 사용 편의를 위한 래퍼 함수: 전체 배열을 정렬한다
    quicksort(arr, 0, len(arr) - 1)  # 전체 범위로 재귀 퀵소트를 호출한다

if __name__ == "__main__":  # 스크립트를 직접 실행할 때만 아래 예제를 실행한다
    a = [3, 6, 8, 10, 1, 2, 1]  # 예제 배열을 정의한다
    print("정렬 전:", a)  # 정렬 전 상태를 출력한다
    quicksort_inplace(a)  # 배열을 제자리에서 정렬한다
    print("정렬 후:", a)  # 정렬된 결과를 출력한다
```
