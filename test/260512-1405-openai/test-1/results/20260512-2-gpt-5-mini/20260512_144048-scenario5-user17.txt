다음은 파이썬으로 구현한 퀵소트이며, 각 코드 줄마다 한글 주석을 추가했습니다.

```python
import random  # 랜덤 피벗 선택을 위해 random 모듈 임포트
def partition(arr, low, high):  # 배열 arr의 인덱스 low~high 구간을 분할하는 함수 정의
    pivot_index = random.randint(low, high)  # low~high 사이에서 무작위로 피벗 인덱스 선택
    arr[pivot_index], arr[high] = arr[high], arr[pivot_index]  # 선택한 피벗을 구간의 끝(high)으로 이동
    pivot = arr[high]  # 피벗 값 저장
    i = low - 1  # 작은 값들의 경계 인덱스를 low-1로 초기화
    for j in range(low, high):  # low부터 high-1까지 순회하면서 요소를 피벗과 비교
        if arr[j] <= pivot:  # 현재 요소가 피벗보다 작거나 같으면
            i += 1  # 작은 값 영역의 끝을 한 칸 늘림
            arr[i], arr[j] = arr[j], arr[i]  # 현재 요소를 작은 값 영역의 뒤로 이동(교환)
    arr[i+1], arr[high] = arr[high], arr[i+1]  # 피벗을 작은 값 영역의 바로 다음 위치로 이동
    return i+1  # 피벗의 최종 위치 인덱스 반환
def quicksort(arr, low, high):  # 퀵소트 재귀 함수 정의(부분 배열 arr[low..high] 정렬)
    if low < high:  # 구간에 두 개 이상의 원소가 있는 경우에만 정렬 수행
        p = partition(arr, low, high)  # 분할하여 피벗의 최종 위치 p를 얻음
        quicksort(arr, low, p-1)  # 피벗 왼쪽 부분 배열 재귀 정렬
        quicksort(arr, p+1, high)  # 피벗 오른쪽 부분 배열 재귀 정렬
if __name__ == "__main__":  # 이 파일을 직접 실행할 때만 아래 예시 코드 실행
    data = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 예시 리스트 정의
    print("정렬 전:", data)  # 정렬 전 상태 출력
    quicksort(data, 0, len(data) - 1)  # 전체 배열에 대해 퀵소트 호출
    print("정렬 후:", data)  # 정렬 후 상태 출력
```
