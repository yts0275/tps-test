아래는 각 줄마다 한국어 주석을 단 파이썬 퀵소트 구현 예시입니다.

```python
def quicksort(arr):  # 퀵소트 함수 정의: 리스트 arr을 받아 정렬된 새 리스트를 반환
    if len(arr) <= 1:  # 리스트 길이가 0 또는 1이면 이미 정렬되어 있음
        return arr  # 그대로 반환하여 재귀 종료
    pivot = arr[len(arr) // 2]  # 피벗을 리스트의 중간 원소로 선택
    left = [x for x in arr if x < pivot]  # 피벗보다 작은 원소들을 left에 모음
    middle = [x for x in arr if x == pivot]  # 피벗과 같은 원소들을 middle에 모음
    right = [x for x in arr if x > pivot]  # 피벗보다 큰 원소들을 right에 모음
    return quicksort(left) + middle + quicksort(right)  # 좌우를 재귀 정렬한 뒤 합쳐서 반환

if __name__ == "__main__":  # 이 파일을 직접 실행할 때만 아래 코드 실행
    data = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 예시 리스트
    print(quicksort(data))  # 정렬 결과를 출력하여 함수 동작 확인
```
