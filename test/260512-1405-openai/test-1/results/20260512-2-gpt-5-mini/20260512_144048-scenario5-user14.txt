아래는 각 코드 줄마다 주석을 단 파이썬 퀵소트 구현입니다.

```python
def quicksort(arr, low, high):                       # 퀵소트를 수행하는 재귀 함수 정의: arr는 리스트, low와 high는 현재 정렬 범위의 인덱스
    if low < high:                                  # 정렬할 구간에 두 개 이상의 원소가 있을 때만 처리
        pivot = partition(arr, low, high)           # 분할을 수행하여 피벗의 최종 위치를 얻음
        quicksort(arr, low, pivot - 1)              # 피벗 왼쪽 부분을 재귀적으로 정렬
        quicksort(arr, pivot + 1, high)             # 피벗 오른쪽 부분을 재귀적으로 정렬

def partition(arr, low, high):                      # 분할 함수 정의: 마지막 원소를 피벗으로 사용
    pivot = arr[high]                               # 피벗 값을 구함 (여기서는 arr[high])
    i = low - 1                                     # 작은 요소들의 경계 인덱스 초기화
    for j in range(low, high):                      # low에서 high-1까지 순회하며 피벗보다 작거나 같은 요소 찾기
        if arr[j] <= pivot:                         # 현재 요소가 피벗보다 작거나 같으면
            i += 1                                  # 경계 인덱스를 하나 증가시키고
            arr[i], arr[j] = arr[j], arr[i]         # arr[i]와 arr[j]를 교환하여 작은 요소를 왼쪽으로 모음
    arr[i + 1], arr[high] = arr[high], arr[i + 1]   # 피벗을 경계 바로 다음 위치로 옮겨서 분할 완료
    return i + 1                                    # 피벗의 최종 위치 인덱스를 반환

def quicksort_inplace(arr):                         # 사용자 편의를 위한 래퍼 함수 정의 (리스트를 제자리 정렬)
    quicksort(arr, 0, len(arr) - 1)                 # 전체 리스트 범위를 지정하여 퀵소트 호출

# 사용 예시
if __name__ == "__main__":                          # 이 파일을 직접 실행할 때만 아래 코드가 동작하도록 함
    data = [3, 6, 8, 10, 1, 2, 1]                    # 정렬할 예시 리스트 정의
    print("정렬 전:", data)                          # 정렬 전 상태 출력
    quicksort_inplace(data)                          # 제자리 퀵소트 수행
    print("정렬 후:", data)                          # 정렬 후 결과 출력
```
