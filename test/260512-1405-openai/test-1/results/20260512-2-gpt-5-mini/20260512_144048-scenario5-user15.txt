아래는 Python으로 구현한 퀵소트 예제이며, 각 코드 줄마다 한국어 주석을 추가했습니다.

```python
def quicksort(arr, low, high):  # 퀵소트 함수 선언: 정렬할 리스트와 정렬 범위 인덱스 받음
    if low < high:  # 정렬할 구간이 유효한지(원소가 2개 이상인지) 검사
        p = partition(arr, low, high)  # 분할 함수 호출로 피벗의 최종 위치 받음
        quicksort(arr, low, p - 1)  # 피벗 왼쪽 부분을 재귀적으로 정렬
        quicksort(arr, p + 1, high)  # 피벗 오른쪽 부분을 재귀적으로 정렬

def partition(arr, low, high):  # 분할 함수 선언(Lomuto 분할 사용)
    pivot = arr[high]  # 피벗으로 구간의 마지막 원소 선택
    i = low - 1  # 작은 원소들의 경계 인덱스를 초기화
    for j in range(low, high):  # low부터 high-1까지 각 원소를 검사
        if arr[j] <= pivot:  # 현재 원소가 피벗 이하이면
            i += 1  # 경계 인덱스를 한 칸 오른쪽으로 이동
            arr[i], arr[j] = arr[j], arr[i]  # 경계 오른쪽으로 원소를 교환하여 작은 그룹에 포함
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 경계 다음 위치로 옮겨 제자리에 둠
    return i + 1  # 피벗의 최종 인덱스 반환

if __name__ == "__main__":  # 이 파일을 직접 실행할 때만 아래 코드 실행
    arr = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 예시 리스트 정의
    quicksort(arr, 0, len(arr) - 1)  # 리스트 전체 범위를 정렬하도록 퀵소트 호출
    print(arr)  # 정렬된 리스트를 출력
```
