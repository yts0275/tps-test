다음은 Python으로 구현한 퀵소트이며, 각 줄마다 한국어 주석을 달아 설명한 코드입니다.

```python
def quicksort(arr, low, high):  # 배열 arr와 정렬할 구간의 시작 인덱스 low, 끝 인덱스 high를 받는 함수 정의
    if low < high:  # 정렬할 구간에 원소가 2개 이상 있어야만 정렬을 진행함을 확인
        p = partition(arr, low, high)  # 피벗을 기준으로 분할하고 피벗의 최종 위치 인덱스를 받음
        quicksort(arr, low, p - 1)  # 피벗의 왼쪽 부분을 재귀적으로 정렬함
        quicksort(arr, p + 1, high)  # 피벗의 오른쪽 부분을 재귀적으로 정렬함

def partition(arr, low, high):  # Lomuto 분할 방식을 사용하는 분할 함수 정의
    pivot = arr[high]  # 구간의 마지막 원소를 피벗으로 선택함
    i = low - 1  # 더 작은 원소들의 끝 위치를 나타내는 인덱스를 초기화함
    for j in range(low, high):  # low부터 high-1까지 순회하면서 피벗과 비교함
        if arr[j] <= pivot:  # 현재 원소가 피벗 이하이면
            i += 1  # 더 작은 원소들의 끝 위치를 하나 증가시킴
            arr[i], arr[j] = arr[j], arr[i]  # 현재 원소를 더 작은 원소들 구간의 다음 위치로 교환함
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 올바른 위치인 i+1로 이동시킴
    return i + 1  # 피벗의 최종 인덱스를 반환함

# 사용 예시:
arr = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 샘플 배열을 정의함
quicksort(arr, 0, len(arr) - 1)  # 배열 전체를 정렬하기 위해 시작 인덱스 0과 끝 인덱스 len(arr)-1을 전달함
print(arr)  # 정렬된 배열을 출력함
```
