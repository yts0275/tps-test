아래는 파이썬으로 구현한 퀵소트(람모토 분할) 예제이며, 코드의 각 줄에 한국어 주석을 달아 설명했습니다.

```python
def partition(arr, low, high):  # partition 함수 정의: 배열과 시작/끝 인덱스를 받음
    pivot = arr[high]  # 피벗을 배열의 마지막 원소로 선택함
    i = low - 1  # 작은 원소들의 경계 인덱스를 초기화함
    for j in range(low, high):  # low부터 high-1까지 반복하여 피벗보다 작은 원소 찾음
        if arr[j] <= pivot:  # 현재 원소가 피벗보다 작거나 같으면
            i += 1  # 경계 인덱스를 오른쪽으로 한 칸 이동함
            arr[i], arr[j] = arr[j], arr[i]  # 작은 원소를 경계 오른쪽으로 교환함
    arr[i+1], arr[high] = arr[high], arr[i+1]  # 피벗을 경계 바로 다음 위치로 옮겨 제자리를 만듦
    return i + 1  # 피벗의 최종 위치 인덱스를 반환함

def quicksort(arr, low, high):  # 퀵소트 재귀 함수 정의: 배열과 정렬할 구간의 시작/끝 인덱스
    if low < high:  # 구간에 두 개 이상의 원소가 있을 때만 정렬 수행
        p = partition(arr, low, high)  # 분할을 수행하여 피벗의 최종 위치를 얻음
        quicksort(arr, low, p - 1)  # 피벗 왼쪽 부분을 재귀적으로 정렬함
        quicksort(arr, p + 1, high)  # 피벗 오른쪽 부분을 재귀적으로 정렬함

if __name__ == "__main__":  # 이 파일이 직접 실행될 때만 아래 코드를 수행함
    data = [33, 10, 55, 71, 29, 3, 18, 92, 44]  # 정렬할 예제 배열을 정의함
    print("정렬 전:", data)  # 정렬 전 배열 상태를 출력함
    quicksort(data, 0, len(data) - 1)  # 배열 전체를 대상으로 퀵소트를 호출함
    print("정렬 후:", data)  # 정렬이 완료된 배열을 출력함
```
