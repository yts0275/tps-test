아래는 파이썬으로 구현한 퀵소트이며, 각 줄마다 한국어 주석을 달아 설명했습니다.

```python
def quicksort(arr, low=0, high=None):  # 퀵소트 함수 정의, 배열과 정렬할 구간(low~high)을 인자로 받음
    if high is None: high = len(arr) - 1  # high가 주어지지 않으면 배열의 마지막 인덱스로 설정함
    if low < high:  # 정렬할 구간에 두 개 이상의 원소가 있을 때만 분할·정복 수행함
        p = partition(arr, low, high)  # 피벗을 기준으로 분할하고 피벗의 최종 위치 인덱스를 얻음
        quicksort(arr, low, p - 1)  # 피벗의 왼쪽 부분 배열에 대해 재귀적으로 퀵소트 수행함
        quicksort(arr, p + 1, high)  # 피벳의 오른쪽 부분 배열에 대해 재귀적으로 퀵소트 수행함
    return arr  # 정렬된(또는 부분적으로 정렬된) 배열을 반환함

def partition(arr, low, high):  # 분할 작업을 수행하는 함수, Lomuto 파티셔닝 사용
    pivot = arr[high]  # 배열의 마지막 원소를 피벗으로 선택함
    i = low - 1  # 작은 값들의 경계 인덱스를 초기화함 (작은 값들의 마지막 인덱스)
    for j in range(low, high):  # low부터 high-1까지 순회하면서 피벗보다 작은 원소를 찾음
        if arr[j] <= pivot:  # 현재 원소가 피벗보다 작거나 같으면
            i += 1  # 경계 인덱스를 하나 증가시킴
            arr[i], arr[j] = arr[j], arr[i]  # 현재 원소를 작은 값들의 영역 끝으로 스왑함
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 작은 값들의 바로 다음 위치로 옮겨 최종 위치로 함
    return i + 1  # 피벗의 최종 위치 인덱스를 반환함

if __name__ == "__main__":  # 이 스크립트를 직접 실행할 때만 아래 코드를 수행함
    a = [3, 6, 8, 10, 1, 2, 1]  # 예시로 정렬할 리스트를 정의함
    print("정렬 전:", a)  # 정렬 전 상태를 출력함
    quicksort(a)  # 리스트 a를 인플레이스(in-place)로 정렬함
    print("정렬 후:", a)  # 정렬 후 상태를 출력함
```
