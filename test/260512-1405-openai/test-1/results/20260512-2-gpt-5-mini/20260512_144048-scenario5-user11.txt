아래는 Python으로 구현한 퀵소트 예제입니다. 각 코드 줄 끝에 한국어 주석을 달았습니다.

```python
def partition(arr, low, high):  # 부분 배열을 분할하는 함수 정의 (피벗 기준으로)
    pivot = arr[high]  # 피벗을 부분 배열의 마지막 요소로 선택
    i = low - 1  # 피벗보다 작은 요소들의 경계 인덱스 초기화
    for j in range(low, high):  # 부분 배열의 처음부터 피벗 바로 전까지 순회
        if arr[j] <= pivot:  # 현재 요소가 피벗보다 작거나 같으면
            i += 1  # 작은 요소들의 경계를 한 칸 오른쪽으로 이동
            arr[i], arr[j] = arr[j], arr[i]  # 현재 요소와 경계 위치 요소를 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 경계 바로 다음 위치로 이동시켜 분할 완료
    return i + 1  # 피벗의 최종 인덱스를 반환


def quicksort(arr, low, high):  # 퀵소트 재귀 함수 정의 (배열과 정렬할 구간 인덱스 전달)
    if low < high:  # 구간에 두 개 이상의 요소가 있을 때만 정렬 수행
        pi = partition(arr, low, high)  # 분할 수행 후 피벗의 인덱스 얻기
        quicksort(arr, low, pi - 1)  # 피벗 왼쪽 부분 배열에 대해 재귀적으로 정렬
        quicksort(arr, pi + 1, high)  # 피벗 오른쪽 부분 배열에 대해 재귀적으로 정렬


if __name__ == "__main__":  # 이 파일을 직접 실행할 때만 아래 코드 실행
    arr = [33, 10, 55, 71, 29, 3, 18, 42]  # 예시로 정렬할 리스트 정의
    print("정렬 전:", arr)  # 정렬 전 리스트 출력
    quicksort(arr, 0, len(arr) - 1)  # 리스트 전체에 대해 퀵소트 호출
    print("정렬 후:", arr)  # 정렬 후 리스트 출력
```
