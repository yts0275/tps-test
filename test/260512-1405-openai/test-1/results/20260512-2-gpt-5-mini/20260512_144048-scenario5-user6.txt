아래는 파이썬으로 구현한 퀵소트 코드입니다. 각 코드 줄마다 한국어 주석을 추가했습니다.

```python
def quick_sort(arr):  # 배열을 제자리(in-place)로 정렬하는 외부 함수(래퍼)
    quick_sort_recursive(arr, 0, len(arr) - 1)  # 전체 범위를 재귀 정렬 함수에 전달하여 정렬 시작

def quick_sort_recursive(arr, low, high):  # low부터 high까지의 구간을 재귀적으로 정렬하는 함수
    if low < high:  # 정렬할 구간이 1개 이하이면 더 이상 할 필요 없음
        pivot_index = partition(arr, low, high)  # 분할 함수로 피벗 위치를 얻음
        quick_sort_recursive(arr, low, pivot_index - 1)  # 피벗 왼쪽 구간을 재귀적으로 정렬
        quick_sort_recursive(arr, pivot_index + 1, high)  # 피벗 오른쪽 구간을 재귀적으로 정렬

def partition(arr, low, high):  # 라무토(Lomuto) 분할 방식을 사용하는 함수
    pivot = arr[high]  # 구간의 마지막 요소를 피벗으로 선택
    i = low - 1  # 작은 요소들의 경계 인덱스(i)를 초기화
    for j in range(low, high):  # low부터 high-1까지 각 요소를 피벗과 비교
        if arr[j] <= pivot:  # 현재 요소가 피벗 이하이면
            i += 1  # 작은 요소들의 경계를 오른쪽으로 이동
            arr[i], arr[j] = arr[j], arr[i]  # 현재 요소를 경계 오른쪽(작은 영역)으로 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 경계 다음 위치로 옮겨 올바른 위치에 둠
    return i + 1  # 피벗의 최종 인덱스를 반환

if __name__ == "__main__":  # 이 파일을 직접 실행했을 때만 아래 코드가 실행되도록 함
    data = [3, 6, 8, 10, 1, 2, 1]  # 예시 정렬 대상 배열을 정의
    quick_sort(data)  # 퀵소트 함수 호출로 배열을 정렬
    print(data)  # 정렬된 배열을 출력
```
