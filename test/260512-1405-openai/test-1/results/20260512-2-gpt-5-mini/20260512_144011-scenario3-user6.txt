아래는 파이썬으로 구현한 in-place 퀵소트(Lomuto 분할)를 각 줄마다 한국어 주석을 단 코드입니다.

```python
def quicksort(arr, low=0, high=None):  # 퀵소트 함수 정의(기본적으로 전체 배열을 정렬하도록 low와 high를 받음)
    if high is None:  # high가 주어지지 않았으면
        high = len(arr) - 1  # 배열의 마지막 인덱스를 high로 설정
    if low < high:  # 정렬할 부분이 최소 두 원소 이상일 때만 처리
        p = partition(arr, low, high)  # 분할함수를 호출해 피벗의 최종 위치를 얻음
        quicksort(arr, low, p - 1)  # 피벗 왼쪽 부분을 재귀적으로 정렬
        quicksort(arr, p + 1, high)  # 피벗 오른쪽 부분을 재귀적으로 정렬
    return arr  # 정렬된 배열(같은 객체)을 반환


def partition(arr, low, high):  # Lomuto 분할 알고리즘을 사용하는 분할 함수 정의
    pivot = arr[high]  # 피벗을 구간의 마지막 원소로 선택
    i = low - 1  # 작은 원소들의 경계 인덱스를 초기화
    for j in range(low, high):  # low부터 high-1까지 반복하면서
        if arr[j] <= pivot:  # 현재 원소가 피벗보다 작거나 같으면
            i += 1  # 작은 원소들의 경계를 한 칸 오른쪽으로 이동
            arr[i], arr[j] = arr[j], arr[i]  # 현재 원소를 작은 구간의 끝으로 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 작은 구간의 바로 다음 위치로 옮김
    return i + 1  # 피벗의 최종 인덱스를 반환


if __name__ == "__main__":  # 이 파일을 직접 실행했을 때만 아래 코드 실행
    arr = [3, 6, 8, 10, 1, 2, 1]  # 예시로 정렬할 배열을 정의
    print("정렬 전:", arr)  # 정렬 전 배열 출력
    sorted_arr = quicksort(arr)  # 퀵소트 호출(제자리 정렬이므로 arr 자체가 바뀜)
    print("정렬 후:", sorted_arr)  # 정렬 후 배열 출력
```
