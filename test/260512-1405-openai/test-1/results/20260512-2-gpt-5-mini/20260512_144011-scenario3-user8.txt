아래는 각 줄마다 주석을 단 Python 퀵소트 구현입니다.

```python
def quicksort(arr, low, high):  # 재귀적 퀵소트 함수 정의: arr의 [low..high] 구간을 정렬
    if low < high:  # 구간에 두 개 이상의 원소가 있을 때만 정렬을 수행
        p = partition(arr, low, high)  # 분할 연산으로 피벗의 최종 위치 p 를 얻음
        quicksort(arr, low, p-1)  # 피벗 왼쪽 부분을 재귀적으로 정렬
        quicksort(arr, p+1, high)  # 피벗 오른쪽 부분을 재귀적으로 정렬
def partition(arr, low, high):  # Lomuto 분할 규칙을 사용하는 분할 함수 정의
    pivot = arr[high]  # 피벗을 구간의 마지막 원소로 선택
    i = low - 1  # 작은 원소들의 경계 인덱스 i 초기화
    for j in range(low, high):  # low부터 high-1까지 각 원소를 검사
        if arr[j] <= pivot:  # 현재 원소가 피벗 이하이면
            i += 1  # 경계 인덱스 증가
            arr[i], arr[j] = arr[j], arr[i]  # 현재 원소를 왼쪽 부분으로 교환
    arr[i+1], arr[high] = arr[high], arr[i+1]  # 피벗을 경계 오른쪽으로 교환하여 제자리로 함
    return i+1  # 피벗의 최종 위치 인덱스 반환
def quicksort_inplace(arr):  # 사용 편의를 위한 호출용 래퍼 함수 정의
    quicksort(arr, 0, len(arr)-1)  # 전체 배열을 정렬하도록 내부 함수 호출
if __name__ == "__main__":  # 스크립트로 직접 실행될 때만 아래 코드 실행
    a = [3, 6, 8, 10, 1, 2, 1]  # 예시 배열 정의
    quicksort_inplace(a)  # 배열을 제자리(in-place)로 정렬
    print(a)  # 정렬 결과 출력
```
