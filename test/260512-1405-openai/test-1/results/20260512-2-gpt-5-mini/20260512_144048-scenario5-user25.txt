다음은 Python으로 구현한 퀵소트이며, 각 줄마다 한국어 주석을 달아두었습니다.

def quicksort(arr, low=0, high=None):  # 퀵소트 함수 정의: 배열 arr와 정렬할 범위 [low, high]
    if high is None:  # high가 지정되지 않았으면
        high = len(arr) - 1  # high를 배열의 마지막 인덱스로 설정
    if low < high:  # 정렬할 요소가 2개 이상일 때만 진행
        p = partition(arr, low, high)  # 분할을 수행하고 피벗의 최종 위치 p를 받음
        quicksort(arr, low, p - 1)  # 피벗 왼쪽 부분 재귀 정렬
        quicksort(arr, p + 1, high)  # 피벗 오른쪽 부분 재귀 정렬
    return arr  # 정렬된 배열을 반환

def partition(arr, low, high):  # 파티션 함수 정의: 고정 피벗 방식(마지막 요소를 피벗)
    pivot = arr[high]  # 피벗으로 배열의 마지막 요소 선택
    i = low - 1  # 작은 요소들의 경계를 나타내는 인덱스 i 초기화
    for j in range(low, high):  # low부터 high-1까지 각 요소를 검사
        if arr[j] <= pivot:  # 현재 요소가 피벗 이하이면
            i += 1  # 작은 요소들의 경계를 한 칸 오른쪽으로 이동
            arr[i], arr[j] = arr[j], arr[i]  # 현재 요소를 작은 그룹 뒤로 이동(교환)
        # 위 조건이 아니면 아무 작업도 하지 않음
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 경계 바로 다음 위치로 옮겨 중앙에 둠
    return i + 1  # 피벗의 최종 인덱스를 반환

if __name__ == "__main__":  # 이 파일을 직접 실행할 때만 아래 코드 실행
    data = [3, 6, 8, 10, 1, 2, 1]  # 예제 입력 배열
    print("정렬 전:", data)  # 정렬 전 상태 출력
    sorted_data = quicksort(data)  # 배열을 퀵소트로 정렬
    print("정렬 후:", sorted_data)  # 정렬 후 결과 출력
