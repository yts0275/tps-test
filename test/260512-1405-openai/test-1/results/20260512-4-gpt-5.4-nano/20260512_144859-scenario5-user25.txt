아래는 **퀵소트(Quick Sort)** 구현 예시(파이썬)입니다. **각 줄마다 주석**을 달았습니다.

```python
def quicksort(arr):  # 퀵소트 함수를 정의
    if len(arr) <= 1:  # 원소가 1개 이하이면 이미 정렬됨
        return arr  # 그대로 반환

    pivot = arr[len(arr) // 2]  # 가운데 원소를 피벗으로 선택
    left = []  # 피벗보다 작은 값들을 담을 리스트
    middle = []  # 피벗과 같은 값들을 담을 리스트
    right = []  # 피벗보다 큰 값들을 담을 리스트

    for x in arr:  # 배열의 모든 원소를 순회
        if x < pivot:  # 현재 원소가 피벗보다 작다면
            left.append(x)  # left에 추가
        elif x > pivot:  # 현재 원소가 피벗보다 크다면
            right.append(x)  # right에 추가
        else:  # x == pivot 인 경우
            middle.append(x)  # middle에 추가

    return quicksort(left) + middle + quicksort(right)  # 왼쪽 정렬 + 피벗들 + 오른쪽 정렬 결과 결합


# 예시 실행
data = [3, 6, 8, 10, 1, 2, 1]  # 테스트용 배열 생성
print(quicksort(data))  # 퀵소트를 수행하고 결과 출력
```
