아래는 **퀵소트(Quick Sort)** 구현 예시(파이썬)입니다. **코드의 각 줄마다 주석**을 달았습니다.

```python
def quicksort(arr, low, high):  # 퀵소트 함수를 정의한다 (배열, 시작 인덱스, 끝 인덱스를 받는다)
    if low >= high:  # 부분 배열의 크기가 0 또는 1이면 정렬할 필요가 없으므로 종료한다
        return  # 재귀 종료

    pivot = arr[high]  # 피벗을 부분 배열의 마지막 값으로 정한다
    i = low - 1  # 피벗보다 작은 값의 끝 위치(경계)를 가리키는 인덱스를 초기화한다

    for j in range(low, high):  # low부터 high-1까지 순회하며 피벗과 비교한다
        if arr[j] <= pivot:  # 현재 값이 피벗보다 작거나 같으면
            i += 1  # 작은 값 구역의 끝을 1칸 확장한다
            arr[i], arr[j] = arr[j], arr[i]  # 작은 값들을 앞으로 모이도록 교환한다

    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 i+1 위치로 보내며 정렬된 기준을 만든다
    pi = i + 1  # 피벗의 최종 위치를 pi에 저장한다

    quicksort(arr, low, pi - 1)  # 피벗보다 작은 구간을 재귀적으로 정렬한다
    quicksort(arr, pi + 1, high)  # 피벗보다 큰 구간을 재귀적으로 정렬한다


# 예시 실행을 위한 코드
data = [10, 7, 8, 9, 1, 5]  # 정렬할 배열을 준비한다
quicksort(data, 0, len(data) - 1)  # 배열 전체를 퀵소트로 정렬한다
print(data)  # 정렬 결과를 출력한다
```
