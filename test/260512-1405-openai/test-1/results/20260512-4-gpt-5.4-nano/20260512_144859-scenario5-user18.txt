아래는 **퀵소트(Quick Sort) 구현 예시**입니다. 요청하신 대로 **각 줄마다 주석**을 달았습니다.

```python
def quick_sort(arr):  # 퀵소트 함수를 정의합니다. (입력: arr)
    if len(arr) <= 1:  # 배열의 길이가 0 또는 1이면(원소가 0개/1개) 더 이상 정렬할 필요가 없습니다.
        return arr  # 정렬된 배열(그대로)을 반환합니다.

    pivot = arr[len(arr) // 2]  # 가운데 원소를 피벗(pivot)으로 선택합니다.
    left = []  # 피벗보다 작은 원소들을 담을 리스트를 만듭니다.
    right = []  # 피벗보다 큰 원소들을 담을 리스트를 만듭니다.
    equal = []  # 피벗과 같은 원소들을 담을 리스트를 만듭니다.

    for x in arr:  # 배열의 각 원소 x에 대해 반복합니다.
        if x < pivot:  # x가 피벗보다 작으면
            left.append(x)  # left에 x를 추가합니다.
        elif x > pivot:  # x가 피벗보다 크면
            right.append(x)  # right에 x를 추가합니다.
        else:  # x가 피벗과 같으면
            equal.append(x)  # equal에 x를 추가합니다.

    return quick_sort(left) + equal + quick_sort(right)  # 재귀로 정렬한 뒤 left + equal + right를 합쳐 반환합니다.


# 사용 예시: 정렬할 배열을 준비합니다.
data = [3, 6, 8, 10, 1, 2, 1]  # 테스트를 위한 정수 배열입니다.

# 퀵소트를 수행하고 결과를 출력합니다.
print(quick_sort(data))  # 정렬 결과를 출력합니다.
```
