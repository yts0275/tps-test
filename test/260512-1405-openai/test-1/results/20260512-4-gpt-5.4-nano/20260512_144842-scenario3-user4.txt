아래는 **퀵소트(Quick Sort) 구현 예시**입니다. 요청하신 대로 **각 줄마다 주석**을 달았습니다. (파이썬)

```python
def quick_sort(arr, low=0, high=None):  # 퀵소트 함수 정의(배열, 시작 인덱스, 끝 인덱스)
    if high is None:  # high를 지정하지 않았을 때의 기본값 처리
        high = len(arr) - 1  # 배열 마지막 인덱스로 설정

    if low >= high:  # 원소가 0개 또는 1개면 정렬할 필요가 없음(종료 조건)
        return arr  # 정렬된 배열 반환

    pivot = arr[high]  # 피벗을 배열의 마지막 원소로 선택
    i = low - 1  # 피벗보다 작은 값들의 마지막 위치를 가리킬 인덱스

    for j in range(low, high):  # low부터 high-1까지 피벗 전 구간을 순회
        if arr[j] <= pivot:  # 현재 값이 피벗보다 작거나 같으면
            i += 1  # 작은 값 구역을 한 칸 확장
            arr[i], arr[j] = arr[j], arr[i]  # i와 j 위치의 값을 교환(파티션 수행)

    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 최종 위치로 이동
    pivot_index = i + 1  # 피벗의 최종 인덱스 저장

    quick_sort(arr, low, pivot_index - 1)  # 피벗 왼쪽(작은 값들) 재귀 정렬
    quick_sort(arr, pivot_index + 1, high)  # 피벗 오른쪽(큰 값들) 재귀 정렬

    return arr  # 정렬 결과 반환


# 예시 실행
data = [10, 7, 8, 9, 1, 5]  # 정렬할 배열
sorted_data = quick_sort(data)  # 퀵소트 실행
print(sorted_data)  # 정렬된 결과 출력
```
