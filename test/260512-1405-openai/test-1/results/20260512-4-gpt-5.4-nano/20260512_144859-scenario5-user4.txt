아래는 **퀵소트(Quick Sort) 구현 코드**이며, **각 줄마다 주석**을 달았습니다. (파이썬 예시)

```python
def quick_sort(arr):  # 퀵소트 함수를 정의한다 (입력: arr, 출력: 정렬된 새 리스트)
    if len(arr) <= 1:  # 원소가 0개 또는 1개면 이미 정렬된 상태이므로 그대로 반환한다
        return arr  # 정렬된 리스트(원본)를 반환한다

    pivot = arr[len(arr) // 2]  # 가운데 원소를 피벗(pivot)으로 선택한다
    left = []  # 피벗보다 작은 값들을 담을 리스트를 만든다
    right = []  # 피벗보다 큰 값들을 담을 리스트를 만든다
    equal = []  # 피벗과 같은 값들을 담을 리스트를 만든다

    for x in arr:  # 입력 배열의 각 원소 x에 대해 반복한다
        if x < pivot:  # x가 피벗보다 작으면
            left.append(x)  # left 리스트에 x를 추가한다
        elif x > pivot:  # x가 피벗보다 크면
            right.append(x)  # right 리스트에 x를 추가한다
        else:  # x가 피벗과 같으면
            equal.append(x)  # equal 리스트에 x를 추가한다

    return quick_sort(left) + equal + quick_sort(right)  # 재귀적으로 정렬한 결과를 합쳐 반환한다
    # 왼쪽(작은 값들) + 피벗과 같은 값들 + 오른쪽(큰 값들)의 순서로 정렬된다


# 사용 예시 코드 시작
data = [3, 6, 8, 10, 1, 2, 1]  # 예시로 사용할 숫자 리스트를 준비한다
print(quick_sort(data))  # 퀵소트를 수행한 결과를 출력한다
```
