아래는 **파이썬( Python )**으로 구현한 **퀵소트(Quick Sort)** 예시이며, **각 줄마다 주석**을 달아두었습니다.

```python
def quicksort(arr):                      # 퀵소트 함수를 정의: 입력 배열 arr
    if len(arr) <= 1:                   # 배열 길이가 0 또는 1이면 이미 정렬됨
        return arr                       # 그대로 반환

    pivot = arr[len(arr) // 2]         # 기준값(pivot)을 배열의 가운데 값으로 선택
    left = []                           # pivot보다 작은 값들을 담을 리스트
    middle = []                         # pivot과 같은 값들을 담을 리스트
    right = []                          # pivot보다 큰 값들을 담을 리스트

    for x in arr:                       # 배열의 각 원소 x에 대해 반복
        if x < pivot:                   # x가 pivot보다 작으면
            left.append(x)              # left 리스트에 추가
        elif x > pivot:                 # x가 pivot보다 크면
            right.append(x)            # right 리스트에 추가
        else:                           # x가 pivot과 같으면
            middle.append(x)          # middle 리스트에 추가

    return quicksort(left) + middle + quicksort(right)  # 재귀로 정렬 후 합쳐 반환


# 사용 예시: 아래 코드를 실행하면 정렬 결과를 확인할 수 있습니다.
if __name__ == "__main__":              # main 실행 여부를 확인
    data = [3, 6, 8, 10, 1, 2, 1]      # 정렬할 예시 데이터
    print(quicksort(data))              # 퀵소트 결과 출력
```
