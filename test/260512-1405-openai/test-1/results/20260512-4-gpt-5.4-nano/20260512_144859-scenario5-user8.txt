아래는 **퀵소트(Quick Sort)** 구현 예시(파이썬)와 **각 줄마다 주석**을 단 코드입니다.

```python
def quicksort(arr):  # 퀵소트 함수(배열 arr)를 받아 정렬된 새 배열을 반환
    if len(arr) <= 1:  # 원소가 0개 또는 1개면 이미 정렬되어 있으므로 그대로 반환
        return arr  # 그대로 반환

    pivot = arr[len(arr) // 2]  # 가운데 원소를 피벗(pivot)으로 선택
    left = []  # 피벗보다 작은 값들을 담을 리스트
    middle = []  # 피벗과 같은 값들을 담을 리스트
    right = []  # 피벗보다 큰 값들을 담을 리스트

    for x in arr:  # 배열의 각 원소 x를 순회
        if x < pivot:  # x가 피벗보다 작으면
            left.append(x)  # left에 추가
        elif x > pivot:  # x가 피벗보다 크면
            right.append(x)  # right에 추가
        else:  # x가 피벗과 같으면
            middle.append(x)  # middle에 추가

    return quicksort(left) + middle + quicksort(right)  # 왼쪽/중간/오른쪽을 정렬 후 합쳐 반환

# 예시 실행
data = [3, 6, 8, 10, 1, 2, 1]  # 테스트용 입력 배열
result = quicksort(data)  # 퀵소트 실행하여 정렬 결과를 result에 저장
print(result)  # 정렬된 배열 출력
```
