아래는 **퀵소트(Quick Sort) 구현 예시(파이썬)**이고, **각 줄마다 주석**을 달아두었습니다.

```python
def quick_sort(arr, left, right):  # 퀵소트 함수 정의 (배열, 왼쪽 인덱스, 오른쪽 인덱스)
    if left >= right:  # 부분 배열의 크기가 1 이하이면 정렬할 것이 없음(종료 조건)
        return  # 종료

    pivot = arr[(left + right) // 2]  # 피벗을 가운데 값으로 선택
    i = left  # i를 왼쪽 포인터로 설정
    j = right  # j를 오른쪽 포인터로 설정

    while i <= j:  # 포인터 i와 j가 교차할 때까지 반복
        while arr[i] < pivot:  # i가 가리키는 값이 피벗보다 작으면 i를 오른쪽으로 이동
            i += 1  # i 증가

        while arr[j] > pivot:  # j가 가리키는 값이 피벗보다 크면 j를 왼쪽으로 이동
            j -= 1  # j 감소

        if i <= j:  # i가 j보다 작거나 같으면 교환이 필요
            arr[i], arr[j] = arr[j], arr[i]  # i와 j 위치의 값을 교환
            i += 1  # 교환 후 i를 다음 위치로 이동
            j -= 1  # 교환 후 j를 다음 위치로 이동

    quick_sort(arr, left, j)  # 왼쪽 파티션을 재귀적으로 정렬
    quick_sort(arr, i, right)  # 오른쪽 파티션을 재귀적으로 정렬


# 테스트용 코드: 배열 준비
data = [10, 7, 8, 9, 1, 5]  # 예시 데이터
quick_sort(data, 0, len(data) - 1)  # 전체 범위에 대해 퀵소트 수행
print(data)  # 정렬된 결과 출력
```
