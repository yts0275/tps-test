아래는 **퀵소트(Quick Sort)** 구현 예시이며, **각 줄마다 주석**을 달았습니다(파이썬).

```python
def quicksort(arr):  # 퀵소트 함수 정의(리스트 arr을 정렬)
    if len(arr) <= 1:  # 원소가 0개 또는 1개면 더 이상 정렬할 필요 없음
        return arr  # 그대로 반환

    pivot = arr[len(arr) // 2]  # 가운데 원소를 피벗(pivot)으로 선택
    left = []  # 피벗보다 작은 원소들을 담을 리스트
    middle = []  # 피벗과 같은 원소들을 담을 리스트
    right = []  # 피벗보다 큰 원소들을 담을 리스트

    for x in arr:  # 원소 x를 arr에서 하나씩 꺼내 반복
        if x < pivot:  # x가 피벗보다 작으면
            left.append(x)  # left에 추가
        elif x == pivot:  # x가 피벗과 같으면
            middle.append(x)  # middle에 추가
        else:  # x가 피벗보다 크면
            right.append(x)  # right에 추가

    return quicksort(left) + middle + quicksort(right)  # 재귀로 left/right 정렬 후 합치고, middle을 가운데에 둠


# 테스트 코드(원하면 삭제해도 됨)
data = [3, 6, 8, 10, 1, 2, 1]  # 예시 입력 리스트
sorted_data = quicksort(data)  # 퀵소트 실행
print(sorted_data)  # 정렬 결과 출력
```
