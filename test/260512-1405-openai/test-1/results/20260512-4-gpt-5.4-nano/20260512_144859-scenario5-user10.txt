아래는 **퀵소트(Quick Sort)**를 파이썬으로 구현한 예시이며, **각 줄마다 주석**을 달았습니다.

```python
def quicksort(arr, low=0, high=None):  # 퀵소트 함수 정의 (배열, 시작 인덱스, 끝 인덱스)
    if high is None:  # 호출 시 high가 지정되지 않았다면
        high = len(arr) - 1  # high를 배열의 마지막 인덱스로 설정

    if low >= high:  # 구간 길이가 0 또는 1이면 정렬할 필요가 없음
        return arr  # 정렬 완료이므로 배열 반환

    pivot = arr[high]  # 기준값(pivot)을 구간의 마지막 원소로 선택
    i = low - 1  # pivot보다 작은 값들이 모일 위치를 가리키는 인덱스 설정

    for j in range(low, high):  # low부터 high-1까지 순회
        if arr[j] <= pivot:  # 현재 값이 pivot 이하라면
            i += 1  # 작은 값 영역을 확장하기 위해 i 증가
            arr[i], arr[j] = arr[j], arr[i]  # i 위치와 j 위치의 값을 교환

    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # pivot을 최종 위치로 옮김
    pi = i + 1  # pivot의 최종 인덱스를 pi에 저장

    quicksort(arr, low, pi - 1)  # pivot 왼쪽(작은 값들) 구간 재귀 정렬
    quicksort(arr, pi + 1, high)  # pivot 오른쪽(큰 값들) 구간 재귀 정렬

    return arr  # 정렬된 배열 반환


# 테스트 코드: 정렬 전 배열을 준비
example = [10, 7, 8, 9, 1, 5]  # 예시 입력 배열
print(quicksort(example))  # 퀵소트 실행 후 결과 출력
```
