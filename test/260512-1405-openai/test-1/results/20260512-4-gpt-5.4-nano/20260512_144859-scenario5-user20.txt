```python
def quicksort(arr, low=0, high=None):
    # quicksort를 arr[low:high+1] 구간에 대해 수행하는 함수

    # high를 지정하지 않았으면 배열의 마지막 인덱스를 사용
    if high is None:
        high = len(arr) - 1  # 배열의 마지막 인덱스

    # 구간에 원소가 0개 또는 1개면 정렬할 필요가 없으므로 종료
    if low >= high:  # 종료 조건
        return

    # 피벗(pivot)을 구간의 마지막 원소로 선택
    pivot = arr[high]  # 기준 값(피벗)

    # i는 "피벗보다 작은 값들이 모일 위치"를 추적하는 인덱스
    i = low - 1  # 아직 아무 값도 없으므로 low-1에서 시작

    # j를 low부터 high-1까지 이동시키며 피벗과 비교
    for j in range(low, high):
        # 현재 원소가 피벗보다 작거나 같으면
        if arr[j] <= pivot:  # 피벗 기준 비교
            i += 1  # 작은 값들이 모일 다음 위치로 이동
            arr[i], arr[j] = arr[j], arr[i]  # i와 j의 원소를 교환(자리 변경)

    # 반복이 끝나면 i+1 위치에 피벗을 놓으면 정렬이 분할됨
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 제자리(분할 경계)로 이동

    # 피벗 기준 왼쪽 구간을 재귀적으로 정렬
    quicksort(arr, low, i)  # 피벗보다 작은 부분

    # 피벗 기준 오른쪽 구간을 재귀적으로 정렬
    quicksort(arr, i + 2, high)  # 피벗보다 큰 부분


# 사용 예시
if __name__ == "__main__":
    arr = [10, 7, 8, 9, 1, 5]
    quicksort(arr)  # 원본 배열을 제자리(in-place)로 정렬
    print(arr)  # 정렬된 결과 출력
```
