아래는 **퀵소트(Quick Sort)** 예시 구현입니다. **각 줄마다 주석**을 달아 드렸어요. (파이썬)

```python
def quicksort(arr):  # 퀵소트 함수 정의: 입력 배열 arr을 정렬해 반환
    if len(arr) <= 1:  # 배열 길이가 0 또는 1이면 이미 정렬됨
        return arr  # 정렬된 배열(그 자체)을 그대로 반환
    pivot = arr[len(arr) // 2]  # 기준값(pivot)을 배열의 가운데 값으로 선택
    left = []  # pivot보다 작은 값들을 담을 리스트 생성
    middle = []  # pivot과 같은 값들을 담을 리스트 생성
    right = []  # pivot보다 큰 값들을 담을 리스트 생성

    for x in arr:  # 배열의 각 원소 x를 하나씩 순회
        if x < pivot:  # x가 pivot보다 작으면
            left.append(x)  # left 리스트에 x 추가
        elif x > pivot:  # x가 pivot보다 크면
            right.append(x)  # right 리스트에 x 추가
        else:  # x가 pivot과 같으면
            middle.append(x)  # middle 리스트에 x 추가

    return quicksort(left) + middle + quicksort(right)  # 왼쪽 정렬 + pivot들 + 오른쪽 정렬 결과를 합쳐 반환


# 테스트용 코드 시작
data = [3, 6, 8, 10, 1, 2, 1]  # 예시 입력 배열
sorted_data = quicksort(data)  # quicksort로 정렬 수행
print(sorted_data)  # 정렬된 결과 출력
```
