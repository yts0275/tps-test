아래는 **퀵소트(Quick Sort) 구현 예시**입니다. 요청하신 대로 **각 줄마다 주석을 달았습니다**(보기 좋게 단일 파일 형태로 작성).  

## ✅ Python 구현 (주석: 각 줄마다)
```python
def quicksort(arr, low, high):  # 퀵소트 함수 정의 (배열, 시작 인덱스, 끝 인덱스)
    if low < high:  # 분할 범위가 유효할 때만 진행
        pivot = partition(arr, low, high)  # 파티션을 수행하고 피벗이 들어간 위치를 반환받음
        quicksort(arr, low, pivot - 1)  # 피벗보다 왼쪽 구간을 재귀적으로 정렬
        quicksort(arr, pivot + 1, high)  # 피벗보다 오른쪽 구간을 재귀적으로 정렬

def partition(arr, low, high):  # 파티션 함수 정의 (피벗을 기준으로 정렬)
    pivot = arr[high]  # 피벗을 구간의 마지막 값으로 선택
    i = low - 1  # pivot보다 작은 값이 모일 위치를 위한 인덱스 초기화
    for j in range(low, high):  # low부터 high-1까지 순회
        if arr[j] <= pivot:  # 현재 값이 피벗보다 작거나 같으면
            i += 1  # 작은 값 영역의 다음 위치로 이동
            arr[i], arr[j] = arr[j], arr[i]  # i 위치와 j 위치의 값을 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 올바른 위치로 이동
    return i + 1  # 피벗의 최종 위치를 반환

def quick_sort(arr):  # 외부에서 호출하기 위한 래퍼 함수
    if not arr:  # 배열이 비어있으면 그대로 반환
        return arr  # 빈 배열 반환
    quicksort(arr, 0, len(arr) - 1)  # 전체 범위를 대상으로 퀵소트 수행
    return arr  # 정렬된 배열 반환

# 테스트 코드 시작
data = [3, 6, 8, 10, 1, 2, 1]  # 예시 입력 배열
print("정렬 전:", data)  # 정렬 전 출력
sorted_data = quick_sort(data)  # 퀵소트 수행
print("정렬 후:", sorted_data)  # 정렬 후 출력
```
