아래는 **퀵소트(Quick Sort)** 구현 예시(파이썬)입니다. **각 줄마다 주석**을 달았습니다.

```python
def quicksort(arr):  # 퀵소트 함수를 정의한다 (입력: arr)
    if len(arr) <= 1:  # 원소가 0개 또는 1개면 정렬할 필요가 없으니
        return arr  # 그대로 반환한다

    pivot = arr[len(arr) // 2]  # 가운데 원소를 피벗으로 선택한다
    left = []  # 피벗보다 작은 원소들을 담을 리스트를 만든다
    middle = []  # 피벗과 같은 원소들을 담을 리스트를 만든다
    right = []  # 피벗보다 큰 원소들을 담을 리스트를 만든다

    for x in arr:  # 배열의 각 원소 x에 대해 반복한다
        if x < pivot:  # x가 피벗보다 작으면
            left.append(x)  # left에 x를 추가한다
        elif x > pivot:  # x가 피벗보다 크면
            right.append(x)  # right에 x를 추가한다
        else:  # x가 피벗과 같으면
            middle.append(x)  # middle에 x를 추가한다

    return quicksort(left) + middle + quicksort(right)  # 정렬된 left + middle + 정렬된 right를 합쳐 반환한다


# 동작 예시를 위한 코드
data = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 예시 배열을 만든다
print(quicksort(data))  # 퀵소트를 수행한 결과를 출력한다
```
