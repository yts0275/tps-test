아래는 **퀵소트(Quick Sort)** 구현 예시이며, **각 줄마다 주석**을 달았습니다. (파이썬)

```python
def quicksort(arr):  # 퀵소트 함수 정의: 입력 배열 arr을 정렬해 반환
    if len(arr) <= 1:  # 배열 길이가 0 또는 1이면 이미 정렬된 상태이므로
        return arr  # 정렬된 배열을 그대로 반환
    pivot = arr[len(arr) // 2]  # 기준값(pivot)을 배열의 가운데 원소로 선택
    left = []  # pivot보다 작은 값들을 담을 리스트 준비
    middle = []  # pivot과 같은 값들을 담을 리스트 준비
    right = []  # pivot보다 큰 값들을 담을 리스트 준비
    
    for x in arr:  # 배열의 각 원소 x에 대해 반복
        if x < pivot:  # x가 pivot보다 작으면
            left.append(x)  # left 리스트에 x를 추가
        elif x > pivot:  # x가 pivot보다 크면
            right.append(x)  # right 리스트에 x를 추가
        else:  # x가 pivot과 같으면
            middle.append(x)  # middle 리스트에 x를 추가
    
    return quicksort(left) + middle + quicksort(right)  # 재귀로 정렬한 결과를 합쳐 반환


# 테스트 예시(원하면 지우셔도 됩니다)
data = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 입력 배열
print(quicksort(data))  # 퀵소트를 수행한 정렬 결과 출력
```
