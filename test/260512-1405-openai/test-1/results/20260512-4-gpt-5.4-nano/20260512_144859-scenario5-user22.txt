아래는 **퀵소트(Quick Sort)** 예시 구현(파이썬)이고, **각 줄마다 주석**을 달았습니다.  

```python
def quicksort(arr):  # 입력 리스트 arr을 퀵소트로 정렬하는 함수 정의
    if len(arr) <= 1:  # 원소가 0개 또는 1개면 이미 정렬되어 있으므로 그대로 반환
        return arr  # 정렬된 리스트(원본)를 반환

    pivot = arr[len(arr) // 2]  # 기준(pivot)값을 가운데 원소로 선택
    left = []  # pivot보다 작은 값들을 담을 리스트 생성
    middle = []  # pivot과 같은 값들을 담을 리스트 생성
    right = []  # pivot보다 큰 값들을 담을 리스트 생성

    for x in arr:  # 배열의 각 원소 x에 대해 반복
        if x < pivot:  # x가 pivot보다 작으면
            left.append(x)  # left 리스트에 x를 추가
        elif x > pivot:  # x가 pivot보다 크면
            right.append(x)  # right 리스트에 x를 추가
        else:  # x가 pivot과 같으면
            middle.append(x)  # middle 리스트에 x를 추가

    return quicksort(left) + middle + quicksort(right)  # 정렬된 left + middle + 정렬된 right를 합쳐 반환


# 테스트 코드 (원하면 지우셔도 됩니다)
data = [3, 6, 8, 10, 1, 2, 1]  # 예시 입력 리스트
sorted_data = quicksort(data)  # 퀵소트를 수행해 정렬된 결과를 sorted_data에 저장
print(sorted_data)  # 정렬 결과 출력
```
