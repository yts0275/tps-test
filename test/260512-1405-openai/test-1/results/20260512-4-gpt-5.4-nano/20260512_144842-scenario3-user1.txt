아래는 **퀵소트(Quick Sort)** 구현 예시와 **각 줄마다 주석**을 달아둔 코드입니다. (파이썬)

```python
def quicksort(arr):  # 입력 배열 arr을 퀵소트로 정렬하는 함수 정의
    if len(arr) <= 1:  # 원소가 0개 또는 1개면 이미 정렬 완료이므로
        return arr  # 그대로 반환
    pivot = arr[len(arr) // 2]  # 가운데 원소를 피벗(pivot)으로 선택
    left = []  # 피벗보다 작은 원소들을 담을 리스트 생성
    middle = []  # 피벗과 같은 원소들을 담을 리스트 생성
    right = []  # 피벗보다 큰 원소들을 담을 리스트 생성
    for x in arr:  # 배열의 각 원소 x를 순회
        if x < pivot:  # x가 피벗보다 작으면
            left.append(x)  # left 리스트에 추가
        elif x > pivot:  # x가 피벗보다 크면
            right.append(x)  # right 리스트에 추가
        else:  # x가 피벗과 같으면
            middle.append(x)  # middle 리스트에 추가
    return quicksort(left) + middle + quicksort(right)  # 정렬된 결과(왼쪽 + 피벗들 + 오른쪽) 합쳐서 반환

# 예시 사용: 정렬 전 배열 준비
data = [7, 3, 5, 2, 9, 1, 8, 6, 4]  # 테스트용 숫자 배열
# 예시 출력: 퀵소트 적용 후 결과 출력
print(quicksort(data))  # 정렬된 배열을 출력
```
