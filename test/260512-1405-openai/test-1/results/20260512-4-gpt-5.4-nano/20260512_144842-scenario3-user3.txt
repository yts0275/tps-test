아래는 **퀵소트(Quick Sort) 구현 예시**이며, **모든 줄마다 주석**을 달아두었습니다. (파이썬)

```python
def quicksort(arr):  # 퀵소트를 수행하는 함수(정렬할 리스트 arr을 받음)
    if len(arr) <= 1:  # 원소가 0개 또는 1개면 이미 정렬된 상태이므로 그대로 반환
        return arr  # 그대로 반환

    pivot = arr[len(arr) // 2]  # 중앙 원소를 피벗(pivot)으로 선택
    left = []  # 피벗보다 작은 값들을 담을 리스트
    middle = []  # 피벗과 같은 값들을 담을 리스트
    right = []  # 피벗보다 큰 값들을 담을 리스트

    for x in arr:  # 배열의 각 원소 x에 대해 반복
        if x < pivot:  # x가 피벗보다 작으면
            left.append(x)  # left 리스트에 추가
        elif x > pivot:  # x가 피벗보다 크면
            right.append(x)  # right 리스트에 추가
        else:  # x가 피벗과 같으면
            middle.append(x)  # middle 리스트에 추가

    return quicksort(left) + middle + quicksort(right)  # 왼쪽 정렬 + 피벗과 같은 값 + 오른쪽 정렬을 합쳐 반환


# 사용 예시(테스트용 코드)
data = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 입력 배열
print(quicksort(data))  # 퀵소트 실행 결과 출력
```
