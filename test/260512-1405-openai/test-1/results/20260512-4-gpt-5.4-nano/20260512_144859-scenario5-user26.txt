아래는 **퀵소트(Quicksort)** 구현 예시(파이썬)이고, **각 줄마다 주석**을 달았습니다.

```python
def quicksort(arr):  # 배열을 받아 퀵소트 결과 배열을 반환하는 함수 정의
    if len(arr) <= 1:  # 원소가 1개 이하이면 이미 정렬된 상태이므로
        return arr  # 그대로 반환
    pivot = arr[len(arr) // 2]  # 가운데 원소를 피벗으로 선택
    left = []  # 피벗보다 작은 값들을 담을 리스트
    middle = []  # 피벗과 같은 값들을 담을 리스트
    right = []  # 피벗보다 큰 값들을 담을 리스트
    for x in arr:  # 입력 배열의 각 원소를 순회
        if x < pivot:  # 현재 원소가 피벗보다 작으면
            left.append(x)  # left에 추가
        elif x > pivot:  # 현재 원소가 피벗보다 크면
            right.append(x)  # right에 추가
        else:  # 현재 원소가 피벗과 같으면
            middle.append(x)  # middle에 추가
    return quicksort(left) + middle + quicksort(right)  # 재귀적으로 정렬 후 합쳐서 반환


# 예시 실행
data = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 데이터 리스트
print(quicksort(data))  # 퀵소트 결과를 출력
```
