아래는 **퀵소트(Quick Sort) 구현 예시**입니다.  
요청하신 대로 **각 줄마다 주석**을 달았습니다.

```python
def quicksort(arr, left=0, right=None):  # 퀵소트 함수 정의 (기본 left는 0, right는 미지정 시 전체로 처리)
    if right is None:  # right가 전달되지 않았다면
        right = len(arr) - 1  # right를 배열의 마지막 인덱스로 설정

    if left >= right:  # 부분 배열의 크기가 0 또는 1이면 정렬할 필요가 없으므로 종료
        return  # 함수 종료

    pivot = arr[(left + right) // 2]  # 피벗을 가운데 값으로 선택
    i, j = left, right  # i는 왼쪽 포인터, j는 오른쪽 포인터로 초기화

    while i <= j:  # 포인터 i가 j를 지나치면 분할 작업 종료
        while arr[i] < pivot:  # i가 피벗보다 작은 동안 i를 오른쪽으로 이동
            i += 1  # i 증가
        while arr[j] > pivot:  # j가 피벗보다 큰 동안 j를 왼쪽으로 이동
            j -= 1  # j 감소

        if i <= j:  # i가 j보다 크지 않은 경우(교차 전) 교환 가능
            arr[i], arr[j] = arr[j], arr[i]  # i와 j 위치의 값을 서로 교환
            i += 1  # i를 한 칸 이동
            j -= 1  # j를 한 칸 이동

    quicksort(arr, left, j)  # 왼쪽 파티션을 재귀적으로 정렬
    quicksort(arr, i, right)  # 오른쪽 파티션을 재귀적으로 정렬
    return arr  # 정렬된 배열 반환


# 사용 예시: 배열 생성
data = [10, 7, 8, 9, 1, 5]  # 예제 데이터
print(quicksort(data))  # 퀵소트 수행 후 결과 출력
```
