아래는 **퀵소트(Quick Sort)** 예시 구현입니다. **각 줄마다 주석**을 달았어요. (파이썬 기준)

```python
def quicksort(arr, left=0, right=None):  # 퀵소트를 수행하는 함수(배열과 구간을 받음)
    if right is None:  # right가 지정되지 않았다면
        right = len(arr) - 1  # 배열의 마지막 인덱스를 right로 설정
    if left >= right:  # 구간 길이가 0 또는 1이면(정렬할 필요 없음)
        return  # 재귀 종료(리턴)

    pivot = arr[(left + right) // 2]  # 피벗을 구간의 가운데 값으로 선택
    i = left  # i 포인터를 구간 시작으로 초기화
    j = right  # j 포인터를 구간 끝으로 초기화

    while i <= j:  # i가 j를 넘을 때까지 반복
        while arr[i] < pivot:  # i가 가리키는 값이 피벗보다 작으면
            i += 1  # i를 오른쪽으로 이동
        while arr[j] > pivot:  # j가 가리키는 값이 피벗보다 크면
            j -= 1  # j를 왼쪽으로 이동

        if i <= j:  # i가 j보다 크지 않다면 교환할 수 있는 쌍이 존재
            arr[i], arr[j] = arr[j], arr[i]  # i와 j의 값을 서로 교환
            i += 1  # 교환 후 i를 증가시켜 다음 후보로 이동
            j -= 1  # 교환 후 j를 감소시켜 다음 후보로 이동

    quicksort(arr, left, j)  # 왼쪽 파티션을 재귀적으로 정렬
    quicksort(arr, i, right)  # 오른쪽 파티션을 재귀적으로 정렬
    return arr  # 정렬된 배열을 반환


# 테스트 코드: 예시 배열을 준비
data = [5, 3, 8, 4, 2, 7, 1, 10]  # 정렬할 입력 배열
sorted_data = quicksort(data)  # 퀵소트 실행(원본 배열도 in-place로 변경됨)
print(sorted_data)  # 정렬 결과 출력
```
