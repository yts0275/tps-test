```python
def quicksort(array):  # 배열을 퀵소트로 정렬하는 메인 함수 정의
    if len(array) <= 1:  # 배열 길이가 1 이하이면 이미 정렬된 상태이므로
        return array  # 그대로 반환하여 기저 사례 처리
    pivot = array[len(array) // 2]  # 배열의 중간 값을 피벗으로 선택
    left = [x for x in array if x < pivot]  # 피벗보다 작은 요소들을 왼쪽 리스트에 모음
    middle = [x for x in array if x == pivot]  # 피벗과 같은 요소들만 중간 리스트에 모음
    right = [x for x in array if x > pivot]  # 피벗보다 큰 요소들을 오른쪽 리스트에 모음
    return quicksort(left) + middle + quicksort(right)  # 재귀적으로 왼쪽과 오른쪽을 정렬한 후 병합하여 반환
```
