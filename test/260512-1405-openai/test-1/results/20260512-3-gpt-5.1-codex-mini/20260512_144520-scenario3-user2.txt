```python
def quicksort(arr):  # 퀵소트 함수를 정의합니다
    if len(arr) <= 1:  # 배열 길이가 0이나 1이면 이미 정렬된 상태이므로
        return arr  # 그대로 반환하여 더 이상 분할하지 않습니다
    pivot = arr[len(arr) // 2]  # 피벗으로 배열 중앙값을 선택합니다
    left = [x for x in arr if x < pivot]  # 피벗보다 작은 요소들을 모아 왼쪽 부분 배열을 만듭니다
    middle = [x for x in arr if x == pivot]  # 피벗과 같은 요소들을 모아 중간 부분 배열을 만듭니다
    right = [x for x in arr if x > pivot]  # 피벗보다 큰 요소들을 모아 오른쪽 부분 배열을 만듭니다
    return quicksort(left) + middle + quicksort(right)  # 왼쪽과 오른쪽을 재귀적으로 정렬한 뒤 병합하여 반환합니다
```
