다음은 파이썬으로 구현한 퀵소트입니다. 각 줄마다 주석을 달아 이해를 돕도록 했습니다.

```python
def quicksort(arr):  # 재귀적으로 정렬할 리스트를 받아 퀵소트를 수행하는 함수 정의
    if len(arr) <= 1:  # 리스트의 길이가 1 이하이면 이미 정렬된 상태이므로 그대로 반환
        return arr  # 더 이상 분할할 필요 없으므로 현재 리스트 그대로 반환
    pivot = arr[len(arr) // 2]  # 리스트의 중간 요소를 피벗으로 선택하여 분할 기준으로 사용
    left = [x for x in arr if x < pivot]  # 피벗보다 작은 값들을 왼쪽 서브리스트에 저장
    middle = [x for x in arr if x == pivot]  # 피벗과 동일한 값들을 가운데 서브리스트에 저장
    right = [x for x in arr if x > pivot]  # 피벗보다 큰 값들을 오른쪽 서브리스트에 저장
    return quicksort(left) + middle + quicksort(right)  # 왼쪽과 오른쪽 서브리스트를 재귀적으로 정렬 후 합쳐 반환

# 예시 사용법
data = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 리스트 예시 데이터
sorted_data = quicksort(data)  # 퀵소트 함수 호출로 정렬 수행
print(sorted_data)  # 정렬된 결과 출력
```
