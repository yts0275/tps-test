```python
def quicksort(arr):  # 퀵소트를 수행할 함수를 정의합니다.
    if len(arr) <= 1:  # 배열의 크기가 1 이하이면 이미 정렬되어 있다고 보고 그대로 반환합니다.
        return arr  # 이 경우 그대로 배열을 반환합니다.
    pivot = arr[len(arr) // 2]  # 피벗으로 중간 위치의 값을 선택하여 기준으로 삼습니다.
    left = [x for x in arr if x < pivot]  # 피벗보다 작은 값들은 왼쪽 그룹에 모읍니다.
    middle = [x for x in arr if x == pivot]  # 피벗과 같은 값들은 중앙 그룹에 모읍니다.
    right = [x for x in arr if x > pivot]  # 피벗보다 큰 값들은 오른쪽 그룹에 모읍니다.
    return quicksort(left) + middle + quicksort(right)  # 왼쪽과 오른쪽 그룹을 재귀적으로 정렬한 뒤 합쳐서 반환합니다.
```
