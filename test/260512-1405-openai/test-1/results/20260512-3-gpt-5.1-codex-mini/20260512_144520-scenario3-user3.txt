```python
def quicksort(arr):  # 퀵소트 함수 정의
    if len(arr) <= 1:  # 배열 길이가 1 이하이면 이미 정렬된 상태
        return arr  # 배열 그대로 반환
    pivot = arr[len(arr) // 2]  # 피벗을 배열 중간 값으로 선택
    left = [x for x in arr if x < pivot]  # 피벗보다 작은 값들로 왼쪽 리스트 구성
    middle = [x for x in arr if x == pivot]  # 피벗과 같은 값들로 가운데 리스트 구성
    right = [x for x in arr if x > pivot]  # 피벗보다 큰 값들로 오른쪽 리스트 구성
    return quicksort(left) + middle + quicksort(right)  # 재귀적으로 정렬 결과 합쳐서 반환
```
