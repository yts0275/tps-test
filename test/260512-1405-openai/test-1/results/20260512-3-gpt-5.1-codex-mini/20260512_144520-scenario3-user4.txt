```python
def quicksort(arr):  # 배열을 받아 퀵소트 함수 시작
    if len(arr) <= 1:  # 배열 길이가 1 이하이면 정렬할 필요 없음
        return arr  # 입력 배열 그대로 반환
    pivot = arr[len(arr) // 2]  # 중간 원소를 피벗으로 선택
    left = [x for x in arr if x < pivot]  # 피벗보다 작은 요소들 추출
    middle = [x for x in arr if x == pivot]  # 피벗과 같은 요소들 추출
    right = [x for x in arr if x > pivot]  # 피벗보다 큰 요소들 추출
    return quicksort(left) + middle + quicksort(right)  # 재귀 호출로 정렬 후 결합
```
