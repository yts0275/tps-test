def quicksort(arr):  # 정렬할 리스트를 인자로 받아 빠른 정렬을 수행하는 함수를 정의합니다.
    if len(arr) <= 1:  # 리스트 길이가 1 이하이면 이미 정렬된 상태이므로 그대로 반환합니다.
        return arr  # 정렬된 리스트를 그대로 반환합니다.
    pivot = arr[len(arr) // 2]  # 리스트 중간값을 피벗으로 선택합니다.
    left = [x for x in arr if x < pivot]  # 피벗보다 작은 원소들을 모아 왼쪽 부분을 만듭니다.
    middle = [x for x in arr if x == pivot]  # 피벗과 같은 원소들을 모아 가운데 부분을 만듭니다.
    right = [x for x in arr if x > pivot]  # 피벗보다 큰 원소들을 모아 오른쪽 부분을 만듭니다.
    return quicksort(left) + middle + quicksort(right)  # 분할 정복 방식으로 왼쪽과 오른쪽을 재귀적으로 정렬한 뒤 결과를 합칩니다.
