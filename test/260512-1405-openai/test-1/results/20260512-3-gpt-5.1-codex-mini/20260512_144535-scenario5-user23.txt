def quicksort(arr):  # 퀵소트 함수 정의
    if len(arr) <= 1:  # 배열 길이가 1 이하이면 이미 정렬된 상태이므로
        return arr[:]  # 원본 배열을 변경하지 않도록 복사본을 반환
    pivot = arr[len(arr) // 2]  # 중앙 요소를 피벗으로 선택
    left = [x for x in arr if x < pivot]  # 피벗보다 작은 요소들
    middle = [x for x in arr if x == pivot]  # 피벗과 같은 요소들
    right = [x for x in arr if x > pivot]  # 피벗보다 큰 요소들
    return quicksort(left) + middle + quicksort(right)  # 재귀적으로 정렬 후 합침

sample = [3, 6, 1, 8, 4, 5]  # 예시 배열 선언
print(quicksort(sample))  # 정렬 결과 출력
