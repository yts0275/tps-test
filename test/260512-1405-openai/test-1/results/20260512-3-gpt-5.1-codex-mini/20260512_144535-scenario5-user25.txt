아래는 각 줄마다 주석을 단 퀵소트 구현 예시입니다.

```python
def quicksort(arr):  # 정렬할 배열을 입력받는 퀵소트 함수 선언
    if len(arr) <= 1:  # 배열 길이가 1 이하이면 더 이상 나눌 필요가 없음
        return arr  # 그대로 배열을 반환하여 재귀 종료
    pivot = arr[len(arr) // 2]  # 배열 중앙 요소를 피벗으로 선택
    left = [x for x in arr if x < pivot]  # 피벗보다 작은 값들을 모아서 왼쪽 하위 배열 생성
    middle = [x for x in arr if x == pivot]  # 피벗과 같은 값들을 모아서 중간 배열 생성
    right = [x for x in arr if x > pivot]  # 피벗보다 큰 값들을 모아서 오른쪽 하위 배열 생성
    return quicksort(left) + middle + quicksort(right)  # 왼쪽, 중간, 오른쪽 결과를 합쳐 반환

data = [3, 6, 2, 7, 5, 1, 4]  # 정렬하려는 초기 데이터 배열 선언
sorted_data = quicksort(data)  # 퀵소트 함수를 호출해서 정렬된 결과 저장
print(sorted_data)  # 정렬된 결과를 출력하여 확인
```
