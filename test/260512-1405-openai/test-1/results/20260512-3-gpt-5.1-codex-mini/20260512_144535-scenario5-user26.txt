```python
def quicksort(arr):  # 배열을 정렬하는 퀵소트 함수의 정의입니다.
    if len(arr) <= 1:  # 배열 길이가 1 이하이면 이미 정렬된 상태로 판단합니다.
        return arr  # 현재 배열을 그대로 반환합니다.
    pivot = arr[len(arr) // 2]  # 피벗으로 사용할 중앙 요소를 선택합니다.
    less = [x for x in arr if x < pivot]  # 피벗보다 작은 요소들을 추출합니다.
    equal = [x for x in arr if x == pivot]  # 피벗과 같은 요소들을 추출합니다.
    greater = [x for x in arr if x > pivot]  # 피벗보다 큰 요소들을 추출합니다.
    return quicksort(less) + equal + quicksort(greater)  # 피벗을 중심으로 재귀적으로 정렬한 결과를 합칩니다.

numbers = [3, 1, 4, 1, 5, 9, 2, 6, 5]  # 예시로 사용할 숫자 배열입니다.
sorted_numbers = quicksort(numbers)  # 퀵소트 함수로 정렬된 결과를 변수에 저장합니다.
print(sorted_numbers)  # 정렬된 결과를 출력합니다.
```
