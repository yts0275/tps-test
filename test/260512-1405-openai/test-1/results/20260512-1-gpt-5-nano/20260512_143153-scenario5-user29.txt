다음은 파이썬으로 구현한 퀵소트이며, 각 줄에 주석을 달았습니다.

```python
def quicksort(arr):  # 배열을 제자리에서 정렬하기 위한 진입점
    _quicksort(arr, 0, len(arr) - 1)  # 맨 왼쪽 인덱스와 맨 오른쪽 인덱스로 재귀 시작
    return arr  # 정렬된 배열을 반환

def _quicksort(arr, low, high):  # 부분 배열을 재귀적으로 정렬하는 내부 함수
    if low < high:  # 남은 원소가 2개 이상일 때만 분할 진행
        pi = partition(arr, low, high)  # 피벗의 최종 위치를 구함
        _quicksort(arr, low, pi - 1)  # 왼쪽 부분 배열 재귀적으로 정렬
        _quicksort(arr, pi + 1, high)  # 오른쪽 부분 배열 재귀적으로 정렬

def partition(arr, low, high):  # 피벗으로 배열을 분할하는 함수
    pivot = arr[high]  # 가장 오른쪽 원소를 피벗으로 선택
    i = low - 1  # 작은 원소의 영역 끝 인덱스
    for j in range(low, high):  # low부터 high-1까지 순회
        if arr[j] <= pivot:  # 현재 원소가 피벗보다 작거나 같으면
            i += 1  # 작은 원소 영역 확장
            arr[i], arr[j] = arr[j], arr[i]  # 원소 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 최종 위치에 배치
    return i + 1  # 피벗의 최종 위치를 반환

if __name__ == "__main__":  # 스크립트 실행 진입점
    arr = [3, 6, 8, 2, 10, 1, 1, 5]  # 정렬할 원소들
    print("원래 배열:", arr)  # 원래 배열 출력
    sorted_arr = quicksort(arr)  # 퀵소트 실행
    print("정렬된 배열:", sorted_arr)  # 정렬된 배열 출력
```
