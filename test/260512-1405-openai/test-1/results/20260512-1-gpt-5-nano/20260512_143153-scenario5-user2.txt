다음은 각 줄에 주석을 다는 형태의 파이썬 퀵소트 구현입니다.

def quicksort(arr):  # 주어진 리스트를 제자리에서 퀵소트로 정렬한다
    def _quicksort(a, low, high):  # 내부 재귀 함수: 부분 배열 [low, high]를 정렬한다
        if low < high:  # 부분 배열에 원소가 두 개 이상 있는지 확인한다
            p = partition(a, low, high)  # 피벗의 최종 위치를 얻는다
            _quicksort(a, low, p - 1)   # 피벗 기준 왼쪽 부분 배열을 재귀적으로 정렬한다
            _quicksort(a, p + 1, high)  # 피벗 기준 오른쪽 부분 배열을 재귀적으로 정렬한다

    def partition(a, low, high):  # 배열을 피벗으로 분할하는 함수
        pivot = a[high]  # 맨 끝 원소를 피벗으로 삼는다
        i = low - 1  # 작은 원소 구역의 끝 인덱스를 나타낸다
        for j in range(low, high):  # low부터 high-1까지 순회한다
            if a[j] <= pivot:  # 현재 원소가 피벗보다 작거나 같으면
                i += 1  # 작은 원소 구역의 끝을 한 칸 늘린다
                a[i], a[j] = a[j], a[i]  # 현재 원소를 작은 구역으로 옮긴다
        a[i+1], a[high] = a[high], a[i+1]  # 피벗을 정확한 위치로 옮긴다
        return i + 1  # 피벗의 최종 인덱스를 반환한다

    _quicksort(arr, 0, len(arr) - 1)  # 배열 전체를 정렬 시작점으로 호출한다
    return arr  # 정렬된 배열을 반환한다

if __name__ == "__main__":  # 이 파일이 독립 실행될 때만 아래 테스트를 수행한다
    data = [3, 1, 4, 1, 5, 9, 2, 6]  # 테스트용 리스트를 만든다
    print(quicksort(data))  # 정렬된 리스트를 출력한다
