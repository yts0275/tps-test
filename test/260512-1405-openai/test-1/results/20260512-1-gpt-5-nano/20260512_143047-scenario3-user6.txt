다음은 파이썬으로 구현한 퀵소트 예시입니다. 각 줄마다 주석을 달아두었습니다.

```python
# 퀵소트 구현 (Lomuto 파티션 방식)

def partition(arr, low, high):  # 파티션 함수: 피벗을 기준으로 배열을 두 부분으로 나눕니다
    pivot = arr[high]  # 피벗으로 배열의 마지막 요소를 선택합니다
    i = low - 1        # 작은 영역의 끝 인덱스를 초기화합니다
    for j in range(low, high):  # low부터 high-1까지 순회합니다
        if arr[j] <= pivot:     # 현재 원소가 피벗보다 작거나 같으면
            i += 1                # 작은 영역의 끝을 한 칸 확장합니다
            arr[i], arr[j] = arr[j], arr[i]  # 현재 원소를 작은 영역으로 이동합니다
    arr[i+1], arr[high] = arr[high], arr[i+1]  # 피벗을 작은 영역의 뒤로 배치합니다
    return i + 1  # 피벗의 최종 위치 인덱스를 반환합니다

def quicksort(arr, low, high):  # 퀵소트 메인 함수 (재귀)
    if low < high:  # 정렬할 범위에 하나 이상의 원소가 있으면
        p = partition(arr, low, high)  # 피벗 위치를 구합니다
        quicksort(arr, low, p - 1)     # 피벗 왼쪽 부분 배열을 재귀적으로 정렬합니다
        quicksort(arr, p + 1, high)    # 피벗 오른쪽 부분 배열을 재귀적으로 정렬합니다

if __name__ == "__main__":  # 예제 실행 부분
    data = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 데이터
    quicksort(data, 0, len(data) - 1)  # 배열 전체에 대해 퀵소트를 수행합니다
    print(data)  # 정렬 결과를 출력합니다
```
