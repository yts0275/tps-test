다음은 파이썬으로 작성한 퀵소트 예시입니다. 각 줄에 주석을 달아 설명했습니다.

```python
def quicksort(arr, low=0, high=None):
    # 정렬할 배열, 시작 인덱스, 끝 인덱스(포함)를 받는다
    if high is None:
        # 처음 호출 시 끝 인덱스를 배열의 마지막으로 설정
        high = len(arr) - 1
    # 구간이 1개 이하로 남으면 이미 정렬된 상태
    if low < high:
        # 파티션 함수로 피봇의 최종 위치를 얻는다
        p = partition(arr, low, high)
        # 피봇 기준으로 왼쪽 부분 배열 재귀적으로 정렬
        quicksort(arr, low, p - 1)
        # 피봇 기준으로 오른쪽 부분 배열 재귀적으로 정렬
        quicksort(arr, p + 1, high)

def partition(arr, low, high):
    # 구간의 마지막 원소를 피봇으로 선택
    pivot = arr[high]
    # 피봇보다 작거나 같은 원소들의 영역 끝 인덱스
    i = low - 1
    # 구간의 시작부터 피봇 직전까지 순회
    for j in range(low, high):
        # 현재 원소가 피봇보다 작거나 같으면 앞쪽으로 모으기
        if arr[j] <= pivot:
            i += 1
            # 앞쪽 영역과 현재 원소를 교환
            arr[i], arr[j] = arr[j], arr[i]
    # 피봇을 앞쪽 영역의 끝 다음 위치로 이동시키기
    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    # 피봇의 최종 위치를 반환
    return i + 1

if __name__ == "__main__":
    # 테스트용 배열
    a = [3, 6, 8, 10, 1, 2, 1]
    # 배열을 제자리에서 퀵소트 수행
    quicksort(a)
    # 정렬된 배열 출력
    print(a)
```
