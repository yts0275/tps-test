다음은 파이썬으로 작성한, 각 줄에 주석을 달은 인-플레이스 퀵소트 구현입니다.

```python
def quicksort_inplace(arr, low, high):  # 배열의 일부분을 제자리에서 정렬하는 재귀 함수
    if low < high:  # 구간에 최소한 두 원소가 있을 때만 정렬 진행
        p = partition(arr, low, high)  # 피벗의 최종 인덱스를 구하는 함수 호출
        quicksort_inplace(arr, low, p - 1)  # 피벗 왼쪽 부분 재귀적으로 정렬
        quicksort_inplace(arr, p + 1, high)  # 피벗 오른쪽 부분 재귀적으로 정렬

def partition(arr, low, high):  # 파티션 함수: 피벗 기준으로 배열을 분할
    pivot = arr[high]  # 피벗으로 배열의 마지막 원소 선택
    i = low - 1  # 작은 원소 구역의 끝 인덱스 초기화
    for j in range(low, high):  # low부터 high-1까지 순회
        if arr[j] <= pivot:  # 현재 원소가 피벗보다 작거나 같으면
            i += 1  # 작은 원소 구역의 끝을 한 칸 앞으로 이동
            arr[i], arr[j] = arr[j], arr[i]  # 현재 원소를 작은 원소 구역으로 스왑
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 중간 위치로 이동
    return i + 1  # 피벗의 최종 인덱스 반환

if __name__ == "__main__":  # 스크립트로 실행될 때만 아래 코드 실행
    arr = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 배열 예시
    quicksort_inplace(arr, 0, len(arr) - 1)  # 배열을 제자리에서 정렬 시작
    print(arr)  # 정렬 결과 출력
```
