다음은 각 줄에 주석을 달아 놓은 파이썬 기반의 간단한 재귀적 퀵소트 구현입니다. 추가 공간을 사용해 입력 리스트를 나눠 새 리스트를 반환합니다.

def quicksort(arr):  # 리스트를 받아 정렬된 새 리스트를 반환하는 퀵소트
    if len(arr) <= 1:  # 원소가 하나 이하이면 이미 정렬됨
        return arr  # 입력 배열 그대로 반환
    else:  # 피벗을 기준으로 분할
        pivot = arr[len(arr) // 2]  # 중간 원소를 피벗으로 선택
        left = [x for x in arr if x < pivot]  # 피봇보다 작은 요소들로 왼쪽 리스트 구성
        middle = [x for x in arr if x == pivot]  # 피벗과 같은 값들로 가운데 리스트 구성
        right = [x for x in arr if x > pivot]  # 피봇보다 큰 요소들로 오른쪽 리스트 구성
        return quicksort(left) + middle + quicksort(right)  # 재귀적으로 정렬하여 합치기

# 사용 예시
lst = [4, 2, 7, 1, 3]  # 정렬할 원소들의 리스트
sorted_lst = quicksort(lst)  # quicksort 호출로 정렬된 새 리스트 얻기
print(sorted_lst)  # 결과 출력
