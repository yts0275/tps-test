다음은 Python으로 작성한 각 줄에 주석이 달린 QuickSort 구현입니다.

```python
def quicksort(arr, low, high):  # QuickSort으로 배열을 in-place 정렬하는 재귀 함수
    if low < high:  # 구간에 두 원소 이상이 있으면 정렬 수행
        pi = partition(arr, low, high)  # 피벗의 최종 인덱스 계산
        quicksort(arr, low, pi - 1)  # 왼쪽 부분배열 재귀적으로 정렬
        quicksort(arr, pi + 1, high)  # 오른쪽 부분배열 재귀적으로 정렬

def partition(arr, low, high):  # 파티션 함수: 피벗 기준으로 배열 분할
    pivot = arr[high]  # 피벗으로 사용할 마지막 원소 선택
    i = low - 1  # 작은 원소의 구분 인덱스 초기화
    for j in range(low, high):  # low부터 high-1까지 순회
        if arr[j] <= pivot:  # 현재 원소가 피벗보다 작거나 같으면
            i += 1  # 작은 원소의 도메인 확장
            arr[i], arr[j] = arr[j], arr[i]  # 원소 위치 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 정확한 위치로 이동
    return i + 1  # 피벗의 최종 인덱스 반환

def sort(arr):  # 배열 전체를 QuickSort로 정렬하는 래퍼 함수
    quicksort(arr, 0, len(arr) - 1)  # 전체 구간에 대해 QuickSort 시작
    return arr  # 정렬된 배열을 반환
```
