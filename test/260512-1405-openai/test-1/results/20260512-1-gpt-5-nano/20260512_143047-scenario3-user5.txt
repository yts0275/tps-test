다음은 파이썬으로 작성한 퀵소트 예시입니다. 각 줄마다 주석을 달았습니다.

```python
def quicksort(arr, low, high):  # 시작 인덱스, 끝 인덱스를 받아 구간 정렬
    if low < high:  # 구간에 원소가 두 개 이상인지 확인
        pi = partition(arr, low, high)  # 피벗의 최종 위치를 얻어 분할
        quicksort(arr, low, pi - 1)  # 왼쪽 부분 배열 재귀적으로 정렬
        quicksort(arr, pi + 1, high)  # 오른쪽 부분 배열 재귀적으로 정렬

def partition(arr, low, high):  # 분할 함수
    pivot = arr[high]  # 피벗으로 배열의 마지막 원소 선택
    i = low - 1  # 작은 원소 구간의 끝 인덱스 초기화
    for j in range(low, high):  # 구간의 각 원소를 순회
        if arr[j] <= pivot:  # 현재 원소가 피벗보다 작거나 같다면
            i += 1  # 작은 원소 구간 크기 증가
            arr[i], arr[j] = arr[j], arr[i]  # 원소 위치 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 올바른 위치로 이동
    return i + 1  # 피벗의 새로운 위치 반환

def quicksort_full(arr):  # 외부에서 쉽게 호출할 수 있는 래퍼 함수
    if arr:  # 배열이 비어 있지 않다면
        quicksort(arr, 0, len(arr) - 1)  # 전체 배열 정렬

if __name__ == "__main__":  # 간단한 실행 예시 테스트 블록
    data = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 배열 예시
    quicksort_full(data)  # 퀵소트 실행
    print(data)  # 정렬 결과 출력
```
