다음은 파이썬으로 만든 QuickSort 구현이며, 줄마다 한 줄씩 주석을 달았습니다.

# QuickSort를 배열에 제자리에서 정렬하는 구현
def quicksort(arr):  # QuickSort 정렬 함수
    if arr is None:  # 입력 배열이 None인지 확인
        return  # None인 경우 정렬하지 않음
    def _partition(a, low, high):  # 내부: 부분 배열의 피벗 분할 함수
        pivot = a[high]  # 피벗으로 배열의 마지막 원소 선택
        i = low - 1  # 작은 값들의 경계 인덱스 초기화
        for j in range(low, high):  # 구간 [low, high-1]를 순회
            if a[j] <= pivot:  # 피벗보다 작거나 같으면
                i += 1  # 경계를 한 칸 오른쪽으로 이동
                a[i], a[j] = a[j], a[i]  # 원소 교환으로 작은 값 앞으로 모으기
        a[i + 1], a[high] = a[high], a[i + 1]  # 피벗을 최종 위치로 이동
        return i + 1  # 피벗의 최종 인덱스 반환
    def _quicksort(a, low, high):  # 내부: 재귀적으로 부분 배열 정렬
        if low < high:  # 구간에 2개 이상 원소가 있을 때만 진행
            pi = _partition(a, low, high)  # 피벗 위치 얻기
            _quicksort(a, low, pi - 1)  # 왼쪽 부분 배열 재귀 정렬
            _quicksort(a, pi + 1, high)  # 오른쪽 부분 배열 재귀 정렬
    _quicksort(arr, 0, len(arr) - 1)  # 전체 배열에 대해 정렬 시작
lst = [3, 6, 8, 2, 10, 1, 9]  # 예시 배열
quicksort(lst)  # 배열을 제자리에서 정렬
print(lst)  # 정렬 결과 출력
