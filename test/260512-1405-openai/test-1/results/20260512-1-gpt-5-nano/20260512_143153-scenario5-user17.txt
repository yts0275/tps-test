다음은 파이썬으로 구현한 퀵소트이며, 각 줄에 한국어 주석을 달아 놓은 버전입니다.

def partition(arr, low, high):  # 피봇 기반 파티션 수행
    pivot = arr[high]  # 피봇으로 배열의 마지막 원소 선택
    i = low - 1  # 작은 원소 구역의 경계 인덱스 초기화
    for j in range(low, high):  # low부터 high-1까지 순회
        if arr[j] <= pivot:  # 현재 원소가 피봇보다 작으면
            i += 1  # 작은 원소 구역을 하나 확장
            arr[i], arr[j] = arr[j], arr[i]  # 원소 위치 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피봇을 올바른 위치로 이동
    return i + 1  # 피봇의 최종 인덱스 반환

def quicksort(arr, low, high):  # 퀵소트 재귀 함수
    if low < high:  # 부분배열에 원소가 2개 이상 있을 때만 수행
        pi = partition(arr, low, high)  # 파티션 인덱스 계산
        quicksort(arr, low, pi - 1)  # 왼쪽 부분배열 재귀 정렬
        quicksort(arr, pi + 1, high)  # 오른쪽 부분배열 재귀 정렬

if __name__ == "__main__":  # 이 스크립트가 직접 실행될 때만 아래를 실행
    arr = [10, 7, 8, 9, 1, 5]  # 정렬 대상 배열 초기값
    quicksort(arr, 0, len(arr) - 1)  # 퀵소트 실행
    print("정렬 결과:", arr)  # 정렬 결과 출력
