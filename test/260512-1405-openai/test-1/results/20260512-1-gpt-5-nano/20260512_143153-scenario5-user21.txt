다음은 각 줄마다 주석이 달린 파이썬 퀵소트 구현입니다.

```python
def partition(a, low, high):  # Lomuto 파티션: 피봇을 마지막 원소로 선택
    pivot = a[high]  # 피봇 값을 배열의 마지막 원소로 지정
    i = low - 1  # 작은 원소들의 마지막 위치를 가리키는 인덱스 초기화
    for j in range(low, high):  # low부터 high-1까지 순회
        if a[j] <= pivot:  # 현재 원소가 피봇보다 작거나 같다면
            i += 1  # i를 하나 증가시켜 작은 원소의 구간 확장
            a[i], a[j] = a[j], a[i]  # i와 j 위치의 원소를 교환
    a[i + 1], a[high] = a[high], a[i + 1]  # 피봇을 올바른 위치에 배치
    return i + 1  # 피봇의 최종 인덱스 반환

def quicksort(a, low, high):  # 재귀적으로 배열을 정렬하는 함수
    if low < high:  # 정렬할 구간에 두 개 이상 원소가 있을 때만 진행
        p = partition(a, low, high)  # 피봇의 위치를 얻음
        quicksort(a, low, p - 1)  # 피봇 왼쪽 부분 재귀적으로 정렬
        quicksort(a, p + 1, high)  # 피봇 오른쪽 부분 재귀적으로 정렬

def quicksort_inplace(a):  # 배열을 제자리에서 퀵소트하는 간단한 래퍼
    if a:  # 배열이 비어있지 않으면
        quicksort(a, 0, len(a) - 1)  # 전체 구간에 대해 퀵소트 수행

arr = [3, 6, 8, 10, 1, 2, 1]  # 정렬할 배열
quicksort_inplace(arr)  # 배열을 제자리에서 정렬
print(arr)  # 출력: [1, 1, 2, 3, 6, 8, 10]
```
