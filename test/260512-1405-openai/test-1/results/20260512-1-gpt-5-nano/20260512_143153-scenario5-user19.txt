```python
def quicksort(arr):  # 배열을 제자리에서 정렬하는 QuickSort의 진입점
    def _quicksort(a, low, high):  # 내부 재귀 함수 정의
        if low >= high:  # 구간에 원소가 0개나 1개일 때 재귀 종료 조건
            return  # 재귀를 멈추고 위로 올라감
        pivot = a[high]  # 구간의 마지막 원소를 피봇으로 선택
        i = low - 1  # 왼쪽 부분 배열의 경계 인덱스 초기화
        for j in range(low, high):  # 구간 [low, high-1]를 순회
            if a[j] <= pivot:  # 현재 원소가 피봇보다 작거나 같으면
                i += 1  # 작은 원소의 경계를 한 칸 증가시킴
                a[i], a[j] = a[j], a[i]  # 왼쪽으로 작은 원소를 이동시키고 교환
        a[i+1], a[high] = a[high], a[i+1]  # 피봇을 올바른 위치로 이동
        _quicksort(a, low, i)  # 피봇보다 작은 부분 배열 재귀 정렬
        _quicksort(a, i+2, high)  # 피봇보다 큰 부분 배열 재귀 정렬
    _quicksort(arr, 0, len(arr) - 1)  # 배열 전체 구간에 대해 정렬 시작
```
