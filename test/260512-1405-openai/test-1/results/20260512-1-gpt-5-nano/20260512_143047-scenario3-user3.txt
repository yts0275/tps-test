다음은 파이썬으로 구현한 퀵소트 예시이며, 각 줄에 주석을 달았습니다.

```python
def quicksort(arr, low, high):  # 배열과 정렬할 인덱스 범위를 받아 QuickSort 실행
    if low < high:  # 범위에 남은 요소가 2개 이상인지 확인
        pi = partition(arr, low, high)  # 현재 범위를 분할하고 피벗 인덱스 얻기
        quicksort(arr, low, pi - 1)  # 피벗 왼쪽 부분 배열 재귀적으로 정렬
        quicksort(arr, pi + 1, high)  # 피벗 오른쪽 부분 배열 재귀적으로 정렬

def partition(arr, low, high):  # 분할 함수
    pivot = arr[high]  # 피벗으로 배열의 마지막 원소 선택
    i = low - 1  # i는 작은 원소들의 최종 위치를 가리키도록 초기화
    for j in range(low, high):  # low부터 high-1까지 순회
        if arr[j] <= pivot:  # 현재 원소가 피벗보다 작거나 같으면
            i += 1  # 작은 원소의 위치를 한 칸 앞으로 이동
            arr[i], arr[j] = arr[j], arr[i]  # 원소 두 개의 위치를 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 올바른 위치로 이동
    return i + 1  # 피벗의 최종 인덱스 반환

if __name__ == "__main__":  # 이 스크립트가 직접 실행될 때만 아래 코드 실행
    arr = [10, 7, 8, 9, 1, 5]  # 정렬할 숫자 배열 선언
    quicksort(arr, 0, len(arr) - 1)  # 배열 전체를 QuickSort로 정렬
    print(arr)  # 정렬 결과 출력
```
