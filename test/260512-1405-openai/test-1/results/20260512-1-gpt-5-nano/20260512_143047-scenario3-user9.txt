다음은 각 줄에 주석을 달아 만든 파이썬 퀵소트 구현입니다.

```python
# QuickSort 구현 예시 - 파이썬
# 리스트를 오름차순으로 정렬하는 퀵소트의 예제(파이썬)
def partition(arr, low, high):  # 피봇 기준으로 파티션 작업 수행
    pivot = arr[high]  # 피봇으로 사용할 마지막 원소 지정
    i = low - 1  # 작은 원소들의 구분 인덱스 초기화
    for j in range(low, high):  # low부터 high-1까지 순회
        if arr[j] <= pivot:  # 현재 원소가 피봇보다 작거나 같으면
            i += 1  # i를 한 칸 증가시킴
            arr[i], arr[j] = arr[j], arr[i]  # 원소를 교환해 작은 값들을 앞쪽으로 이동
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피봇을 정확한 위치로 이동
    return i + 1  # 피봇의 최종 인덱스 반환

def quicksort(arr, low, high):  # 재귀적으로 부분 배열 정렬
    if low < high:  # 배열에 두 개 이상의 원소가 남아있으면
        pi = partition(arr, low, high)  # 파티션 수행 후 피봇 인덱스 얻음
        quicksort(arr, low, pi - 1)  # 왼쪽 부분 배열 재귀적으로 정렬
        quicksort(arr, pi + 1, high)  # 오른쪽 부분 배열 재귀적으로 정렬

if __name__ == "__main__":  # 스크립트가 직접 실행될 때만 실행
    data = [33, 10, 55, 71, 29, 3, 18, 42]  # 정렬할 원소들
    quicksort(data, 0, len(data) - 1)  # 전체 배열에 대해 퀵소트 수행
    print(data)  # 정렬 결과 출력
```
