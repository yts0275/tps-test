다음은 파이썬으로 작성한, 모든 줄에 주석이 달린 퀵소트 구현입니다.

```python
# 퀵소트 구현 (Python, in-place)
def quicksort(arr, low, high):  # arr: 정렬 대상 리스트, low: 시작 인덱스, high: 끝 인덱스
    if low < high:  # 남은 구간이 1개 이상일 때만 수행
        pi = partition(arr, low, high)  # 피벗 인덱스를 얻음
        quicksort(arr, low, pi - 1)  # 피벗 왼쪽 부분을 정렬
        quicksort(arr, pi + 1, high)  # 피벗 오른쪽 부분을 정렬

def partition(arr, low, high):  # 분할 함수: 피벗을 기준으로 배열을 재배치
    pivot = arr[high]  # 가장 오른쪽 원소를 피벗으로 선택
    i = low - 1  # 피벗보다 작은 구간의 끝 인덱스 초기화
    for j in range(low, high):  # low부터 high-1까지 순회
        if arr[j] <= pivot:  # 현재 원소가 피벗보다 작거나 같으면
            i += 1  # 작아진 구간의 끝을 한 칸 증가
            arr[i], arr[j] = arr[j], arr[i]  # 두 원소의 위치를 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 중간으로 올려 놓음
    return i + 1  # 피벗의 최종 위치를 반환

# 사용 예시
if __name__ == "__main__":  # 파일이 직접 실행될 때만 아래 예시를 수행
    data = [33, 10, 19, 27, 5, 1, 3, 8, 12]  # 정렬할 리스트(샘플)
    print("원본:", data)  # 원래 배열 출력
    quicksort(data, 0, len(data) - 1)  # 배열 전체를 정렬
    print("정렬 후:", data)  # 정렬된 배열 출력
```
