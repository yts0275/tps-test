def partition(arr, low, high):  # 배열에서 피벗을 기준으로 분할하는 함수
    pivot = arr[high]  # 피벗으로 배열의 마지막 원소를 선택
    i = low - 1  # 작은 요소들을 모아둘 구역의 끝 인덱스 초기화
    for j in range(low, high):  # 현재 원소를 차례대로 검사하는 루프
        if arr[j] <= pivot:  # 현재 원소가 피벗보다 작거나 같으면
            i += 1  # 구역의 크기를 1 증가시킴
            arr[i], arr[j] = arr[j], arr[i]  # 현재 원소를 구역의 끝과 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피벗을 구역의 중앙에 위치시키고 교환
    return i + 1  # 피벗의 최종 위치 인덱스 반환

def quicksort(arr, low, high):  # 배열을 제시된 인덱스 범위에서 퀵소트하는 재귀 함수
    if low < high:  # 범위에 두 개 이상의 원소가 있을 때만 실행
        p = partition(arr, low, high)  # 피벗의 최종 위치를 얻음
        quicksort(arr, low, p - 1)  # 피벗 좌측 부분 배열 재귀적으로 정렬
        quicksort(arr, p + 1, high)  # 피벗 우측 부분 배열 재귀적으로 정렬

if __name__ == "__main__":  # 스크립트로 실행될 때만 아래 예제를 동작시킴
    arr = [3, 6, 8, 10, 1, 2, 1]  # 예시 배열
    quicksort(arr, 0, len(arr) - 1)  # 배열 전체를 퀵소트로 정렬 시작
    print(arr)  # 정렬된 배열 출력
