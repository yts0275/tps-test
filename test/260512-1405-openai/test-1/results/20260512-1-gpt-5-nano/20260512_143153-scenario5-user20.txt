다음은 파이썬으로 작성한 퀵소트의 예시입니다. 각 줄마다 한국어 주석을 달아 두었습니다. 이 코드는 배열을 제자리에서 정렬합니다.

```python
def quicksort(arr, low=0, high=None):  # QuickSort 래퍼: 배열을 제자리에서 정렬
    if high is None:  # high가 주어지지 않으면 배열의 마지막 인덱스로 초기화
        high = len(arr) - 1  # 배열의 마지막 인덱스

    if low < high:  # 부분배열에 두 원소 이상이 남아 있을 때만
        p = partition(arr, low, high)  # 분할을 수행하고 피봇 인덱스 얻기
        quicksort(arr, low, p - 1)  # 피봇 왼쪽 부분 재귀적으로 정렬
        quicksort(arr, p + 1, high)  # 피봇 오른쪽 부분 재귀적으로 정렬

def partition(arr, low, high):  # Lomuto 분할 방식: 피봇은 arr[high]
    pivot = arr[high]  # 피봇 값은 배열의 마지막 원소
    i = low - 1  # 작은 원소 경계 인덱스

    for j in range(low, high):  # low부터 high-1까지 순회
        if arr[j] <= pivot:  # 현재 원소가 피봇보다 작거나 같으면
            i += 1  # 작은 원소 경계 한 칸 확장
            arr[i], arr[j] = arr[j], arr[i]  # 교환: 작은 원소를 왼쪽으로 모으기

    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피봇을 올바른 위치로 배치
    return i + 1  # 피봇의 최종 인덱스 반환

# 사용 예시(주석 처리): 주석으로 남겨두면 바로 실행 가능
```
