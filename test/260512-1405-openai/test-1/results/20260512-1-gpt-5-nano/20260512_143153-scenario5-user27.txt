다음은 파이썬으로 구현한 퀵소트 예시입니다. 각 줄에 주석을 달았습니다.

```python
def quicksort(arr, low, high):  # QuickSort 함수 정의: arr의 [low, high] 범위를 정렬
    if low < high:  # 범위에 두 개 이상의 원소가 있는지 확인
        pi = partition(arr, low, high)  # 파티션으로 피봇 위치 계산
        quicksort(arr, low, pi - 1)  # 피봇 왼쪽 서브배열 재귀 정렬
        quicksort(arr, pi + 1, high)  # 피봇 오른쪽 서브배열 재귀 정렬

def partition(arr, low, high):  # 파티션 함수 정의
    pivot = arr[high]  # 피봇으로 배열의 마지막 원소 선택
    i = low - 1  # i는 작은 원소의 구역 끝 인덱스
    for j in range(low, high):  # low에서 high-1까지 순회
        if arr[j] <= pivot:  # 현재 원소가 피봇보다 작거나 같으면
            i += 1  # 작은 원소 구역 확장
            arr[i], arr[j] = arr[j], arr[i]  # 두 원소를 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피봇을 올바른 위치로 이동
    return i + 1  # 피봇의 최종 위치 반환

arr = [9, 3, 7, 1, 5, 8, 2]  # 테스트용 배열
quicksort(arr, 0, len(arr) - 1)  # 배열 전체를 정렬
print(arr)  # 정렬 결과 출력
```
