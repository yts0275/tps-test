def partition(arr, low, high):  # 피봇으로 배열 분할하는 함수
    pivot = arr[high]  # 마지막 원소를 피봇으로 선택
    i = low - 1  # 소요된 작은 원소들의 끝 인덱스 초기화
    for j in range(low, high):  # low부터 high-1까지 순회
        if arr[j] <= pivot:  # 현재 원소가 피봇보다 작거나 같으면
            i += 1  # 작은 원소의 구간 확장
            arr[i], arr[j] = arr[j], arr[i]  # 해당 위치의 원소를 교환
    pivot_index = i + 1  # 피봇의 최종 위치 계산
    arr[pivot_index], arr[high] = arr[high], arr[pivot_index]  # 피봇을 위치에 배치
    return pivot_index  # 피봇의 위치 반환

def quicksort(arr, low, high):  # 퀵소트 메인 함수
    if low < high:  # 부분배열에 원소가 두 개 이상 있을 때만 작동
        pi = partition(arr, low, high)  # 피봇 위치 얻기
        quicksort(arr, low, pi - 1)  # 왼쪽 부분배열 재귀적으로 정렬
        quicksort(arr, pi + 1, high)  # 오른쪽 부분배열 재귀적으로 정렬

if __name__ == "__main__":  # 테스트용 실행 블록
    arr = [10, 7, 8, 9, 1, 5]  # 예시 배열
    print("정렬 전:", arr)  # 정렬 전 출력
    quicksort(arr, 0, len(arr) - 1)  # 제자리에서 정렬 수행
    print("정렬 후:", arr)  # 정렬 후 출력
