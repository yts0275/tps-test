def quicksort_in_place(arr, low, high):  # in-place 퀵소트 구현 시작
    if low < high:  # 부분 배열에 요소가 남아 있는지 검사
        pi = partition(arr, low, high)  # 피봇 위치 계산
        quicksort_in_place(arr, low, pi - 1)  # 피봇보다 작은 부분 재귀 정렬
        quicksort_in_place(arr, pi + 1, high)  # 피봇보다 큰 부분 재귀 정렬

def partition(arr, low, high):  # 피봇으로 마지막 원소를 선택하는 파티션 함수
    pivot = arr[high]  # 피봇 원소 저장
    i = low - 1  # 작은 원소의 구분 인덱스 초기화
    for j in range(low, high):  # low부터 high-1까지 순회
        if arr[j] <= pivot:  # 현재 원소가 피봇보다 작거나 같으면
            i += 1  # 작은 영역 확장
            arr[i], arr[j] = arr[j], arr[i]  # 원소 교환
    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # 피봇을 올바른 위치로 이동
    return i + 1  # 피봇의 최종 인덱스 반환

def quicksort(arr):  # 외부 호출용 래퍼 함수
    if arr:  # 배열이 비어있지 않으면
        quicksort_in_place(arr, 0, len(arr) - 1)  # 전체 배열에 대해 정렬 수행

if __name__ == "__main__":  # 스크립트로 실행될 때 테스트 코드 실행
    data = [3, 6, 8, 10, 1, 2, 1]  # 테스트용 숫자 리스트
    quicksort(data)  # 리스트를 제자리에서 정렬
    print(data)  # 정렬 결과 출력
