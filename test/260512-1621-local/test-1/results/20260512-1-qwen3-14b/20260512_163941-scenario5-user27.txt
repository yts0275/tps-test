def quicksort(arr):
    # 기저 조건: 배열의 길이가 1 이하이면 이미 정렬된 상태이므로 그대로 반환
    if len(arr) <= 1:
        return arr

    # 피봇 선택: 배열의 중앙 요소를 피봇으로 설정
    pivot = arr[len(arr) // 2]

    # 피봇을 기준으로 배열을 분할
    # 왼쪽 배열: 피봇보다 작은 요소들
    left = [x for x in arr if x < pivot]
    # 오른쪽 배열: 피봇보다 크거나 같은 요소들
    right = [x for x in arr if x >= pivot]

    # 재귀적으로 왼쪽과 오른쪽 배열을 정렬하고, 피봇을 중간에 삽입하여 합친다
