def quicksort(arr, low, high):

    if low < high:
        # 배열을 분할하고 피벗의 인덱스를 반환

        # 피벗 왼쪽의 요소를 재귀적으로 정렬

        # 피벗 오른쪽의 요소를 재귀적으로 정렬
        quicksort(arr, pi + 1, high)

def partition(arr, low, high):
    # 피벗으로 사용할 요소(현재 배열의 마지막 요소)
    pivot = arr[high]

            i += 1
            # i와 j의 요소를 교환
            arr[i], arr[j] = arr[j], arr[i]

    # 피벗 요소를 적절한 위치로 이동
    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    # 피벗의 최종 위치를 반환
    return i + 1
