def quicksort(arr, start, end):
    # 기저 조건: start가 end보다 작을 때만 실행
    if start < end:
        # 파티션을 실행하여 pivot의 위치를 결정하고, pivot_index를 반환
        pivot_index = partition(arr, start, end)
        # pivot을 기준으로 왼쪽 부분 배열 재귀적으로 정렬
        quicksort(arr, start, pivot_index - 1)
        # pivot을 기준으로 오른쪽 부분 배열 재귀적으로 정렬
        quicksort(arr, pivot_index + 1, end)

def partition(arr, start, end):
    # pivot을 배열의 마지막 요소로 선택
    pivot = arr[end]
    # pivot보다 작은 요소의 위치를 추적하는 인덱스
    i = start - 1
    # 배열의 start부터 end-1까지 반복
    for j in range(start, end):
        # 현재 요소가 pivot보다 작으면 i를 증가시키고 요소를 교환
        if arr[j] < pivot:
            i += 1
            arr[i], arr[j] = arr[j], arr[i]
    # pivot을 적절한 위치로 이동
    arr[i + 1], arr[end] = arr[end], arr[i + 1]
    # pivot의 최종 위치 반환
    return i + 1

# 예시 배열

n = len(arr)
# 정렬 실행
quicksort(arr, 0, n - 1)

print("정렬된 배열:", arr)
