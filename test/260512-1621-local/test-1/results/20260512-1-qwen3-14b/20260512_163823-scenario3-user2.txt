```python
# 퀵소트(QuickSort) 알고리즘을 구현합니다.
# 이 알고리즘은 분할 정복(Divide and Conquer) 기법을 사용하여 배열을 정렬합니다.

def quicksort(arr):
    # 기저 조건: 배열의 길이가 0 또는 1일 경우, 배열을 그대로 반환합니다.
    if len(arr) <= 1:
        return arr

    # 피봇(pivot)을 선택합니다. 여기서는 마지막 요소를 피봇으로 사용합니다.
    pivot = arr[-1]
    # 피봇보다 작은 요소들을 저장할 리스트
    left = []
    # 피봇보다 큰 요소들을 저장할 리스트
    right = []

    # 배열의 요소를 피봇을 기준으로 좌우 두 부분으로 분할합니다.
    for i in range(len(arr) - 1):
        if arr[i] < pivot:
            left.append(arr[i])  # 피봇보다 작으면 왼쪽 그룹에 추가
        else:
            right.append(arr[i])  # 피봇보다 크거나 같으면 오른쪽 그룹에 추가

    # 분할된 왼쪽, 피봇, 오른쪽을 재귀적으로 정렬하고 합칩니다.
    return quicksort(left) + [pivot] + quicksort(right)

# 예시 배열
arr = [3, 6, 8, 10, 1, 2, 1]
# 퀵소트 실행
sorted_arr = quicksort(arr)
# 정렬된 결과 출력
print("정렬된 배열:", sorted_arr)
```
