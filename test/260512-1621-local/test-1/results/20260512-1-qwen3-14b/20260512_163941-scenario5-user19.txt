```python
# 퀵소트(Quick Sort) 알고리즘 구현
# 배열을 오름차순으로 정렬하는 알고리즘

def quicksort(arr, low, high):
    # low: 정렬할 배열의 시작 인덱스
    # high: 정렬할 배열의 끝 인덱스
    if low < high:
        # 파티션을 나누고, 피벗의 인덱스를 반환
        pi = partition(arr, low, high)
        
        # 피벗을 기준으로 좌측 배열 재귀 호출
        quicksort(arr, low, pi - 1)
        
        # 피벗을 기준으로 우측 배열 재귀 호출
        quicksort(arr, pi + 1, high)

def partition(arr, low, high):
    # 피벗 선택: high 인덱스의 요소를 기준으로 함
    pivot = arr[high]
    
    # 작은 요소의 마지막 인덱스 초기화
    i = low - 1
    
    # low에서 high-1까지 반복
    for j in range(low, high):
        # 현재 요소가 피벗보다 작으면
        if arr[j] < pivot:
            # i를 증가시키고, i와 j의 요소를 교환
            i += 1
            arr[i], arr[j] = arr[j], arr[i]
    
    # 마지막으로 피벗을 올바른 위치로 이동
    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    
    # 피벗의 최종 인덱스 반환
    return i + 1

# 예제 배열 생성
arr = [10, 7, 8, 9, 1, 5]
print("정렬 전 배열:", arr)

# 퀵소트 실행
quicksort(arr, 0, len(arr) - 1)

# 정렬된 배열 출력
print("정렬 후 배열:", arr)
```

   - 재귀적으로 배열을 정렬
   - `low`와 `high`은 배열의 범위를 나타냄
   - `partition` 함수를 통해 배열을 분할하고, 분할된 배열에 대해 재귀 호출

2. **partition 함수**:
   - 피벗(기준 요소) 선택: 배열의 마지막 요소
   - `i`는 작은 요소의 마지막 인덱스를 추적
   - `j`는 배열을 순회하며 피벗보다 작은 요소를 찾음
   - 작은 요소를 `i` 위치로 이동
   - 마지막으로 피벗을 올바른 위치로 이동
