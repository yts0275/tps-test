# 퀵소트(Quick Sort) 알고리즘 구현

def quicksort(arr, low, high):
    # 기저 조건: low가 high보다 크면 종료 (하나의 요소만 있을 때)
    if low < high:
        # 파티션 함수를 호출하여 피벗의 위치를 얻음
        # 파티션 함수는 배열을 피벗을 기준으로 두 부분으로 나누고, 피벗의 최종 위치를 반환
        pi = partition(arr, low, high)
        
        # 피벗 기준으로 왼쪽 부분 배열을 재귀적으로 정렬
        quicksort(arr, low, pi - 1)
        
        # 피벗 기준으로 오른쪽 부분 배열을 재귀적으로 정렬
        quicksort(arr, pi + 1, high)

def partition(arr, low, high):
    # 피벗은 배열의 마지막 요소로 선택 (high 인덱스)
    pivot = arr[high]
    
    # i는 "작은 요소의 위치"를 추적하는 인덱스
    # 초기값은 low-1로 설정 (low보다 왼쪽에 있다고 가정)
    i = low - 1
    
    # low부터 high-1까지의 요소를 순회
    for j in range(low, high):
        # 현재 요소가 피벗보다 작으면
        if arr[j] < pivot:
            # i를 1 증가시키고, arr[i]와 arr[j]를 교환
            # 이 과정을 통해 작은 요소를 왼쪽으로 이동
            i += 1
            arr[i], arr[j] = arr[j], arr[i]
    
    # 순회가 끝난 후, 피벗을 적절한 위치로 이동
    # i+1 위치에 피벗을 배치 (i는 마지막으로 작은 요소의 위치)
    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    
    # 피벗의 최종 위치를 반환
    return i + 1

# 예시 사용

print("정렬된 배열:", arr)
