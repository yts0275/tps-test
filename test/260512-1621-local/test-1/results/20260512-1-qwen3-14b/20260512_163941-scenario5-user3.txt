    # 배열의 범위가 유효할 때만 실행
    if low < high:
        # 파티션을 생성하고 피벗의 인덱스를 반환
        pi = partition(arr, low, high)
        
        # 피벗 왼쪽의 배열을 재귀적으로 정렬
        quicksort(arr, low, pi - 1)
        
        # 피벗 오른쪽의 배열을 재귀적으로 정렬
        quicksort(arr, pi + 1, high)

def partition(arr, low, high):
    # 피벗으로 마지막 요소를 선택
    pivot = arr[high]
    
    # 피벗보다 작은 요소의 마지막 인덱스
    i = low - 1
    
    # 배열을 탐색하면서 피벗보다 작은 요소를 왼쪽으로 이동
    for j in range(low, high):
        if arr[j] <= pivot:
            # 작은 요소를 왼쪽으로 이동
            i += 1
            arr[i], arr[j] = arr[j], arr[i]
    
    # 피벗을 올바른 위치로 이동
    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    
    # 피벗의 최종 인덱스 반환
    return i + 1

# 예제 실행
arr = [10, 7, 8, 9, 1, 5]
n = len(arr)
quicksort(arr, 0, n - 1)
print("정렬된 배열:", arr)

1. **`quicksort()`** 함수는:
   - 배열의 범위(`low`, `high`)가 유효할 때만 실행
   - `partition()`을 호출하여 피벗을 기준으로 배열을 분할
   - 분할된 왼쪽/오른쪽 배열을 재귀적으로 정렬

2. **`partition()`** 함수는:
   - 피벗(마지막 요소)을 선택
   - 배열을 탐색하며 피벗보다 작은 요소를 왼쪽으로 이동
   - 피벗을 올바른 위치로 이동하고 해당 인덱스를 반환

3. **시간 복잡도**:
   - 평균: O(n log n)
   - 최악: O(n²) (이미 정렬된 배열일 때)

4. **공간 복잡도**:
   - O(log n) (재귀 스택 사용)
