# 퀵소트(Quick Sort) 알고리즘을 Python으로 구현한 예제입니다.
# 이 알고리즘은 분할 정복(Divide and Conquer) 전략을 사용하여 배열을 정렬합니다.

def quicksort(arr, low, high):
    """
    퀵소트의 메인 함수입니다.
    - arr: 정렬할 배열
    - low: 배열의 시작 인덱스
    - high: 배열의 끝 인덱스
    """
    if low < high:  # 배열이 하나 이상의 요소를 가질 때
        # 파티션을 나누고, 피봇의 위치를 반환
        pi = partition(arr, low, high)  # pivot index

        # 피봇 기준으로 좌우 서브배열을 재귀적으로 정렬

        quicksort(arr, pi + 1, high)  # 오른쪽 서브배열

def partition(arr, low, high):
    """
    배열을 피봇을 기준으로 두 부분으로 나누는 함수입니다.
    - arr: 정렬할 배열
    - low: 배열의 시작 인덱스
    - high: 배열의 끝 인덱스
    """
    # 피봇은 마지막 요소로 선택
    pivot = arr[high]  
    i = low - 1  # 작은 요소를 저장할 인덱스

    # low부터 high-1까지 반복
    for j in range(low, high):
        # 현재 요소가 피봇보다 작으면
        if arr[j] < pivot:
            # i를 증가시키고, arr[i]와 arr[j]를 교환
            i += 1
            arr[i], arr[j] = arr[j], arr[i]

    # 피봇을 올바른 위치로 이동 (i+1)
    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    return i + 1  # 피봇의 인덱스를 반환

# 예시 배열
arr = [10, 7, 8, 9, 1, 5]
print("정렬 전 배열:", arr)

# 퀵소트 실행
quicksort(arr, 0, len(arr) - 1)

# 정렬된 배열 출력
print("정렬 후 배열:", arr)

1. `quicksort` 함수는 재귀적으로 실행되며, 배열을 피봇 기준으로 분할합니다.
2. `partition` 함수는 배열을 피봇을 중심으로 두 부분으로 나누고, 피봇의 최종 위치를 반환합니다.
3. 피봇은 항상 배열의 마지막 요소로 선택됩니다 (이를 "리버스 피봇"이라고 합니다).
4. `i` 변수는 "작은 요소"를 저장할 위치를 나타내며, `j` 변수는 배열을 순회하는 포인터입니다.
5. 최종적으로 피봇을 올바른 위치로 이동하고, 이 위치를 기준으로 좌우 서브배열을 재귀적으로 정렬합니다.
