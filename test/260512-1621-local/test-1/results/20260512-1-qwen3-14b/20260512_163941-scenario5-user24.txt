def quicksort(arr, low, high):
    # 배열이 두 개 이상의 요소를 포함할 때만 실행
    if low < high:
        # 파티션을 분할하고, 피벗의 최종 위치를 반환
        pi = partition(arr, low, high)
        
        # 피벗 왼쪽의 부분 배열을 재귀적으로 정렬
        quicksort(arr, low, pi - 1)

        quicksort(arr, pi + 1, high)

def partition(arr, low, high):
    # 피벗은 고정된 마지막 요소로 선택
    pivot = arr[high]
    
    # i는 작은 요소들의 마지막 인덱스를 나타냄
    i = low - 1
    
    # high보다 작은 요소를 왼쪽으로 이동
    for j in range(low, high):
        if arr[j] <= pivot:
            # i를 증가시키고, arr[i]와 arr[j]를 교환
            i += 1
            arr[i], arr[j] = arr[j], arr[i]
    
    # 피벗을 정확한 위치로 이동
    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    
    # 피벗의 최종 위치 반환
    return i + 1
