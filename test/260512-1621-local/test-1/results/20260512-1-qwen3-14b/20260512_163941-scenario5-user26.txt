```python
# 퀵소트(Quick Sort) 알고리즘 구현
# 파이썬에서 리스트를 직접 조작하여 퀵소트를 구현

def quicksort(arr, low, high):
    # 기저 조건: low가 high보다 크면 종료
    if low < high:
        # 파티션을 생성하고 피봇의 최종 위치를 반환
        pi = partition(arr, low, high)
        
        # 피봇을 기준으로 왼쪽 부분 배열 재귀적으로 정렬
        quicksort(arr, low, pi - 1)
        
        # 피봇을 기준으로 오른쪽 부분 배열 재귀적으로 정렬
        quicksort(arr, pi + 1, high)

def partition(arr, low, high):
    # 피봇은 배열의 마지막 요소로 선택
    pivot = arr[high]
    
    # i는 작은 요소의 마지막 위치를 가리킴
    i = low - 1
    
    # j는 배열을 순회하는 포인터
    for j in range(low, high):
        # 현재 요소가 피봇보다 작으면 i를 증가시키고 요소 교환
        if arr[j] < pivot:
            i += 1
            arr[i], arr[j] = arr[j], arr[i]
    
    # 피봇을 올바른 위치로 이동 (i+1)
    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    
    # 피봇의 최종 위치 반환
    return i + 1

# 예제 사용
if __name__ == "__main__":
    sample = [10, 7, 8, 9, 1, 5]
    print("정렬 전 배열:", sample)
    
    # 퀵소트 실행
    quicksort(sample, 0, len(sample) - 1)
    
    print("정렬 후 배열:", sample)
```

### 주석 설명
1. `quicksort` 함수:
   - `low`: 배열의 시작 인덱스
   - `high`: 배열의 끝 인덱스
   - `pi`: 파티션을 생성한 후 피봇의 위치
   - 재귀적으로 왼쪽/오른쪽 부분 배열을 정렬

2. `partition` 함수:
   - `pivot`: 배열의 마지막 요소 선택
   - `i`: 작은 요소의 마지막 위치 추적
   - `j`: 배열을 순회하며 요소 비교
   - `arr[i+1]`에 피봇을 올바른 위치로 이동

3. 예제 실행:
   - `sample` 배열을 생성하고 퀵소트 실행 후 결과 출력
