def quicksort(arr):
    # 기저 조건: 배열의 길이가 0 또는 1이면 이미 정렬되어 있으므로 그대로 반환
    if len(arr) <= 1:
        return arr
    
    # 피봇 선택: 배열의 첫 번째 요소를 피봇으로 사용
    pivot = arr[0]
    
    # 피봇보다 작은 요소를 담을 리스트
    left = []
    # 피봇과 같은 요소를 담을 리스트
    middle = []
    # 피봇보다 큰 요소를 담을 리스트
    right = []
    
    # 배열의 모든 요소를 탐색하여 left, middle, right에 분류
    for num in arr:
        if num < pivot:
            left.append(num)
        elif num == pivot:
            middle.append(num)
        else:
            right.append(num)
    
    # left, middle, right에 대해 재귀적으로 quicksort 실행 후 결과를 결합
    return quicksort(left) + middle + quicksort(right)

# 예제 사용
