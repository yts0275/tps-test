# 퀵 정렬(Quick Sort) 알고리즘 구현
def quicksort(arr):
    # 기저 조건: 배열의 길이가 1 이하일 경우 이미 정렬된 상태이므로 그대로 반환
    if len(arr) <= 1:
        return arr
    
    # 피벗(pivot) 선택: 배열의 마지막 요소를 피벗으로 사용
    pivot = arr[-1]
    
    # 피벗보다 작은 요소를 포함한 왼쪽 배열 생성
    left = [x for x in arr[:-1] if x < pivot]
    
    # 피벗보다 큰 또는 같은 요소를 포함한 오른쪽 배열 생성
    right = [x for x in arr[:-1] if x >= pivot]
    
    # 재귀적으로 왼쪽 배열과 오른쪽 배열을 퀵 정렬 수행
    # 정렬된 왼쪽 배열 + 피벗 + 정렬된 오른쪽 배열을 결합
    return quicksort(left) + [pivot] + quicksort(right)
