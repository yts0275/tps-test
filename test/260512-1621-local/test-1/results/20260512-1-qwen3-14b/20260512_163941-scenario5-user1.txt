def quicksort(arr, low, high):
    # 기저 조건: 배열의 요소가 0개 또는 1개일 때, 이미 정렬된 상태
    if low < high:

def partition(arr, low, high):
    # 피봇으로 마지막 요소를 선택
    pivot = arr[high]
    
    # i는 "작은 요소의 위치"를 추적하는 포인터
    i = low - 1
    
    # j는 배열을 순회하는 포인터
    for j in range(low, high):
        # 현재 요소가 피봇보다 작으면 i를 증가시키고 요소를 교환
        if arr[j] < pivot:
            i += 1
            arr[i], arr[j] = arr[j], arr[i]
    
    # 피봇을 올바른 위치로 이동 (i+1)
    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    
    # 피봇의 최종 위치 반환
    return i + 1
