def quicksort(arr):
    # 기저 조건: 배열의 길이가 0 또는 1일 경우 이미 정렬되어 있으므로 그대로 반환

        return arr
    
    # 피벗 선택: 배열의 마지막 요소를 피벗으로 선택
    pivot = arr[-1]
    
    # 왼쪽 배열: 피벗보다 작은 요소들을 추출
    left = [x for x in arr[:-1] if x < pivot]
    
    # 오른쪽 배열: 피벗보다 크거나 같은 요소들을 추출
    right = [x for x in arr[:-1] if x >= pivot]
    
    # 재귀적 호출: 왼쪽 배열과 오른쪽 배열에 대해 퀵소트 수행

    return quicksort(left) + [pivot] + quicksort(right)
