def quicksort(arr, low, high):
    # 기저 조건: 요소가 하나 이하일 때 종료
    if low < high:
        # 파티션 함수를 호출하여 피봇의 위치를 얻음
        pi = partition(arr, low, high)
        
        # 피봇을 기준으로 좌측과 우측 배열을 재귀적으로 정렬
        quicksort(arr, low, pi - 1)  # 좌측 배열 정렬
        quicksort(arr, pi + 1, high)  # 우측 배열 정렬

def partition(arr, low, high):
    # 피봇을 배열의 마지막 요소로 선택
    pivot = arr[high]
    
    # 피봇보다 작은 값을 가진 요소의 위치를 추적할 인덱스
    i = low - 1  # 초기값은 low - 1 (배열 왼쪽 끝 왼쪽)
    
    # low부터 high-1까지 반복 (피봇은 제외)
    for j in range(low, high):
        # 현재 요소가 피봇보다 작으면 i를 증가시키고 요소를 교환
        if arr[j] < pivot:
            i += 1
            arr[i], arr[j] = arr[j], arr[i]
    
    # 피봇을 i+1 위치로 이동 (모든 작은 요소가 왼쪽에, 큰 요소가 오른쪽에)
    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    
    # 피봇의 최종 위치 반환
    return i + 1

# 예제 실행
arr = [10, 7, 8, 9, 1, 5]
print("정렬 전 배열:", arr)

   - 배열 요소가 1개 이하일 경우 (`low < high` 조건 실패) 종료
   - `partition` 함수를 호출하여 피봇의 위치를 찾고, 좌우 배열을 재귀적으로 정렬
