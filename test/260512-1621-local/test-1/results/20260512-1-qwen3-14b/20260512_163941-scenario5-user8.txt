def quicksort(arr, start, end):
    # 기저 조건: 배열이 하나 이하의 요소를 가지면 종료
    if start >= end:
        return

    # 피벗 선택: 배열의 중앙 요소를 피벗으로 선택
    mid = (start + end) // 2
    pivot = arr[mid]

    # 피벗을 배열의 끝으로 이동 (분할 작업을 용이하게 하기 위해)
    arr[mid], arr[end] = arr[end], arr[mid]
    pivot = arr[end]

    # 분할 작업: i는 왼쪽 포인터, j는 오른쪽 포인터
    i = start
    j = end - 1

    # 포인터 이동
    while i <= j:
        # 왼쪽 포인터 i를 오른쪽으로 이동 (피벗보다 작은 요소 찾기)
        while arr[i] < pivot:
            i += 1

        # 오른쪽 포인터 j를 왼쪽으로 이동 (피벗보다 큰 요소 찾기)
        while arr[j] > pivot:
            j -= 1

        # 포인터가 교차하면 종료
        if i > j:
            break

        # 교차하지 않았다면, i와 j의 요소를 교환
        arr[i], arr[j] = arr[j], arr[i]

    # 피벗을 정확한 위치로 이동 (i와 end 위치 교환)
    arr[i], arr[end] = arr[end], arr[i]

    # 재귀적으로 좌측과 우측 배열에 대해 퀵 정렬 수행
    quicksort(arr, start, i - 1)
    quicksort(arr, i + 1, end)

# 예시 사용

print("정렬된 배열:", arr)
