# 퀵소트 알고리즘 구현 (Python)
# 주어진 리스트를 오름차순으로 정렬하는 함수

def quicksort(arr):
    # 기본 조건: 리스트의 길이가 0 또는 1일 경우 이미 정렬된 상태
    if len(arr) <= 1:
        return arr
    
    # 피벗 선택 (리스트의 중앙 요소)
    mid = len(arr) // 2
    pivot = arr[mid]
    
    # 피벗을 제외한 나머지 요소를 분할
    # left: 피벗보다 작은 요소
    # middle: 피벗과 같은 요소
    # right: 피벗보다 큰 요소
    left = []
    middle = []
    right = []
    
    # 요소를 분할하는 반복문
    for num in arr:
        if num < pivot:
            left.append(num)
        elif num == pivot:
            middle.append(num)
        else:
            right.append(num)
    
    # 재귀적 호출: left와 right 리스트에 대해 퀵소트 수행
    # middle 리스트는 이미 정렬되어 있으므로 그대로 합침
    return quicksort(left) + middle + quicksort(right)

# 예시 사용
