```python
def quicksort(arr):
    # 기저 조건: 배열의 길이가 0 또는 1이면 이미 정렬된 상태
    if len(arr) <= 1:
        return arr

    # 피벗 선택: 배열의 마지막 요소를 피벗으로 선택
    pivot = arr[-1]

    # 피벗을 기준으로 배열을 분할
    # 1. 피벗보다 작은 요소들
    less_than = [x for x in arr[:-1] if x < pivot]
    # 2. 피벗과 같은 요소들
    equal_to = [x for x in arr[:-1] if x == pivot]
    # 3. 피벗보다 큰 요소들
    greater_than = [x for x in arr[:-1] if x > pivot]

    # 재귀적으로 작은 요소와 큰 요소에 대해 퀵소트 실행
    sorted_less = quicksort(less_than)
    sorted_greater = quicksort(greater_than)

    # 정렬된 작은 요소 + 피벗 + 정렬된 큰 요소를 연결
    return sorted_less + equal_to + sorted_greater
```
