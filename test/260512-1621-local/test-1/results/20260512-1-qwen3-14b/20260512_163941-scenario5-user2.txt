def quicksort(arr, low, high):
    # 기저 조건: 요소가 0개 또는 1개일 때 종료
    if low >= high:
        return

    # 피벗 선택 (마지막 요소 선택)
    pivot = arr[high]
    
    # i는 작은 요소의 위치를 가리킴
    i = low - 1
    
    # 배열을 순회하며 피벗보다 작은 요소를 왼쪽으로 이동
    for j in range(low, high):
        if arr[j] < pivot:
            i += 1
            # 작은 요소와 i 위치의 요소 교환
            arr[i], arr[j] = arr[j], arr[i]
    
    # 피벗을 올바른 위치로 이동
    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    
    # 재귀적으로 좌우 배열 정렬
    quicksort(arr, low, i)
    quicksort(arr, i + 2, high)

# 예시 배열 정렬
