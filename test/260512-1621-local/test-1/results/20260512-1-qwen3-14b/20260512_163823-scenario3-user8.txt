def quicksort(arr):
    # 기저 조건: 배열이 비어 있거나 하나의 요소만 있을 때, 그대로 반환
    if len(arr) <= 1:
        return arr
    
    # 피벗 선택: 배열의 중앙 요소를 피벗으로 선택
    pivot = arr[len(arr) // 2]
    
    # 피벗보다 작은 요소들을 담을 리스트
    left = [x for x in arr if x < pivot]
    
    # 피벗보다 큰 또는 같은 요소들을 담을 리스트
    right = [x for x in arr if x >= pivot]
    
    # 재귀적으로 왼쪽과 오른쪽 부분 배열을 정렬하고, 결과를 결합
    return quicksort(left) + [pivot] + quicksort(right)
