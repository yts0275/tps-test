# 퀵소트(Quick Sort) 알고리즘 구현

def quick_sort(arr, low, high):
    # 베이스 케이스: low >= high이면 배열이 정렬되어 있음
    if low < high:
        # 파티션을 나누고 피벗의 인덱스를 반환
        pi = partition(arr, low, high)
        
        # 피벗 기준으로 좌측 배열 정렬
        quick_sort(arr, low, pi - 1)
        
        # 피벗 기준으로 우측 배열 정렬
        quick_sort(arr, pi + 1, high)

def partition(arr, low, high):
    # 피벗을 선택 (일반적으로 마지막 요소)
    pivot = arr[high]
    
    # 피벗보다 작은 요소들의 마지막 인덱스
    i = low - 1
    
    # 배열을 탐색
    for j in range(low, high):
        # 현재 요소가 피벗보다 작으면
        if arr[j] < pivot:
            # i를 증가시키고 요소 교환
            i += 1
            arr[i], arr[j] = arr[j], arr[i]
    
    # 피벗을 올바른 위치로 이동
    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    
    # 피벗의 최종 인덱스 반환
    return i + 1

# 테스트
