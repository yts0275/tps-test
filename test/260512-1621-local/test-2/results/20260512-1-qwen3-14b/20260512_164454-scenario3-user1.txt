def quicksort(arr, start, end):
    # 베이스 케이스: start가 end보다 크면 종료
    if start >= end:
        return

    # 피벗을 선택 (마지막 요소를 피벗으로 사용)
    pivot = arr[end]
    
    # 피벗보다 작은 요소들의 위치를 추적할 인덱스
    pivot_index = start

    # 배열을 탐색하면서 피벗보다 작은 요소를 찾고 교환
    for i in range(start, end):
        if arr[i] < pivot:
            # 피벗보다 작은 요소를 pivot_index 위치로 이동
            arr[pivot_index], arr[i] = arr[i], arr[pivot_index]
            pivot_index += 1  # pivot_index를 한칸 이동

    # 피벗을 정확한 위치로 이동
    arr[pivot_index], arr[end] = arr[end], arr[pivot_index]

    # 왼쪽 부분 배열에 대해 재귀 호출
    quicksort(arr, start, pivot_index - 1)
    
    # 오른쪽 부분 배열에 대해 재귀 호출
    quicksort(arr, pivot_index + 1, end)

# 예시 사용
