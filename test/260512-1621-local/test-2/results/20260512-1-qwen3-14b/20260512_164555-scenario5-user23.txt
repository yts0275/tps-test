```python
# 퀵소트(Quick Sort) 알고리즘 구현 (Python)
# 시간 복잡도: 평균 O(n log n), 최악 O(n²)
# 공간 복잡도: O(log n) (재귀 스택 사용)

def quicksort(arr, low, high):
    # 베이스 케이스: low >= high일 때 종료 (1개 요소 또는 빈 배열)
    if low < high:
        # 파티션 함수를 호출하여 피벗의 위치를 찾고
        # 파티션을 기준으로 왼쪽/오른쪽 배열을 재귀적으로 정렬
        pi = partition(arr, low, high)  # pi는 피벗의 최종 인덱스
        quicksort(arr, low, pi - 1)     # 왼쪽 배열 재귀 정렬
        quicksort(arr, pi + 1, high)    # 오른쪽 배열 재귀 정렬

def partition(arr, low, high):
    # 피벗으로 사용할 요소 선택 (이 예제에서는 마지막 요소)
    pivot = arr[high]
    i = low - 1  # 작은 요소의 인덱스 (초기 -1)

    # low부터 high-1까지 반복
    for j in range(low, high):
        # 현재 요소가 피벗보다 작으면
        if arr[j] < pivot:
            # i를 1 증가시키고 arr[i]와 arr[j]를 교환
            i += 1
            arr[i], arr[j] = arr[j], arr[i]

    # 피벗을 정확한 위치로 이동 (i+1)
    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    # 피벗의 최종 인덱스 반환
    return i + 1

# 사용 예제
if __name__ == "__main__":
    arr = [10, 7, 8, 9, 1, 5]
    print("정렬 전 배열:", arr)
    quicksort(arr, 0, len(arr) - 1)
    print("정렬 후 배열:", arr)
```

2. **partition 함수**:

   - `for` 루프: 배열을 순회하며 피벗보다 작은 요소를 왼쪽으로 이동
