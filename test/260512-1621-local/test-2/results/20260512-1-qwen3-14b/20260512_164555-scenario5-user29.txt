```python
# 퀵소트(Quick Sort) 알고리즘 구현
def quicksort(arr):
    # 기저 조건: 배열의 길이가 0 또는 1일 경우, 이미 정렬된 상태이므로 그대로 반환
    if len(arr) <= 1:
        return arr
    
    # 피벗(pivot) 선택: 배열의 중간 요소를 피봇으로 선택 (성능 향상)
    pivot = arr[len(arr) // 2]
    
    # 파티션 분할:
    # 1. 피봇보다 작은 요소들로 구성된 배열
    left = [x for x in arr if x < pivot]
    # 2. 피봇보다 큰 요소들로 구성된 배열
    right = [x for x in arr if x > pivot]
    # 3. 피봇과 같은 요소들 (중복 처리)
    equal = [x for x in arr if x == pivot]
    
    # 재귀적 호출: 좌측 배열과 우측 배열을 각각 퀵소트 적용
    # 좌측 배열: quicksort(left) -> 정렬된 작은 요소들
    # 우측 배열: quicksort(right) -> 정렬된 큰 요소들
    # 피봇과 동일한 요소들은 그대로 유지 (equal)
    return quicksort(left) + equal + quicksort(right)

# 예시 사용법

    sorted_example = quicksort(example)
    print("정렬 후:", sorted_example)
```
