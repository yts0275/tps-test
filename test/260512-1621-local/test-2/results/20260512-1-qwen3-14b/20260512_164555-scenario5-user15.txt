# 퀵소트 알고리즘을 구현합니다.
def quicksort(arr):
    # 재귀 종료 조건: 배열의 길이가 1 이하일 때는 이미 정렬되어 있음
    if len(arr) <= 1:
        return arr
    
    # 피봇을 선택합니다. 여기서는 배열의 중앙 요소를 피봇으로 사용
    pivot = arr[len(arr) // 2]
    
    # 피봇보다 작은 요소, 같은 요소, 큰 요소로 배열을 분할
    left = [x for x in arr if x < pivot]  # 피봇보다 작은 요소들
    middle = [x for x in arr if x == pivot]  # 피봇과 같은 요소들
    right = [x for x in arr if x > pivot]  # 피봇보다 큰 요소들
    
    # 재귀적으로 좌측과 우측 배열을 정렬하고, 중간 요소를 합쳐 결과를 반환
    # left 배열을 퀵소트로 정렬, middle은 이미 정렬됨, right 배열을 퀵소트로 정렬
    return quicksort(left) + middle + quicksort(right)

# 예시 사용: 정렬되지 않은 배열을 입력하여 퀵소트 실행
unsorted_array = [3, 6, 8, 10, 1, 2, 1]
sorted_array = quicksort(unsorted_array)
print("정렬된 배열:", sorted_array)
