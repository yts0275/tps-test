def quicksort(arr, low, high):
    # 배열이 하나의 요소를 가진 경우 또는 비어있는 경우 재귀 종료
    if low < high:
        # 파티션을 통해 피벗의 정확한 위치를 찾고, 그 위치를 반환
        pi = partition(arr, low, high)
        
        # 피벗을 기준으로 왼쪽 부분 배열을 정렬
        quicksort(arr, low, pi - 1)
        
        # 피벗을 기준으로 오른쪽 부분 배열을 정렬
        quicksort(arr, pi + 1, high)

def partition(arr, low, high):
    # 피벗으로 마지막 요소 선택
    pivot = arr[high]
    
    # 피벗보다 작은 요소들을 왼쪽으로 옮길 인덱스
    i = low - 1
    
    # 배열을 탐색하며 피벗보다 작은 요소를 찾음
    for j in range(low, high):
        if arr[j] < pivot:
            # 작은 요소를 왼쪽으로 이동
            i += 1
            arr[i], arr[j] = arr[j], arr[i]
    
    # 피벗을 올바른 위치로 이동
    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    
    # 피벗의 최종 위치 반환
    return i + 1

# 예제 실행
if __name__ == "__main__":
    arr = [5, 3, 8, 4, 2]
    print("정렬 전 배열:", arr)
    quicksort(arr, 0, len(arr) - 1)
    print("정렬 후 배열:", arr)

1. `quicksort` 함수:
   - 배열을 재귀적으로 정렬하는 메인 함수
   - `low`는 배열의 시작 인덱스, `high`는 배열의 끝 인덱스
   - `partition` 함수를 통해 배열을 분할하고, 분할된 두 부분을 다시 정렬

2. `partition` 함수:
   - 피벗(기준 요소)을 선택하고, 배열을 두 부분으로 나눔
   - `i`는 피벗보다 작은 요소들이 위치할 인덱스
   - `j`는 배열을 순회하며 피벗보다 작은 요소를 찾는 인덱스
   - `i+1` 위치에 피벗을 배치하고, 이 위치를 반환
