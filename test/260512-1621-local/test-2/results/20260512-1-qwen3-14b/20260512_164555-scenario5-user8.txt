def quicksort(arr, low, high):
    # 기저 조건: low >= high이면 재귀 종료
    if low < high:
        # 파티션 함수를 호출하여 피봇의 위치를 구함
        pi = partition(arr, low, high)
        # 피봇 왼쪽 부분 배열 정렬
        quicksort(arr, low, pi - 1)
        # 피봇 오른쪽 부분 배열 정렬
        quicksort(arr, pi + 1, high)

def partition(arr, low, high):
    # 피봇은 배열의 가장 오른쪽 요소로 선택
    pivot = arr[high]
    # 작은 요소의 마지막 인덱스를 나타내는 포인터
    i = low - 1

    # 배열의 low부터 high-1까지 순회
    for j in range(low, high):
        # 현재 요소가 피봇보다 작으면
        if arr[j] < pivot:
            # i를 증가시키고, arr[i]와 arr[j]를 교환
            i += 1
            arr[i], arr[j] = arr[j], arr[i]

    # 피봇을 올바른 위치로 이동
    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    # 피봇의 최종 인덱스 반환
    return i + 1
