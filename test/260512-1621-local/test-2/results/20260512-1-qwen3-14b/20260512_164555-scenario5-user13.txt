def quicksort(arr, low, high):
    # 범위가 유효하지 않으면 종료
    if low < high:
        # 파티션을 통해 피벗의 올바른 위치를 찾고, 그 위치를 반환
        pivot_index = partition(arr, low, high)
        
        # 피벗을 기준으로 왼쪽 부분 배열을 재귀적으로 정렬
        quicksort(arr, low, pivot_index - 1)
        
        # 피벗을 기준으로 오른쪽 부분 배열을 재귀적으로 정렬
        quicksort(arr, pivot_index + 1, high)

def partition(arr, low, high):
    # 피벗으로 사용할 요소를 선택 (마지막 요소)
    pivot = arr[high]
    
    # 피벗보다 작은 요소들을 이동시키기 위한 인덱스
    i = low - 1
    
    # 배열을 순회하며 피벗보다 작은 요소를 찾음
    for j in range(low, high):
        # 현재 요소가 피벗보다 작으면
        if arr[j] < pivot:
            # i를 증가시키고, i와 j 위치의 요소를 교환
            i += 1
            arr[i], arr[j] = arr[j], arr[i]
    
    # 피벗을 올바른 위치(현재 i+1)로 이동
    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    
    # 피벗의 최종 위치를 반환
    return i + 1

# 테스트용 예시
