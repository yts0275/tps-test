```python
# 퀵소트(Quick Sort) 알고리즘을 구현한 함수
def quick_sort(arr):
    # 기저 조건: 배열의 길이가 0 또는 1일 경우 이미 정렬된 상태이므로 그대로 반환
    if len(arr) <= 1:
        return arr

    # 피벗(pivot)을 선택: 여기서는 배열의 마지막 요소를 피벗으로 사용
    pivot = arr[-1]

    # 피벗보다 작은 요소들을 담을 리스트
    left = []
    # 피벗보다 큰 요소들을 담을 리스트
    right = []

    # 배열의 요소를 순회하며 피벗과 비교
    for i in range(len(arr) - 1):
        # 현재 요소가 피벗보다 작을 경우 left 리스트에 추가
        if arr[i] < pivot:
            left.append(arr[i])
        # 현재 요소가 피벗보다 클 경우 right 리스트에 추가
        else:
            right.append(arr[i])

    # 재귀적으로 left 리스트와 right 리스트를 정렬하고, 피벗을 중간에 삽입하여 결과 반환
    return quick_sort(left) + [pivot] + quick_sort(right)

# 예시 배열 정의
example_arr = [5, 3, 8, 4, 2, 7, 1, 6]

# 퀵소트 실행
sorted_arr = quick_sort(example_arr)

# 정렬된 결과 출력
print("정렬된 배열:", sorted_arr)
```
