# 퀵소트(Quick Sort) 알고리즘 구현

def quicksort(arr, low, high):
    # 배열이 하나의 요소 이하인 경우 종료
    if low < high:
        # 파티션을 나누고, 피벗의 위치를 반환
        pivot_index = partition(arr, low, high)
        
        # 피벗 왼쪽 부분 배열을 재귀적으로 정렬
        quicksort(arr, low, pivot_index - 1)
        
        # 피벗 오른쪽 부분 배열을 재귀적으로 정렬
        quicksort(arr, pivot_index + 1, high)

def partition(arr, low, high):
    # 피벗을 마지막 요소로 선택
    pivot = arr[high]
    
    # i는 작은 요소의 마지막 위치를 가리킴
    i = low - 1
    
    # j는 배열을 탐색하는 포인터
    for j in range(low, high):
        # 현재 요소가 피벗보다 작으면 i를 증가시키고 요소를 교환
        if arr[j] < pivot:
            i += 1
            arr[i], arr[j] = arr[j], arr[i]
    
    # 피벗을 정확한 위치로 이동
    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    
    # 피벗의 최종 위치를 반환
    return i + 1

# 예제 실행
arr = [10, 7, 8, 9, 1, 5]
print("정렬 전 배열:", arr)
quicksort(arr, 0, len(arr) - 1)
print("정렬 후 배열:", arr)
