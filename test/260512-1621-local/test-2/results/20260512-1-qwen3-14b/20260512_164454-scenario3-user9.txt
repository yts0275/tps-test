def partition(arr, low, high):
    # 피벗으로 마지막 요소를 선택
    pivot = arr[high]
    
    # 피벗보다 작은 요소의 위치를 추적하는 인덱스
    i = low - 1
    
    # 배열을 순회하며 피벗보다 작은 요소를 찾음
    for j in range(low, high):
        if arr[j] < pivot:
            # 작은 요소를 i 위치와 교환
            i += 1
            arr[i], arr[j] = arr[j], arr[i]

    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    
    # 피벗의 최종 인덱스 반환
    return i + 1
