def quicksort(arr):
    # 기저 조건: 배열의 길이가 0 또는 1일 경우 이미 정렬된 상태
    if len(arr) <= 1:
        return arr

    # 피벗 선택: 배열의 중앙 요소를 피벗으로 선택
    pivot = arr[len(arr) // 2]

    # 피벗보다 작은 요소들을 담을 왼쪽 배열
    left = [x for x in arr if x < pivot]
    # 피벗보다 큰 요소들을 담을 오른쪽 배열
    right = [x for x in arr if x > pivot]

    # 재귀적으로 왼쪽 배열과 오른쪽 배열을 정렬하고, 피벗을 중간에 삽입
    return quicksort(left) + [pivot] + quicksort(right)

# 예시 사용

sorted_example = quicksort(example)
