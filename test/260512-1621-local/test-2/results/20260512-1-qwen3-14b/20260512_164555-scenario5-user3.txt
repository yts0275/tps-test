```python
# 퀵소트(Quick Sort) 알고리즘을 구현하는 함수 정의
def quicksort(arr):
    # 기저 조건: 배열의 길이가 0 또는 1일 경우 이미 정렬된 상태
    if len(arr) <= 1:
        return arr

    # 피벗(pivot)을 선택 (이 예제에서는 중앙의 요소를 선택)
    pivot = arr[len(arr) // 2]

    # 피벗보다 작은 요소를 담을 리스트
    left = [x for x in arr if x < pivot]
    # 피벗보다 큰 요소를 담을 리스트
    right = [x for x in arr if x > pivot]

    # 피벗을 기준으로 왼쪽과 오른쪽 배열을 재귀적으로 정렬
    # 왼쪽 배열과 오른쪽 배열을 정렬한 후 피벗을 중간에 삽입
    return quicksort(left) + [pivot] + quicksort(right)

# 예제 배열
example = [3, 6, 8, 10, 1, 2, 1]

# 퀵소트 실행
sorted_example = quicksort(example)

# 결과 출력
print("정렬된 배열:", sorted_example)
```
