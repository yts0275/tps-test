# 퀵소트 알고리즘 구현 (Python)
# 배열을 정렬하는 퀵소트 함수
def quicksort(arr, low, high):
    # 종료 조건: low가 high보다 크거나 같을 때 재귀 종료
    if low < high:
        # 파티션을 수행하여 피봇의 위치를 반환
        pi = partition(arr, low, high)
        # 피봇 왼쪽의 배열을 재귀적으로 정렬
        quicksort(arr, low, pi - 1)
        # 피봇 오른쪽의 배열을 재귀적으로 정렬
        quicksort(arr, pi + 1, high)

# 배열을 파티션하는 함수 (피봇 기준으로 분할)
def partition(arr, low, high):
    # 피봇을 배열의 마지막 요소로 선택
    pivot = arr[high]
    # 피봇보다 작은 요소들의 마지막 위치를 저장할 포인터
    i = low - 1
    # low부터 high-1까지 반복
    for j in range(low, high):
        # 현재 요소가 피봇보다 작으면
        if arr[j] < pivot:
            # i를 증가시키고 해당 요소와 교환
            i += 1
            arr[i], arr[j] = arr[j], arr[i]
    # 마지막으로 피봇을 적절한 위치로 이동
    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    # 피봇의 최종 위치 반환
    return i + 1

# 테스트 코드
if __name__ == "__main__":

    # 배열 복사 (원본 보호)
    arr_copy = arr[:]

2. **partition 함수**:
