def quicksort(arr, low, high):
    # 기저 조건: low가 high보다 크거나 같으면 종료
    if low < high:
        # 파티션 함수를 호출하여 pivot의 위치를 반환
        pivot_index = partition(arr, low, high)
        # pivot의 왼쪽 부분 배열을 재귀적으로 정렬
        quicksort(arr, low, pivot_index - 1)
        # pivot의 오른쪽 부분 배열을 재귀적으로 정렬
        quicksort(arr, pivot_index + 1, high)

def partition(arr, low, high):
    # pivot으로 사용할 요소를 선택 (마지막 요소)
    pivot = arr[high]
    # pivot보다 작은 요소의 위치를 추적하는 인덱스
    i = low - 1
    # low부터 high-1까지 반복
    for j in range(low, high):
        # 현재 요소가 pivot보다 작으면
        if arr[j] < pivot:
            # i를 증가시키고 arr[i]와 arr[j]를 교환
            i += 1
            arr[i], arr[j] = arr[j], arr[i]
    # pivot을 올바른 위치로 이동 (i+1)
    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    # pivot의 최종 인덱스 반환
    return i + 1

# 예시 배열
arr = [10, 7, 8, 9, 1, 5]
# 배열의 첫 인덱스와 마지막 인덱스를 전달
quicksort(arr, 0, len(arr) - 1)
# 정렬된 배열 출력
print("정렬된 배열:", arr)

   - `low < high`일 때만 실행 (기저 조건)
   - `partition` 함수를 호출하여 pivot의 위치를 찾고, pivot 좌우를 재귀적으로 정렬

   - `pivot`은 배열의 마지막 요소

   - `i`와 `j`의 요소를 교환하여 작은 요소를 왼쪽으로 이동
   - 마지막에 pivot을 올바른 위치로 이동하고 인덱스 반환
