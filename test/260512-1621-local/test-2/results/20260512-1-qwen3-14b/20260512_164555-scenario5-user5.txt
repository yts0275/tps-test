# 퀵소트 알고리즘 구현 (Python)
# 배열을 정렬하는 함수: quicksort(array, start, end)
# array: 정렬할 배열
# start: 시작 인덱스
# end: 끝 인덱스

def quicksort(array, start, end):
    # 배열의 시작 인덱스가 끝 인덱스보다 작을 때만 실행 (종료 조건)
    if start < end:
        # 파티션 함수를 호출하여 피벗의 위치를 얻음
        pivot_index = partition(array, start, end)
        # 피벗 왼쪽 배열을 재귀적으로 정렬

        # 피벗 오른쪽 배열을 재귀적으로 정렬

# 파티션 함수: 배열을 피벗 기준으로 분할하고 피벗의 최종 위치를 반환
def partition(array, start, end):
    # 피벗을 선택 (보통 마지막 요소를 선택)
    pivot = array[end]
    # 피벗보다 작은 요소들이 놓일 위치를 지정
    i = start - 1  # i는 작은 요소들의 마지막 인덱스

    # start부터 end-1까지 순회
    for j in range(start, end):
        # 현재 요소가 피벗보다 작으면
        if array[j] < pivot:
            # i를 증가시키고, array[i]와 array[j]를 교환
            i += 1
            array[i], array[j] = array[j], array[i]

    # 마지막으로 피벗을 정확한 위치로 이동
    array[i + 1], array[end] = array[end], array[i + 1]
    # 피벗의 최종 인덱스 반환
    return i + 1

# 예시 사용

   - `j` 변수는 배열을 순회하며 피벗보다 작은 요소를 찾음
