def quicksort(arr, low, high):
    # 배열이 하나의 원소 이하일 경우 종료
    if low < high:
        # 피봇을 선택 (이 예제에서는 마지막 원소 사용)
        pivot = arr[high]
        
        # 피봇을 기준으로 배열을 분할
        # i는 작은 값의 마지막 인덱스를 가리킴
        i = low - 1
        
        # 배열을 순회하면서 피봇보다 작은 값을 찾음
        for j in range(low, high):
            if arr[j] < pivot:
                # 작은 값을 i+1 위치로 이동
                i += 1
                arr[i], arr[j] = arr[j], arr[i]
        
        # 피봇을 올바른 위치로 이동
        arr[i+1], arr[high] = arr[high], arr[i+1]
        
        # 피봇을 기준으로 좌우 서브배열을 재귀적으로 정렬
        quicksort(arr, low, i)
        quicksort(arr, i+2, high)
