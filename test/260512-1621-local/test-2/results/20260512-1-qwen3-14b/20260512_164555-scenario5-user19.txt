def quicksort(arr):
    # 배열이 0개 또는 1개 요소일 경우 이미 정렬되어 있으므로 바로 반환
    if len(arr) <= 1:
        return arr

    # 피벗 요소 선택: 배열의 중앙 요소를 선택 (다른 선택도 가능)
    pivot = arr[len(arr) // 2]
    # 피벗보다 작은 요소를 담을 리스트
    left = []
    # 피벗보다 큰 요소를 담을 리스트
    right = []
    # 피벗과 같은 요소를 담을 리스트 (중복 요소가 있을 경우)
    equal = []

    # 배열을 반복하면서 요소를 left, equal, right에 분류
    for num in arr:
        if num < pivot:
            left.append(num)
        elif num > pivot:
            right.append(num)
        else:
            equal.append(num)

    # 분할된 left, equal, right 배열을 재귀적으로 정렬하고, 합치기
    # left 배열을 정렬 -> equal 배열을 정렬 -> right 배열을 정렬 -> 모두 합치면 전체 배열이 정렬됨
    return quicksort(left) + equal + quicksort(right)

# 예시 사용
example = [3, 6, 8, 10, 1, 2, 1]
