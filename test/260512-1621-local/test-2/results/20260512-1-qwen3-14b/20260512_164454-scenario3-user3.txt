def quicksort(arr, low, high):
    # base case: if the array has one element or less, it's already sorted
    if low < high:
        # partition the array and get the pivot index
        pivot_index = partition(arr, low, high)
        # recursively sort the elements before the pivot
        quicksort(arr, low, pivot_index - 1)
        # recursively sort the elements after the pivot
        quicksort(arr, pivot_index + 1, high)

def partition(arr, low, high):
    # choose the last element as pivot
    pivot = arr[high]
    # initialize pointer for the greater element
    i = low - 1  # i marks the boundary of elements less than pivot

    # iterate through the array
    for j in range(low, high):
        # if current element is less than or equal to pivot
        if arr[j] <= pivot:
            # move the boundary of elements less than pivot
            i += 1
            # swap the boundary element with current element
            arr[i], arr[j] = arr[j], arr[i]

    # swap the pivot element with the element at i+1 (correct position for pivot)
    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    # return the partition index
    return i + 1
