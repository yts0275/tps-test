def quicksort(arr):
    # 기저 조건: 배열의 길이가 0 또는 1일 경우 이미 정렬된 상태
    if len(arr) <= 1:
        return arr
    
    # 피벗 선택: 배열의 중간 요소를 피봇으로 선택 (중간 요소를 선택하는 방법)
    mid = len(arr) // 2
    pivot = arr[mid]
    
    # 피벗보다 작은 요소를 담는 리스트
    left = [x for x in arr if x < pivot]
    
    # 피벗보다 큰 또는 같은 요소를 담는 리스트
    right = [x for x in arr if x >= pivot]
    
    # 재귀적으로 왼쪽과 오른쪽 리스트를 정렬하고, 피벗을 중간에 삽입하여 합친다
    return quicksort(left) + [pivot] + quicksort(right)

# 예제 사용
example_list = [3, 6, 8, 10, 1, 2, 1]
sorted_list = quicksort(example_list)
print("정렬된 배열:", sorted_list)
